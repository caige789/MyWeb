   107→export default function SnakeGame({ onBack }: SnakeGameProps) {
   108→  const canvasRef = useRef<HTMLCanvasElement>(null);
   109→  const gameLoopRef = useRef<number | null>(null);
   110→  const snakeRef = useRef<Position[]>([]);
   111→  const directionRef = useRef<Direction>(Direction.Right);
   112→  const nextDirectionRef = useRef<Direction>(Direction.Right);
   113→  const foodRef = useRef<Food>({ x: 15, y: 10, type: FoodType.Normal });
   114→  const powerUpRef = useRef<PowerUp | null>(null);
   115→  const powerUpTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   116→  const scoreRef = useRef<number>(0);
   117→  const foodsEatenRef = useRef<number>(0);
   118→  const difficultyRef = useRef<number>(0);
   119→  const gameStatusRef = useRef<GameStatus>('idle');
   120→  const touchStartRef = useRef<{ x: number; y: number } | null>(null);
   121→  const containerRef = useRef<HTMLDivElement>(null);
   122→  const dprRef = useRef(1);
   123→  const scaleRef = useRef(1);
   124→  // Active effects
   125→  const hasShieldRef = useRef<boolean>(false);
   126→  const isDoubleScoreRef = useRef<boolean>(false);
   127→  const isSpeedBoostedRef = useRef<boolean>(false);
   128→  const isSlowedRef = useRef<boolean>(false);
   129→  const currentSpeedRef = useRef<number>(BASE_SPEED);
   130→  // Power-up spawn timer
   131→  const powerUpSpawnTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);
   132→
   133→  const [score, setScore] = useState<number>(0);
   134→  const [highScore, setHighScore] = useState<number>(0);
   135→  const [gameStatus, setGameStatus] = useState<GameStatus>('idle');
   136→  const [isNewRecord, setIsNewRecord] = useState<boolean>(false);
   137→  const [loadingHighScore, setLoadingHighScore] = useState<boolean>(true);
   138→  const [paused, setPaused] = useState<boolean>(false);
   139→  const pausedRef = useRef<boolean>(false);
   140→  // UI for active effects
   141→  const [activeEffects, setActiveEffects] = useState<string[]>([]);
   142→  const [difficulty, setDifficulty] = useState(0);
   143→
   144→  const updateActiveEffectsUI = useCallback(() => {
   145→    const effects: string[] = [];
   146→    if (hasShieldRef.current) effects.push('🛡 护盾');
   147→    if (isDoubleScoreRef.current) effects.push('★ 双倍分数');
   148→    if (isSpeedBoostedRef.current) effects.push('⚡ 加速');
   149→    if (isSlowedRef.current) effects.push('❄ 减速');
   150→    setActiveEffects(effects);
   151→  }, []);
   152→
   153→  const getSpeed = useCallback(() => {
   154→    const base = BASE_SPEED - difficultyRef.current * SPEED_REDUCTION_PER_LEVEL;
   155→    let speed = Math.max(base, BASE_SPEED - SPEED_REDUCTION_PER_LEVEL * MAX_DIFFICULTY);
   156→    if (isSpeedBoostedRef.current) speed = Math.max(speed * 0.55, 40);
   157→    if (isSlowedRef.current) speed = speed * 2;
   158→    return speed;
   159→  }, []);
   160→
   161→  const restartGameLoop = useCallback((speed?: number) => {
   162→    if (gameLoopRef.current) {
   163→      clearInterval(gameLoopRef.current);
   164→      gameLoopRef.current = null;
   165→    }
   166→    if (gameStatusRef.current === 'playing' && !pausedRef.current) {
   167→      gameLoopRef.current = window.setInterval(
   168→        () => { /* noop, actual step called from gameStep */ },
   169→        speed ?? getSpeed()
   170→      );
   171→    }
   172→  }, [getSpeed]);
   173→
   174→  const clearAllTimers = useCallback(() => {
   175→    if (gameLoopRef.current) {
   176→      clearInterval(gameLoopRef.current);
   177→      gameLoopRef.current = null;
   178→    }
   179→    if (powerUpTimerRef.current) {
   180→      clearTimeout(powerUpTimerRef.current);
   181→      powerUpTimerRef.current = null;
   182→    }
   183→    if (powerUpSpawnTimerRef.current) {
   184→      clearInterval(powerUpSpawnTimerRef.current);
   185→      powerUpSpawnTimerRef.current = null;
   186→    }
   187→  }, []);
   188→
   189→  const applyPowerUpEffect = useCallback((kind: PowerUpKind) => {
   190→    if (kind === PowerUpKind.Shield) {
   191→      hasShieldRef.current = true;
   192→      updateActiveEffectsUI();
   193→      return;
   194→    }
   195→
   196→    let duration = 0;
   197→    if (kind === PowerUpKind.Speed) {
   198→      isSpeedBoostedRef.current = true;
   199→      isSlowedRef.current = false;
   200→      duration = 3000;
   201→    } else if (kind === PowerUpKind.Slow) {
   202→      isSlowedRef.current = true;
   203→      isSpeedBoostedRef.current = false;
   204→      duration = 5000;
   205→    } else if (kind === PowerUpKind.DoubleScore) {
   206→      isDoubleScoreRef.current = true;
   207→      duration = 10000;
   208→    }
   209→
   210→    updateActiveEffectsUI();
   211→
   212→    if (powerUpTimerRef.current) {
   213→      clearTimeout(powerUpTimerRef.current);
   214→    }
   215→    powerUpTimerRef.current = setTimeout(() => {
   216→      if (kind === PowerUpKind.Speed) isSpeedBoostedRef.current = false;
   217→      if (kind === PowerUpKind.Slow) isSlowedRef.current = false;
   218→      if (kind === PowerUpKind.DoubleScore) isDoubleScoreRef.current = false;
   219→      powerUpTimerRef.current = null;
   220→      updateActiveEffectsUI();
   221→      // Restart loop with updated speed
   222→      const newSpeed = getSpeed();
   223→      currentSpeedRef.current = newSpeed;
   224→      if (gameStatusRef.current === 'playing' && !pausedRef.current) {
   225→        if (gameLoopRef.current) clearInterval(gameLoopRef.current);
   226→        gameLoopRef.current = window.setInterval(gameStepRef.current, newSpeed);
   227→      }
   228→    }, duration);
   229→  }, [updateActiveEffectsUI, getSpeed]);
   230→
   231→  const initSnake = useCallback(() => {
   232→    const centerX = Math.floor(GRID_SIZE / 2);
   233→    const centerY = Math.floor(GRID_SIZE / 2);
   234→    snakeRef.current = [
   235→      { x: centerX, y: centerY },
   236→      { x: centerX - 1, y: centerY },
   237→      { x: centerX - 2, y: centerY },
   238→    ];
   239→  }, []);
   240→
   241→  // Check if a position is occupied by snake or power-up
   242→  const isOccupied = useCallback((pos: Position, exclude?: Position) => {
   243→    for (const seg of snakeRef.current) {
   244→      if (seg.x === pos.x && seg.y === pos.y) return true;
   245→    }
   246→    if (powerUpRef.current && powerUpRef.current.x === pos.x && powerUpRef.current.y === pos.y) return true;
   247→    if (exclude && exclude.x === pos.x && exclude.y === pos.y) return false;
   248→    return false;
   249→  }, []);
   250→
   251→  const spawnFood = useCallback(() => {
   252→    const snake = snakeRef.current;
   253→    let pos: Position;
   254→    let attempts = 0;
   255→    do {
   256→      pos = {
   257→        x: Math.floor(Math.random() * GRID_SIZE),
   258→        y: Math.floor(Math.random() * GRID_SIZE),
   259→      };
   260→      attempts++;
   261→    } while (snake.some((seg) => seg.x === pos.x && seg.y === pos.y) && attempts < 200);
   262→
   263→    // Decide food type: 10% gold, 8% purple, 82% normal
   264→    const roll = Math.random();
   265→    let type = FoodType.Normal;
   266→    if (roll < 0.10) type = FoodType.Gold;
   267→    else if (roll < 0.18) type = FoodType.Purple;
   268→
   269→    foodRef.current = { x: pos.x, y: pos.y, type };
   270→  }, []);
   271→
   272→  const spawnPowerUp = useCallback(() => {
   273→    if (powerUpRef.current) return; // Only one power-up at a time
   274→    if (gameStatusRef.current !== 'playing' || pausedRef.current) return;
   275→
   276→    let pos: Position;
   277→    let attempts = 0;
   278→    do {
   279→      pos = {
   280→        x: Math.floor(Math.random() * GRID_SIZE),
   281→        y: Math.floor(Math.random() * GRID_SIZE),
   282→      };
   283→      attempts++;
   284→    } while (isOccupied(pos) || (foodRef.current.x === pos.x && foodRef.current.y === pos.y) && attempts < 200);
   285→
   286→    if (attempts >= 200) return;
   287→
   288→    const kinds = [PowerUpKind.Speed, PowerUpKind.Shield, PowerUpKind.Slow, PowerUpKind.DoubleScore];
   289→    const kind = kinds[Math.floor(Math.random() * kinds.length)];
   290→    powerUpRef.current = { x: pos.x, y: pos.y, kind, frame: 0 };
   291→  }, [isOccupied]);
   292→
   293→  const draw = useCallback(() => {
   294→    const canvas = canvasRef.current;
   295→    if (!canvas) return;
   296→    const ctx = canvas.getContext('2d');
   297→    if (!ctx) return;
   298→
   299→    ctx.setTransform(dprRef.current * scaleRef.current, 0, 0, dprRef.current * scaleRef.current, 0, 0);
   300→    const size = LOGIC_SIZE;
   301→    const cellSize = LOGIC_SIZE / GRID_SIZE;
   302→
   303→    // Background
   304→    ctx.fillStyle = '#1a1a2e';
   305→    ctx.fillRect(0, 0, size, size);
   306→
   307→    // Grid
   308→    ctx.strokeStyle = 'rgba(255, 255, 255, 0.06)';
   309→    ctx.lineWidth = 0.5;
   310→    for (let i = 0; i <= GRID_SIZE; i++) {
   311→      const pos = i * cellSize;
   312→      ctx.beginPath(); ctx.moveTo(pos, 0); ctx.lineTo(pos, size); ctx.stroke();
   313→      ctx.beginPath(); ctx.moveTo(0, pos); ctx.lineTo(size, pos); ctx.stroke();
   314→    }
   315→
   316→    // Draw power-up
   317→    const pu = powerUpRef.current;
   318→    if (pu) {
   319→      const config = POWERUP_CONFIG[pu.kind];
   320→      const cx = pu.x * cellSize + cellSize / 2;
   321→      const cy = pu.y * cellSize + cellSize / 2;
   322→      const radius = Math.max(cellSize / 2 - 1, 1);
   323→      const pulse = Math.sin(pu.frame * 0.15) * 0.25 + 0.75;
   324→
   325→      ctx.shadowColor = config.glow;
   326→      ctx.shadowBlur = 10 * pulse;
   327→      ctx.fillStyle = config.color;
   328→      ctx.globalAlpha = pulse;
   329→      ctx.beginPath();
   330→      ctx.arc(cx, cy, radius, 0, Math.PI * 2);
   331→      ctx.fill();
   332→      ctx.globalAlpha = 1;
   333→      ctx.shadowBlur = 0;
   334→
   335→      // Symbol
   336→      ctx.fillStyle = '#ffffff';
   337→      ctx.font = `bold ${Math.round(cellSize * 0.6)}px sans-serif`;
   338→      ctx.textAlign = 'center';
   339→      ctx.textBaseline = 'middle';
   340→      ctx.fillText(config.symbol, cx, cy + 1);
   341→    }
   342→
   343→    // Draw food
   344→    const food = foodRef.current;
   345→    const foodCx = food.x * cellSize + cellSize / 2;
   346→    const foodCy = food.y * cellSize + cellSize / 2;
   347→    const foodR = Math.max(cellSize / 2 - 2, 1);
   348→    const fConfig = FOOD_CONFIG[food.type];
   349→
   350→    ctx.shadowColor = fConfig.glow;
   351→    ctx.shadowBlur = food.type === FoodType.Gold ? 14 : 8;
   352→    ctx.fillStyle = fConfig.color;
   353→    ctx.beginPath();
   354→    ctx.arc(foodCx, foodCy, foodR, 0, Math.PI * 2);
   355→    ctx.fill();
   356→    ctx.shadowBlur = 0;
   357→
   358→    // Food label for special foods
   359→    if (food.type !== FoodType.Normal && fConfig.label) {
   360→      ctx.fillStyle = '#ffffff';
   361→      ctx.font = `bold ${Math.round(cellSize * 0.5)}px sans-serif`;
   362→      ctx.textAlign = 'center';
   363→      ctx.textBaseline = 'middle';
   364→      ctx.fillText(fConfig.label, foodCx, foodCy + 1);
   365→    }
   366→
   367→    // Draw snake
   368→    const snake = snakeRef.current;
   369→    snake.forEach((seg, index) => {
   370→      const x = seg.x * cellSize;
   371→      const y = seg.y * cellSize;
   372→      const padding = 1;
   373→      const radius = 3;
   374→
   375→      if (index === 0) {
   376→        ctx.shadowColor = '#4ade80';
   377→        ctx.shadowBlur = 6;
   378→        ctx.fillStyle = '#4ade80';
   379→      } else {
   380→        ctx.shadowBlur = 0;
   381→        const ratio = 1 - (index / snake.length) * 0.5;
   382→        const g = Math.round(180 * ratio + 60);
   383→        ctx.fillStyle = `rgb(34, ${g}, 34)`;
   384→      }
   385→
   386→      // Shield glow on head
   387→      if (index === 0 && hasShieldRef.current) {
   388→        ctx.shadowColor = '#38bdf8';
   389→        ctx.shadowBlur = 12;
   390→        ctx.strokeStyle = 'rgba(56, 189, 248, 0.6)';
   391→        ctx.lineWidth = 2;
   392→        ctx.beginPath();
   393→        ctx.arc(x + cellSize / 2, y + cellSize / 2, cellSize / 2 + 2, 0, Math.PI * 2);
   394→        ctx.stroke();
   395→        ctx.shadowColor = '#4ade80';
   396→        ctx.shadowBlur = 6;
   397→      }
   398→
   399→      const rx = x + padding;
   400→      const ry = y + padding;
   401→      const rw = cellSize - padding * 2;
   402→      const rh = cellSize - padding * 2;
   403→      ctx.beginPath();
   404→      ctx.moveTo(rx + radius, ry);
   405→      ctx.lineTo(rx + rw - radius, ry);
   406→      ctx.quadraticCurveTo(rx + rw, ry, rx + rw, ry + radius);
   407→      ctx.lineTo(rx + rw, ry + rh - radius);
   408→      ctx.quadraticCurveTo(rx + rw, ry + rh, rx + rw - radius, ry + rh);
   409→      ctx.lineTo(rx + radius, ry + rh);
   410→      ctx.quadraticCurveTo(rx, ry + rh, rx, ry + rh - radius);
   411→      ctx.lineTo(rx, ry + radius);
   412→      ctx.quadraticCurveTo(rx, ry, rx + radius, ry);
   413→      ctx.closePath();
   414→      ctx.fill();
   415→    });
   416→
   417→    ctx.shadowBlur = 0;
   418→
   419→    // Game over overlay
   420→    if (gameStatusRef.current === 'over') {
   421→      ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';
   422→      ctx.fillRect(0, 0, size, size);
   423→      ctx.fillStyle = '#ffffff';
   424→      ctx.font = `bold ${Math.round(cellSize * 1.5)}px sans-serif`;
   425→      ctx.textAlign = 'center';
   426→      ctx.textBaseline = 'middle';
   427→      ctx.fillText('游戏结束', size / 2, size / 2 - cellSize);
   428→      ctx.font = `${Math.round(cellSize * 0.9)}px sans-serif`;
   429→      ctx.fillStyle = '#fbbf24';
   430→      ctx.fillText(`得分: ${scoreRef.current}`, size / 2, size / 2 + cellSize);
   431→    }
   432→
   433→    // Paused overlay
   434→    if (pausedRef.current && gameStatusRef.current === 'playing') {
   435→      ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';
   436→      ctx.fillRect(0, 0, size, size);
   437→      ctx.fillStyle = '#ffffff';
   438→      ctx.font = `bold ${Math.round(cellSize * 1.5)}px sans-serif`;
   439→      ctx.textAlign = 'center';
   440→      ctx.textBaseline = 'middle';
   441→      ctx.fillText('已暂停', size / 2, size / 2);
   442→    } else if (gameStatusRef.current === 'idle') {
   443→      ctx.fillStyle = 'rgba(0, 0, 0, 0.5)';
   444→      ctx.fillRect(0, 0, size, size);
   445→      ctx.fillStyle = '#ffffff';
   446→      ctx.font = `bold ${Math.round(cellSize * 1.2)}px sans-serif`;
   447→      ctx.textAlign = 'center';
   448→      ctx.textBaseline = 'middle';
   449→      ctx.fillText('按「开始游戏」', size / 2, size / 2);
   450→    }
   451→  }, []);
   452→
   453→  // We need a ref to gameStep for the power-up timer callback
   454→  const gameStepRef = useRef<() => void>(() => {});
   455→
   456→  const endGame = useCallback(async () => {
   457→    gameStatusRef.current = 'over';
   458→    setGameStatus('over');
   459→    clearAllTimers();
   460→    draw();
   461→
   462→    if (scoreRef.current > highScore) {
   463→      setIsNewRecord(true);
   464→      try {
   465→        const res = await fetch('/api/games/scores', {
   466→          method: 'POST',
   467→          headers: { 'Content-Type': 'application/json' },
   468→          body: JSON.stringify({ game: GAME_NAME, score: scoreRef.current }),
   469→        });
   470→        if (res.ok) {
   471→          const data = await res.json();
   472→          if (data.data?.score !== undefined) setHighScore(data.data.score);
   473→        }
   474→      } catch { /* silent */ }
   475→    } else {
   476→      setIsNewRecord(false);
   477→    }
   478→  }, [highScore, draw, clearAllTimers]);
   479→
   480→  const gameStep = useCallback(() => {
   481→    const snake = snakeRef.current;
   482→    const head = { ...snake[0] };
   483→
   484→    directionRef.current = nextDirectionRef.current;
   485→    switch (directionRef.current) {
   486→      case Direction.Up: head.y -= 1; break;
   487→      case Direction.Down: head.y += 1; break;
   488→      case Direction.Left: head.x -= 1; break;
   489→      case Direction.Right: head.x += 1; break;
   490→    }
   491→
   492→    // Wall wrap (pass-through mode)
   493→    if (head.x < 0) head.x = GRID_SIZE - 1;
   494→    else if (head.x >= GRID_SIZE) head.x = 0;
   495→    if (head.y < 0) head.y = GRID_SIZE - 1;
   496→    else if (head.y >= GRID_SIZE) head.y = 0;
   497→
   498→    // Self collision
   499→    if (snake.some((seg) => seg.x === head.x && seg.y === head.y)) {
   500→      if (hasShieldRef.current) {
   501→        hasShieldRef.current = false;
   502→        updateActiveEffectsUI();
   503→        // Don't move, skip this frame
   504→        draw();
   505→        return;
   506→      }
   507→      endGame();
   508→      return;
   509→    }
   510→
   511→    snake.unshift(head);
   512→
   513→    // Check power-up collision
   514→    if (powerUpRef.current && head.x === powerUpRef.current.x && head.y === powerUpRef.current.y) {
   515→      applyPowerUpEffect(powerUpRef.current.kind);
   516→      powerUpRef.current = null;
   517→      // Restart loop if speed changed
   518→      if (isSpeedBoostedRef.current || isSlowedRef.current) {
   519→        const newSpeed = getSpeed();
   520→        currentSpeedRef.current = newSpeed;
   521→        if (gameLoopRef.current) clearInterval(gameLoopRef.current);
   522→        gameLoopRef.current = window.setInterval(gameStepRef.current, newSpeed);
   523→      }
   524→    } else {
   525→      // Update power-up animation frame
   526→      if (powerUpRef.current) powerUpRef.current.frame += 1;
   527→    }
   528→
   529→    // Check food collision
   530→    const food = foodRef.current;
   531→    if (head.x === food.x && head.y === food.y) {
   532→      let gained = 0;
   533→      if (food.type === FoodType.Gold) {
   534→        gained = 30;
   535→      } else if (food.type === FoodType.Purple) {
   536→        gained = 10;
   537→        // Shrink snake by 2 segments (min length 1)
   538→        const removeCount = Math.min(2, snake.length - 1);
   539→        for (let i = 0; i < removeCount; i++) snake.pop();
   540→      } else {
   541→        gained = SCORE_PER_FOOD;
   542→      }
   543→
   544→      if (isDoubleScoreRef.current) gained *= 2;
   545→      scoreRef.current += gained;
   546→      setScore(scoreRef.current);
   547→
   548→      // Difficulty scaling
   549→      if (food.type !== FoodType.Purple) {
   550→        foodsEatenRef.current += 1;
   551→        const newDiff = Math.min(
   552→          Math.floor(foodsEatenRef.current / FOODS_PER_DIFFICULTY),
   553→          MAX_DIFFICULTY
   554→        );
   555→        if (newDiff !== difficultyRef.current) {
   556→          difficultyRef.current = newDiff;
   557→          setDifficulty(newDiff);
   558→          const newSpeed = getSpeed();
   559→          currentSpeedRef.current = newSpeed;
   560→          if (gameLoopRef.current) clearInterval(gameLoopRef.current);
   561→          gameLoopRef.current = window.setInterval(gameStepRef.current, newSpeed);
   562→        }
   563→      }
   564→
   565→      spawnFood();
   566→    } else {
   567→      snake.pop();
   568→    }
   569→
   570→    draw();
   571→  }, [draw, spawnFood, endGame, applyPowerUpEffect, getSpeed, updateActiveEffectsUI]);
   572→
   573→  // Keep gameStepRef updated
   574→  useEffect(() => {
   575→    gameStepRef.current = gameStep;
   576→  }, [gameStep]);
   577→
   578→  const startGame = useCallback(() => {
   579→    clearAllTimers();
   580→    initSnake();
   581→    spawnFood();
   582→    directionRef.current = Direction.Right;
   583→    nextDirectionRef.current = Direction.Right;
   584→    scoreRef.current = 0;
   585→    setScore(0);
   586→    foodsEatenRef.current = 0;
   587→    difficultyRef.current = 0;
   588→    setDifficulty(0);
   589→    gameStatusRef.current = 'playing';
   590→    setGameStatus('playing');
   591→    setIsNewRecord(false);
   592→    pausedRef.current = false;
   593→    setPaused(false);
   594→    powerUpRef.current = null;
   595→    hasShieldRef.current = false;
   596→    isDoubleScoreRef.current = false;
   597→    isSpeedBoostedRef.current = false;
   598→    isSlowedRef.current = false;
   599→    currentSpeedRef.current = BASE_SPEED;
   600→    setActiveEffects([]);
   601→
   602→    draw();
   603→
   604→    const speed = getSpeed();
   605→    currentSpeedRef.current = speed;
   606→    gameLoopRef.current = window.setInterval(gameStep, speed);
   607→
   608→    // Spawn power-ups periodically (every 8-15 seconds)
   609→    const spawnPowerUpPeriodic = () => {
   610→      spawnPowerUp();
   611→      // Randomize next spawn: 8-15 seconds
   612→      const nextDelay = 8000 + Math.random() * 7000;
   613→      if (powerUpSpawnTimerRef.current) clearInterval(powerUpSpawnTimerRef.current);
   614→      powerUpSpawnTimerRef.current = setTimeout(() => {
   615→        spawnPowerUpPeriodic();
   616→      }, nextDelay);
   617→    };
   618→    // First power-up after 5-10 seconds
   619→    const firstDelay = 5000 + Math.random() * 5000;
   620→    powerUpSpawnTimerRef.current = setTimeout(spawnPowerUpPeriodic, firstDelay);
   621→  }, [initSnake, spawnFood, draw, gameStep, getSpeed, clearAllTimers, spawnPowerUp]);
   622→
   623→  const handleKeyDown = useCallback(
   624→    (e: KeyboardEvent) => {
   625→      if (gameStatusRef.current !== 'playing') return;
   626→      if (e.key === 'Escape') {
   627→        e.preventDefault();
   628→        pausedRef.current = !pausedRef.current;
   629→        setPaused(pausedRef.current);
   630→        if (pausedRef.current) {
   631→          if (gameLoopRef.current) {
   632→            clearInterval(gameLoopRef.current);
   633→            gameLoopRef.current = null;
   634→          }
   635→        } else {
   636→          const speed = currentSpeedRef.current;
   637→          gameLoopRef.current = window.setInterval(gameStepRef.current, speed);
   638→        }
   639→        draw();
   640→        return;
   641→      }
   642→      if (pausedRef.current) return;
   643→
   644→      const current = directionRef.current;
   645→      switch (e.key) {
   646→        case 'ArrowUp': case 'w': case 'W':
   647→          if (current !== Direction.Down) nextDirectionRef.current = Direction.Up;
   648→          e.preventDefault(); break;
   649→        case 'ArrowDown': case 's': case 'S':
   650→          if (current !== Direction.Up) nextDirectionRef.current = Direction.Down;
   651→          e.preventDefault(); break;
   652→        case 'ArrowLeft': case 'a': case 'A':
   653→          if (current !== Direction.Right) nextDirectionRef.current = Direction.Left;
   654→          e.preventDefault(); break;
   655→        case 'ArrowRight': case 'd': case 'D':
   656→          if (current !== Direction.Left) nextDirectionRef.current = Direction.Right;
   657→          e.preventDefault(); break;
   658→      }
   659→    },
   660→    [draw]
   661→  );
   662→
   663→  const handleTouchStart = useCallback((e: React.TouchEvent) => {
   664→    const touch = e.touches[0];
   665→    touchStartRef.current = { x: touch.clientX, y: touch.clientY };
   666→  }, []);
   667→
   668→  const handleTouchEnd = useCallback(
   669→    (e: React.TouchEvent) => {
   670→      if (gameStatusRef.current !== 'playing') return;
   671→      if (pausedRef.current) return;
   672→
   673→      const touchStart = touchStartRef.current;
   674→      if (!touchStart) return;
   675→
   676→      const touch = e.changedTouches[0];
   677→      const dx = touch.clientX - touchStart.x;
   678→      const dy = touch.clientY - touchStart.y;
   679→
   680→      const minSwipeDistance = 20;
   681→      if (Math.abs(dx) < minSwipeDistance && Math.abs(dy) < minSwipeDistance) return;
   682→
   683→      const current = directionRef.current;
   684→      if (Math.abs(dx) > Math.abs(dy)) {
   685→        if (dx > 0 && current !== Direction.Left) nextDirectionRef.current = Direction.Right;
   686→        else if (dx < 0 && current !== Direction.Right) nextDirectionRef.current = Direction.Left;
   687→      } else {
   688→        if (dy > 0 && current !== Direction.Up) nextDirectionRef.current = Direction.Down;
   689→        else if (dy < 0 && current !== Direction.Down) nextDirectionRef.current = Direction.Up;
   690→      }
   691→    },
   692→    []
   693→  );
   694→
   695→  useEffect(() => {
   696→    const fetchHighScore = async () => {
   697→      try {
   698→        const res = await fetch('/api/games/scores');
   699→        if (res.ok) {
   700→          const data = await res.json();
   701→          if (Array.isArray(data.data)) {
   702→            const snakeScore = data.data.find(
   703→              (item: { game: string; score: number }) => item.game === GAME_NAME
   704→            );
   705→            if (snakeScore) setHighScore(snakeScore.score);
   706→          }
   707→        }
   708→      } catch { /* silent */ }
   709→      finally { setLoadingHighScore(false); }
   710→    };
   711→    fetchHighScore();
   712→  }, []);
   713→
   714→  useEffect(() => {
   715→    window.addEventListener('keydown', handleKeyDown);
   716→    return () => window.removeEventListener('keydown', handleKeyDown);
   717→  }, [handleKeyDown]);
   718→
   719→  useEffect(() => {
   720→    return () => { clearAllTimers(); };
   721→  }, [clearAllTimers]);
   722→
   723→  useEffect(() => {
   724→    const container = containerRef.current;
   725→    const canvas = canvasRef.current;
   726→    if (!container || !canvas) return;
   727→
   728→    const update = () => {
   729→      const dpr = Math.min(window.devicePixelRatio || 1, 2);
   730→      const displaySize = Math.min(container.clientWidth, LOGIC_SIZE);
   731→      const scale = displaySize / LOGIC_SIZE;
   732→
   733→      canvas.width = displaySize * dpr;
   734→      canvas.height = displaySize * dpr;
   735→      canvas.style.width = `${displaySize}px`;
   736→      canvas.style.height = `${displaySize}px`;
   737→
   738→      dprRef.current = dpr;
   739→      scaleRef.current = scale;
   740→      draw();
   741→    };
   742→
   743→    const ro = new ResizeObserver(update);
   744→    ro.observe(container);
   745→    const onVis = () => { if (document.visibilityState === 'visible') update(); };
   746→    document.addEventListener('visibilitychange', onVis);
   747→
   748→    update();
   749→    return () => { ro.disconnect(); document.removeEventListener('visibilitychange', onVis); };
   750→  }, [draw]);
   751→
   752→  return (
   753→    <div className="flex flex-col items-center w-full max-w-md mx-auto px-2 py-2 gap-2">
   754→      <div className="flex items-center justify-between w-full">
   755→        <Button
   756→          variant="ghost"
   757→          size="sm"
   758→          onClick={onBack}
   759→          className="text-muted-foreground hover:text-foreground"
   760→        >
   761→          <ArrowLeft className="h-4 w-4 mr-1" />
   762→          返回大厅
   763→        </Button>
   764→        <h2 className="text-lg font-bold tracking-wide">🐍 贪吃蛇</h2>
   765→        {gameStatus === 'playing' && <GameControlsHelp info={snakeControlsInfo} variant="button" />}
   766→        {gameStatus !== 'playing' && <div className="w-8" />}
   767→      </div>
   768→
   769→      <Card className="w-full border-border/50 bg-card/80">
   770→        <CardContent className="flex items-center justify-between p-3">
   771→          <div className="flex flex-col items-center gap-0.5">
   772→            <span className="text-xs text-muted-foreground">分数</span>
   773→            <span className="text-xl font-bold text-primary tabular-nums">{score}</span>
   774→          </div>
   775→          <div className="h-10 w-px bg-border" />
   776→          <div className="flex flex-col items-center gap-0.5">
   777→            <span className="text-xs text-muted-foreground flex items-center gap-1">
   778→              <Trophy className="h-3 w-3 text-yellow-500" />
   779→              最高分
   780→            </span>
   781→            <span className="text-xl font-bold text-yellow-500 tabular-nums">
   782→              {loadingHighScore ? '...' : highScore}
   783→            </span>
   784→          </div>
   785→          <div className="h-10 w-px bg-border" />
   786→          <div className="flex flex-col items-center gap-0.5">
   787→            <span className="text-xs text-muted-foreground">难度</span>
   788→            <span className="text-xl font-bold text-orange-400 tabular-nums">
   789→              {difficulty + 1}/{MAX_DIFFICULTY + 1}
   790→            </span>
   791→          </div>
   792→        </CardContent>
   793→      </Card>
   794→
   795→      {activeEffects.length > 0 && (
   796→        <div className="flex flex-wrap gap-1.5 justify-center">
   797→          {activeEffects.map((effect, i) => (
   798→            <span
   799→              key={i}
   800→              className="text-xs px-2 py-1 rounded-full bg-secondary text-secondary-foreground"
   801→            >
   802→              {effect}
   803→            </span>
   804→          ))}
   805→        </div>
   806→      )}
   807→
   808→      <div
   809→        ref={containerRef}
   810→        className="w-full flex justify-center"
   811→        onTouchStart={handleTouchStart}
   812→        onTouchEnd={handleTouchEnd}
   813→      >
   814→        <canvas
   815→          ref={canvasRef}
   816→          className="rounded-lg border border-border/30 shadow-lg"
   817→          style={{ touchAction: 'none' }}
   818→        />
   819→      </div>
   820→
   821→      <div className="flex flex-col items-center gap-3 w-full">
   822→        {gameStatus === 'over' && isNewRecord && (
   823→          <p className="text-sm font-semibold text-yellow-500 animate-pulse">
   824→            🎉 新纪录！
   825→          </p>
   826→        )}
   827→
   828→        <Button
   829→          onClick={startGame}
   830→          size="lg"
   831→          className="w-full max-w-xs text-base font-semibold"
   832→          disabled={gameStatus === 'playing'}
   833→        >
   834→          <RotateCcw className="h-4 w-4 mr-2" />
   835→          {gameStatus === 'playing' ? '游戏中...' : '开始游戏'}
   836→        </Button>
   837→      </div>
   838→
   839→      <div className="grid grid-cols-3 gap-2 w-40 mt-2 md:hidden">
   840→        <div />
   841→        <Button
   842→          variant="outline"
   843→          size="icon"
   844→          className="h-12 w-12 rounded-lg active:bg-primary/20"
   845→          onTouchStart={(e) => {
   846→            e.stopPropagation();
   847→            if (directionRef.current !== Direction.Down) nextDirectionRef.current = Direction.Up;
   848→          }}
   849→          disabled={gameStatus !== 'playing' || paused}
   850→        >
   851→          ↑
   852→        </Button>
   853→        <div />
   854→        <Button
   855→          variant="outline"
   856→          size="icon"
   857→          className="h-12 w-12 rounded-lg active:bg-primary/20"
   858→          onTouchStart={(e) => {
   859→            e.stopPropagation();
   860→            if (directionRef.current !== Direction.Right) nextDirectionRef.current = Direction.Left;
   861→          }}
   862→          disabled={gameStatus !== 'playing' || paused}
   863→        >
   864→          ←
   865→        </Button>
   866→        <Button
   867→          variant="outline"
   868→          size="icon"
   869→          className="h-12 w-12 rounded-lg active:bg-primary/20"
