   154→function createStars(count: number): Star[] {
   155→  const stars: Star[] = [];
   156→  for (let i = 0; i < count; i++) {
   157→    stars.push({
   158→      x: Math.random() * LOGIC_W,
   159→      y: Math.random() * LOGIC_H,
   160→      r: Math.random() * 1.5 + 0.5,
   161→      alpha: Math.random() * 0.6 + 0.4,
   162→      twinkleSpeed: Math.random() * 0.03 + 0.01,
   163→      phase: Math.random() * Math.PI * 2,
   164→    });
   165→  }
   166→  return stars;
   167→}
   168→
   169→function createExplosion(x: number, y: number, color: string, count: number): Particle[] {
   170→  const particles: Particle[] = [];
   171→  for (let i = 0; i < count; i++) {
   172→    const angle = Math.random() * Math.PI * 2;
   173→    const speed = Math.random() * 3 + 1;
   174→    particles.push({
   175→      x, y,
   176→      vx: Math.cos(angle) * speed,
   177→      vy: Math.sin(angle) * speed,
   178→      life: Math.random() * 20 + 15,
   179→      maxLife: 35,
   180→      color,
   181→      r: Math.random() * 3 + 1,
   182→    });
   183→  }
   184→  return particles;
   185→}

   187→function rectCollide(
   188→  ax: number, ay: number, aw: number, ah: number,
   189→  bx: number, by: number, bw: number, bh: number,
   190→): boolean {
   191→  return Math.abs(ax - bx) < (aw + bw) / 2 && Math.abs(ay - by) < (ah + bh) / 2;
   192→}

   196→function drawPlayer(ctx: CanvasRenderingContext2D, p: Player, frame: number) {
   197→  ctx.save();
   198→  ctx.translate(p.x, p.y);
   199→
   200→  // Invincibility blink
   201→  if (p.invincibleTimer > 0 && Math.floor(frame / 3) % 2 === 0) {
   202→    ctx.globalAlpha = 0.35;
   203→  }
   204→
   205→  // Tail flame animation
   206→  const flameLen = 12 + Math.sin(frame * 0.5) * 4;
   207→  const flameW = 8 + Math.sin(frame * 0.7) * 2;
   208→  const flameGrad = ctx.createLinearGradient(0, PLAYER_H / 2, 0, PLAYER_H / 2 + flameLen);
   209→  flameGrad.addColorStop(0, '#38bdf8');
   210→  flameGrad.addColorStop(0.4, '#f97316');
   211→  flameGrad.addColorStop(1, 'rgba(239,68,68,0)');
   212→  ctx.fillStyle = flameGrad;
   213→  ctx.beginPath();
   214→  ctx.moveTo(-flameW, PLAYER_H / 2 - 2);
   215→  ctx.lineTo(flameW, PLAYER_H / 2 - 2);
   216→  ctx.lineTo(0, PLAYER_H / 2 + flameLen);
   217→  ctx.closePath();
   218→  ctx.fill();
   219→
   220→  // Aircraft body
   221→  ctx.fillStyle = '#e0f2fe';
   222→  ctx.beginPath();
   223→  ctx.moveTo(0, -PLAYER_H / 2);
   224→  ctx.lineTo(-PLAYER_W / 2, PLAYER_H / 2 - 6);
   225→  ctx.lineTo(-PLAYER_W / 4, PLAYER_H / 2);
   226→  ctx.lineTo(PLAYER_W / 4, PLAYER_H / 2);
   227→  ctx.lineTo(PLAYER_W / 2, PLAYER_H / 2 - 6);
   228→  ctx.closePath();
   229→  ctx.fill();
   230→
   231→  // Wing decoration lines
   232→  ctx.strokeStyle = '#38bdf8';
   233→  ctx.lineWidth = 1.5;
   234→  ctx.beginPath();
   235→  ctx.moveTo(-PLAYER_W / 2 + 4, PLAYER_H / 2 - 10);
   236→  ctx.lineTo(0, -PLAYER_H / 2 + 10);
   237→  ctx.lineTo(PLAYER_W / 2 - 4, PLAYER_H / 2 - 10);
   238→  ctx.stroke();
   239→
   240→  // Cockpit
   241→  ctx.fillStyle = '#7dd3fc';
   242→  ctx.beginPath();
   243→  ctx.ellipse(0, -PLAYER_H / 4, 4, 7, 0, 0, Math.PI * 2);
   244→  ctx.fill();
   245→
   246→  // Shield bubble
   247→  if (p.shieldTimer > 0) {
   248→    const shieldAlpha = p.shieldTimer < 60 ? (Math.sin(frame * 0.3) * 0.3 + 0.4) : 0.5;
   249→    ctx.strokeStyle = `rgba(6, 182, 212, ${shieldAlpha})`;
   250→    ctx.lineWidth = 2.5;
   251→    ctx.beginPath();
   252→  ctx.ellipse(0, 0, PLAYER_W / 2 + 8, PLAYER_H / 2 + 8, 0, 0, Math.PI * 2);
   253→    ctx.stroke();
   254→  }
   255→
   256→  ctx.restore();
   257→}

   259→function drawEnemy(ctx: CanvasRenderingContext2D, e: Enemy) {
   260→  ctx.save();
   261→  ctx.translate(e.x, e.y);
   262→
   263→  if (e.type === EnemyType.Normal) {
   264→    // Red inverted triangle
   265→    ctx.fillStyle = '#ef4444';
   266→    ctx.beginPath();
   267→    ctx.moveTo(0, e.h / 2);
   268→    ctx.lineTo(-e.w / 2, -e.h / 2);
   269→    ctx.lineTo(e.w / 2, -e.h / 2);
   270→    ctx.closePath();
   271→    ctx.fill();
   272→    ctx.strokeStyle = '#fca5a5';
   273→    ctx.lineWidth = 1;
   274→    ctx.beginPath();
   275→    ctx.moveTo(0, e.h / 2 - 6);
   276→    ctx.lineTo(-e.w / 4, -e.h / 4);
   277→    ctx.moveTo(0, e.h / 2 - 6);
   278→    ctx.lineTo(e.w / 4, -e.h / 4);
   279→    ctx.stroke();
   280→  } else if (e.type === EnemyType.Fast) {
   281→    // Yellow arrow shape
   282→    ctx.fillStyle = '#f59e0b';
   283→    ctx.beginPath();
   284→    ctx.moveTo(0, e.h / 2);
   285→    ctx.lineTo(-e.w / 2, -e.h / 3);
   286→    ctx.lineTo(-e.w / 4, -e.h / 2);
   287→    ctx.lineTo(e.w / 4, -e.h / 2);
   288→    ctx.lineTo(e.w / 2, -e.h / 3);
   289→    ctx.closePath();
   290→    ctx.fill();
   291→    ctx.fillStyle = '#fcd34d';
   292→    ctx.beginPath();
   293→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   294→    ctx.fill();
   295→  } else if (e.type === EnemyType.Tank) {
   296→    // Purple hexagon
   297→    const r = e.w / 2;
   298→    ctx.fillStyle = '#a855f7';
   299→    ctx.beginPath();
   300→    for (let i = 0; i < 6; i++) {
   301→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   302→      const px = Math.cos(angle) * r;
   303→      const py = Math.sin(angle) * r;
   304→      if (i === 0) ctx.moveTo(px, py);
   305→      else ctx.lineTo(px, py);
   306→    }
   307→    ctx.closePath();
   308→    ctx.fill();
   309→    ctx.strokeStyle = '#d8b4fe';
   310→    ctx.lineWidth = 1.5;
   311→    ctx.beginPath();
   312→    for (let i = 0; i < 6; i++) {
   313→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   314→      const px = Math.cos(angle) * (r * 0.55);
   315→      const py = Math.sin(angle) * (r * 0.55);
   316→      if (i === 0) ctx.moveTo(px, py);
   317→      else ctx.lineTo(px, py);
   318→    }
   319→    ctx.closePath();
   320→    ctx.stroke();
   321→    ctx.fillStyle = '#e9d5ff';
   322→    ctx.beginPath();
   323→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   324→    ctx.fill();
   325→  }
   326→
   327→  ctx.restore();
   328→
   329→  // HP bar (only when damaged)
   330→  if (e.hp < e.maxHp) {
   331→    const barW = e.w * 0.8;
   332→    const barH = 3;
   333→    const barX = e.x - barW / 2;
   334→    const barY = e.y - e.h / 2 - 6;
   335→    ctx.fillStyle = 'rgba(255,255,255,0.2)';
   336→    ctx.fillRect(barX, barY, barW, barH);
   337→    ctx.fillStyle = '#22c55e';
   338→    ctx.fillRect(barX, barY, barW * (e.hp / e.maxHp), barH);
   339→  }
   340→}

   342→function drawBoss(ctx: CanvasRenderingContext2D, boss: Boss, frame: number) {
   343→  ctx.save();
   344→  ctx.translate(boss.x, boss.y);
   345→
   346→  // Flash white on hit
   347→  if (boss.flashTimer > 0) {
   348→    ctx.globalAlpha = 0.5 + Math.sin(frame) * 0.5;
   349→    // White overlay circle
   350→    ctx.fillStyle = 'rgba(255,255,255,0.6)';
   351→    ctx.beginPath();
   352→    ctx.ellipse(0, 0, BOSS_W / 2 + 4, BOSS_H / 2 + 4, 0, 0, Math.PI * 2);
   353→    ctx.fill();
   354→    ctx.globalAlpha = 1;
   355→  }
   356→
   357→  // Boss emoji
   358→  ctx.font = `${BOSS_H}px sans-serif`;
   359→  ctx.textAlign = 'center';
   360→  ctx.textBaseline = 'middle';
   361→  ctx.fillText(boss.emoji, 0, 2);
   362→
   363→  ctx.restore();
   364→
   365→  // Boss HP bar at top of canvas
   366→  const barW = LOGIC_W * 0.6;
   367→  const barH = 8;
   368→  const barX = (LOGIC_W - barW) / 2;
   369→  const barY = 6;
   370→
   371→  ctx.fillStyle = 'rgba(255,255,255,0.15)';
   372→  ctx.fillRect(barX - 1, barY - 1, barW + 2, barH + 2);
   373→
   374→  const hpRatio = boss.hp / boss.maxHp;
   375→  const hpColor = hpRatio > 0.5 ? '#ef4444' : hpRatio > 0.25 ? '#f97316' : '#fbbf24';
   376→  ctx.fillStyle = hpColor;
   377→  ctx.fillRect(barX, barY, barW * hpRatio, barH);
   378→
   379→  ctx.strokeStyle = 'rgba(255,255,255,0.4)';
   380→  ctx.lineWidth = 1;
   381→  ctx.strokeRect(barX - 1, barY - 1, barW + 2, barH + 2);
   382→
   383→  ctx.fillStyle = '#ffffff';
   384→  ctx.font = 'bold 11px sans-serif';
   385→  ctx.textAlign = 'center';
   386→  ctx.textBaseline = 'top';
   387→  ctx.fillText('BOSS', LOGIC_W / 2, barY + barH + 2);
   388→}

   390→function drawPowerUp(ctx: CanvasRenderingContext2D, pw: PowerUp) {
   391→  const glow = 0.3 + Math.sin(pw.phase) * 0.2;
   392→  const color = POWERUP_COLORS[pw.type];
   393→
   394→  // Outer glow
   395→  ctx.save();
   396→  ctx.globalAlpha = glow;
   397→  ctx.fillStyle = color;
   398→  ctx.beginPath();
   399→  ctx.arc(pw.x, pw.y, 16, 0, Math.PI * 2);
   400→  ctx.fill();
   401→  ctx.restore();
   402→
   403→  // Inner circle
   404→  ctx.fillStyle = color;
   405→  ctx.beginPath();
   406→  ctx.arc(pw.x, pw.y, 12, 0, Math.PI * 2);
   407→  ctx.fill();
   408→
   409→  // Emoji icon
   410→  ctx.font = '14px sans-serif';
   411→  ctx.textAlign = 'center';
   412→  ctx.textBaseline = 'middle';
   413→  ctx.fillText(POWERUP_EMOJIS[pw.type], pw.x, pw.y);
   414→}

   416→function drawEnemyBullet(ctx: CanvasRenderingContext2D, b: EnemyBullet) {
   417→  ctx.save();
   418→  ctx.shadowColor = '#ef4444';
   419→  ctx.shadowBlur = 5;

   422→  ctx.arc(b.x, b.y, 4, 0, Math.PI * 2);
   423→  ctx.fill();
   424→  ctx.shadowBlur = 0;
   425→  ctx.restore();
   426→}

   489→  /** Damage the player - lose life, downgrade weapon */
   490→  const damagePlayer = useCallback(() => {
   491→    const p = playerRef.current;
   492→    if (p.shieldTimer > 0 || p.invincibleTimer > 0) return;
   493→

   512→    const lv = levelRef.current;
   513→    const rand = Math.random();
   514→    let type: EnemyType;
   515→
   516→    const tankChance = Math.min(0.1 + lv * 0.04, 0.35);

   518→
   519→    if (rand < tankChance) {
   520→      type = EnemyType.Tank;
   521→    } else if (rand < tankChance + fastChance) {
   522→      type = EnemyType.Fast;
   523→    } else {
   524→      type = EnemyType.Normal;
   525→    }
   526→
   527→    const cfg = ENEMY_CONFIG[type];
   528→    const speedMul = 1 + (lv - 1) * 0.12;
   529→    const x = Math.random() * (LOGIC_W - cfg.w - 20) + cfg.w / 2 + 10;
   530→
   531→    enemiesRef.current.push({
   532→      x, y: -cfg.h,
   533→      type,
   534→      hp: cfg.hp,
   535→      maxHp: cfg.hp,
   536→      speed: cfg.speed * speedMul,
   537→      w: cfg.w,
   538→      h: cfg.h,
   539→      score: cfg.score,
   540→    });
   541→  }, []);
   542→
   543→  /** Spawn a boss */
   544→  const spawnBoss = useCallback(() => {
   545→    const idx = bossSpawnedCountRef.current;
   546→    const emoji = BOSS_EMOJIS[idx % BOSS_EMOJIS.length];
   547→    const hp = 30 + idx * 15;
   548→    const score = 200 + idx * 100;
   549→
   550→    bossRef.current = {
   551→      x: LOGIC_W / 2,
   552→      y: -BOSS_H,
   553→      hp,
   554→      maxHp: hp,
   555→      bossIndex: idx,
   556→      emoji,
   557→      flashTimer: 0,
   558→      shootTimer: 0,
   559→      dirX: 1.5,
   560→      score,
   561→      entered: false,
   562→    };
   563→    bossSpawnedCountRef.current++;
   564→    bossKillCounterRef.current = 0;
   565→  }, []);
   566→
   567→  /** Fire player bullets based on weapon level */
   568→  const shoot = useCallback(() => {
   569→    const p = playerRef.current;
   570→    const speed = 8;
   571→    const bx = p.x;
   572→    const by = p.y - PLAYER_H / 2;
   573→    const bullets = bulletsRef.current;
   574→
   575→    switch (p.weaponLevel) {
   576→      case 1:
   577→        bullets.push({ x: bx, y: by, vx: 0, vy: -speed });
   578→        break;
   579→      case 2:
   580→        bullets.push(
   581→          { x: bx - 8, y: by, vx: 0, vy: -speed },
   582→          { x: bx + 8, y: by, vx: 0, vy: -speed },
   583→        );
   584→        break;
   585→      case 3: {
   586→        const a = Math.PI / 12; // 15 degrees
   587→        const svx = Math.sin(a) * speed;
   588→        const svy = -Math.cos(a) * speed;
   589→        bullets.push(
   590→          { x: bx, y: by, vx: 0, vy: -speed },
   591→          { x: bx, y: by, vx: svx, vy: svy },
   592→          { x: bx, y: by, vx: -svx, vy: svy },
   593→        );
   594→        break;
   595→      }
   596→      case 4: {
   597→        const a1 = Math.PI / 18; // 10 degrees
   598→        const a2 = Math.PI / 9;  // 20 degrees
   599→        bullets.push(
   600→          { x: bx, y: by, vx: 0, vy: -speed },
   601→          { x: bx, y: by, vx: Math.sin(a1) * speed, vy: -Math.cos(a1) * speed },
   602→          { x: bx, y: by, vx: -Math.sin(a1) * speed, vy: -Math.cos(a1) * speed },
   603→          { x: bx, y: by, vx: Math.sin(a2) * speed, vy: -Math.cos(a2) * speed },
   604→          { x: bx, y: by, vx: -Math.sin(a2) * speed, vy: -Math.cos(a2) * speed },
   605→        );
   606→        break;
   607→      }
   608→      default:
   609→        break;
   610→    }
   611→  }, []);
   612→
   613→  /** Get shoot interval based on weapon level */
   614→  const getShootInterval = useCallback(() => {
   615→    return Math.max(BASE_SHOOT_INTERVAL - playerRef.current.weaponLevel + 1, 6);
   616→  }, []);
   617→
   618→  /** Get enemy spawn interval based on difficulty */
   619→  const getSpawnInterval = useCallback(() => {
   620→    const lv = levelRef.current;
   621→    return Math.max(45 - (lv - 1) * 4, 15);
   622→  }, []);
   623→
   624→  /** Activate bomb powerup effect */
   625→  const activateBomb = useCallback(() => {
   626→    screenFlashRef.current = SCREEN_FLASH_FRAMES;
   627→
   628→    // Destroy all regular enemies
   629→    for (const e of enemiesRef.current) {
   630→      scoreRef.current += e.score;
   631→      const expColor = e.type === EnemyType.Fast ? '#f59e0b'
   632→        : e.type === EnemyType.Tank ? '#a855f7' : '#ef4444';
   633→      particlesRef.current.push(...createExplosion(e.x, e.y, expColor, 10));
   634→    }
   635→    setScore(scoreRef.current);
   636→    enemiesRef.current = [];
   637→
   638→    // Clear enemy bullets
   639→    enemyBulletsRef.current = [];
   640→
   641→    // Damage boss by 30% if present
   642→    const boss = bossRef.current;
   643→    if (boss) {
   644→      const dmg = Math.ceil(boss.maxHp * 0.3);
   645→      boss.hp = Math.max(1, boss.hp - dmg);
   646→      boss.flashTimer = 10;
   647→      particlesRef.current.push(...createExplosion(boss.x, boss.y, '#fbbf24', 20));
   648→
   649→      if (boss.hp <= 1) {
   650→        // Boss killed by bomb
   651→        scoreRef.current += boss.score;
   652→        setScore(scoreRef.current);
   653→        particlesRef.current.push(...createExplosion(boss.x, boss.y, '#f97316', 30));
   654→        // Drop powerups
   655→        const dropTypes = [PowerUpType.Weapon, PowerUpType.Heal, PowerUpType.Shield];
   656→        for (const dt of dropTypes) {
   657→          powerupsRef.current.push({
   658→            x: boss.x + (Math.random() - 0.5) * 60,
   659→            y: boss.y + (Math.random() - 0.5) * 30,
   660→            type: dt,
   661→            phase: Math.random() * Math.PI * 2,
   662→          });
   663→        }
   664→        bossRef.current = null;
   665→      }
   666→    }
   667→  }, []);
   668→
   669→  /** Main game loop */
   670→  const gameLoop = useCallback(() => {
   671→    const canvas = canvasRef.current;
   672→    if (!canvas) return;
   673→    const ctx = canvas.getContext('2d');
   674→    if (!ctx) return;
   675→
   676→    const dpr = window.devicePixelRatio || 1;
   677→    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   678→
   679→    const frame = frameRef.current;
   680→    frameRef.current++;
   681→    const p = playerRef.current;
   682→    const boss = bossRef.current;
   683→
   684→    // ==== UPDATE LOGIC ====
   685→
   686→    // Twinkling stars
   687→    starsRef.current.forEach((s) => { s.phase += s.twinkleSpeed; });
   688→
   689→    if (statusRef.current === 'playing' && !pausedRef.current) {
   690→      // Move player toward input
   691→      if (inputXRef.current !== null) {
   692→        const target = inputXRef.current;
   693→        const diff = target - p.x;
   694→        if (Math.abs(diff) > 2) {
   695→          p.x += diff * 0.15;
   696→        } else {
   697→          p.x = target;
   698→        }
   699→      }
   700→      p.x = Math.max(PLAYER_W / 2, Math.min(LOGIC_W - PLAYER_W / 2, p.x));
   701→
   702→      // Update timers
   703→      if (p.shieldTimer > 0) p.shieldTimer--;
   704→      if (p.invincibleTimer > 0) p.invincibleTimer--;
   705→      if (screenFlashRef.current > 0) screenFlashRef.current--;
   706→
   707→      // Shooting
   708→      const isMobile = isMobileRef.current;
   709→      const interval = getShootInterval();
   710→      if (isMobile) {
   711→        if (frame - lastShootRef.current >= interval) {
   712→          shoot();
   713→          lastShootRef.current = frame;
   714→        }
   715→      } else {
   716→        if (shootRequestRef.current) {
   717→          if (frame - lastShootRef.current >= interval) {
   718→            shoot();
   719→            lastShootRef.current = frame;
   720→          }
   721→        }
   722→      }
   723→      shootRequestRef.current = false;
   724→
   725→      // Spawn regular enemies (reduced rate during boss)
   726→      const spawnInterval = boss ? getSpawnInterval() * 1.5 : getSpawnInterval();
   727→      if (frame % Math.round(spawnInterval) === 0) {
   728→        spawnEnemy();
   729→      }
   730→
   731→      // Check boss spawn (every 30 kills, no boss currently active)
   732→      if (bossKillCounterRef.current >= BOSS_SPAWN_KILLS && !boss) {
   733→        spawnBoss();
   734→      }
   735→
   736→      // Update player bullets
   737→      bulletsRef.current = bulletsRef.current.filter((b) => {
   738→        b.x += b.vx;
   739→        b.y += b.vy;
   740→        return b.y > -BULLET_H && b.x > -BULLET_W && b.x < LOGIC_W + BULLET_W;
   741→      });
   742→
   743→      // Update enemy bullets
   744→      enemyBulletsRef.current = enemyBulletsRef.current.filter((b) => {
   745→        b.x += b.vx;
   746→        b.y += b.vy;
   747→        return b.y < LOGIC_H + 10 && b.x > -10 && b.x < LOGIC_W + 10;
   748→      });
   749→
   750→      // Update regular enemies
   751→      enemiesRef.current = enemiesRef.current.filter((e) => {
   752→        e.y += e.speed;
   753→        return e.y < LOGIC_H + e.h;
   754→      });
   755→
   756→      // Update boss
   757→      if (boss) {
   758→        if (!boss.entered) {
   759→          boss.y += 1.5;
   760→          if (boss.y >= 80) {
   761→            boss.y = 80;
   762→            boss.entered = true;
   763→          }
   764→        } else {
   765→          // Side-to-side movement
   766→          boss.x += boss.dirX;
   767→          if (boss.x > LOGIC_W - BOSS_W / 2 - 10) boss.dirX = -Math.abs(boss.dirX);
   768→          if (boss.x < BOSS_W / 2 + 10) boss.dirX = Math.abs(boss.dirX);
   769→
   770→          // Boss shooting
   771→          boss.shootTimer++;
   772→          if (boss.shootTimer >= BOSS_SHOOT_INTERVAL) {
   773→            boss.shootTimer = 0;
   774→            for (let i = -1; i <= 1; i++) {
   775→              const angle = Math.PI / 2 + i * BOSS_SPREAD_ANGLE;
   776→              enemyBulletsRef.current.push({
   777→                x: boss.x,
   778→                y: boss.y + BOSS_H / 2,
   779→                vx: Math.cos(angle) * BOSS_BULLET_SPEED,
   780→                vy: Math.sin(angle) * BOSS_BULLET_SPEED,
   781→              });
   782→            }
   783→          }
   784→        }
   785→        if (boss.flashTimer > 0) boss.flashTimer--;
   786→      }
   787→
   788→      // Update powerups
   789→      powerupsRef.current = powerupsRef.current.filter((pw) => {
   790→        pw.y += 2;
   791→        pw.phase += 0.1;
   792→        return pw.y < LOGIC_H + 20;
   793→      });
   794→
   795→      // === COLLISION: player bullets vs enemies ===
   796→      const bullets = bulletsRef.current;
   797→      const enemies = enemiesRef.current;
   798→      const hitBullets = new Set<number>();
   799→      const hitEnemies = new Set<number>();
   800→
   801→      for (let bi = 0; bi < bullets.length; bi++) {
   802→        const b = bullets[bi];
   803→        for (let ei = 0; ei < enemies.length; ei++) {
   804→          if (hitEnemies.has(ei)) continue;
   805→          const e = enemies[ei];
   806→          if (rectCollide(b.x, b.y, BULLET_W, BULLET_H, e.x, e.y, e.w, e.h)) {
   807→            hitBullets.add(bi);
   808→            e.hp--;
   809→            if (e.hp <= 0) {
   810→              hitEnemies.add(ei);
   811→              scoreRef.current += e.score;
   812→              killCountRef.current++;
   813→              bossKillCounterRef.current++;
   814→              setScore(scoreRef.current);
   815→
   816→              const expColor = e.type === EnemyType.Fast ? '#f59e0b'
   817→                : e.type === EnemyType.Tank ? '#a855f7' : '#ef4444';
   818→              const expCount = e.type === EnemyType.Tank ? 18 : e.type === EnemyType.Normal ? 8 : 6;
   819→              particlesRef.current.push(...createExplosion(e.x, e.y, expColor, expCount));
   820→
   821→              // Powerup drop (20%)
   822→              if (Math.random() < POWERUP_CHANCE) {
   823→                const types = [PowerUpType.Weapon, PowerUpType.Heal, PowerUpType.Bomb, PowerUpType.Shield];
   824→                const weights = [0.35, 0.25, 0.15, 0.25];
   825→                let r = Math.random();
   826→                let chosen = PowerUpType.Shield;
   827→                for (let ti = 0; ti < types.length; ti++) {
   828→                  r -= weights[ti];
   829→                  if (r <= 0) { chosen = types[ti]; break; }
   830→                }
   831→                powerupsRef.current.push({ x: e.x, y: e.y, type: chosen, phase: 0 });
   832→              }
   833→
   834→              // Difficulty level up
   835→              if (killCountRef.current % LEVEL_UP_KILLS === 0) {
   836→                levelRef.current++;
   837→                setLevel(levelRef.current);
   838→              }
   839→            } else {
   840→              particlesRef.current.push(...createExplosion(b.x, b.y, '#fbbf24', 3));
   841→            }
   842→            break;
   843→          }
   844→        }
   845→      }
   846→
   847→      bulletsRef.current = bullets.filter((_, i) => !hitBullets.has(i));
   848→      enemiesRef.current = enemies.filter((_, i) => !hitEnemies.has(i));
   849→
   850→      // === COLLISION: player bullets vs boss ===
   851→      if (boss && statusRef.current === 'playing') {
   852→        const bossBullets = bulletsRef.current;
   853→        const bossHitBullets = new Set<number>();
   854→        for (let bi = 0; bi < bossBullets.length; bi++) {
   855→          const b = bossBullets[bi];
   856→          if (rectCollide(b.x, b.y, BULLET_W, BULLET_H, boss.x, boss.y, BOSS_W, BOSS_H)) {
   857→            bossHitBullets.add(bi);
   858→            boss.hp--;
   859→            boss.flashTimer = 4;
   860→            particlesRef.current.push(...createExplosion(b.x, b.y, '#fbbf24', 2));
   861→
   862→            if (boss.hp <= 0) {
   863→              // Boss defeated!
   864→              scoreRef.current += boss.score;
   865→              setScore(scoreRef.current);
   866→              particlesRef.current.push(...createExplosion(boss.x, boss.y, '#f97316', 35));
   867→
   868→              // Drop 3 powerups
   869→              const dropTypes = [PowerUpType.Weapon, PowerUpType.Heal, PowerUpType.Shield];
   870→              for (const dt of dropTypes) {
   871→                powerupsRef.current.push({
   872→                  x: boss.x + (Math.random() - 0.5) * 80,
   873→                  y: boss.y + (Math.random() - 0.5) * 30,
   874→                  type: dt,
   875→                  phase: Math.random() * Math.PI * 2,
   876→                });
   877→              }
   878→              bossRef.current = null;
   879→              break;
   880→            }
   881→          }
   882→        }
   883→        if (bossRef.current !== null) {
   884→          bulletsRef.current = bossBullets.filter((_, i) => !bossHitBullets.has(i));
   885→        } else {
   886→          // Boss was just killed, remove all hit bullets
   887→          bulletsRef.current = bossBullets.filter((_, i) => !bossHitBullets.has(i));
   888→        }
   889→      }
   890→
   891→      // === COLLISION: enemy bullets vs player ===
   892→      if (statusRef.current === 'playing') {
   893→        for (const eb of enemyBulletsRef.current) {
   894→          if (rectCollide(p.x, p.y, PLAYER_W * 0.5, PLAYER_H * 0.5, eb.x, eb.y, 8, 8)) {
   895→            damagePlayer();
   896→            break;
   897→          }
   898→        }
   899→      }
   900→
   901→      // === COLLISION: enemies vs player ===
   902→      if (statusRef.current === 'playing') {
   903→        for (const e of enemiesRef.current) {
   904→          if (rectCollide(p.x, p.y, PLAYER_W * 0.6, PLAYER_H * 0.6, e.x, e.y, e.w, e.h)) {
   905→            damagePlayer();
   906→            // Destroy the enemy on contact
   907→            e.hp = 0;
   908→            break;
   909→          }
   910→        }
   911→        enemiesRef.current = enemiesRef.current.filter((e) => e.hp > 0);
   912→      }
   913→
   914→      // === COLLISION: boss vs player ===
   915→      if (boss && statusRef.current === 'playing') {
   916→        if (rectCollide(p.x, p.y, PLAYER_W * 0.6, PLAYER_H * 0.6, boss.x, boss.y, BOSS_W, BOSS_H)) {
   917→          damagePlayer();
   918→        }
   919→      }
   920→
   921→      // === COLLISION: powerups vs player ===
   922→      if (statusRef.current === 'playing') {
   923→        powerupsRef.current = powerupsRef.current.filter((pw) => {
   924→          if (rectCollide(p.x, p.y, PLAYER_W, PLAYER_H, pw.x, pw.y, 24, 24)) {
   925→            switch (pw.type) {
   926→              case PowerUpType.Weapon:
   927→                if (p.weaponLevel < MAX_WEAPON_LEVEL) {
   928→                  p.weaponLevel++;
   929→                  setWeaponLevel(p.weaponLevel);
   930→                }
   931→                break;
   932→              case PowerUpType.Heal:
   933→                if (p.lives < MAX_LIVES) {
   934→                  p.lives++;
   935→                  setLives(p.lives);
   936→                }
   937→                break;
   938→              case PowerUpType.Bomb:
   939→                activateBomb();
   940→                break;
   941→              case PowerUpType.Shield:
   942→                p.shieldTimer = SHIELD_DURATION;
   943→                break;
   944→            }
   945→            const pColor = POWERUP_COLORS[pw.type];
   946→            particlesRef.current.push(...createExplosion(pw.x, pw.y, pColor, 10));
   947→            return false;
   948→          }
   949→          return true;
   950→        });
   951→      }
   952→    }
   953→
   954→    // Submit score on game over (outside the playing block to avoid TS narrowing)
   955→    if (statusRef.current === 'over' && !scoreSubmittedRef.current) {
   956→      scoreSubmittedRef.current = true;
   957→      if (scoreRef.current > highScoreRef.current) {
   958→        setIsNewRecord(true);
   959→        fetch('/api/games/scores', {
   960→          method: 'POST',
   961→          headers: { 'Content-Type': 'application/json' },
   962→          body: JSON.stringify({ game: GAME_NAME, score: scoreRef.current }),
   963→        })
   964→          .then((r) => r.json())
   965→          .then((data) => {
   966→            if (data.data?.score !== undefined) {
   967→              setHighScore(data.data.score);
   968→              highScoreRef.current = data.data.score;
   969→            }
   970→          })
   971→          .catch(() => {});
   972→      }
   973→    }
   974→
   975→    // Update particles
   976→    particlesRef.current = particlesRef.current.filter((pt) => {
   977→      pt.x += pt.vx;
   978→      pt.y += pt.vy;
   979→      pt.life--;
   980→      pt.vx *= 0.97;
   981→      pt.vy *= 0.97;
   982→      return pt.life > 0;
   983→    });
   984→
   985→    // ==== DRAW LOGIC ====
   986→
   987→    // Deep space background
   988→    const bgGrad = ctx.createLinearGradient(0, 0, 0, LOGIC_H);
   989→    bgGrad.addColorStop(0, '#0a0e27');
   990→    bgGrad.addColorStop(0.5, '#0f1638');
   991→    bgGrad.addColorStop(1, '#0a0e27');
   992→    ctx.fillStyle = bgGrad;
   993→    ctx.fillRect(0, 0, LOGIC_W, LOGIC_H);
   994→
   995→    // Stars
   996→    starsRef.current.forEach((s) => {
   997→      const alpha = s.alpha * (0.5 + Math.sin(s.phase) * 0.5);
   998→      ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
   999→      ctx.beginPath();
  1000→      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
  1001→      ctx.fill();
  1002→    });
  1003→
  1004→    // Powerups
  1005→    powerupsRef.current.forEach((pw) => drawPowerUp(ctx, pw));
  1006→
  1007→    // Player bullets (cyan glow)
  1008→    ctx.shadowColor = '#38bdf8';
  1009→    ctx.shadowBlur = 6;
  1010→    ctx.fillStyle = '#7dd3fc';
  1011→    bulletsRef.current.forEach((b) => {
  1012→      ctx.fillRect(b.x - BULLET_W / 2, b.y - BULLET_H / 2, BULLET_W, BULLET_H);
  1013→    });
  1014→    ctx.shadowBlur = 0;
  1015→
  1016→    // Enemy bullets
  1017→    enemyBulletsRef.current.forEach((b) => drawEnemyBullet(ctx, b));
  1018→
  1019→    // Regular enemies
  1020→    enemiesRef.current.forEach((e) => drawEnemy(ctx, e));
  1021→
  1022→    // Boss
  1023→    if (bossRef.current) {
  1024→      drawBoss(ctx, bossRef.current, frame);
  1025→    }
  1026→
  1027→    // Particles
  1028→    particlesRef.current.forEach((pt) => {
  1029→      const alpha = pt.life / pt.maxLife;
  1030→      ctx.globalAlpha = alpha;
  1031→      ctx.fillStyle = pt.color;
  1032→      ctx.beginPath();
  1033→      ctx.arc(pt.x, pt.y, pt.r * alpha, 0, Math.PI * 2);
  1034→      ctx.fill();
  1035→    });
  1036→    ctx.globalAlpha = 1;
  1037→
  1038→    // Player (not drawn when dead)
  1039→    if (statusRef.current !== 'over') {
  1040→      drawPlayer(ctx, p, frame);
  1041→    }
  1042→
  1043→    // Canvas HUD (lives + weapon level)
  1044→    if (statusRef.current === 'playing' || statusRef.current === 'over') {
  1045→      drawCanvasHUD(ctx, p, !!bossRef.current);
  1046→    }
  1047→
  1048→    // Screen flash effect (bomb)
  1049→    if (screenFlashRef.current > 0) {
  1050→      const flashAlpha = (screenFlashRef.current / SCREEN_FLASH_FRAMES) * 0.6;
