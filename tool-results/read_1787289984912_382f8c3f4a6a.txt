   165→function getWaveZombies(wave: number, countMul: number): { type: ZombieType; delay: number }[] {
   166→  const zombies: { type: ZombieType; delay: number }[] = [];
   167→  let delay = 0;
   168→  const baseCounts: number[][] = [
   169→    [4, 1, 0, 0, 0, 0, 0],
   170→    [3, 2, 0, 1, 1, 0, 0],
   171→    [2, 2, 1, 1, 1, 1, 0],
   172→    [2, 3, 1, 1, 2, 1, 1],
   173→    [1, 3, 2, 2, 2, 2, 2],
   174→  ];
   175→  const counts = baseCounts[Math.min(wave, baseCounts.length - 1)];
   176→  const types: ZombieType[] = ['normal', 'cone', 'bucket', 'flag', 'newspaper', 'football', 'gargantuar'];
   177→  for (let i = 0; i < types.length; i++) {
   178→    const count = Math.round(counts[i] * countMul);
   179→    for (let j = 0; j < count; j++) {
   180→      zombies.push({ type: types[i], delay });
   181→      delay += 2000 + Math.random() * 3000;
   182→    }
   183→  }
   184→  for (let i = zombies.length - 1; i > 0; i--) {
   185→    const j = Math.floor(Math.random() * (i + 1));
   186→    [zombies[i], zombies[j]] = [zombies[j], zombies[i]];
   187→  }
   188→  return zombies;
   189→}
