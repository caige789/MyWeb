   259→function createStars(count: number): Star[] {
   260→  const stars: Star[] = [];
   261→  for (let i = 0; i < count; i++) {
   262→    stars.push({
   263→      x: Math.random() * LOGIC_W,
   264→      y: Math.random() * LOGIC_H,
   265→      r: Math.random() * 1.5 + 0.5,
   266→      alpha: Math.random() * 0.6 + 0.4,
   267→      twinkleSpeed: Math.random() * 0.03 + 0.01,
   268→      phase: Math.random() * Math.PI * 2,
   269→    });
   270→  }
   271→  return stars;
   272→}
   273→
   274→function createExplosion(x: number, y: number, color: string, count: number): Particle[] {
   275→  const particles: Particle[] = [];
   276→  for (let i = 0; i < count; i++) {
   277→    const angle = Math.random() * Math.PI * 2;
   278→    const speed = Math.random() * 3 + 1;
   279→    particles.push({
   280→      x, y,
   281→      vx: Math.cos(angle) * speed,
   282→      vy: Math.sin(angle) * speed,
   283→      life: Math.random() * 20 + 15,
   284→      maxLife: 35,
   285→      color,
   286→      r: Math.random() * 3 + 1,
   287→    });
   288→  }
   289→  return particles;
   290→}

   309→function drawPlayer(ctx: CanvasRenderingContext2D, p: Player, frame: number, skinEmoji: string, skinAttackType: string) {
   310→  ctx.save();
   311→  ctx.translate(p.x, p.y);
   312→
   313→  if (p.invincibleTimer > 0 && Math.floor(frame / 3) % 2 === 0) {
   314→    ctx.globalAlpha = 0.35;
   315→  }
   316→
   317→  // Tail flame
   318→  let flameColor1 = '#38bdf8';
   319→  let flameColor2 = '#f97316';
   320→  if (skinAttackType === 'Fire') { flameColor1 = '#f97316'; flameColor2 = '#ef4444'; }
   321→  else if (skinAttackType === 'Ice') { flameColor1 = '#06b6d4'; flameColor2 = '#a5f3fc'; }
   322→  else if (skinAttackType === 'Star') { flameColor1 = '#fbbf24'; flameColor2 = '#f472b6'; }
   323→  else if (skinAttackType === 'Shadow') { flameColor1 = '#7c3aed'; flameColor2 = '#1e1b4b'; }
   324→  else if (skinAttackType === 'Lightning') { flameColor1 = '#fbbf24'; flameColor2 = '#fde047'; }
   325→
   326→  const flameLen = 12 + Math.sin(frame * 0.5) * 4;
   327→  const flameW = 8 + Math.sin(frame * 0.7) * 2;
   328→  const flameGrad = ctx.createLinearGradient(0, PLAYER_H / 2, 0, PLAYER_H / 2 + flameLen);
   329→  flameGrad.addColorStop(0, flameColor1);
   330→  flameGrad.addColorStop(0.4, flameColor2);
   331→  flameGrad.addColorStop(1, 'rgba(0,0,0,0)');
   332→  ctx.fillStyle = flameGrad;
   333→  ctx.beginPath();
   334→  ctx.moveTo(-flameW, PLAYER_H / 2 - 2);
   335→  ctx.lineTo(flameW, PLAYER_H / 2 - 2);
   336→  ctx.lineTo(0, PLAYER_H / 2 + flameLen);
   337→  ctx.closePath();
   338→  ctx.fill();
   339→
   340→  // Skin emoji
   341→  ctx.font = '32px sans-serif';
   342→  ctx.textAlign = 'center';
   343→  ctx.textBaseline = 'middle';
   344→  ctx.fillText(skinEmoji, 0, -2);
   345→
   346→  // Shield bubble
   347→  if (p.shieldTimer > 0) {
   348→    const shieldAlpha = p.shieldTimer < 60 ? (Math.sin(frame * 0.3) * 0.3 + 0.4) : 0.5;
   349→    ctx.strokeStyle = `rgba(6, 182, 212, ${shieldAlpha})`;
   350→    ctx.lineWidth = 2.5;
   351→    ctx.beginPath();
   352→    ctx.ellipse(0, 0, PLAYER_W / 2 + 8, PLAYER_H / 2 + 8, 0, 0, Math.PI * 2);
   353→    ctx.stroke();
   354→  }
   355→
   356→  ctx.restore();
   357→}

   359→function drawEnemy(ctx: CanvasRenderingContext2D, e: Enemy) {
   360→  ctx.save();
   361→  ctx.translate(e.x, e.y);
   362→
   363→  if (e.type === EnemyType.Normal) {
   364→    ctx.fillStyle = '#ef4444';
   365→    ctx.beginPath();
   366→    ctx.moveTo(0, e.h / 2);
   367→    ctx.lineTo(-e.w / 2, -e.h / 2);
   368→    ctx.lineTo(e.w / 2, -e.h / 2);
   369→    ctx.closePath();
   370→    ctx.fill();
   371→    ctx.strokeStyle = '#fca5a5';
   372→    ctx.lineWidth = 1;
   373→    ctx.beginPath();
   374→    ctx.moveTo(0, e.h / 2 - 6);
   375→    ctx.lineTo(-e.w / 4, -e.h / 4);
   376→    ctx.moveTo(0, e.h / 2 - 6);
   377→    ctx.lineTo(e.w / 4, -e.h / 4);
   378→    ctx.stroke();
   379→  } else if (e.type === EnemyType.Fast) {
   380→    ctx.fillStyle = '#f59e0b';
   381→    ctx.beginPath();
   382→    ctx.moveTo(0, e.h / 2);
   383→    ctx.lineTo(-e.w / 2, -e.h / 3);
   384→    ctx.lineTo(-e.w / 4, -e.h / 2);
   385→    ctx.lineTo(e.w / 4, -e.h / 2);
   386→    ctx.lineTo(e.w / 2, -e.h / 3);
   387→    ctx.closePath();
   388→    ctx.fill();
   389→    ctx.fillStyle = '#fcd34d';
   390→    ctx.beginPath();
   391→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   392→    ctx.fill();
   393→  } else if (e.type === EnemyType.Tank) {
   394→    const r = e.w / 2;
   395→    ctx.fillStyle = '#a855f7';
   396→    ctx.beginPath();
   397→    for (let i = 0; i < 6; i++) {
   398→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   399→      const px = Math.cos(angle) * r;
   400→      const py = Math.sin(angle) * r;
   401→      if (i === 0) ctx.moveTo(px, py);
   402→      else ctx.lineTo(px, py);
   403→    }
   404→    ctx.closePath();
   405→    ctx.fill();
   406→    ctx.strokeStyle = '#d8b4fe';
   407→    ctx.lineWidth = 1.5;
   408→    ctx.beginPath();
   409→    for (let i = 0; i < 6; i++) {
   410→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   411→      const px = Math.cos(angle) * (r * 0.55);
   412→      const py = Math.sin(angle) * (r * 0.55);
   413→      if (i === 0) ctx.moveTo(px, py);
   414→      else ctx.lineTo(px, py);
   415→    }
   416→    ctx.closePath();
   417→    ctx.stroke();
   418→    ctx.fillStyle = '#e9d5ff';
   419→    ctx.beginPath();
   420→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   421→    ctx.fill();
   422→  }
   423→
   424→  // Slow indicator
   425→  if (e.slowTimer > 0) {
   426→    ctx.strokeStyle = 'rgba(6, 182, 212, 0.6)';
   427→    ctx.lineWidth = 2;
   428→    ctx.beginPath();
   429→    ctx.arc(0, 0, e.w / 2 + 4, 0, Math.PI * 2);
   430→    ctx.stroke();
   431→  }
   432→
   433→  ctx.restore();
   434→
   435→  // HP bar
   436→  if (e.hp < e.maxHp) {
   437→    const barW = e.w * 0.8;
   438→    const barH = 3;
   439→    const barX = e.x - barW / 2;
   440→    const barY = e.y - e.h / 2 - 6;
   441→    ctx.fillStyle = 'rgba(255,255,255,0.2)';
   442→    ctx.fillRect(barX, barY, barW, barH);
   443→    ctx.fillStyle = '#22c55e';
   444→    ctx.fillRect(barX, barY, barW * (e.hp / e.maxHp), barH);
   445→  }
   446→}

   448→function drawBoss(ctx: CanvasRenderingContext2D, boss: Boss, frame: number) {
   449→  ctx.save();
   450→  ctx.translate(boss.x, boss.y);
   451→
   452→  if (boss.flashTimer > 0) {
   453→    ctx.globalAlpha = 0.5 + Math.sin(frame) * 0.5;
   454→    ctx.fillStyle = 'rgba(255,255,255,0.6)';
   455→    ctx.beginPath();
   456→    ctx.ellipse(0, 0, BOSS_W / 2 + 4, BOSS_H / 2 + 4, 0, 0, Math.PI * 2);
   457→    ctx.fill();
   458→    ctx.globalAlpha = 1;
   459→  }
   460→
   461→  if (boss.enraged) {
   462→    const enrageAlpha = 0.15 + Math.sin(frame * 0.2) * 0.1;
   463→    ctx.fillStyle = `rgba(255, 0, 0, ${enrageAlpha})`;
   464→    ctx.beginPath();
   465→    ctx.ellipse(0, 0, BOSS_W / 2 + 12, BOSS_H / 2 + 12, 0, 0, Math.PI * 2);
   466→    ctx.fill();
   467→  }
   468→
   469→  ctx.font = `${BOSS_H}px sans-serif`;
   470→  ctx.textAlign = 'center';

   517→function drawPowerUp(ctx: CanvasRenderingContext2D, pw: PowerUp) {
   518→  const glow = 0.3 + Math.sin(pw.phase) * 0.2;

   523→  ctx.fillStyle = color;
   524→  ctx.beginPath();
   525→  ctx.arc(pw.x, pw.y, 16, 0, Math.PI * 2);
   526→  ctx.fill();
   527→  ctx.restore();
   528→
   529→  ctx.fillStyle = color;
   530→  ctx.beginPath();
   531→  ctx.arc(pw.x, pw.y, 12, 0, Math.PI * 2);
   532→  ctx.fill();
   533→
   534→  ctx.font = '14px sans-serif';
   535→  ctx.textAlign = 'center';
   536→  ctx.textBaseline = 'middle';
   537→  ctx.fillText(POWERUP_EMOJIS[pw.type], pw.x, pw.y);
