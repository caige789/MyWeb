   133→   133→/** Current piece info */
   134→   134→interface Piece {
   135→   135→  type: string;
   136→   136→  shape: number[][];
   137→   137→  x: number;
   138→   138→  y: number;
   139→   139→}

   141→   141→type GameStatus = 'idle' | 'playing' | 'over';

   143→   143→/* ==================== Utility functions ==================== */
   144→   144→
   145→   145→function randomType(): string {
   146→   146→  return PIECE_TYPES[Math.floor(Math.random() * PIECE_TYPES.length)];
   147→   147→}
   148→   148→
   149→   149→function cloneShape(shape: number[][]): number[][] {
   150→   150→  return shape.map(row => [...row]);
   151→   151→}
   152→   152→
   153→   153→function rotateShape(shape: number[][]): number[][] {
   154→   154→  const n = shape.length;
   155→   155→  const rotated: number[][] = [];
   156→   156→  for (let i = 0; i < n; i++) {
   157→   157→    rotated[i] = [];
   158→   158→    for (let j = 0; j < n; j++) {
   159→   159→      rotated[i][j] = shape[n - 1 - j][i];
   160→   160→    }
   161→   161→  }
   162→   162→  return rotated;
   163→   163→}
   164→   164→
   165→   165→function createPiece(type: string): Piece {
   166→   166→  return {
   167→   167→    type,
   168→   168→    shape: cloneShape(SHAPES[type]),
   169→   169→    x: Math.floor((COLS - SHAPES[type][0].length) / 2),
   170→   170→    y: 0,
   171→   171→  };
   172→   172→}
   173→   173→
   174→   174→function isValid(
   175→   175→  shape: number[][],
   176→   176→  board: (string | null)[][],
   177→   177→  px: number,
   178→   178→  py: number
   179→   179→): boolean {
   180→   180→  for (let r = 0; r < shape.length; r++) {
   181→   181→    for (let c = 0; c < shape[r].length; c++) {
   182→   182→      if (shape[r][c]) {
   183→   183→        const nx = px + c;
   184→   184→        const ny = py + r;
   185→   185→        if (nx < 0 || nx >= COLS || ny >= ROWS) return false;
   186→   186→        if (ny >= 0 && board[ny][nx] !== null) return false;
   187→   187→      }
   188→   188→    }
   189→   189→  }
   190→   190→  return true;
   191→   191→}

   195→   195→export default function TetrisGame({ onBack }: TetrisGameProps) {
   196→   196→  const isMobile = useIsMobile();
   197→   197→
   198→   198→  /* --- Refs --- */

   202→   202→  const boardRef = useRef<(string | null)[][]>(
   203→   203→    Array.from({ length: ROWS }, () => Array(COLS).fill(null))
   204→   204→  );
   205→   205→  const currentRef = useRef<Piece | null>(null);
   206→   206→  const nextRef = useRef<string>(randomType());
   207→   207→  const scoreRef = useRef<number>(0);
   208→   208→  const levelRef = useRef<number>(1);
   209→   209→  const linesRef = useRef<number>(0);
   210→   210→  const statusRef = useRef<GameStatus>('idle');
   211→   211→  const lastDropRef = useRef<number>(0);
   212→   212→  const speedRef = useRef<SpeedKey>('normal');
   213→   213→  const flashRowsRef = useRef<number[]>([]);
   214→   214→  const flashStartRef = useRef<number>(0);
   215→   215→  const gameOverCheckedRef = useRef<boolean>(false);
   216→   216→  const scaleRef = useRef<number>(1);
   217→   217→
   218→   218→  /* --- React state (for UI rendering) --- */
   219→   219→  const [score, setScore] = useState(0);

   221→   221→  const [highScore, setHighScore] = useState(0);
   222→   222→  const [gameStatus, setGameStatus] = useState<GameStatus>('idle');
   223→   223→  const [speed, setSpeed] = useState<SpeedKey>('normal');
   224→   224→  const [isNewRecord, setIsNewRecord] = useState(false);

   229→   229→
   230→   230→  /* ==================== Drop interval ==================== */
   231→   231→  const getDropInterval = useCallback((): number => {
   232→   232→    const base = SPEED_CONFIG[speedRef.current].base;
   233→   233→    const lv = levelRef.current;

   236→   236→
   237→   237→  /* ==================== Reset board ==================== */
   238→   238→  const resetBoard = useCallback(() => {
   239→   239→    boardRef.current = Array.from({ length: ROWS }, () =>
   240→   240→      Array(COLS).fill(null)
   241→   241→    );
   242→   242→  }, []);

   264→   264→      for (let c = 0; c < piece.shape[r].length; c++) {
   265→   265→        if (piece.shape[r][c]) {
   266→   266→          const ny = piece.y + r;
   267→   267→          const nx = piece.x + c;
   268→   268→          if (ny >= 0 && ny < ROWS && nx >= 0 && nx < COLS) {
   269→   269→            board[ny][nx] = piece.type;
   270→   270→          }
   271→   271→        }
   272→   272→      }
   273→   273→    }
   274→   274→  }, []);
   275→   275→
   276→   276→  /* ==================== Clear lines ==================== */
   277→   277→  const clearLines = useCallback((): number => {
   278→   278→    const board = boardRef.current;
   279→   279→    const fullRows: number[] = [];
   280→   280→    for (let r = 0; r < ROWS; r++) {
   281→   281→      if (board[r].every(cell => cell !== null)) {
   282→   282→        fullRows.push(r);
   283→   283→      }
   284→   284→    }
   285→   285→    if (fullRows.length === 0) return 0;
   286→   286→    flashRowsRef.current = fullRows;
   287→   287→    flashStartRef.current = performance.now();
   288→   288→    return fullRows.length;
   289→   289→  }, []);
   290→   290→
   291→   291→  /** Remove full rows and add empty rows at top */
   292→   292→  const removeLines = useCallback(() => {
   293→   293→    const board = boardRef.current;
   294→   294→    const fullRows = flashRowsRef.current;
   295→   295→    const newBoard = board.filter((_, idx) => !fullRows.includes(idx));
   296→   296→    while (newBoard.length < ROWS) {
   297→   297→      newBoard.unshift(Array(COLS).fill(null));
   298→   298→    }
   299→   299→    boardRef.current = newBoard;
   300→   300→    flashRowsRef.current = [];
   301→   301→  }, []);
   302→   302→
   303→   303→  /* ==================== Move down ==================== */
   304→   304→  const moveDown = useCallback(() => {
   305→   305→    const piece = currentRef.current;
   306→   306→    if (!piece || statusRef.current !== 'playing') return;
   307→   307→    const board = boardRef.current;
   308→   308→    if (isValid(piece.shape, board, piece.x, piece.y + 1)) {
   309→   309→      piece.y += 1;
   310→   310→    } else {
   311→   311→      lockPiece();
   312→   312→      const cleared = clearLines();
   313→   313→      if (cleared > 0) {
   314→   314→        const pts = LINE_SCORES[cleared] || cleared * 100;
   315→   315→        scoreRef.current += pts;
   316→   316→        linesRef.current += cleared;
   317→   317→        const newLv = Math.floor(linesRef.current / 10) + 1;
   318→   318→        levelRef.current = newLv;
   319→   319→        setScore(scoreRef.current);
   320→   320→        setLevel(newLv);
   321→   321→      }
   322→   322→      if (cleared === 0) {
   323→   323→        spawnPiece();
   324→   324→      }
   325→   325→    }
   326→   326→  }, [lockPiece, clearLines, spawnPiece]);
   327→   327→
   328→   328→  /* ==================== Move left ==================== */
   329→   329→  const moveLeft = useCallback(() => {
   330→   330→    const piece = currentRef.current;
   331→   331→    if (!piece || statusRef.current !== 'playing') return;
   332→   332→    if (isValid(piece.shape, boardRef.current, piece.x - 1, piece.y)) {
   333→   333→      piece.x -= 1;
   334→   334→    }
   335→   335→  }, []);
   336→   336→
   337→   337→  /* ==================== Move right ==================== */
   338→   338→  const moveRight = useCallback(() => {
   339→   339→    const piece = currentRef.current;
   340→   340→    if (!piece || statusRef.current !== 'playing') return;
   341→   341→    if (isValid(piece.shape, boardRef.current, piece.x + 1, piece.y)) {
   342→   342→      piece.x += 1;
   343→   343→    }
   344→   344→  }, []);
   345→   345→
   346→   346→  /* ==================== Rotate ==================== */
   347→   347→  const rotate = useCallback(() => {
   348→   348→    const piece = currentRef.current;
   349→   349→    if (!piece || statusRef.current !== 'playing') return;
   350→   350→    if (piece.type === 'O') return;
   351→   351→    const rotated = rotateShape(piece.shape);
   352→   352→    if (isValid(rotated, boardRef.current, piece.x, piece.y)) {
   353→   353→      piece.shape = rotated;
   354→   354→      return;
   355→   355→    }
   356→   356→    const kicks = [1, -1, 2, -2];
   357→   357→    for (const kick of kicks) {
   358→   358→      if (isValid(rotated, boardRef.current, piece.x + kick, piece.y)) {
   359→   359→        piece.shape = rotated;
   360→   360→        piece.x += kick;
   361→   361→        return;
   362→   362→      }
   363→   363→    }
   364→   364→  }, []);
   365→   365→
   366→   366→  /* ==================== Hard drop ==================== */
   367→   367→  const hardDrop = useCallback(() => {
   368→   368→    const piece = currentRef.current;
   369→   369→    if (!piece || statusRef.current !== 'playing') return;
   370→   370→    const board = boardRef.current;
   371→   371→    const originalY = piece.y;
   372→   372→    let dropY = piece.y;
   373→   373→    while (isValid(piece.shape, board, piece.x, dropY + 1)) {
   374→   374→      dropY += 1;
   375→   375→    }
   376→   376→    piece.y = dropY;
   377→   377→    scoreRef.current += (dropY - originalY) * 2;
   378→   378→    setScore(scoreRef.current);
   379→   379→    lockPiece();
   380→   380→    const cleared = clearLines();
   381→   381→    if (cleared > 0) {
   382→   382→      const pts = LINE_SCORES[cleared] || cleared * 100;
   383→   383→      scoreRef.current += pts;
   384→   384→      linesRef.current += cleared;
   385→   385→      const newLv = Math.floor(linesRef.current / 10) + 1;
   386→   386→      levelRef.current = newLv;
   387→   387→      setScore(scoreRef.current);
   388→   388→      setLevel(newLv);
   389→   389→    }
   390→   390→    if (cleared === 0) {
   391→   391→      spawnPiece();
   392→   392→    }
   393→   393→  }, [lockPiece, clearLines, spawnPiece]);
   394→   394→
   395→   395→  /* ==================== Ghost Y ==================== */
   396→   396→  const getGhostY = useCallback((): number => {
   397→   397→    const piece = currentRef.current;
   398→   398→    if (!piece) return 0;
   399→   399→    const board = boardRef.current;
   400→   400→    let ghostY = piece.y;
   401→   401→    while (isValid(piece.shape, board, piece.x, ghostY + 1)) {
   402→   402→      ghostY += 1;
   403→   403→    }
   404→   404→    return ghostY;
   405→   405→  }, []);
   406→   406→
   407→   407→  /* ==================== Canvas rendering ==================== */
   408→   408→  const draw = useCallback(
   409→   409→    (ctx: CanvasRenderingContext2D, now: number) => {
   410→   410→      const s = scaleRef.current;
   411→   411→      const dpr = window.devicePixelRatio || 1;
   412→   412→      const cw = COLS * CELL;
   413→   413→      const ch = ROWS * CELL;
   414→   414→      const board = boardRef.current;
   415→   415→      const piece = currentRef.current;
   416→   416→
   417→   417→      // Set transform to map game coordinates to bitmap pixels
   418→   418→      ctx.setTransform(s * dpr, 0, 0, s * dpr, 0, 0);
   419→   419→      ctx.clearRect(0, 0, cw, ch);
   420→   420→
   421→   421→      // Deep background
   422→   422→      ctx.fillStyle = '#1a1a2e';
   423→   423→      ctx.fillRect(0, 0, cw, ch);
   424→   424→
   425→   425→      // Grid lines
   426→   426→      ctx.strokeStyle = 'rgba(255,255,255,0.06)';
   427→   427→      ctx.lineWidth = 0.5;
   428→   428→      for (let r = 0; r <= ROWS; r++) {
   429→   429→        ctx.beginPath();
   430→   430→        ctx.moveTo(0, r * CELL);
   431→   431→        ctx.lineTo(cw, r * CELL);
   432→   432→        ctx.stroke();
   433→   433→      }
   434→   434→      for (let c = 0; c <= COLS; c++) {
   435→   435→        ctx.beginPath();
   436→   436→        ctx.moveTo(c * CELL, 0);
   437→   437→        ctx.lineTo(c * CELL, ch);
   438→   438→        ctx.stroke();
   439→   439→      }
   440→   440→
   441→   441→      // Draw locked cells
   442→   442→      for (let r = 0; r < ROWS; r++) {
   443→   443→        for (let c = 0; c < COLS; c++) {
   444→   444→          if (board[r][c] !== null) {
   445→   445→            drawCell(ctx, c, r, board[r][c]!, 1);
   446→   446→          }
   447→   447→        }
   448→   448→      }
   449→   449→
   450→   450→      // Line clear flash animation
   451→   451→      if (flashRowsRef.current.length > 0) {
   452→   452→        const elapsed = now - flashStartRef.current;
   453→   453→        const flashDuration = 300;
   454→   454→        if (elapsed < flashDuration) {
   455→   455→          const alpha = 0.5 + 0.5 * Math.sin((elapsed / flashDuration) * Math.PI * 4);
   456→   456→          ctx.fillStyle = `rgba(255,255,255,${alpha})`;
   457→   457→          for (const row of flashRowsRef.current) {
   458→   458→            ctx.fillRect(0, row * CELL, cw, CELL);
   459→   459→          }
   460→   460→        } else {
   461→   461→          removeLines();
   462→   462→          spawnPiece();
   463→   463→        }
   464→   464→      }
   465→   465→
   466→   466→      // Draw ghost piece
   467→   467→      if (piece && statusRef.current === 'playing' && flashRowsRef.current.length === 0) {
   468→   468→        const ghostY = getGhostY();
   469→   469→        if (ghostY !== piece.y) {
   470→   470→          for (let r = 0; r < piece.shape.length; r++) {
   471→   471→            for (let c = 0; c < piece.shape[r].length; c++) {
   472→   472→              if (piece.shape[r][c]) {
   473→   473→                const px = (piece.x + c) * CELL;
   474→   474→                const py = (ghostY + r) * CELL;
   475→   475→                ctx.strokeStyle = COLORS[piece.type];
