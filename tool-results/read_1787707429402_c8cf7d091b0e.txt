    59→// Create monster with floor-scaled stats
    60→function makeMonster(type: MonsterType, x: number, y: number, floor: number): Monster {
    61→  const f = floor;
    62→  const s = {
    63→    slime:    { hp: 6 + f * 2,  atk: 1 + f,  def: 0,                 emoji: '🟢' },
    64→    skeleton: { hp: 8 + f * 3,  atk: 2 + f,  def: Math.floor(f / 2), emoji: '💀' },
    65→    ghost:    { hp: 7 + f * 2,  atk: 3 + f,  def: 0,                 emoji: '👻' },
    66→    dragon:   { hp: 20 + f * 5, atk: 5 + f,  def: 1,                 emoji: '🐉' },
    67→  }[type];
    68→  return { x, y, type, ...s, maxHp: s.hp };
    69→}

    71→// Generate dungeon for a given floor
    72→function generateDungeon(floor: number) {
    73→  const grid: number[][] = Array.from({ length: GRID }, () => Array(GRID).fill(WALL));
    74→  const targetRooms = Math.min(4 + Math.floor((floor + 1) / 2), 7);
    75→  const rooms: { x: number; y: number; w: number; h: number }[] = [];
    76→
    77→  // Place rooms with no overlap (1-tile padding)
    78→  for (let attempt = 0; attempt < 300 && rooms.length < targetRooms; attempt++) {
    79→    const w = 3 + Math.floor(Math.random() * 4);
    80→    const h = 3 + Math.floor(Math.random() * 4);
    81→    const x = 1 + Math.floor(Math.random() * (GRID - w - 2));
    82→    const y = 1 + Math.floor(Math.random() * (GRID - h - 2));
    83→    let ok = true;
    84→    for (const r of rooms) {
    85→      if (x - 1 < r.x + r.w && x + w + 1 > r.x && y - 1 < r.y + r.h && y + h + 1 > r.y) {
    86→        ok = false; break;
    87→      }
    88→    }
    89→    if (ok) rooms.push({ x, y, w, h });
    90→  }
    91→
    92→  if (rooms.length < 2) return generateDungeon(floor);
    93→
    94→  // Carve rooms into the grid
    95→  for (const r of rooms) {
    96→    for (let ry = r.y; ry < r.y + r.h; ry++)
    97→      for (let rx = r.x; rx < r.x + r.w; rx++)
    98→        grid[ry][rx] = FLOOR;
    99→  }
   100→
   101→  // Connect consecutive rooms with L-shaped corridors
   102→  for (let i = 1; i < rooms.length; i++) {
   103→    const a = rooms[i - 1], b = rooms[i];
   104→    const ax = Math.floor(a.x + a.w / 2), ay = Math.floor(a.y + a.h / 2);
   105→    const bx = Math.floor(b.x + b.w / 2), by = Math.floor(b.y + b.h / 2);
   106→    if (Math.random() > 0.5) {
   107→      for (let x = Math.min(ax, bx); x <= Math.max(ax, bx); x++) grid[ay][x] = FLOOR;
   108→      for (let y = Math.min(ay, by); y <= Math.max(ay, by); y++) grid[y][bx] = FLOOR;
   109→    } else {
   110→      for (let y = Math.min(ay, by); y <= Math.max(ay, by); y++) grid[y][ax] = FLOOR;
   111→      for (let x = Math.min(ax, bx); x <= Math.max(ax, bx); x++) grid[by][x] = FLOOR;
   112→    }
   113→  }
   114→
   115→  // Player start: center of first room
   116→  const sr = rooms[0];
   117→  const px = Math.floor(sr.x + sr.w / 2);
   118→  const py = Math.floor(sr.y + sr.h / 2);
   119→
   120→  // Door: center of last room
   121→  const lr = rooms[rooms.length - 1];
   122→  const doorX = Math.floor(lr.x + lr.w / 2);
   123→  const doorY = Math.floor(lr.y + lr.h / 2);
   124→
   125→  // Collect available floor tiles (excluding start room, player pos, door pos)
   126→  const available: { x: number; y: number }[] = [];
   127→  for (let cy = 0; cy < GRID; cy++) {
   128→    for (let cx = 0; cx < GRID; cx++) {
   129→      if (grid[cy][cx] !== FLOOR) continue;
   130→      if (cx === px && cy === py) continue;
   131→      if (cx === doorX && cy === doorY) continue;
   132→      if (cx >= sr.x && cx < sr.x + sr.w && cy >= sr.y && cy < sr.y + sr.h) continue;
   133→      available.push({ x: cx, y: cy });
   134→    }
   135→  }
   136→
   137→  // Shuffle available tiles
   138→  for (let i = available.length - 1; i > 0; i--) {
   139→    const j = Math.floor(Math.random() * (i + 1));
   140→    [available[i], available[j]] = [available[j], available[i]];
   141→  }
   142→
   143→  let idx = 0;
   144→
   145→  // Place key
   146→  const keyPos = available[idx++];
   147→  const items: DItem[] = [
   148→    { x: keyPos.x, y: keyPos.y, type: 'key', emoji: '🔑' },
   149→  ];
   150→
   151→  // Place other items (health potions, swords, shields)
   152→  const numItems = 2 + Math.floor(floor / 2);
   153→  const itemPool: ItemType[] = ['health', 'sword', 'shield'];
   154→  for (let i = 0; i < numItems && idx < available.length; i++) {
   155→    const type = itemPool[Math.floor(Math.random() * itemPool.length)];
   156→    const pos = available[idx++];
   157→    const emoji = type === 'health' ? '❤️' : type === 'sword' ? '⚔️' : '🛡️';
   158→    items.push({ x: pos.x, y: pos.y, type, emoji });
   159→  }
   160→
   161→  // Place monsters
   162→  const monsters: Monster[] = [];
   163→  const numMonsters = 3 + floor * 2;
   164→  for (let i = 0; i < numMonsters && idx < available.length; i++) {
   165→    const pos = available[idx++];
   166→    const r = Math.random();
   167→    let type: MonsterType;
   168→    if (floor >= 3 && r < 0.1) type = 'dragon';
   169→    else if (floor >= 3 && r < 0.3) type = 'ghost';
   170→    else if (floor >= 2 && r < 0.55) type = 'skeleton';
   171→    else type = 'slime';
   172→    monsters.push(makeMonster(type, pos.x, pos.y, floor));
   173→  }
   174→
   175→  // Guarantee at least 1 dragon on floor 3+
   176→  if (floor >= 3 && !monsters.some(m => m.type === 'dragon')) {
   177→    const ri = monsters.findIndex(m => m.type === 'slime' || m.type === 'skeleton');
   178→    if (ri >= 0) {
   179→      monsters[ri] = makeMonster('dragon', monsters[ri].x, monsters[ri].y, floor);
   180→    }
   181→  }
   182→
   183→  return { grid, px, py, doorX, doorY, monsters, items };
   184→}

   186→export default function DungeonGame({ onBack }: { onBack: () => void }) {
   187→  const canvasRef = useRef<HTMLCanvasElement>(null);
   188→  const rafRef = useRef(0);
   189→  const [phase, setPhase] = useState<Phase>('idle');
   190→  const [floor, setFloor] = useState(1);
   191→  const [hp, setHp] = useState(30);
   192→  const [maxHp, setMaxHp] = useState(30);
   193→  const [score, setScore] = useState(0);
   194→  const [bestScore, setBestScore] = useState(0);
   195→  const [hasKey, setHasKey] = useState(false);
   196→  const [paused, setPaused] = useState(false);
   197→  const [victory, setVictory] = useState(false);
   198→  const [monstersKilled, setMonstersKilled] = useState(0);
   199→
   200→  const g = useRef({
   201→    grid: [] as number[][],
   202→    floor: 1,
   203→    px: 1, py: 1,
   204→    hp: 30, maxHp: 30, atk: 5, def: 1,
   205→    hasKey: false,
   206→    monsters: [] as Monster[],
   207→    items: [] as DItem[],
   208→    doorX: 1, doorY: 1,
   209→    seen: new Set<string>(),
   210→    monstersKilled: 0,
   211→    score: 0,
   212→    phase: 'idle' as Phase,
   213→    paused: false,
   214→    victory: false,
   215→    floats: [] as FloatText[],
   216→  }).current;
   217→
   218→  const fetchBest = useCallback(async () => {
   219→    try {
   220→      const r = await fetch('/api/games/scores');
   221→      const d = await r.json();
   222→      if (d.data) {
   223→        const e = d.data.find((s: { game: string; score: number }) => s.game === GAME_NAME);
   224→        setBestScore(e ? e.score : 0);
   225→      }
   226→    } catch { /* */ }
   227→  }, []);
   228→
   229→  useEffect(() => { fetchBest(); }, [fetchBest]);
   230→
   231→  const submitScore = useCallback(async (s: number) => {
   232→    try {
   233→      await fetch('/api/games/scores', {
   234→        method: 'POST',
   235→        headers: { 'Content-Type': 'application/json' },
   236→        body: JSON.stringify({ game: GAME_NAME, score: s }),
   237→      });
   238→    } catch { /* */ }
   239→    fetchBest();
   240→  }, [fetchBest]);
   241→
   242→  const addFloat = useCallback((x: number, y: number, text: string, color: string, life = 40) => {
   243→    g.floats.push({ x: x * CELL + CELL / 2, y: y * CELL, text, color, life, maxLife: life });
   244→  }, []);
   245→
   246→  const endGame = useCallback((win: boolean) => {
   247→    g.phase = 'over'; g.victory = win;
   248→    const s = g.floor * 100 + g.monstersKilled * 10;
   249→    g.score = s;
   250→    setPhase('over'); setScore(s); setVictory(win);
   251→    submitScore(s);
   252→  }, [submitScore]);
   253→
   254→  const initFloor = useCallback((fl: number, keepStats: boolean) => {
   255→    const d = generateDungeon(fl);
   256→    g.grid = d.grid; g.px = d.px; g.py = d.py;
   257→    g.doorX = d.doorX; g.doorY = d.doorY;
   258→    g.monsters = d.monsters; g.items = d.items;
   259→    g.seen = new Set<string>();
   260→
   261→    if (!keepStats) {
   262→      g.hp = 30; g.maxHp = 30; g.atk = 5; g.def = 1;
   263→      g.hasKey = false; g.monstersKilled = 0; g.victory = false;
   264→    } else {
   265→      g.hasKey = false;
   266→      const heal = Math.floor(g.maxHp * 0.3);
   267→      g.hp = Math.min(g.hp + heal, g.maxHp);
   268→    }
   269→
   270→    g.floor = fl;
   271→    // Reveal area around player start
   272→    for (let dy = -VISION; dy <= VISION; dy++)
   273→      for (let dx = -VISION; dx <= VISION; dx++) {
   274→        const nx = g.px + dx, ny = g.py + dy;
   275→        if (nx >= 0 && nx < GRID && ny >= 0 && ny < GRID) g.seen.add(`${nx},${ny}`);
   276→      }
   277→    g.floats.push({ x: W / 2, y: H / 2, text: `第 ${fl} 层`, color: '#fbbf24', life: 60, maxLife: 60 });
   278→
   279→    setFloor(fl); setHp(g.hp); setMaxHp(g.maxHp); setHasKey(false);
   280→    setScore(g.floor * 100 + g.monstersKilled * 10);
   281→  }, []);
   282→
   283→  const startGame = useCallback(() => {
   284→    g.phase = 'playing'; g.paused = false; g.floats = [];
   285→    setPhase('playing'); setPaused(false); setVictory(false);
   286→    initFloor(1, false);
   287→  }, [initFloor]);
   288→
   289→  // Main movement and interaction logic
   290→  const movePlayer = useCallback((dx: number, dy: number) => {
   291→    if (g.phase !== 'playing' || g.paused) return;
   292→    const nx = g.px + dx, ny = g.py + dy;
   293→    if (nx < 0 || nx >= GRID || ny < 0 || ny >= GRID) return;
   294→    if (g.grid[ny][nx] === WALL) return;
   295→
   296→    // Check for monster at target position
   297→    const mi = g.monsters.findIndex(m => m.x === nx && m.y === ny);
   298→    if (mi >= 0) {
   299→      const m = g.monsters[mi];
   300→      const dmg = Math.max(1, g.atk - m.def);
   301→      m.hp -= dmg;
   302→      addFloat(nx, ny, `-${dmg}`, '#fbbf24');
   303→
   304→      if (m.hp <= 0) {
   305→        g.monsters.splice(mi, 1);
   306→        g.monstersKilled++;
   307→        setMonstersKilled(g.monstersKilled);
   308→        addFloat(nx, ny - 0.5, '击杀!', '#22c55e', 50);
   309→        setScore(g.floor * 100 + g.monstersKilled * 10);
   310→      } else {
   311→        const cd = Math.max(1, m.atk - g.def);
   312→        g.hp -= cd;
   313→        addFloat(g.px, g.py, `-${cd}`, '#ef4444');
   314→        setHp(g.hp);
   315→        if (g.hp <= 0) { g.hp = 0; setHp(0); endGame(false); return; }
   316→      }
   317→      return;
   318→    }
   319→
   320→    // Move player to new position
   321→    g.px = nx; g.py = ny;
   322→    for (let ddy = -VISION; ddy <= VISION; ddy++)
   323→      for (let ddx = -VISION; ddx <= VISION; ddx++) {
   324→        const rrx = nx + ddx, rry = ny + ddy;
   325→        if (rrx >= 0 && rrx < GRID && rry >= 0 && rry < GRID) g.seen.add(`${rrx},${rry}`);
   326→      }
   327→
   328→    // Check for items at new position
   329→    const ii = g.items.findIndex(it => it.x === nx && it.y === ny);
   330→    if (ii >= 0) {
   331→      const item = g.items[ii];
   332→      g.items.splice(ii, 1);
   333→      if (item.type === 'key') {
   334→        g.hasKey = true; setHasKey(true);
   335→        addFloat(nx, ny, '获得钥匙!', '#fbbf24', 50);
   336→      } else if (item.type === 'health') {
   337→        const heal = 15;
   338→        g.hp = Math.min(g.hp + heal, g.maxHp);
   339→        setHp(g.hp);
   340→        addFloat(nx, ny, `+${heal} 生命`, '#22c55e');
   341→      } else if (item.type === 'sword') {
   342→        g.atk += 2;
   343→        addFloat(nx, ny, '+2 攻击', '#f97316');
   344→      } else if (item.type === 'shield') {
   345→        g.def += 1;
   346→        addFloat(nx, ny, '+1 防御', '#3b82f6');
   347→      }
   348→    }
   349→
   350→    // Check for door at new position
   351→    if (nx === g.doorX && ny === g.doorY) {
   352→      if (g.hasKey) {
   353→        if (g.floor >= 5) {
   354→          endGame(true);
   355→        } else {
   356→          initFloor(g.floor + 1, true);
   357→        }
   358→      } else {
   359→        addFloat(nx, ny, '需要钥匙!', '#ef4444', 50);
   360→      }
   361→    }
   362→  }, [addFloat, endGame, initFloor]);
   363→
   364→  // Canvas buffer resize for high-DPI and responsive sizing
   365→  useEffect(() => {
   366→    if (phase !== 'playing') return;
   367→    const canvas = canvasRef.current;
   368→    if (!canvas) return;
   369→    const resize = () => {
   370→      const dpr = Math.min(window.devicePixelRatio || 1, 2);
   371→      const rect = canvas.getBoundingClientRect();
   372→      canvas.width = rect.width * dpr;
   373→      canvas.height = rect.height * dpr;
   374→    };
   375→    resize();
   376→    window.addEventListener('resize', resize);
   377→    return () => { window.removeEventListener('resize', resize); };
   378→  }, [phase]);
   379→
   380→  // Canvas render loop
   381→  useEffect(() => {
   382→    if (phase !== 'playing') return;
   383→    const canvas = canvasRef.current;
   384→    if (!canvas) return;
   385→    const ctx = canvas.getContext('2d');
   386→    if (!ctx) return;
   387→
   388→    function isVis(x: number, y: number) {
   389→      return Math.abs(x - g.px) <= VISION && Math.abs(y - g.py) <= VISION;
   390→    }
   391→
   392→    function loop() {
   393→      if (g.phase !== 'playing') return;
   394→      if (!ctx || !canvas) return;
   395→
   396→      // Scale logical W x H to canvas buffer
   397→      ctx.setTransform(canvas.width / W, 0, 0, canvas.height / H, 0, 0);
   398→
   399→      ctx.fillStyle = '#0a0a1a';
   400→      ctx.fillRect(0, 0, W, H);
   401→
   402→      // Draw all seen tiles
   403→      for (let y = 0; y < GRID; y++) {
   404→        for (let x = 0; x < GRID; x++) {
   405→          const k = `${x},${y}`;
   406→          if (!g.seen.has(k)) continue;
   407→          const sx = x * CELL, sy = y * CELL;
   408→          const vis = isVis(x, y);
   409→          const v = ((x * 7 + y * 13) % 5) * 3;
   410→
   411→          if (g.grid[y][x] === WALL) {
   412→            ctx.fillStyle = vis ? `rgb(${58 + v},${58 + v},${72 + v})` : `rgb(${38 + v},${38 + v},${48 + v})`;
   413→            ctx.fillRect(sx, sy, CELL, CELL);
   414→            ctx.strokeStyle = vis ? 'rgba(255,255,255,0.08)' : 'rgba(255,255,255,0.03)';
   415→            ctx.lineWidth = 0.5;
   416→            ctx.strokeRect(sx + 0.5, sy + 0.5, CELL - 1, CELL - 1);
   417→          } else {
   418→            ctx.fillStyle = vis ? `rgb(${120 + v},${108 + v},${88 + v})` : `rgb(${80 + v},${70 + v},${55 + v})`;
   419→            ctx.fillRect(sx, sy, CELL, CELL);
   420→            ctx.strokeStyle = vis ? 'rgba(0,0,0,0.15)' : 'rgba(0,0,0,0.1)';
   421→            ctx.lineWidth = 0.5;
   422→            ctx.strokeRect(sx + 0.5, sy + 0.5, CELL - 1, CELL - 1);
   423→          }
   424→        }
   425→      }
   426→
   427→      // Draw door (if seen)
   428→      if (g.seen.has(`${g.doorX},${g.doorY}`)) {
   429→        const vis = isVis(g.doorX, g.doorY);
   430→        ctx.globalAlpha = vis ? 1 : 0.5;
   431→        ctx.font = `${CELL - 6}px serif`;
   432→        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
   433→        ctx.fillText('🚪', g.doorX * CELL + CELL / 2, g.doorY * CELL + CELL / 2);
   434→        ctx.globalAlpha = 1;
   435→      }
   436→
   437→      // Draw items (if seen)
   438→      for (const item of g.items) {
   439→        if (!g.seen.has(`${item.x},${item.y}`)) continue;
   440→        const vis = isVis(item.x, item.y);
   441→        ctx.globalAlpha = vis ? 1 : 0.5;
   442→        ctx.font = `${CELL - 6}px serif`;
   443→        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
   444→        ctx.fillText(item.emoji, item.x * CELL + CELL / 2, item.y * CELL + CELL / 2);
   445→        ctx.globalAlpha = 1;
   446→      }
   447→
   448→      // Draw monsters (visible only)
   449→      for (const m of g.monsters) {
   450→        if (!isVis(m.x, m.y)) continue;
   451→        const mx = m.x * CELL, my = m.y * CELL;
   452→        ctx.font = `${CELL - 4}px serif`;
   453→        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
   454→        ctx.fillText(m.emoji, mx + CELL / 2, my + CELL / 2);
   455→        if (m.hp < m.maxHp) {
   456→          const bw = CELL - 4, bh = 3;
   457→          ctx.fillStyle = '#333';
   458→          ctx.fillRect(mx + 2, my + 1, bw, bh);
   459→          ctx.fillStyle = '#ef4444';
   460→          ctx.fillRect(mx + 2, my + 1, bw * (m.hp / m.maxHp), bh);
   461→        }
   462→      }
