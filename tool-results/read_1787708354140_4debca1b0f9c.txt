    61→    61→// Generate maze using recursive backtracking (iterative stack to avoid stack overflow)
    62→    62→function generateMaze(rows: number, cols: number): number[][] {
    63→    63→  // Initialize grid with all walls
    64→    64→  const grid: number[][] = Array.from({ length: rows }, () =>
    65→    65→    Array.from({ length: cols }, () => WALL)
    66→    66→  );
    67→    67→
    68→    68→  // Mark starting cell as path
    69→    69→  const startR = 1;
    70→    70→  const startC = 1;
    71→    71→  grid[startR][startC] = PATH;
    72→    72→
    73→    73→  // Stack-based DFS
    74→    74→  const stack: [number, number][] = [[startR, startC]];
    75→    75→  const directions = [
    76→    76→    [-2, 0], // up
    77→    77→    [2, 0],  // down
    78→    78→    [0, -2], // left
    79→    79→    [0, 2],  // right
    80→    80→  ];
    81→    81→
    82→    82→  while (stack.length > 0) {
    83→    83→    const [cr, cc] = stack[stack.length - 1];
    84→    84→
    85→    85→    // Shuffle directions
    86→    86→    const shuffled = [...directions].sort(() => Math.random() - 0.5);
    87→    87→
    88→    88→    let found = false;
    89→    89→    for (const [dr, dc] of shuffled) {
    90→    90→      const nr = cr + dr;
    91→    91→      const nc = cc + dc;
    92→    92→
    93→    93→      if (nr > 0 && nr < rows - 1 && nc > 0 && nc < cols - 1 && grid[nr][nc] === WALL) {
    94→    94→        // Carve the wall between current and neighbor
    95→    95→        grid[cr + dr / 2][cc + dc / 2] = PATH;
    96→    96→        // Mark neighbor as path
    97→    97→        grid[nr][nc] = PATH;
    98→    98→        stack.push([nr, nc]);
    99→    99→        found = true;
   100→   100→        break;
   101→   101→      }
   102→   102→    }
   103→   103→
   104→   104→    if (!found) {
   105→   105→      stack.pop();
   106→   106→    }
   107→   107→  }
   108→   108→
   109→   109→  // Ensure start and end are open
   110→   110→  grid[1][1] = PATH;
   111→   111→  grid[rows - 2][cols - 2] = PATH;
   112→   112→
   113→   113→  return grid;
   114→   114→}

   440→   440→      let dr = 0;
   441→   441→      let dc = 0;
   442→   442→      switch (e.key) {
   443→   443→        case 'ArrowUp': case 'w': case 'W': dr = -1; break;

   446→   446→        case 'ArrowRight': case 'd': case 'D': dc = 1; break;

   448→   448→      }

   451→   451→    };

   454→   454→    return () => window.removeEventListener('keydown', handleKeyDown);

   456→   456→
   457→   457→  // Swipe controls on canvas

   462→   462→
