   128→   128→      y: Math.random() * LOGIC_H,

   138→   138→/** 绘制玩家飞机 */
   139→   139→function drawPlayer(ctx: CanvasRenderingContext2D, p: Player, frame: number) {
   140→   140→  ctx.save();
   141→   141→  ctx.translate(p.x, p.y);
   142→   142→
   143→   143→  // 绘制尾焰动画
   144→   144→  const flameLen = 12 + Math.sin(frame * 0.5) * 4;
   145→   145→  const flameW = 8 + Math.sin(frame * 0.7) * 2;
   146→   146→  const flameGrad = ctx.createLinearGradient(0, PLAYER_H / 2, 0, PLAYER_H / 2 + flameLen);
   147→   147→  flameGrad.addColorStop(0, '#60a5fa');
   148→   148→  flameGrad.addColorStop(0.4, '#f97316');
   149→   149→  flameGrad.addColorStop(1, 'rgba(239,68,68,0)');
   150→   150→  ctx.fillStyle = flameGrad;
   151→   151→  ctx.beginPath();
   152→   152→  ctx.moveTo(-flameW, PLAYER_H / 2 - 2);
   153→   153→  ctx.lineTo(flameW, PLAYER_H / 2 - 2);
   154→   154→  ctx.lineTo(0, PLAYER_H / 2 + flameLen);
   155→   155→  ctx.closePath();
   156→   156→  ctx.fill();
   157→   157→
   158→   158→  // 绘制飞机机身
   159→   159→  ctx.fillStyle = '#e0f2fe';
   160→   160→  ctx.beginPath();
   161→   161→  ctx.moveTo(0, -PLAYER_H / 2);
   162→   162→  ctx.lineTo(-PLAYER_W / 2, PLAYER_H / 2 - 6);
   163→   163→  ctx.lineTo(-PLAYER_W / 4, PLAYER_H / 2);
   164→   164→  ctx.lineTo(PLAYER_W / 4, PLAYER_H / 2);
   165→   165→  ctx.lineTo(PLAYER_W / 2, PLAYER_H / 2 - 6);
   166→   166→  ctx.closePath();
   167→   167→  ctx.fill();
   168→   168→
   169→   169→  // 机翼装饰线
   170→   170→  ctx.strokeStyle = '#3b82f6';
   171→   171→  ctx.lineWidth = 1.5;
   172→   172→  ctx.beginPath();
   173→   173→  ctx.moveTo(-PLAYER_W / 2 + 4, PLAYER_H / 2 - 10);
   174→   174→  ctx.lineTo(0, -PLAYER_H / 2 + 10);
   175→   175→  ctx.lineTo(PLAYER_W / 2 - 4, PLAYER_H / 2 - 10);
   176→   176→  ctx.stroke();
   177→   177→
   178→   178→  // 座舱
   179→   179→  ctx.fillStyle = '#93c5fd';
   180→   180→  ctx.beginPath();
   181→   181→  ctx.ellipse(0, -PLAYER_H / 4, 4, 7, 0, 0, Math.PI * 2);
   182→   182→  ctx.fill();
   183→   183→
   184→   184→  // 如果有护盾，绘制护盾效果
   185→   185→  if (p.shieldTimer > 0) {
   186→   186→    const shieldAlpha = p.shieldTimer < 60 ? (Math.sin(frame * 0.3) * 0.3 + 0.4) : 0.5;
   187→   187→    ctx.strokeStyle = `rgba(74, 222, 128, ${shieldAlpha})`;
   188→   188→    ctx.lineWidth = 2.5;
   189→   189→    ctx.beginPath();
   190→   190→    ctx.ellipse(0, 0, PLAYER_W / 2 + 8, PLAYER_H / 2 + 8, 0, 0, Math.PI * 2);
   191→   191→    ctx.stroke();
   192→   192→  }
   193→   193→
   194→   194→  ctx.restore();
   195→   195→}

   245→   245→/** 绘制大型敌机（紫色六边形） */
   246→   246→function drawLargeEnemy(ctx: CanvasRenderingContext2D, e: Enemy) {
   247→   247→  ctx.save();
   248→   248→  ctx.translate(e.x, e.y);
   249→   249→  const r = e.w / 2;
   250→   250→  ctx.fillStyle = '#a855f7';
   251→   251→  ctx.beginPath();
   252→   252→  for (let i = 0; i < 6; i++) {
   253→   253→    const angle = (Math.PI / 3) * i - Math.PI / 2;
   254→   254→    const px = Math.cos(angle) * r;
   255→   255→    const py = Math.sin(angle) * r;
   256→   256→    if (i === 0) ctx.moveTo(px, py);
   257→   257→    else ctx.lineTo(px, py);
   258→   258→  }
   259→   259→  ctx.closePath();
   260→   260→  ctx.fill();
   261→   261→  // 内部装饰
   262→   262→  ctx.strokeStyle = '#d8b4fe';
   263→   263→  ctx.lineWidth = 1.5;
   264→   264→  ctx.beginPath();
   265→   265→  for (let i = 0; i < 6; i++) {
   266→   266→    const angle = (Math.PI / 3) * i - Math.PI / 2;
   267→   267→    const px = Math.cos(angle) * (r * 0.55);
   268→   268→    const py = Math.sin(angle) * (r * 0.55);
   269→   269→    if (i === 0) ctx.moveTo(px, py);
   270→   270→    else ctx.lineTo(px, py);
   271→   271→  }
   272→   272→  ctx.closePath();
   273→   273→  ctx.stroke();
   274→   274→  // 中心点
   275→   275→  ctx.fillStyle = '#e9d5ff';
   276→   276→  ctx.beginPath();
   277→   277→  ctx.arc(0, 0, 3, 0, Math.PI * 2);
   278→   278→  ctx.fill();
   279→   279→  ctx.restore();
   280→   280→}

   282→   282→/** 根据敌机类型分发绘制 */
   283→   283→function drawEnemy(ctx: CanvasRenderingContext2D, e: Enemy) {
   284→   284→  switch (e.type) {
   285→   285→    case EnemyType.Small: drawSmallEnemy(ctx, e); break;
   286→   286→    case EnemyType.Medium: drawMediumEnemy(ctx, e); break;
   287→   287→    case EnemyType.Large: drawLargeEnemy(ctx, e); break;
   288→   288→  }
   289→   289→  // 血条（非满血时显示）
   290→   290→  if (e.hp < e.maxHp) {
   291→   291→    const barW = e.w * 0.8;
   292→   292→    const barH = 3;
   293→   293→    const barX = e.x - barW / 2;
   294→   294→    const barY = e.y - e.h / 2 - 6;
   295→   295→    ctx.fillStyle = 'rgba(255,255,255,0.2)';
   296→   296→    ctx.fillRect(barX, barY, barW, barH);
   297→   297→    ctx.fillStyle = '#22c55e';
   298→   298→    ctx.fillRect(barX, barY, barW * (e.hp / e.maxHp), barH);
   299→   299→  }
   300→   300→}

   303→   303→function createExplosion(x: number, y: number, color: string, count: number): Particle[] {
   304→   304→  const particles: Particle[] = [];
   305→   305→  for (let i = 0; i < count; i++) {
   306→   306→    const angle = Math.random() * Math.PI * 2;
   307→   307→    const speed = Math.random() * 3 + 1;
   308→   308→    particles.push({
   309→   309→      x,
   310→   310→      y,
   311→   311→      vx: Math.cos(angle) * speed,
   312→   312→      vy: Math.sin(angle) * speed,
   313→   313→      life: Math.random() * 20 + 15,
   314→   314→      maxLife: 35,
   315→   315→      color,
   316→   316→      r: Math.random() * 3 + 1,
   317→   317→    });
   318→   318→  }
   319→   319→  return particles;
   320→   320→}

   322→   322→/** 矩形碰撞检测 */
   323→   323→function rectCollide(
   324→   324→  ax: number, ay: number, aw: number, ah: number,
   325→   325→  bx: number, by: number, bw: number, bh: number
   326→   326→): boolean {
   327→   327→  return Math.abs(ax - bx) < (aw + bw) / 2 && Math.abs(ay - by) < (ah + bh) / 2;
