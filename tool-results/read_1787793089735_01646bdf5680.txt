   130→function generateBricks(level: number, canvasW: number): Brick[] {
   131→  const bricks: Brick[] = [];
   132→  const brickW = (canvasW - (BRICK_COLS + 1) * BRICK_GAP) / BRICK_COLS;
   133→
   134→  if (level === 1) {
   135→    // Level 1: 5 rows standard
   136→    for (let row = 0; row < 5; row++) {
   137→      for (let col = 0; col < BRICK_COLS; col++) {
   138→        bricks.push({
   139→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   140→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   141→          w: brickW, h: BRICK_H, typeIndex: row, alive: true,
   142→        });
   143→      }
   144→    }
   145→  } else if (level === 2) {
   146→    // Level 2: 6 rows with diamond cutouts
   147→    for (let row = 0; row < 6; row++) {
   148→      const ti = Math.min(row, BRICK_TYPES.length - 1);
   149→      for (let col = 0; col < BRICK_COLS; col++) {
   150→        if (row === 2 && (col === 0 || col === 7)) continue;
   151→        if (row === 3 && (col === 3 || col === 4)) continue;
   152→        bricks.push({
   153→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   154→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   155→          w: brickW, h: BRICK_H, typeIndex: ti, alive: true,
   156→        });
   157→      }
   158→    }
   159→  } else if (level === 3) {
   160→    // Level 3: 7 rows, checkerboard pattern
   161→    for (let row = 0; row < 7; row++) {
   162→      const ti = Math.min(row, BRICK_TYPES.length - 1);
   163→      for (let col = 0; col < BRICK_COLS; col++) {
   164→        if ((row + col) % 2 === 0) continue;
   165→        bricks.push({
   166→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   167→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   168→          w: brickW, h: BRICK_H, typeIndex: ti, alive: true,
   169→        });
   170→      }
   171→    }
   172→  } else if (level === 4) {
   173→    // Level 4: 8 rows, X pattern cutout
   174→    for (let row = 0; row < 8; row++) {
   175→      const ti = Math.min(row, BRICK_TYPES.length - 1);
   176→      for (let col = 0; col < BRICK_COLS; col++) {
   177→        // X-shaped cutout
   178→        const diag1 = row - col;
   179→        const diag2 = row - (7 - col);
   180→        if (Math.abs(diag1) <= 0 && row < 8) continue;
   181→        if (Math.abs(diag2) <= 0 && row < 8) continue;
   182→        bricks.push({
   183→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   184→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   185→          w: brickW, h: BRICK_H, typeIndex: ti, alive: true,
   186→        });
   187→      }
   188→    }
   189→  } else {
   190→    // Level 5: 8 rows, dense, fortress pattern
   191→    for (let row = 0; row < 8; row++) {
   192→      const ti = Math.min(row, BRICK_TYPES.length - 1);
   193→      for (let col = 0; col < BRICK_COLS; col++) {
   194→        // Arrow cutout
   195→        if (row === 1 && col >= 3 && col <= 4) continue;
   196→        if (row === 2 && col === 3) continue;
   197→        if (row === 3 && col === 2) continue;
   198→        if (row === 3 && col === 5) continue;
   199→        if (row === 4 && col === 2) continue;
   200→        if (row === 4 && col === 5) continue;
   201→        if (row === 5 && col === 3) continue;
   202→        if (row === 6 && col >= 3 && col <= 4) continue;
   203→        bricks.push({
   204→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   205→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   206→          w: brickW, h: BRICK_H, typeIndex: ti, alive: true,
   207→        });
   208→      }
   209→    }
   210→  }
   211→  return bricks;
   212→}

   214→function roundRect(
   215→  ctx: CanvasRenderingContext2D,
   216→  x: number, y: number, w: number, h: number, r: number
   217→) {
   218→  const rr = Math.min(r, w / 2, h / 2);
   219→  ctx.beginPath();
   220→  ctx.moveTo(x + rr, y);
   221→  ctx.lineTo(x + w - rr, y);
   222→  ctx.quadraticCurveTo(x + w, y, x + w, y + rr);
   223→  ctx.lineTo(x + w, y + h - rr);
   224→  ctx.quadraticCurveTo(x + w, y + h, x + w - rr, y + h);
   225→  ctx.lineTo(x + rr, y + h);
   226→  ctx.quadraticCurveTo(x, y + h, x, y + h - rr);
   227→  ctx.lineTo(x, y + rr);
   228→  ctx.quadraticCurveTo(x, y, x + rr, y);
   229→  ctx.closePath();
   230→}

   344→    powerUpsRef.current.push({
   345→      x: brick.x + brick.w / 2,

   436→  const update = useCallback(() => {

   440→    const balls = ballsRef.current;
   441→    const bricks = bricksRef.current;
   442→    const powerUps = powerUpsRef.current;
   443→    const particles = particlesRef.current;
   444→    const trails = trailsRef.current;
   445→    const keys = keysRef.current;
   446→
   447→    // Paddle keyboard control
   448→    const moveSpeed = 7;
   449→    if (keys.has('ArrowLeft') || keys.has('a') || keys.has('A')) {
   450→      paddle.x = Math.max(0, paddle.x - moveSpeed);
   451→    }
   452→    if (keys.has('ArrowRight') || keys.has('d') || keys.has('D')) {
   453→      paddle.x = Math.min(canvasW - paddle.w, paddle.x + moveSpeed);
   454→    }
   455→
   456→    // Update stuck balls to follow paddle
   457→    balls.forEach(ball => {
   458→      if (ball.stuck) {
   459→        ball.x = paddle.x + paddle.w / 2 + ball.stuckOffset;
   460→        ball.y = paddleY - PADDLE_H / 2 - ball.radius - 1;
   461→      }
   462→    });
   463→
   464→    // Update balls
   465→    const ballsToRemove: number[] = [];
   466→    let hitBrickThisFrame = false;
   467→
   468→    balls.forEach((ball, idx) => {
   469→      if (ball.stuck) return;
   470→
   471→      trails.push({ x: ball.x, y: ball.y, alpha: 1 });
   472→
   473→      ball.x += ball.dx;
   474→      ball.y += ball.dy;
   475→
   476→      // Wall collisions
   477→      if (ball.x - ball.radius <= 0) {
   478→        ball.x = ball.radius;
   479→        ball.dx = Math.abs(ball.dx);
   480→      }
   481→      if (ball.x + ball.radius >= canvasW) {
   482→        ball.x = canvasW - ball.radius;
   483→        ball.dx = -Math.abs(ball.dx);
   484→      }
   485→      if (ball.y - ball.radius <= 0) {
   486→        ball.y = ball.radius;
   487→        ball.dy = Math.abs(ball.dy);
   488→      }
   489→
   490→      // Bottom out
   491→      if (ball.y - ball.radius > CANVAS_H) {
   492→        ballsToRemove.push(idx);
   493→        // Reset combo
   494→        comboCountRef.current = 0;
   495→        setCombo(0);
   496→        return;
   497→      }
   498→
   499→      // Paddle collision
   500→      const pLeft = paddle.x;
   501→      const pRight = paddle.x + paddle.w;
   502→      const pTop = paddleY - PADDLE_H / 2;
   503→      const pBottom = paddleY + PADDLE_H / 2;
   504→      if (
   505→        ball.dy > 0 &&
   506→        ball.y + ball.radius >= pTop &&
   507→        ball.y + ball.radius <= pBottom + 4 &&
   508→        ball.x >= pLeft - ball.radius &&
   509→        ball.x <= pRight + ball.radius
   510→      ) {
   511→        if (isMagnetRef.current) {
   512→          // Magnet: stick ball to paddle
   513→          ball.stuck = true;
   514→          ball.stuckOffset = ball.x - (paddle.x + paddle.w / 2);
   515→          ball.y = pTop - ball.radius - 1;
   516→        } else {
   517→          ball.y = pTop - ball.radius;
   518→          const hitPos = (ball.x - (paddle.x + paddle.w / 2)) / (paddle.w / 2);
   519→          const maxAngle = Math.PI / 3;
   520→          const angle = hitPos * maxAngle;
   521→          const spd = ball.speed;
   522→          ball.dx = Math.sin(angle) * spd;
   523→          ball.dy = -Math.cos(angle) * spd;
   524→        }
   525→      }
   526→
   527→      // Brick collision
   528→      bricks.forEach((brick) => {
   529→        if (!brick.alive) return;
   530→        const bLeft = brick.x;
   531→        const bRight = brick.x + brick.w;
   532→        const bTop = brick.y;
   533→        const bBottom = brick.y + brick.h;
   534→
   535→        if (
   536→          ball.x + ball.radius > bLeft &&
   537→          ball.x - ball.radius < bRight &&
   538→          ball.y + ball.radius > bTop &&
   539→          ball.y - ball.radius < bBottom
   540→        ) {
   541→          brick.alive = false;
   542→          hitBrickThisFrame = true;
   543→
   544→          // Combo system
   545→          comboCountRef.current += 1;
   546→          setCombo(comboCountRef.current);
   547→
   548→          // Score with combo multiplier
   549→          const baseScore = BRICK_TYPES[brick.typeIndex].score;
   550→          const comboMultiplier = 1 + Math.floor(comboCountRef.current / 3) * 0.5;
   551→          scoreRef.current += Math.round(baseScore * comboMultiplier);
   552→          setScore(scoreRef.current);
   553→
   554→          spawnParticles(brick);
   555→          trySpawnPowerUp(brick);
   556→
   557→          // Fireball: no bounce, pierce through
   558→          if (!isFireballRef.current) {
   559→            const overlapLeft = ball.x + ball.radius - bLeft;
   560→            const overlapRight = bRight - (ball.x - ball.radius);
   561→            const overlapTop = ball.y + ball.radius - bTop;
   562→            const overlapBottom = bBottom - (ball.y - ball.radius);
   563→            const minOverlapX = Math.min(overlapLeft, overlapRight);
   564→            const minOverlapY = Math.min(overlapTop, overlapBottom);
   565→
   566→            if (minOverlapX < minOverlapY) {
   567→              ball.dx = -ball.dx;
   568→            } else {
   569→              ball.dy = -ball.dy;
   570→            }
   571→          }
   572→        }
   573→      });
   574→    });
   575→
   576→    // Remove out-of-bounds balls
   577→    for (let i = ballsToRemove.length - 1; i >= 0; i--) {
   578→      balls.splice(ballsToRemove[i], 1);
   579→    }
   580→
   581→    // All balls lost
   582→    if (balls.length === 0) {
   583→      livesRef.current -= 1;
   584→      setLives(livesRef.current);
   585→      comboCountRef.current = 0;
   586→      setCombo(0);
   587→      if (livesRef.current <= 0) {
   588→        gameStatusRef.current = 'over';
   589→        setGameStatus('over');
   590→      } else {
   591→        ballsRef.current = [createBall(canvasW)];
   592→      }
   593→    }
   594→
   595→    // Level clear
   596→    if (bricks.every((b) => !b.alive) && gameStatusRef.current === 'playing') {
   597→      if (levelRef.current >= MAX_LEVEL) {
   598→        gameStatusRef.current = 'over';
   599→        setGameStatus('over');
   600→      } else {
   601→        gameStatusRef.current = 'levelUp';
   602→        setGameStatus('levelUp');
   603→        setTimeout(() => {
   604→          initLevel(levelRef.current + 1);
   605→          gameStatusRef.current = 'playing';
   606→          setGameStatus('playing');
   607→        }, 1500);
   608→      }
   609→    }
   610→
   611→    // Update power-ups
   612→    const puToRemove: number[] = [];
   613→    powerUps.forEach((pu, idx) => {
   614→      pu.y += POWERUP_SPEED;
   615→      pu.frame += 1;
   616→
   617→      if (
   618→        pu.y + POWERUP_SIZE / 2 >= paddleY - PADDLE_H / 2 &&
   619→        pu.y - POWERUP_SIZE / 2 <= paddleY + PADDLE_H / 2 &&
   620→        pu.x + POWERUP_SIZE / 2 >= paddle.x &&
   621→        pu.x - POWERUP_SIZE / 2 <= paddle.x + paddle.w
   622→      ) {
   623→        applyPowerUp(pu);
   624→        puToRemove.push(idx);
   625→      }
   626→
   627→      if (pu.y > CANVAS_H + 20) puToRemove.push(idx);
   628→    });
   629→    for (let i = puToRemove.length - 1; i >= 0; i--) {
   630→      powerUps.splice(puToRemove[i], 1);
   631→    }
   632→
   633→    // Update particles
   634→    for (let i = particles.length - 1; i >= 0; i--) {
   635→      const p = particles[i];
   636→      p.x += p.dx;
   637→      p.y += p.dy;
   638→      p.dy += 0.08;
   639→      p.life -= 1;
   640→      if (p.life <= 0) particles.splice(i, 1);
   641→    }
   642→
   643→    // Update trails
   644→    for (let i = trails.length - 1; i >= 0; i--) {
   645→      trails[i].alpha -= 0.08;
   646→      if (trails[i].alpha <= 0) trails.splice(i, 1);
   647→    }
   648→  }, [createBall, spawnParticles, trySpawnPowerUp, applyPowerUp, getPaddleY, initLevel]);

   718→      paddleGrad.addColorStop(1, '#2563eb');

   721→    roundRect(ctx, paddle.x, paddleY - PADDLE_H / 2, paddle.w, PADDLE_H, 6);
   722→    ctx.fill();

   724→    ctx.fillStyle = 'rgba(255,255,255,0.25)';
   725→    roundRect(ctx, paddle.x + 4, paddleY - PADDLE_H / 2 + 1, paddle.w - 8, PADDLE_H / 3, 3);
   726→    ctx.fill();

   728→    if (isMagnetRef.current) {
   729→      ctx.strokeStyle = 'rgba(34, 211, 238, 0.5)';
   730→      ctx.lineWidth = 2;
   731→      roundRect(ctx, paddle.x - 2, paddleY - PADDLE_H / 2 - 2, paddle.w + 4, PADDLE_H + 4, 8);

   745→        ctx.fillStyle = '#ffffff';
   746→        ctx.beginPath();
   747→        ctx.arc(ball.x, ball.y, ball.radius * 0.7, 0, Math.PI * 2);

   750→        ctx.shadowColor = '#ffffff';
   751→        ctx.shadowBlur = 12;
   752→        ctx.fillStyle = '#ffffff';
   753→        ctx.beginPath();
   754→        ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);

   756→        ctx.fillStyle = 'rgba(255,255,255,0.9)';

   774→      ctx.shadowBlur = 0;
   775→      ctx.fillStyle = '#ffffff';
   776→      ctx.font = 'bold 14px sans-serif';
   777→      ctx.textAlign = 'center';

   793→    const comboCount = comboCountRef.current;

   795→      const multiplier = 1 + Math.floor(comboCount / 3) * 0.5;
   796→      ctx.fillStyle = '#fbbf24';
   797→      ctx.font = 'bold 18px sans-serif';
   798→      ctx.textAlign = 'right';
   799→      ctx.textBaseline = 'top';
   800→      ctx.fillText(`${comboCount} COMBO! x${multiplier.toFixed(1)}`, w - 10, 10);
   801→    }
   802→
   803→    // Level up overlay
   804→    if (gameStatusRef.current === 'levelUp') {
   805→      ctx.fillStyle = 'rgba(0,0,0,0.5)';
   806→      ctx.fillRect(0, 0, w, h);
   807→      ctx.fillStyle = '#fbbf24';
   808→      ctx.font = 'bold 36px sans-serif';
   809→      ctx.textAlign = 'center';
   810→      ctx.textBaseline = 'middle';
   811→      ctx.fillText(`第 ${levelRef.current} 关通过！`, w / 2, h / 2 - 20);
   812→      ctx.fillStyle = '#ffffff';
   813→      ctx.font = '20px sans-serif';
   814→      ctx.fillText('准备进入下一关...', w / 2, h / 2 + 25);
   815→    }
   816→
   817→    // Pause overlay
   818→    if (pausedRef.current && gameStatusRef.current === 'playing') {
   819→      ctx.fillStyle = 'rgba(0,0,0,0.6)';
   820→      ctx.fillRect(0, 0, w, h);
   821→      ctx.fillStyle = '#ffffff';
   822→      ctx.font = 'bold 32px sans-serif';
   823→      ctx.textAlign = 'center';
   824→      ctx.textBaseline = 'middle';
   825→      ctx.fillText('已暂停', w / 2, h / 2 - 10);
   826→      ctx.font = '16px sans-serif';
   827→      ctx.fillStyle = '#a1a1aa';
   828→      ctx.fillText('按 Esc 继续', w / 2, h / 2 + 25);
   829→    }
   830→
   831→    // Idle overlay
   832→    if (gameStatusRef.current === 'idle') {
   833→      ctx.fillStyle = 'rgba(0,0,0,0.55)';
   834→      ctx.fillRect(0, 0, w, h);
   835→      ctx.fillStyle = '#ffffff';
   836→      ctx.font = 'bold 28px sans-serif';
   837→      ctx.textAlign = 'center';
   838→      ctx.textBaseline = 'middle';
   839→      ctx.fillText('打砖块', w / 2, h / 2 - 30);
   840→      ctx.font = '16px sans-serif';
   841→      ctx.fillStyle = '#a1a1aa';
   842→      ctx.fillText('按「开始游戏」开始', w / 2, h / 2 + 15);
   843→    }
   844→

   846→    if (gameStatusRef.current === 'over') {
   847→      ctx.fillStyle = 'rgba(0,0,0,0.65)';
   848→      ctx.fillRect(0, 0, w, h);
   849→      const isWin = livesRef.current > 0;
   850→      ctx.fillStyle = isWin ? '#4ade80' : '#ef4444';
   851→      ctx.font = 'bold 36px sans-serif';
   852→      ctx.textAlign = 'center';
   853→      ctx.textBaseline = 'middle';
   854→      ctx.fillText(isWin ? '恭喜通关！' : '游戏结束', w / 2, h / 2 - 40);
   855→      ctx.fillStyle = '#fbbf24';
   856→      ctx.font = 'bold 24px sans-serif';
   857→      ctx.fillText(`最终得分: ${scoreRef.current}`, w / 2, h / 2 + 10);
   858→      if (isWin) {
   859→        ctx.fillStyle = '#a1a1aa';
   860→        ctx.font = '16px sans-serif';
   861→        ctx.fillText('所有关卡已通关！', w / 2, h / 2 + 45);
   862→      }

   869→    }

   879→

   884→          method: 'POST',
   885→          headers: { 'Content-Type': 'application/json' },
   886→          body: JSON.stringify({ game: GAME_NAME, score: scoreRef.current }),

   889→          const data = await res.json();
   890→          if (data.data?.score !== undefined) setHighScore(data.data.score);
   891→        }
   892→      } catch { /* silent */ }
   893→    } else {
   894→      setIsNewRecord(false);
   895→    }
   896→  }, [highScore]);
   897→
   898→  useEffect(() => {
   899→    if (gameStatus === 'over') endGame();
   900→  }, [gameStatus, endGame]);
   901→
   902→  const startGame = useCallback(() => {

   905→    setScore(0);
   906→    setLives(MAX_LIVES);
   907→    setIsNewRecord(false);
   908→    comboCountRef.current = 0;
   909→    setCombo(0);
   910→    initLevel(1);
   911→    gameStatusRef.current = 'playing';
   912→    setGameStatus('playing');
   913→
   914→    if (animFrameRef.current) cancelAnimationFrame(animFrameRef.current);

   917→
   918→  // Keyboard events
   919→  useEffect(() => {

   921→      if (e.key === 'Escape' && gameStatusRef.current === 'playing') {

   930→        launchStuckBalls();

   940→    };

   942→    window.addEventListener('keyup', onKeyUp);

   944→      window.removeEventListener('keydown', onKeyDown);
   945→      window.removeEventListener('keyup', onKeyUp);
   946→    };
   947→  }, [launchStuckBalls]);
   948→
   949→  // Mouse control
   950→  useEffect(() => {
   951→    const canvas = canvasRef.current;
   952→    if (!canvas) return;

   954→      if (gameStatusRef.current !== 'playing') return;
   955→      const rect = canvas.getBoundingClientRect();
   956→      const scaleX = canvasWRef.current / rect.width;
   957→      const mouseX = (e.clientX - rect.left) * scaleX;
   958→      const paddle = paddleRef.current;
   959→      paddle.x = Math.max(0, Math.min(canvasWRef.current - paddle.w, mouseX - paddle.w / 2));
   960→    };
   961→    canvas.addEventListener('mousemove', onMouseMove);
   962→    return () => canvas.removeEventListener('mousemove', onMouseMove);
   963→  }, []);
   964→
   965→  // Touch control
   966→  useEffect(() => {
   967→    const canvas = canvasRef.current;
   968→    if (!canvas) return;

   970→      if (gameStatusRef.current !== 'playing') return;
   971→      e.preventDefault();

   973→      const rect = canvas.getBoundingClientRect();
   974→      const scaleX = canvasWRef.current / rect.width;
   975→      const touchX = (touch.clientX - rect.left) * scaleX;

   982→        launchStuckBalls();

   986→    canvas.addEventListener('touchstart', onTouchStart, { passive: true });

   988→      canvas.removeEventListener('touchmove', onTouchMove);
   989→      canvas.removeEventListener('touchstart', onTouchStart);

   992→

   997→        const res = await fetch('/api/games/scores');

   999→          const data = await res.json();

  1002→              (item: { game: string; score: number }) => item.game === GAME_NAME
  1003→            );

  1035→  // Cleanup
