   137→   137→/** 当前方块信息 */
   138→   138→interface Piece {
   139→   139→  type: string;
   140→   140→  shape: number[][];

   142→   142→  y: number;

   148→   148→/* ==================== 工具函数 ==================== */
   149→   149→
   150→   150→/** 随机生成一个方块类型 */
   151→   151→function randomType(): string {
   152→   152→  return PIECE_TYPES[Math.floor(Math.random() * PIECE_TYPES.length)];
   153→   153→}
   154→   154→
   155→   155→/** 深拷贝二维数组 */
   156→   156→function cloneShape(shape: number[][]): number[][] {
   157→   157→  return shape.map(row => [...row]);
   158→   158→}
   159→   159→
   160→   160→/** 顺时针旋转矩阵90度 */
   161→   161→function rotateShape(shape: number[][]): number[][] {
   162→   162→  const n = shape.length;
   163→   163→  const rotated: number[][] = [];
   164→   164→  for (let i = 0; i < n; i++) {
   165→   165→    rotated[i] = [];
   166→   166→    for (let j = 0; j < n; j++) {
   167→   167→      rotated[i][j] = shape[n - 1 - j][i];
   168→   168→    }
   169→   169→  }
   170→   170→  return rotated;
   171→   171→}
   172→   172→
   173→   173→/** 创建新方块 */
   174→   174→function createPiece(type: string): Piece {
   175→   175→  return {
   176→   176→    type,
   177→   177→    shape: cloneShape(SHAPES[type]),
   178→   178→    x: Math.floor((COLS - SHAPES[type][0].length) / 2),
   179→   179→    y: 0,
   180→   180→  };
   181→   181→}
   182→   182→
   183→   183→/** 碰撞检测：检查方块在指定位置是否合法 */
   184→   184→function isValid(
   185→   185→  shape: number[][],
   186→   186→  board: (string | null)[][],
   187→   187→  px: number,
   188→   188→  py: number
   189→   189→): boolean {
   190→   190→  for (let r = 0; r < shape.length; r++) {
   191→   191→    for (let c = 0; c < shape[r].length; c++) {
   192→   192→      if (shape[r][c]) {
   193→   193→        const nx = px + c;
   194→   194→        const ny = py + r;
   195→   195→        if (nx < 0 || nx >= COLS || ny >= ROWS) return false;
   196→   196→        if (ny >= 0 && board[ny][nx] !== null) return false;
   197→   197→      }
   198→   198→    }
   199→   199→  }
   200→   200→  return true;
   201→   201→}

   221→   221→  const speedRef = useRef<SpeedKey>('normal');
   222→   222→  const flashRowsRef = useRef<number[]>([]);
   223→   223→  const flashStartRef = useRef<number>(0);
   224→   224→  const gameOverCheckedRef = useRef<boolean>(false);
   225→   225→  const scaleRef = useRef<number>(1);
   226→   226→  const displayWidthRef = useRef<number>(COLS * CELL);
   227→   227→
   228→   228→  /* --- React状态（用于UI渲染） --- */
   229→   229→  const [score, setScore] = useState(0);
   230→   230→  const [level, setLevel] = useState(1);
   231→   231→  const [highScore, setHighScore] = useState(0);

   245→   245→    displayWidthRef.current = boardW * s;

   247→   247→  }, [isMobile]);

   256→   256→
   257→   257→  /* ==================== 初始化空棋盘 ==================== */
   258→   258→  const resetBoard = useCallback(() => {
   259→   259→    boardRef.current = Array.from({ length: ROWS }, () =>
   260→   260→      Array(COLS).fill(null)
   261→   261→    );
   262→   262→  }, []);
   263→   263→
   264→   264→  /* ==================== 生成新方块 ==================== */
   265→   265→  const spawnPiece = useCallback(() => {
   266→   266→    const type = nextRef.current;
   267→   267→    nextRef.current = randomType();
   268→   268→    const piece = createPiece(type);
   269→   269→    currentRef.current = piece;
   270→   270→    // 检查是否能放置新方块，若不能则游戏结束
   271→   271→    if (!isValid(piece.shape, boardRef.current, piece.x, piece.y)) {
   272→   272→      statusRef.current = 'over';
   273→   273→      setGameStatus('over');
   274→   274→      gameOverCheckedRef.current = false;

   276→   276→  }, []);
   277→   277→
   278→   278→  /* ==================== 固定方块到棋盘 ==================== */
   279→   279→  const lockPiece = useCallback(() => {
   280→   280→    const piece = currentRef.current;
   281→   281→    if (!piece) return;
   282→   282→    const board = boardRef.current;
   283→   283→    for (let r = 0; r < piece.shape.length; r++) {
   284→   284→      for (let c = 0; c < piece.shape[r].length; c++) {
   285→   285→        if (piece.shape[r][c]) {
   286→   286→          const ny = piece.y + r;
   287→   287→          const nx = piece.x + c;
   288→   288→          if (ny >= 0 && ny < ROWS && nx >= 0 && nx < COLS) {
   289→   289→            board[ny][nx] = piece.type;
   290→   290→          }
   291→   291→        }
   292→   292→      }
   293→   293→    }
   294→   294→  }, []);
   295→   295→
   296→   296→  /* ==================== 消行检测与处理 ==================== */
   297→   297→  const clearLines = useCallback((): number => {
   298→   298→    const board = boardRef.current;
   299→   299→    const fullRows: number[] = [];
   300→   300→    for (let r = 0; r < ROWS; r++) {
   301→   301→      if (board[r].every(cell => cell !== null)) {
   302→   302→        fullRows.push(r);
   303→   303→      }
   304→   304→    }
   305→   305→    if (fullRows.length === 0) return 0;
   306→   306→    // 触发闪白动画
   307→   307→    flashRowsRef.current = fullRows;
   308→   308→    flashStartRef.current = performance.now();
   309→   309→    // 实际消除行（动画结束后由主循环处理）
   310→   310→    return fullRows.length;
   311→   311→  }, []);
   312→   312→
   313→   313→  /** 实际移除已满行并添加空行到顶部 */
   314→   314→  const removeLines = useCallback(() => {
   315→   315→    const board = boardRef.current;
   316→   316→    const fullRows = flashRowsRef.current;
   317→   317→    // 从下往上移除满行
   318→   318→    const newBoard = board.filter((_, idx) => !fullRows.includes(idx));
   319→   319→    // 顶部补空行
   320→   320→    while (newBoard.length < ROWS) {
   321→   321→      newBoard.unshift(Array(COLS).fill(null));
   322→   322→    }
   323→   323→    boardRef.current = newBoard;
   324→   324→    flashRowsRef.current = [];
   325→   325→  }, []);
   326→   326→
   327→   327→  /* ==================== 方块下移 ==================== */
   328→   328→  const moveDown = useCallback(() => {
   329→   329→    const piece = currentRef.current;
   330→   330→    if (!piece || statusRef.current !== 'playing') return;
   331→   331→    const board = boardRef.current;
   332→   332→    if (isValid(piece.shape, board, piece.x, piece.y + 1)) {
   333→   333→      piece.y += 1;
   334→   334→    } else {
   335→   335→      // 方块触底，固定并处理消行
   336→   336→      lockPiece();
   337→   337→      const cleared = clearLines();
   338→   338→      if (cleared > 0) {
   339→   339→        const pts = LINE_SCORES[cleared] || cleared * 100;
   340→   340→        scoreRef.current += pts;
   341→   341→        linesRef.current += cleared;
   342→   342→        // 每10行升一级
   343→   343→        const newLv = Math.floor(linesRef.current / 10) + 1;
   344→   344→        levelRef.current = newLv;
   345→   345→        setScore(scoreRef.current);
   346→   346→        setLevel(newLv);
   347→   347→      }
   348→   348→      // 如果没有消行动画，直接生成新方块
   349→   349→      if (cleared === 0) {
   350→   350→        spawnPiece();
   351→   351→      }
   352→   352→      // 消行动画中会在主循环的渲染阶段处理
   353→   353→    }
   354→   354→  }, [lockPiece, clearLines, spawnPiece]);
   355→   355→
   356→   356→  /* ==================== 左移 ==================== */
   357→   357→  const moveLeft = useCallback(() => {
   358→   358→    const piece = currentRef.current;
   359→   359→    if (!piece || statusRef.current !== 'playing') return;
   360→   360→    if (isValid(piece.shape, boardRef.current, piece.x - 1, piece.y)) {
   361→   361→      piece.x -= 1;
   362→   362→    }
   363→   363→  }, []);
   364→   364→
   365→   365→  /* ==================== 右移 ==================== */
   366→   366→  const moveRight = useCallback(() => {
   367→   367→    const piece = currentRef.current;
   368→   368→    if (!piece || statusRef.current !== 'playing') return;
   369→   369→    if (isValid(piece.shape, boardRef.current, piece.x + 1, piece.y)) {
   370→   370→      piece.x += 1;
   371→   371→    }
   372→   372→  }, []);
   373→   373→
   374→   374→  /* ==================== 旋转 ==================== */
   375→   375→  const rotate = useCallback(() => {
   376→   376→    const piece = currentRef.current;
   377→   377→    if (!piece || statusRef.current !== 'playing') return;
   378→   378→    if (piece.type === 'O') return; // O方块不需要旋转
   379→   379→    const rotated = rotateShape(piece.shape);
   380→   380→    // 尝试基本旋转
   381→   381→    if (isValid(rotated, boardRef.current, piece.x, piece.y)) {
   382→   382→      piece.shape = rotated;
   383→   383→      return;
   384→   384→    }
   385→   385→    // 墙踢：尝试左右偏移
   386→   386→    const kicks = [1, -1, 2, -2];
   387→   387→    for (const kick of kicks) {
   388→   388→      if (isValid(rotated, boardRef.current, piece.x + kick, piece.y)) {
   389→   389→        piece.shape = rotated;
   390→   390→        piece.x += kick;
   391→   391→        return;
   392→   392→      }
   393→   393→    }
   394→   394→  }, []);
   395→   395→
   396→   396→  /* ==================== 硬降 ==================== */
   397→   397→  const hardDrop = useCallback(() => {
   398→   398→    const piece = currentRef.current;
   399→   399→    if (!piece || statusRef.current !== 'playing') return;
   400→   400→    const board = boardRef.current;
   401→   401→    const originalY = piece.y;
   402→   402→    let dropY = piece.y;
   403→   403→    while (isValid(piece.shape, board, piece.x, dropY + 1)) {
   404→   404→      dropY += 1;
   405→   405→    }
   406→   406→    piece.y = dropY;
   407→   407→    // 加上硬降距离的额外分数
   408→   408→    scoreRef.current += (dropY - originalY) * 2;
   409→   409→    setScore(scoreRef.current);
   410→   410→    // 直接触底固定
   411→   411→    lockPiece();
   412→   412→    const cleared = clearLines();
   413→   413→    if (cleared > 0) {
   414→   414→      const pts = LINE_SCORES[cleared] || cleared * 100;
   415→   415→      scoreRef.current += pts;
   416→   416→      linesRef.current += cleared;
   417→   417→      const newLv = Math.floor(linesRef.current / 10) + 1;
   418→   418→      levelRef.current = newLv;
   419→   419→      setScore(scoreRef.current);
   420→   420→      setLevel(newLv);
   421→   421→    }
   422→   422→    if (cleared === 0) {
   423→   423→      spawnPiece();
   424→   424→    }
   425→   425→  }, [lockPiece, clearLines, spawnPiece]);
   426→   426→
   427→   427→  /* ==================== 计算Ghost位置 ==================== */
   428→   428→  const getGhostY = useCallback((): number => {
   429→   429→    const piece = currentRef.current;
   430→   430→    if (!piece) return 0;
   431→   431→    const board = boardRef.current;
   432→   432→    let ghostY = piece.y;
   433→   433→    while (isValid(piece.shape, board, piece.x, ghostY + 1)) {
   434→   434→      ghostY += 1;
   435→   435→    }
   436→   436→    return ghostY;
   437→   437→  }, []);
   438→   438→
   439→   439→  /* ==================== Canvas rendering ==================== */
   440→   440→  const draw = useCallback(
   441→   441→    (ctx: CanvasRenderingContext2D, now: number) => {
   442→   442→      const s = scaleRef.current;
   443→   443→      const cw = COLS * CELL;
   444→   444→      const ch = ROWS * CELL;
   445→   445→      const board = boardRef.current;
   446→   446→      const piece = currentRef.current;
   447→   447→
   448→   448→      // 清空画布
   449→   449→      ctx.clearRect(0, 0, ctx.canvas.width, ctx.canvas.height);
   450→   450→      ctx.save();
   451→   451→      ctx.scale(s, s);
   452→   452→
   453→   453→      // 深色背景
   454→   454→      ctx.fillStyle = '#1a1a2e';
   455→   455→      ctx.fillRect(0, 0, cw, ch);
   456→   456→
   457→   457→      // 网格线

   516→   516→              }

   611→   611→  /** 绘制单个方块单元格（带高光和描边效果） */
   612→   612→  function drawCell(
   613→   613→    ctx: CanvasRenderingContext2D,
   614→   614→    x: number,
   615→   615→    y: number,
   616→   616→    type: string,
   617→   617→    alpha: number
   618→   618→  ) {
   619→   619→    const px = x * CELL;
   620→   620→    const py = y * CELL;

   625→   625→    // 描边

   629→   629→    // 顶部高光条
   630→   630→    ctx.fillStyle = HIGHLIGHT_COLORS[type];
   631→   631→    ctx.globalAlpha = alpha * 0.45;
   632→   632→    ctx.fillRect(px + 3, py + 3, CELL - 6, 5);
   633→   633→    // 左侧高光条
   634→   634→    ctx.fillRect(px + 3, py + 3, 3, CELL - 6);

   636→   636→  }
