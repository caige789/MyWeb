    63→// Generate maze using recursive backtracking (iterative stack to avoid stack overflow)
    64→function generateMaze(rows: number, cols: number): number[][] {
    65→  // Initialize grid with all walls
    66→  const grid: number[][] = Array.from({ length: rows }, () =>
    67→    Array.from({ length: cols }, () => WALL)
    68→  );
    69→
    70→  // Mark starting cell as path
    71→  const startR = 1;
    72→  const startC = 1;
    73→  grid[startR][startC] = PATH;
    74→
    75→  // Stack-based DFS
    76→  const stack: [number, number][] = [[startR, startC]];
    77→  const directions = [
    78→    [-2, 0], // up
    79→    [2, 0],  // down
    80→    [0, -2], // left
    81→    [0, 2],  // right
    82→  ];
    83→
    84→  while (stack.length > 0) {
    85→    const [cr, cc] = stack[stack.length - 1];
    86→
    87→    // Shuffle directions
    88→    const shuffled = [...directions].sort(() => Math.random() - 0.5);
    89→
    90→    let found = false;
    91→    for (const [dr, dc] of shuffled) {
    92→      const nr = cr + dr;
    93→      const nc = cc + dc;
    94→
    95→      if (nr > 0 && nr < rows - 1 && nc > 0 && nc < cols - 1 && grid[nr][nc] === WALL) {
    96→        // Carve the wall between current and neighbor
    97→        grid[cr + dr / 2][cc + dc / 2] = PATH;
    98→        // Mark neighbor as path
    99→        grid[nr][nc] = PATH;
   100→        stack.push([nr, nc]);
   101→        found = true;
   102→        break;
   103→      }
   104→    }
   105→
   106→    if (!found) {
   107→      stack.pop();
   108→    }
   109→  }
   110→
   111→  // Ensure start and end are open
   112→  grid[1][1] = PATH;
   113→  grid[rows - 2][cols - 2] = PATH;
   114→
   115→  return grid;
   116→}

   184→  
   185→    const config = DIFFICULTY_CONFIG[difficultyRef.current];
   186→

   188→    const exitC = cols - 2;

   190→    // Clear
   191→    ctx.fillStyle = '#0f172a';
   192→    ctx.fillRect(0, 0, size, size);
   193→
   194→    // Draw cells
   195→    for (let r = 0; r < rows; r++) {
   196→      for (let c = 0; c < cols; c++) {

   258→        ctx.lineWidth = Math.max(cellSize * 0.3, 2);

   260→        ctx.beginPath();

   262→        ctx.lineTo(x, y);
   263→        ctx.stroke();
   264→
   265→        ctx.globalAlpha = 1;
   266→      }
   267→    }
   268→
   269→    // Draw exit flag

   271→      const fx = exitC * cellSize;

   299→    // Draw player
   300→    const px = player.c * cellSize + cellSize / 2;
   301→    const py = player.r * cellSize + cellSize / 2;
   302→    const radius = Math.max(cellSize * 0.35, 3);
   303→
   304→    // Player glow

   306→    ctx.shadowBlur = 10;
   307→    ctx.fillStyle = '#4ade80';

   309→    ctx.arc(px, py, radius, 0, Math.PI * 2);

   311→    ctx.shadowBlur = 0;
   312→
   313→    // Player inner circle

   317→    ctx.fill();

   325→      if (gamePhaseRef.current !== 'playing') return;

   327→        draw();
   328→        animFrameRef.current = requestAnimationFrame(gameLoopFnRef.current);

   330→      }

   369→    if (gamePhaseRef.current !== 'playing') return;
   370→    if (pausedRef.current) return;
   371→
   372→    const maze = mazeRef.current;
   373→    const player = playerRef.current;
   374→    const nr = player.r + dr;
   375→    const nc = player.c + dc;
   376→
   377→    // Bounds check

   379→    // Wall check
   380→    if (maze[nr][nc] === WALL) return;
   381→
   382→    player.r = nr;
   383→    player.c = nc;
   384→    stepsRef.current += 1;
   385→    setSteps(stepsRef.current);
   386→
   387→    const key = `${nr},${nc}`;

   399→      const finalTime = (Date.now() - startTimeRef.current) / 1000;

   467→  const handleTouchEnd = useCallback((e: React.TouchEvent) => {

   477→    if (Math.abs(dx) < minDist && Math.abs(dy) < minDist) return;
   478→
   479→    if (Math.abs(dx) > Math.abs(dy)) {
   480→      movePlayer(0, dx > 0 ? 1 : -1);
   481→    } else {
   482→      movePlayer(dy > 0 ? 1 : -1, 0);
   483→    }
   484→    touchStartRef.current = null;
   485→  }, [movePlayer]);
   486→
   487→  // Canvas size setup (high DPI)

   489→    const container = containerRef.current;
   490→    const canvas = canvasRef.current;
   491→    if (!container || !canvas) return;
   492→
   493→    const update = () => {
   494→      const dpr = Math.min(window.devicePixelRatio || 1, 2);
   495→      const displaySize = Math.min(container.clientWidth, LOGIC_SIZE);
   496→      const scale = displaySize / LOGIC_SIZE;
   497→
   498→      canvas.width = displaySize * dpr;
   499→      canvas.height = displaySize * dpr;
   500→      canvas.style.width = `${displaySize}px`;
   501→      canvas.style.height = `${displaySize}px`;
   502→
   503→      dprRef.current = dpr;
   504→      scaleRef.current = scale;
   505→      draw();
   506→    };
   507→
   508→    const ro = new ResizeObserver(update);
   509→    ro.observe(container);
   510→    const onVis = () => { if (document.visibilityState === 'visible') update(); };
   511→    document.addEventListener('visibilitychange', onVis);
   512→
   513→    update();
   514→    return () => { ro.disconnect(); document.removeEventListener('visibilitychange', onVis); };
   515→  }, [draw, phase]);
   516→
   517→  // Cleanup
   518→  useEffect(() => {
   519→    return () => cancelAnimationFrame(animFrameRef.current);
   520→  }, []);
   521→

   523→  const toggleFog = useCallback(() => {
   524→    fogEnabledRef.current = !fogEnabledRef.current;
   525→    setFogEnabled(fogEnabledRef.current);
   526→    if (gamePhaseRef.current === 'playing') draw();
   527→  }, [draw]);
   528→
   529→  // --- Format time display ---
   530→  const formatTime = (t: number) => {
   531→    const mins = Math.floor(t / 60);
   532→    const secs = Math.floor(t % 60);
   533→    return `${mins.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
   534→  };
   535→
   536→  // ==================== SETTINGS PHASE ====================
   537→  if (phase === 'settings') {

   540→        <div className="flex items-center justify-between w-full px-2 pt-2 pb-1">
   541→          <Button variant="ghost" size="sm" onClick={onBack} className="text-muted-foreground hover:text-foreground">
   542→            <ArrowLeft className="h-4 w-4 mr-1" />
   543→            返回
   544→          </Button>

   546→          <div className="w-16" />
   547→        </div>
   548→
   549→        <div className="w-full px-2 py-3 flex flex-col gap-6">
   550→          <Card className="border-border/50 bg-card/80">
   551→            <CardContent className="p-4 flex flex-col gap-4">
