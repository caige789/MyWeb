    61→// Generate maze using recursive backtracking (iterative stack to avoid stack overflow)
    62→function generateMaze(rows: number, cols: number): number[][] {
    63→  // Initialize grid with all walls
    64→  const grid: number[][] = Array.from({ length: rows }, () =>
    65→    Array.from({ length: cols }, () => WALL)
    66→  );
    67→
    68→  // Mark starting cell as path
    69→  const startR = 1;
    70→  const startC = 1;
    71→  grid[startR][startC] = PATH;
    72→
    73→  // Stack-based DFS
    74→  const stack: [number, number][] = [[startR, startC]];
    75→  const directions = [
    76→    [-2, 0], // up
    77→    [2, 0],  // down
    78→    [0, -2], // left
    79→    [0, 2],  // right
    80→  ];
    81→
    82→  while (stack.length > 0) {
    83→    const [cr, cc] = stack[stack.length - 1];
    84→
    85→    // Shuffle directions
    86→    const shuffled = [...directions].sort(() => Math.random() - 0.5);
    87→
    88→    let found = false;
    89→    for (const [dr, dc] of shuffled) {
    90→      const nr = cr + dr;
    91→      const nc = cc + dc;
    92→
    93→      if (nr > 0 && nr < rows - 1 && nc > 0 && nc < cols - 1 && grid[nr][nc] === WALL) {
    94→        // Carve the wall between current and neighbor
    95→        grid[cr + dr / 2][cc + dc / 2] = PATH;
    96→        // Mark neighbor as path
    97→        grid[nr][nc] = PATH;
    98→        stack.push([nr, nc]);
    99→        found = true;
   100→        break;
   101→      }
   102→    }
   103→
   104→    if (!found) {
   105→      stack.pop();
   106→    }
   107→  }
   108→
   109→  // Ensure start and end are open
   110→  grid[1][1] = PATH;
   111→  grid[rows - 2][cols - 2] = PATH;
   112→
   113→  return grid;
   114→}

   494→      const dpr = window.devicePixelRatio || 1;

   496→      canvas.height = size * dpr;

   499→
   500→      const ctx = canvas.getContext('2d');
   501→      if (ctx) ctx.scale(dpr, dpr);
   502→
   503→      canvasDisplaySizeRef.current = size;

   505→    };

   511→
   512→  // Cleanup

   523→

   529→  };
   530→

   532→  if (phase === 'settings') {
