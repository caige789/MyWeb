   130→function generateBricks(level: number, canvasW: number): Brick[] {
   131→  const bricks: Brick[] = [];
   132→  const brickW = (canvasW - (BRICK_COLS + 1) * BRICK_GAP) / BRICK_COLS;
   133→
   134→  if (level === 1) {
   135→    // Level 1: 5 rows standard
   136→    for (let row = 0; row < 5; row++) {
   137→      for (let col = 0; col < BRICK_COLS; col++) {
   138→        bricks.push({
   139→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   140→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   141→          w: brickW, h: BRICK_H, typeIndex: row, alive: true,
   142→        });
   143→      }
   144→    }
   145→  } else if (level === 2) {
   146→    // Level 2: 6 rows with diamond cutouts
   147→    for (let row = 0; row < 6; row++) {
   148→      const ti = Math.min(row, BRICK_TYPES.length - 1);
   149→      for (let col = 0; col < BRICK_COLS; col++) {
   150→        if (row === 2 && (col === 0 || col === 7)) continue;
   151→        if (row === 3 && (col === 3 || col === 4)) continue;
   152→        bricks.push({
   153→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   154→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   155→          w: brickW, h: BRICK_H, typeIndex: ti, alive: true,
   156→        });
   157→      }
   158→    }
   159→  } else if (level === 3) {
   160→    // Level 3: 7 rows, checkerboard pattern
   161→    for (let row = 0; row < 7; row++) {
   162→      const ti = Math.min(row, BRICK_TYPES.length - 1);
   163→      for (let col = 0; col < BRICK_COLS; col++) {
   164→        if ((row + col) % 2 === 0) continue;
   165→        bricks.push({
   166→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   167→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   168→          w: brickW, h: BRICK_H, typeIndex: ti, alive: true,
   169→        });
   170→      }
   171→    }
   172→  } else if (level === 4) {
   173→    // Level 4: 8 rows, X pattern cutout
   174→    for (let row = 0; row < 8; row++) {
   175→      const ti = Math.min(row, BRICK_TYPES.length - 1);
   176→      for (let col = 0; col < BRICK_COLS; col++) {
   177→        // X-shaped cutout
   178→        const diag1 = row - col;
   179→        const diag2 = row - (7 - col);
   180→        if (Math.abs(diag1) <= 0 && row < 8) continue;
   181→        if (Math.abs(diag2) <= 0 && row < 8) continue;
   182→        bricks.push({
   183→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   184→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   185→          w: brickW, h: BRICK_H, typeIndex: ti, alive: true,
   186→        });
   187→      }
   188→    }
   189→  } else {
   190→    // Level 5: 8 rows, dense, fortress pattern
   191→    for (let row = 0; row < 8; row++) {
   192→      const ti = Math.min(row, BRICK_TYPES.length - 1);
   193→      for (let col = 0; col < BRICK_COLS; col++) {
   194→        // Arrow cutout
   195→        if (row === 1 && col >= 3 && col <= 4) continue;
   196→        if (row === 2 && col === 3) continue;
   197→        if (row === 3 && col === 2) continue;
   198→        if (row === 3 && col === 5) continue;
   199→        if (row === 4 && col === 2) continue;
   200→        if (row === 4 && col === 5) continue;
   201→        if (row === 5 && col === 3) continue;
   202→        if (row === 6 && col >= 3 && col <= 4) continue;
   203→        bricks.push({
   204→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   205→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   206→          w: brickW, h: BRICK_H, typeIndex: ti, alive: true,
   207→        });
   208→      }
   209→    }
   210→  }
   211→  return bricks;
   212→}

   214→function roundRect(
   215→  ctx: CanvasRenderingContext2D,
   216→  x: number, y: number, w: number, h: number, r: number
   217→) {
   218→  const rr = Math.min(r, w / 2, h / 2);
   219→  ctx.beginPath();
   220→  ctx.moveTo(x + rr, y);
   221→  ctx.lineTo(x + w - rr, y);
   222→  ctx.quadraticCurveTo(x + w, y, x + w, y + rr);
   223→  ctx.lineTo(x + w, y + h - rr);
   224→  ctx.quadraticCurveTo(x + w, y + h, x + w - rr, y + h);
   225→  ctx.lineTo(x + rr, y + h);
   226→  ctx.quadraticCurveTo(x, y + h, x, y + h - rr);
   227→  ctx.lineTo(x, y + rr);
   228→  ctx.quadraticCurveTo(x, y, x + rr, y);
   229→  ctx.closePath();
   230→}

   438→  const update = useCallback(() => {

   448→

   450→    const moveSpeed = 7;
   451→    if (keys.has('ArrowLeft') || keys.has('a') || keys.has('A')) {
   452→      paddle.x = Math.max(0, paddle.x - moveSpeed);
   453→    }
   454→    if (keys.has('ArrowRight') || keys.has('d') || keys.has('D')) {
   455→      paddle.x = Math.min(canvasW - paddle.w, paddle.x + moveSpeed);
   456→    }
   457→
   458→    // Update stuck balls to follow paddle
   459→    balls.forEach(ball => {
   460→      if (ball.stuck) {
   461→        ball.x = paddle.x + paddle.w / 2 + ball.stuckOffset;
   462→        ball.y = paddleY - PADDLE_H / 2 - ball.radius - 1;
   463→      }
   464→    });
   465→
   466→    // Update balls
   467→    const ballsToRemove: number[] = [];
   468→    let hitBrickThisFrame = false;
   469→
   470→    balls.forEach((ball, idx) => {
   471→      if (ball.stuck) return;
   472→
   473→      trails.push({ x: ball.x, y: ball.y, alpha: 1 });
   474→
   475→      ball.x += ball.dx;
   476→      ball.y += ball.dy;
   477→
   478→      // Wall collisions
   479→      if (ball.x - ball.radius <= 0) {
   480→        ball.x = ball.radius;
   481→        ball.dx = Math.abs(ball.dx);
   482→      }
   483→      if (ball.x + ball.radius >= canvasW) {
   484→        ball.x = canvasW - ball.radius;
   485→        ball.dx = -Math.abs(ball.dx);
   486→      }
   487→      if (ball.y - ball.radius <= 0) {
   488→        ball.y = ball.radius;
   489→        ball.dy = Math.abs(ball.dy);
   490→      }
   491→
   492→      // Bottom out
   493→      if (ball.y - ball.radius > CANVAS_H) {
   494→        ballsToRemove.push(idx);
   495→        // Reset combo
   496→        comboCountRef.current = 0;
   497→        setCombo(0);
   498→        return;
   499→      }
   500→
   501→      // Paddle collision
   502→      const pLeft = paddle.x;
   503→      const pRight = paddle.x + paddle.w;
   504→      const pTop = paddleY - PADDLE_H / 2;
   505→      const pBottom = paddleY + PADDLE_H / 2;
   506→      if (
   507→        ball.dy > 0 &&
   508→        ball.y + ball.radius >= pTop &&
   509→        ball.y + ball.radius <= pBottom + 4 &&
   510→        ball.x >= pLeft - ball.radius &&
   511→        ball.x <= pRight + ball.radius
   512→      ) {
   513→        if (isMagnetRef.current) {
   514→          // Magnet: stick ball to paddle
   515→          ball.stuck = true;
   516→          ball.stuckOffset = ball.x - (paddle.x + paddle.w / 2);
   517→          ball.y = pTop - ball.radius - 1;
   518→        } else {
   519→          ball.y = pTop - ball.radius;
   520→          const hitPos = (ball.x - (paddle.x + paddle.w / 2)) / (paddle.w / 2);
   521→          const maxAngle = Math.PI / 3;
   522→          const angle = hitPos * maxAngle;
   523→          const spd = ball.speed;
   524→          ball.dx = Math.sin(angle) * spd;
   525→          ball.dy = -Math.cos(angle) * spd;
   526→        }
   527→      }
   528→
   529→      // Brick collision
   530→      bricks.forEach((brick) => {
   531→        if (!brick.alive) return;
   532→        const bLeft = brick.x;
   533→        const bRight = brick.x + brick.w;
   534→        const bTop = brick.y;
   535→        const bBottom = brick.y + brick.h;
   536→
   537→        if (
   538→          ball.x + ball.radius > bLeft &&
   539→          ball.x - ball.radius < bRight &&
   540→          ball.y + ball.radius > bTop &&
   541→          ball.y - ball.radius < bBottom
   542→        ) {
   543→          brick.alive = false;
   544→          hitBrickThisFrame = true;
   545→
   546→          // Combo system
   547→          comboCountRef.current += 1;
   548→          setCombo(comboCountRef.current);
   549→
   550→          // Score with combo multiplier
   551→          const baseScore = BRICK_TYPES[brick.typeIndex].score;
   552→          const comboMultiplier = 1 + Math.floor(comboCountRef.current / 3) * 0.5;
   553→          scoreRef.current += Math.round(baseScore * comboMultiplier);
   554→          setScore(scoreRef.current);
   555→
   556→          spawnParticles(brick);
   557→          trySpawnPowerUp(brick);
   558→
   559→          // Fireball: no bounce, pierce through
   560→          if (!isFireballRef.current) {
   561→            const overlapLeft = ball.x + ball.radius - bLeft;
   562→            const overlapRight = bRight - (ball.x - ball.radius);
   563→            const overlapTop = ball.y + ball.radius - bTop;
   564→            const overlapBottom = bBottom - (ball.y - ball.radius);
   565→            const minOverlapX = Math.min(overlapLeft, overlapRight);
   566→            const minOverlapY = Math.min(overlapTop, overlapBottom);
   567→
   568→            if (minOverlapX < minOverlapY) {
   569→              ball.dx = -ball.dx;
   570→            } else {
   571→              ball.dy = -ball.dy;
   572→            }
   573→          }
   574→        }
   575→      });
   576→    });
   577→
   578→    // Remove out-of-bounds balls
   579→    for (let i = ballsToRemove.length - 1; i >= 0; i--) {
   580→      balls.splice(ballsToRemove[i], 1);
   581→    }
   582→
   583→    // All balls lost
   584→    if (balls.length === 0) {
   585→      livesRef.current -= 1;
   586→      setLives(livesRef.current);
   587→      comboCountRef.current = 0;
   588→      setCombo(0);
   589→      if (livesRef.current <= 0) {
   590→        gameStatusRef.current = 'over';
   591→        setGameStatus('over');
   592→      } else {
   593→        ballsRef.current = [createBall(canvasW)];
   594→      }
   595→    }
   596→
   597→    // Level clear
   598→    if (bricks.every((b) => !b.alive) && gameStatusRef.current === 'playing') {
   599→      if (levelRef.current >= MAX_LEVEL) {
   600→        gameStatusRef.current = 'over';
   601→        setGameStatus('over');
   602→      } else {
   603→        gameStatusRef.current = 'levelUp';
   604→        setGameStatus('levelUp');
   605→        setTimeout(() => {
   606→          initLevel(levelRef.current + 1);
   607→          gameStatusRef.current = 'playing';
   608→          setGameStatus('playing');
   609→        }, 1500);
   610→      }
   611→    }
   612→
   613→    // Update power-ups
   614→    const puToRemove: number[] = [];
   615→    powerUps.forEach((pu, idx) => {
   616→      pu.y += POWERUP_SPEED;
   617→      pu.frame += 1;
   618→
   619→      if (
   620→        pu.y + POWERUP_SIZE / 2 >= paddleY - PADDLE_H / 2 &&
   621→        pu.y - POWERUP_SIZE / 2 <= paddleY + PADDLE_H / 2 &&
   622→        pu.x + POWERUP_SIZE / 2 >= paddle.x &&
   623→        pu.x - POWERUP_SIZE / 2 <= paddle.x + paddle.w
   624→      ) {
   625→        applyPowerUp(pu);
   626→        puToRemove.push(idx);
   627→      }
   628→
   629→      if (pu.y > CANVAS_H + 20) puToRemove.push(idx);
   630→    });
   631→    for (let i = puToRemove.length - 1; i >= 0; i--) {
   632→      powerUps.splice(puToRemove[i], 1);
   633→    }
   634→
   635→    // Update particles
   636→    for (let i = particles.length - 1; i >= 0; i--) {
   637→      const p = particles[i];
   638→      p.x += p.dx;
   639→      p.y += p.dy;
   640→      p.dy += 0.08;
   641→      p.life -= 1;
   642→      if (p.life <= 0) particles.splice(i, 1);
   643→    }
   644→
   645→    // Update trails
   646→    for (let i = trails.length - 1; i >= 0; i--) {
   647→      trails[i].alpha -= 0.08;
   648→      if (trails[i].alpha <= 0) trails.splice(i, 1);
   649→    }
   650→  }, [createBall, spawnParticles, trySpawnPowerUp, applyPowerUp, getPaddleY, initLevel]);

   782→      ctx.font = 'bold 14px sans-serif';
   783→      ctx.textAlign = 'center';

   793→      ctx.fillStyle = p.color;
   794→      ctx.fillRect(p.x - p.size / 2, p.y - p.size / 2, p.size, p.size);

   799→    const comboCount = comboCountRef.current;

   801→      const multiplier = 1 + Math.floor(comboCount / 3) * 0.5;
   802→      ctx.fillStyle = '#fbbf24';
   803→      ctx.font = 'bold 18px sans-serif';
   804→      ctx.textAlign = 'right';
   805→      ctx.textBaseline = 'top';
   806→      ctx.fillText(`${comboCount} COMBO! x${multiplier.toFixed(1)}`, w - 10, 10);
   807→    }
   808→
   809→    // Level up overlay
   810→    if (gameStatusRef.current === 'levelUp') {
   811→      ctx.fillStyle = 'rgba(0,0,0,0.5)';
   812→      ctx.fillRect(0, 0, w, h);
   813→      ctx.fillStyle = '#fbbf24';
   814→      ctx.font = 'bold 36px sans-serif';
   815→      ctx.textAlign = 'center';
   816→      ctx.textBaseline = 'middle';
   817→      ctx.fillText(`第 ${levelRef.current} 关通过！`, w / 2, h / 2 - 20);
   818→      ctx.fillStyle = '#ffffff';
   819→      ctx.font = '20px sans-serif';
   820→      ctx.fillText('准备进入下一关...', w / 2, h / 2 + 25);
   821→    }
   822→

   826→      ctx.fillRect(0, 0, w, h);
   827→      ctx.fillStyle = '#ffffff';
   828→      ctx.font = 'bold 32px sans-serif';
   829→      ctx.textAlign = 'center';
   830→      ctx.textBaseline = 'middle';
   831→      ctx.fillText('已暂停', w / 2, h / 2 - 10);

   833→      ctx.fillStyle = '#a1a1aa';

   839→      ctx.fillStyle = 'rgba(0,0,0,0.55)';

   841→      ctx.fillStyle = '#ffffff';
   842→      ctx.font = 'bold 28px sans-serif';
   843→      ctx.textAlign = 'center';
   844→      ctx.textBaseline = 'middle';

   862→      ctx.font = 'bold 24px sans-serif';

   924→  // Keyboard events
