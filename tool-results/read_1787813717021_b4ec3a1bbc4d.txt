   168→   168→function getWaveZombies(wave: number, countMul: number): { type: ZombieType; delay: number }[] {
   169→   169→  const zombies: { type: ZombieType; delay: number }[] = [];
   170→   170→  let delay = 0;
   171→   171→  const baseCounts: number[][] = [
   172→   172→    [4, 1, 0, 0, 0, 0, 0],
   173→   173→    [3, 2, 0, 1, 1, 0, 0],
   174→   174→    [2, 2, 1, 1, 1, 1, 0],
   175→   175→    [2, 3, 1, 1, 2, 1, 1],
   176→   176→    [1, 3, 2, 2, 2, 2, 2],
   177→   177→  ];
   178→   178→  const counts = baseCounts[Math.min(wave, baseCounts.length - 1)];
   179→   179→  const types: ZombieType[] = ['normal', 'cone', 'bucket', 'flag', 'newspaper', 'football', 'gargantuar'];
   180→   180→  for (let i = 0; i < types.length; i++) {
   181→   181→    const count = Math.round(counts[i] * countMul);
   182→   182→    for (let j = 0; j < count; j++) {
   183→   183→      zombies.push({ type: types[i], delay });
   184→   184→      delay += 2000 + Math.random() * 3000;
   185→   185→    }
   186→   186→  }
   187→   187→  for (let i = zombies.length - 1; i > 0; i--) {
   188→   188→    const j = Math.floor(Math.random() * (i + 1));
   189→   189→    [zombies[i], zombies[j]] = [zombies[j], zombies[i]];
   190→   190→  }
   191→   191→  return zombies;
   192→   192→}
