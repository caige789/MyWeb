   159→   159→function getWaveZombies(wave: number, countMul: number): { type: ZombieType; delay: number }[] {
   160→   160→  const zombies: { type: ZombieType; delay: number }[] = [];
   161→   161→  let delay = 0;
   162→   162→  const baseCounts: number[][] = [
   163→   163→    [4, 1, 0, 0, 0, 0, 0],
   164→   164→    [3, 2, 0, 1, 1, 0, 0],
   165→   165→    [2, 2, 1, 1, 1, 1, 0],
   166→   166→    [2, 3, 1, 1, 2, 1, 1],
   167→   167→    [1, 3, 2, 2, 2, 2, 2],
   168→   168→  ];
   169→   169→  const counts = baseCounts[Math.min(wave, baseCounts.length - 1)];
   170→   170→  const types: ZombieType[] = ['normal', 'cone', 'bucket', 'flag', 'newspaper', 'football', 'gargantuar'];
   171→   171→  for (let i = 0; i < types.length; i++) {
   172→   172→    const count = Math.round(counts[i] * countMul);
   173→   173→    for (let j = 0; j < count; j++) {
   174→   174→      zombies.push({ type: types[i], delay });
   175→   175→      delay += 2000 + Math.random() * 3000;
   176→   176→    }
   177→   177→  }
   178→   178→  for (let i = zombies.length - 1; i > 0; i--) {
   179→   179→    const j = Math.floor(Math.random() * (i + 1));
   180→   180→    [zombies[i], zombies[j]] = [zombies[j], zombies[i]];
   181→   181→  }
   182→   182→  return zombies;
   183→   183→}
