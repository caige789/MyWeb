    42→const GAME_NAME = 'adventure';

    64→  h: number;

    84→  x: number;

    89→  facing: number;

    92→  invincible: boolean;

    95→  animTimer: number;

    97→/** 游戏状态 */
    98→type GameStatus = 'idle' | 'playing' | 'over';
    99→
   100→export default function AdventureGame({ onBack }: AdventureGameProps) {
   101→  const canvasRef = useRef<HTMLCanvasElement>(null);
   102→  const containerRef = useRef<HTMLDivElement>(null);
   103→  const rafRef = useRef<number>(0);
   104→  const gameStateRef = useRef<GameStatus>('idle');
   105→  const playerRef = useRef<Player>(createPlayer());
   106→  const platformsRef = useRef<Platform[]>([]);
   107→  const coinsRef = useRef<Coin[]>([]);
   108→  const monstersRef = useRef<Monster[]>([]);
   109→  const floatTextsRef = useRef<FloatText[]>([]);
   110→  const scoreRef = useRef(0);
   111→  const livesRef = useRef(3);
   112→  const cameraXRef = useRef(0);
   113→  const generatedUpToRef = useRef(0);
   114→  const keysRef = useRef<Set<string>>(new Set());
   115→  const lastTimeRef = useRef(0);
   116→  const distanceRef = useRef(0);
   117→  const scaleRef = useRef(1);
   118→
   119→  const [score, setScore] = useState(0);
   120→  const [highScore, setHighScore] = useState(0);
   121→  const [lives, setLives] = useState(3);
   122→  const [gameStatus, setGameStatus] = useState<GameStatus>('idle');
   123→  const [isNewRecord, setIsNewRecord] = useState(false);
   124→  const [isMobile, setIsMobile] = useState(false);
   125→  const touchLeftRef = useRef(false);
   126→  const touchRightRef = useRef(false);
   127→  const touchJumpRef = useRef(false);
   128→  const touchAttackRef = useRef(false);
   129→  const jumpPressedRef = useRef(false);
   130→
   131→  /* 检测移动端 */
   132→  useEffect(() => {
   133→    const check = () => setIsMobile(window.innerWidth < 768 || 'ontouchstart' in window);
   134→    check();
   135→    window.addEventListener('resize', check);
   136→    return () => window.removeEventListener('resize', check);
   137→  }, []);
   138→
   139→  /* 创建玩家初始状态 */
   140→  function createPlayer(): Player {
   141→    return {
   142→      x: 100,
   143→      y: GROUND_Y - PLAYER_H,
   144→      vx: 0,
   145→      vy: 0,
   146→      onGround: false,
   147→      facing: 1,
   148→      attacking: false,
   149→      attackTimer: 0,
   150→      invincible: false,
   151→      invincibleTimer: 0,
   152→      animFrame: 0,
   153→      animTimer: 0,
   154→    };
   155→  }
   156→
   157→  /* 初始化地图 */
   158→  const initGame = useCallback(() => {
   159→    playerRef.current = createPlayer();
   160→    platformsRef.current = [];
   161→    coinsRef.current = [];
   162→    monstersRef.current = [];
   163→    floatTextsRef.current = [];
   164→    scoreRef.current = 0;
   165→    livesRef.current = 3;
   166→    cameraXRef.current = 0;
   167→    generatedUpToRef.current = 0;
   168→    distanceRef.current = 0;
   169→    setScore(0);
   170→    setLives(3);
   171→    setIsNewRecord(false);
   172→    /* 生成起始地面和初始地图 */

   176→  /* 生成初始地图（包含起始地面平台） */
   177→  function generateInitialMap() {
   178→    const platforms = platformsRef.current;
   179→    const coins = coinsRef.current;
   180→    const monsters = monstersRef.current;
   181→    /* 起始地面 */
   182→    platforms.push({ x: 0, y: GROUND_Y, w: 400, h: 60 });
   183→    /* 在起始地面上放一些金币引导玩家 */
   184→    for (let i = 0; i < 5; i++) {
   185→      coins.push({ x: 200 + i * 40, y: GROUND_Y - 40, collected: false, animPhase: Math.random() * Math.PI * 2 });
   186→    }
   187→    /* 生成后续地图块 */
   188→    generatedUpToRef.current = 400;
   189→    generateChunks(400, 400 + CHUNK_W * 3);
   190→  }
   191→
   192→  /* 程序化生成地图块 */
   193→  function generateChunks(fromX: number, toX: number) {
   194→    const platforms = platformsRef.current;
   195→    const coins = coinsRef.current;
   196→    const monsters = monstersRef.current;
   197→    let x = fromX;
   198→    while (x < toX) {
   199→      const rand = Math.random();
   200→      if (rand < 0.35) {
   201→        /* 地面段 */
   202→        const w = 120 + Math.random() * 200;
   203→        platforms.push({ x, y: GROUND_Y, w, h: 60 });
   204→        /* 在地面上放怪物 */
   205→        if (Math.random() < 0.5) {
   206→          const kind: MonsterKind = Math.random() < 0.6 ? 'slime' : 'skeleton';
   207→          monsters.push(createMonster(kind, x + 30, GROUND_Y));
   208→        }
   209→        /* 在地面上方放金币 */
   210→        const numCoins = Math.floor(Math.random() * 4) + 1;
   211→        for (let i = 0; i < numCoins; i++) {
   212→          coins.push({ x: x + 20 + i * 30, y: GROUND_Y - 40, collected: false, animPhase: Math.random() * Math.PI * 2 });
   213→        }
   214→        x += w + 30 + Math.random() * 50;
   215→      } else if (rand < 0.65) {
   216→        /* 浮空平台 */
   217→        const w = 80 + Math.random() * 100;
   218→        const py = GROUND_Y - 80 - Math.random() * 120;
   219→        platforms.push({ x, y: py, w, h: 20 });
   220→        /* 平台上方放金币 */
   221→        coins.push({ x: x + w / 2, y: py - 35, collected: false, animPhase: Math.random() * Math.PI * 2 });
   222→        /* 偶尔放怪物在浮空平台上 */
   223→        if (Math.random() < 0.3) {
   224→          monsters.push(createMonster('slime', x + 10, py));
   225→        }
   226→        x += w + 50 + Math.random() * 80;
   227→      } else if (rand < 0.8) {
   228→        /* 阶梯平台组 */
   229→        const steps = 2 + Math.floor(Math.random() * 2);
   230→        let sx = x;
   231→        for (let s = 0; s < steps; s++) {
   232→          const w = 60 + Math.random() * 40;
   233→          const py = GROUND_Y - 50 - s * 55;
   234→          platforms.push({ x: sx, y: py, w, h: 18 });
   235→          coins.push({ x: sx + w / 2, y: py - 30, collected: false, animPhase: Math.random() * Math.PI * 2 });
   236→          sx += w + 20;
   237→        }
   238→        x = sx + 40;
   239→      } else {
   240→        /* 空隙（深渊） */
   241→        x += 60 + Math.random() * 60;
   242→        /* 空隙后放地面 */
   243→        const w = 150 + Math.random() * 200;
   244→        platforms.push({ x, y: GROUND_Y, w, h: 60 });
   245→        if (Math.random() < 0.6) {
   246→          const kind: MonsterKind = Math.random() < 0.6 ? 'slime' : 'skeleton';
   247→          monsters.push(createMonster(kind, x + 20, GROUND_Y));
   248→        }
   249→        const numCoins = Math.floor(Math.random() * 3) + 1;
   250→        for (let i = 0; i < numCoins; i++) {
   251→          coins.push({ x: x + 30 + i * 35, y: GROUND_Y - 40, collected: false, animPhase: Math.random() * Math.PI * 2 });
   252→        }
   253→        x += w + 30 + Math.random() * 50;
   254→      }
   255→    }
   256→    generatedUpToRef.current = Math.max(generatedUpToRef.current, x);
   257→  }
   258→
   259→  /* 创建怪物 */
   260→  function createMonster(kind: MonsterKind, x: number, groundY: number): Monster {
   261→    const isSlime = kind === 'slime';
   262→    const w = isSlime ? 28 : 26;
   263→    const h = isSlime ? 22 : 36;
   264→    const speed = isSlime ? 0.8 : 1.5;
   265→    const hp = isSlime ? 1 : 2;
   266→    return {
   267→      x,
   268→      y: groundY - h,
   269→      w,
   270→      h,
   271→      kind,
   272→      hp,
   273→      maxHp: hp,
   274→      vx: speed * (Math.random() < 0.5 ? 1 : -1),
   275→      alive: true,
   276→      animPhase: Math.random() * Math.PI * 2,
   277→      onPlatform: null,
   278→    };
   279→  }
   280→
   281→  /* 获取最高分 */
   282→  useEffect(() => {
   283→    fetch('/api/games/scores')
   284→      .then(r => r.json())
   285→      .then(res => {
   286→        if (res.code === 0 && res.data) {
   287→          const found = res.data.find((s: { game: string; score: number }) => s.game === GAME_NAME);
   288→          if (found) setHighScore(found.score);
   289→        }
   290→      })
   291→      .catch(() => {});
   292→  }, []);
   293→
   294→  /* 更新最高分 */
   295→  const submitScore = useCallback((finalScore: number) => {
   296→    fetch('/api/games/scores', {
   297→      method: 'POST',
   298→      headers: { 'Content-Type': 'application/json' },
   299→      body: JSON.stringify({ game: GAME_NAME, score: finalScore }),
   300→    })
   301→      .then(r => r.json())
   302→      .then(res => {
   303→        if (res.code === 0 && res.data) {
   304→          if (finalScore >= res.data.score) {
   305→            setIsNewRecord(true);
   306→            setHighScore(finalScore);
   307→          }
   308→        }
   309→      })
   310→      .catch(() => {});
   311→  }, []);
   312→
   313→  /* 游戏结束 */
   314→  const gameOver = useCallback(() => {
   315→    gameStateRef.current = 'over';
   316→    setGameStatus('over');
   317→    submitScore(scoreRef.current);
   318→  }, [submitScore]);
   319→
   320→  /* 玩家受伤 */
   321→  const hurtPlayer = useCallback(() => {
   322→    const p = playerRef.current;
   323→    if (p.invincible) return;
   324→    livesRef.current--;
   325→    setLives(livesRef.current);
   326→    if (livesRef.current <= 0) {
   327→      gameOver();
   328→    } else {
   329→      p.invincible = true;
   330→      p.invincibleTimer = INVINCIBLE_TIME;
   331→    }
   332→  }, [gameOver]);
   333→
   334→  /* 玩家重生（坠落后） */
   335→  const respawnPlayer = useCallback(() => {
   336→    const p = playerRef.current;
   337→    const cam = cameraXRef.current;
   338→    /* 在摄像机附近找一个平台重生 */
   339→    const platforms = platformsRef.current;
   340→    let bestPlatform: Platform | null = null;
   341→    let bestDist = Infinity;
   342→    for (const pl of platforms) {
   343→      if (pl.x + pl.w > cam && pl.x < cam + CANVAS_W) {
   344→        const center = pl.x + pl.w / 2;
   345→        const dist = Math.abs(center - (cam + CANVAS_W / 3));
   346→        if (dist < bestDist) {
   347→          bestDist = dist;
   348→          bestPlatform = pl;
   349→        }
   350→      }
   351→    }
   352→    if (bestPlatform) {
   353→      p.x = bestPlatform.x + bestPlatform.w / 2 - PLAYER_W / 2;
   354→      p.y = bestPlatform.y - PLAYER_H;
   355→    } else {
   356→      p.x = cam + 100;
   357→      p.y = GROUND_Y - PLAYER_H;
   358→    }
   359→    p.vx = 0;
   360→    p.vy = 0;
   361→    p.onGround = false;
   362→    p.invincible = true;
   363→    p.invincibleTimer = INVINCIBLE_TIME;
   364→  }, []);
   365→
   366→  /* 添加飘字 */
   367→  const addFloatText = useCallback((x: number, y: number, text: string, color: string) => {
   368→    floatTextsRef.current.push({ x, y, text, color, life: 60, maxLife: 60 });
   369→  }, []);
   370→
   371→  /* 攻击处理 */
   372→  const playerAttack = useCallback(() => {
   373→    const p = playerRef.current;
   374→    if (p.attacking) return;
   375→    p.attacking = true;
   376→    p.attackTimer = ATTACK_DURATION;
   377→  }, []);
   378→
   379→  /* 跳跃处理 */
   380→  const playerJump = useCallback(() => {
   381→    const p = playerRef.current;
   382→    if (!p.onGround) return;
   383→    p.vy = JUMP_FORCE;
   384→    p.onGround = false;
   385→  }, []);
   386→
   387→  /* 主更新循环 */
   388→  const update = useCallback((dt: number) => {
   389→    const p = playerRef.current;
   390→    const keys = keysRef.current;
   391→    const platforms = platformsRef.current;
   392→    const coins = coinsRef.current;
   393→    const monsters = monstersRef.current;
   394→    const cam = cameraXRef.current;
   395→
   396→    /* 读取输入 */
   397→    let moveLeft = keys.has('ArrowLeft') || keys.has('a') || keys.has('A') || touchLeftRef.current;
   398→    let moveRight = keys.has('ArrowRight') || keys.has('d') || keys.has('D') || touchRightRef.current;
   399→    let wantJump = keys.has('ArrowUp') || keys.has('w') || keys.has('W') || keys.has(' ') || touchJumpRef.current;
   400→    let wantAttack = keys.has('j') || keys.has('J') || touchAttackRef.current;
   401→
   402→    /* 移动 */
   403→    if (moveLeft) {
   404→      p.vx = -MOVE_SPEED;
   405→      p.facing = -1;
   406→    } else if (moveRight) {
   407→      p.vx = MOVE_SPEED;
   408→      p.facing = 1;
   409→    } else {
   410→      p.vx *= 0.7;
   411→      if (Math.abs(p.vx) < 0.3) p.vx = 0;
   412→    }
   413→
   414→    /* 跳跃 */
   415→    if (wantJump && !jumpPressedRef.current) {
   416→      playerJump();
   417→      jumpPressedRef.current = true;
   418→    }
   419→    if (!wantJump) {
   420→      jumpPressedRef.current = false;
   421→    }
   422→
   423→    /* 攻击 */
   424→    if (wantAttack && !p.attacking) {
   425→      playerAttack();
   426→    }
   427→
   428→    /* 攻击计时器 */
   429→    if (p.attacking) {
   430→      p.attackTimer--;
   431→      if (p.attackTimer <= 0) {
   432→        p.attacking = false;
   433→      }
   434→    }
   435→
   436→    /* 无敌计时 */
   437→    if (p.invincible) {
   438→      p.invincibleTimer -= dt;
   439→      if (p.invincibleTimer <= 0) {
   440→        p.invincible = false;
   441→      }
   442→    }
   443→
   444→    /* 动画帧 */
   445→    p.animTimer += dt;
   446→    if (p.animTimer > 120) {
   447→      p.animTimer = 0;
   448→      p.animFrame = (p.animFrame + 1) % 4;
   449→    }
   450→
   451→    /* 物理：重力 */
   452→    p.vy += GRAVITY;
   453→    if (p.vy > MAX_FALL) p.vy = MAX_FALL;
   454→
   455→    /* 水平移动 */
   456→    p.x += p.vx;
   457→
   458→    /* 水平碰撞检测 */
   459→    for (const pl of platforms) {
   460→      if (pl.x + pl.w <= cam - 50 || pl.x >= cam + CANVAS_W + 50) continue;
   461→      if (
   462→        p.x + PLAYER_W > pl.x &&
   463→        p.x < pl.x + pl.w &&
   464→        p.y + PLAYER_H > pl.y + 4 &&
   465→        p.y < pl.y + pl.h
   466→      ) {
   467→        if (p.vx > 0) {
   468→          p.x = pl.x - PLAYER_W;
   469→        } else if (p.vx < 0) {
   470→          p.x = pl.x + pl.w;
   471→        }
   472→        p.vx = 0;
   473→      }
   474→    }
   475→
   476→    /* 垂直移动 */
   477→    p.y += p.vy;
   478→    p.onGround = false;
   479→
   480→    /* 垂直碰撞检测 */
   481→    for (const pl of platforms) {
   482→      if (pl.x + pl.w <= cam - 50 || pl.x >= cam + CANVAS_W + 50) continue;
   483→      if (
   484→        p.x + PLAYER_W > pl.x + 2 &&
   485→        p.x < pl.x + pl.w - 2 &&
   486→        p.y + PLAYER_H > pl.y &&
   487→        p.y + PLAYER_H < pl.y + pl.h + 10 &&
   488→        p.vy >= 0
   489→      ) {
   490→        p.y = pl.y - PLAYER_H;
   491→        p.vy = 0;
   492→        p.onGround = true;
   493→      }
   494→    }
   495→
   496→    /* 相机跟随（玩家靠近右边界时相机移动） */
   497→    const screenX = p.x - cam;
   498→    if (screenX > CANVAS_W * 0.65) {
   499→      cameraXRef.current = p.x - CANVAS_W * 0.65;
   500→    }
   501→    /* 相机不往左退 */
   502→    if (cameraXRef.current < 0) cameraXRef.current = 0;
   503→
   504→    /* 地图自动缓慢滚动（跑酷感） */
   505→    if (cameraXRef.current < p.x - 50) {
   506→      cameraXRef.current += SCROLL_SPEED * 0.5;
   507→    }
   508→
   509→    /* 距离计算 */
   510→    distanceRef.current = Math.floor(p.x / 100);
   511→
   512→    /* 程序化生成更多地图 */
   513→    const genThreshold = cameraXRef.current + CANVAS_W + CHUNK_W;
   514→    if (generatedUpToRef.current < genThreshold) {
   515→      generateChunks(generatedUpToRef.current, genThreshold + CHUNK_W);
   516→    }
   517→
   518→    /* 清理远离的物体 */
   519→    const cleanX = cam - 300;
   520→    platformsRef.current = platforms.filter(pl => pl.x + pl.w > cleanX);
   521→    coinsRef.current = coins.filter(c => c.x > cleanX);
   522→    monstersRef.current = monsters.filter(m => m.x + m.w > cleanX);
   523→
   524→    /* 收集金币 */
   525→    const pcx = p.x + PLAYER_W / 2;
   526→    const pcy = p.y + PLAYER_H / 2;
   527→    for (const c of coins) {
   528→      if (c.collected) continue;
   529→      const dx = pcx - c.x;
   530→      const dy = pcy - c.y;
   531→      if (dx * dx + dy * dy < 500) {
   532→        c.collected = true;
   533→        scoreRef.current += 10;
   534→        setScore(scoreRef.current);
   535→        addFloatText(c.x, c.y, '+10', '#fbbf24');
   536→      }
   537→    }
   538→
   539→    /* 怪物AI */
   540→    for (const m of monsters) {
   541→      if (!m.alive) continue;
   542→      m.animPhase += 0.08;
   543→
   544→      /* 找到怪物脚下的平台 */
   545→      m.onPlatform = null;
   546→      for (const pl of platforms) {
   547→        if (
   548→          m.x + m.w > pl.x &&
   549→          m.x < pl.x + pl.w &&
   550→          Math.abs((m.y + m.h) - pl.y) < 5
   551→        ) {
   552→          m.onPlatform = pl;
   553→          break;
   554→        }
   555→      }
   556→
   557→      /* 移动 */
   558→      m.x += m.vx;
   559→
   560→      /* 平台边缘检测（巡逻转向） */
   561→      if (m.onPlatform) {
   562→        const pl = m.onPlatform;
   563→        const margin = 5;
   564→        if (m.x <= pl.x + margin) {
   565→          m.x = pl.x + margin;
   566→          m.vx = Math.abs(m.vx);
   567→        } else if (m.x + m.w >= pl.x + pl.w - margin) {
   568→          m.x = pl.x + pl.w - m.w - margin;
   569→          m.vx = -Math.abs(m.vx);
   570→        }
   571→      }
   572→
   573→      /* 攻击判定：玩家攻击范围内 */
   574→      if (p.attacking && p.attackTimer > ATTACK_DURATION - 10) {
   575→        const atkX = p.x + (p.facing > 0 ? PLAYER_W : 0);
   576→        const atkY = p.y + PLAYER_H / 2;
   577→        const mdx = (m.x + m.w / 2) - atkX;
   578→        const mdy = (m.y + m.h / 2) - atkY;
   579→        const dist = Math.sqrt(mdx * mdx + mdy * mdy);
   580→        if (dist < ATTACK_RANGE && p.facing * mdx >= -10) {
   581→          m.hp--;
   582→          if (m.hp <= 0) {
   583→            m.alive = false;
   584→            const pts = m.kind === 'slime' ? 20 : 50;
   585→            scoreRef.current += pts;
   586→            setScore(scoreRef.current);
   587→            addFloatText(m.x + m.w / 2, m.y, `+${pts}`, m.kind === 'slime' ? '#4ade80' : '#f87171');
   588→          }
   589→        }
   590→      }
   591→
   592→      /* 踩怪判定：玩家从上方落下 */
   593→      if (m.alive && !p.invincible) {
   594→        const pBottom = p.y + PLAYER_H;
   595→        const mTop = m.y;
   596→        const prevBottom = pBottom - p.vy;
   597→        if (
   598→          p.vy > 0 &&
   599→          prevBottom <= mTop + 5 &&
   600→          pBottom >= mTop &&
   601→          p.x + PLAYER_W > m.x + 3 &&
   602→          p.x < m.x + m.w - 3
   603→        ) {
   604→          m.hp--;
   605→          p.vy = JUMP_FORCE * 0.6;
   606→          if (m.hp <= 0) {
   607→            m.alive = false;
   608→            const pts = m.kind === 'slime' ? 20 : 50;
   609→            scoreRef.current += pts;
   610→            setScore(scoreRef.current);
   611→            addFloatText(m.x + m.w / 2, m.y, `+${pts}`, m.kind === 'slime' ? '#4ade80' : '#f87171');
   612→          }
   613→          continue;
   614→        }
   615→      }
   616→
   617→      /* 碰撞伤害：怪物碰到玩家 */
   618→      if (m.alive && !p.invincible) {
   619→        if (
   620→          p.x + PLAYER_W > m.x + 2 &&
   621→          p.x < m.x + m.w - 2 &&
   622→          p.y + PLAYER_H > m.y + 2 &&
   623→          p.y < m.y + m.h - 2
   624→        ) {
   625→          hurtPlayer();
   626→        }
   627→      }
   628→    }
   629→
   630→    /* 坠落深渊 */
   631→    if (p.y > CANVAS_H + 50) {
   632→      livesRef.current--;
   633→      setLives(livesRef.current);
   634→      if (livesRef.current <= 0) {
   635→        gameOver();
   636→      } else {
   637→        respawnPlayer();
   638→      }
   639→    }
   640→
   641→    /* 防止玩家跑出左边界 */
   642→    if (p.x < cameraXRef.current - 20) {
   643→      p.x = cameraXRef.current - 20;
   644→    }
   645→
   646→    /* 更新飘字 */
   647→    floatTextsRef.current = floatTextsRef.current.filter(ft => {
   648→      ft.life--;
   649→      ft.y -= 1.2;
   650→      return ft.life > 0;
   651→    });
   652→  }, [playerJump, playerAttack, hurtPlayer, respawnPlayer, addFloatText, gameOver]);
