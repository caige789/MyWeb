    92→    92→/** 在[fromX, toX)区间程序化生成地图 */
    93→    93→function generateChunk(
    94→    94→  platforms: Platform[], coins: Coin[], monsters: Monster[],
    95→    95→  fromX: number, toX: number
    96→    96→): number {
    97→    97→  let x = fromX;
    98→    98→  while (x < toX) {
    99→    99→    const r = Math.random();
   100→   100→    if (r < 0.35) {
   101→   101→      /* 地面段 */
   102→   102→      const w = 120 + Math.random() * 200;
   103→   103→      platforms.push({ x, y: GROUND_Y, w, h: 60 });
   104→   104→      if (Math.random() < 0.5) {
   105→   105→        monsters.push(makeMonster(Math.random() < 0.6 ? 'slime' : 'skeleton', x + 30, GROUND_Y));
   106→   106→      }
   107→   107→      const nc = Math.floor(Math.random() * 4) + 1;
   108→   108→      for (let i = 0; i < nc; i++) {
   109→   109→        coins.push({ x: x + 20 + i * 30, y: GROUND_Y - 40, collected: false, phase: Math.random() * 6.28 });
   110→   110→      }
   111→   111→      x += w + 30 + Math.random() * 50;
   112→   112→    } else if (r < 0.65) {
   113→   113→      /* 浮空平台 */
   114→   114→      const w = 80 + Math.random() * 100;
   115→   115→      const py = GROUND_Y - 80 - Math.random() * 120;
   116→   116→      platforms.push({ x, y: py, w, h: 20 });
   117→   117→      coins.push({ x: x + w / 2, y: py - 35, collected: false, phase: Math.random() * 6.28 });
   118→   118→      if (Math.random() < 0.3) monsters.push(makeMonster('slime', x + 10, py));
   119→   119→      x += w + 50 + Math.random() * 80;
   120→   120→    } else if (r < 0.8) {
   121→   121→      /* 阶梯 */
   122→   122→      const steps = 2 + Math.floor(Math.random() * 2);
   123→   123→      let sx = x;
   124→   124→      for (let s = 0; s < steps; s++) {
   125→   125→        const sw = 60 + Math.random() * 40;
   126→   126→        const py = GROUND_Y - 50 - s * 55;
   127→   127→        platforms.push({ x: sx, y: py, w: sw, h: 18 });
   128→   128→        coins.push({ x: sx + sw / 2, y: py - 30, collected: false, phase: Math.random() * 6.28 });
   129→   129→        sx += sw + 20;
   130→   130→      }
   131→   131→      x = sx + 40;
   132→   132→    } else {
   133→   133→      /* 深渊+地面 */
   134→   134→      x += 60 + Math.random() * 60;
   135→   135→      const w = 150 + Math.random() * 200;
   136→   136→      platforms.push({ x, y: GROUND_Y, w, h: 60 });
   137→   137→      if (Math.random() < 0.6) {
   138→   138→        monsters.push(makeMonster(Math.random() < 0.6 ? 'slime' : 'skeleton', x + 20, GROUND_Y));
   139→   139→      }
   140→   140→      const nc = Math.floor(Math.random() * 3) + 1;
   141→   141→      for (let i = 0; i < nc; i++) {
   142→   142→        coins.push({ x: x + 30 + i * 35, y: GROUND_Y - 40, collected: false, phase: Math.random() * 6.28 });
   143→   143→      }
   144→   144→      x += w + 30 + Math.random() * 50;
   145→   145→    }
   146→   146→  }
   147→   147→  return x;
   148→   148→}

   297→   297→function drawPlayer(ctx: CanvasRenderingContext2D, p: PlayerState, cam: number) {
   298→   298→  if (p.invincible && Math.sin(performance.now() / 60) > 0) return;
   299→   299→  const x = p.x - cam;
   300→   300→  const y = p.y;
   301→   301→  const running = Math.abs(p.vx) > 0.5;
   302→   302→  const air = !p.onGround;
   303→   303→  ctx.save();
   304→   304→  ctx.translate(x + PLAYER_W / 2, y + PLAYER_H / 2);
   305→   305→  ctx.scale(p.facing, 1);
   306→   306→  ctx.translate(-PLAYER_W / 2, -PLAYER_H / 2);
   307→   307→  /* 腿 */
   308→   308→  const ls = running && !air ? Math.sin(p.frame * Math.PI / 2) * 6 : 0;
   309→   309→  ctx.fillStyle = '#1e40af';
   310→   310→  ctx.fillRect(6, 30, 6, 10 + (air ? -3 : ls));
   311→   311→  ctx.fillRect(16, 30, 6, 10 + (air ? 3 : -ls));
   312→   312→  ctx.fillStyle = '#78350f';
   313→   313→  ctx.fillRect(4, 38 + (air ? -3 : ls), 9, 3);
   314→   314→  ctx.fillRect(15, 38 + (air ? 3 : -ls), 9, 3);
   315→   315→  /* 身体 */
   316→   316→  ctx.fillStyle = '#3b82f6';
   317→   317→  ctx.fillRect(5, 16, 18, 16);
   318→   318→  ctx.fillStyle = '#92400e';
   319→   319→  ctx.fillRect(5, 28, 18, 3);
   320→   320→  ctx.fillStyle = '#fbbf24';
   321→   321→  ctx.fillRect(12, 27, 5, 5);
   322→   322→  /* 手臂 */
   323→   323→  const as = running && !air ? Math.sin(p.frame * Math.PI / 2) * 8 : 0;
   324→   324→  if (p.attacking) {
   325→   325→    ctx.fillStyle = '#2563eb';
   326→   326→    ctx.fillRect(18, 18, 16, 5);

   329→   329→    ctx.fillStyle = '#2563eb';
   330→   330→    ctx.fillRect(18, 18, 5, 12 + as);
   331→   331→  }
   332→   332→  ctx.fillStyle = '#2563eb';
   333→   333→  ctx.fillRect(0, 18, 5, 12 - as);
   334→   334→  /* 头 */
   335→   335→  ctx.fillStyle = '#fbbf24';
   336→   336→  ctx.fillRect(5, 2, 18, 15);
   337→   337→  ctx.fillStyle = '#78350f';

   339→   339→  ctx.fillRect(3, 2, 4, 8);
   340→   340→  ctx.fillStyle = '#1a1a2e';
   341→   341→  ctx.fillRect(16, 6, 3, 3);
   342→   342→  ctx.fillStyle = '#dc2626';
   343→   343→  ctx.fillRect(15, 12, 5, 1);
   344→   344→  ctx.restore();
   345→   345→}

   407→   407→          if (found) setHighScore(found.score);
   408→   408→        }
   409→   409→      })
   410→   410→      .catch(() => { /* 静默处理 */ });
   411→   411→  }, []);
   412→   412→
   413→   413→  /* 提交分数 */
   414→   414→  const submitScore = (finalScore: number) => {
   415→   415→    fetch('/api/games/scores', {
   416→   416→      method: 'POST',
   417→   417→      headers: { 'Content-Type': 'application/json' },
   418→   418→      body: JSON.stringify({ game: GAME_NAME, score: finalScore }),
   419→   419→    })
   420→   420→      .then(r => r.json())
   421→   421→      .then(res => {
   422→   422→        if (res.code === 0 && res.data && finalScore >= res.data.score) {
   423→   423→          setIsNewRecord(true);
   424→   424→          setHighScore(finalScore);
   425→   425→        }
   426→   426→      })
   427→   427→      .catch(() => { /* 静默处理 */ });
   428→   428→  };
   429→   429→
   430→   430→  /* 初始化游戏数据 */
   431→   431→  const initGameData = () => {
   432→   432→    const g = gRef.current;
   433→   433→    g.player = makePlayer();
   434→   434→    g.platforms = [];
   435→   435→    g.coins = [];
   436→   436→    g.monsters = [];
   437→   437→    g.floats = [];
   438→   438→    g.score = 0;
   439→   439→    g.lives = 3;
   440→   440→    g.cam = 0;
   441→   441→    g.genUpTo = 0;
   442→   442→    g.distance = 0;
   443→   443→    g.lastTime = 0;
   444→   444→    g.status = 'playing';
   445→   445→    /* 起始地面 */
   446→   446→    g.platforms.push({ x: 0, y: GROUND_Y, w: 400, h: 60 });
   447→   447→    for (let i = 0; i < 5; i++) {
   448→   448→      g.coins.push({ x: 200 + i * 40, y: GROUND_Y - 40, collected: false, phase: Math.random() * 6.28 });
   449→   449→    }
   450→   450→    g.genUpTo = generateChunk(g.platforms, g.coins, g.monsters, 400, 400 + CHUNK_W * 3);
   451→   451→    setScore(0);
   452→   452→    setLives(3);
   453→   453→    setIsNewRecord(false);
   454→   454→    setGameStatus('playing');
   455→   455→  };
   456→   456→
   457→   457→  /* 游戏结束 */
   458→   458→  const endGame = (g: GameCore) => {
   459→   459→    g.status = 'over';
   460→   460→    setGameStatus('over');
   461→   461→    submitScore(g.score);
   462→   462→  };
   463→   463→
   464→   464→  /* 玩家受伤 */
   465→   465→  const hurtPlayer = (g: GameCore) => {
   466→   466→    if (g.player.invincible) return;
   467→   467→    g.lives--;
   468→   468→    setLives(g.lives);
   469→   469→    if (g.lives <= 0) {
   470→   470→      endGame(g);
   471→   471→    } else {
   472→   472→      g.player.invincible = true;
   473→   473→      g.player.invTimer = INVINCIBLE_TIME;
   474→   474→    }
   475→   475→  };
   476→   476→
   477→   477→  /* 重生 */
   478→   478→  const respawn = (g: GameCore) => {
   479→   479→    let best: Platform | null = null;
   480→   480→    let bd = Infinity;
   481→   481→    for (const pl of g.platforms) {
   482→   482→      if (pl.x + pl.w > g.cam && pl.x < g.cam + CANVAS_W) {
   483→   483→        const d = Math.abs(pl.x + pl.w / 2 - (g.cam + CANVAS_W / 3));
   484→   484→        if (d < bd) { bd = d; best = pl; }
   485→   485→      }
   486→   486→    }
   487→   487→    if (best) {
   488→   488→      g.player.x = best.x + best.w / 2 - PLAYER_W / 2;
   489→   489→      g.player.y = best.y - PLAYER_H;
   490→   490→    } else {
   491→   491→      g.player.x = g.cam + 100;
   492→   492→      g.player.y = GROUND_Y - PLAYER_H;
   493→   493→    }
   494→   494→    g.player.vx = 0;
   495→   495→    g.player.vy = 0;
   496→   496→    g.player.onGround = false;
   497→   497→    g.player.invincible = true;
   498→   498→    g.player.invTimer = INVINCIBLE_TIME;
   499→   499→  };
   500→   500→
   501→   501→  /* 核心更新逻辑（纯函数，接收g作为参数） */
   502→   502→  const tick = (g: GameCore, dt: number) => {
   503→   503→    const p = g.player;
   504→   504→    const k = g.keys;
   505→   505→
   506→   506→    /* 读取输入 */
   507→   507→    const ml = k.has('ArrowLeft') || k.has('a') || k.has('A') || g.touchL;
   508→   508→    const mr = k.has('ArrowRight') || k.has('d') || k.has('D') || g.touchR;
   509→   509→    const wj = k.has('ArrowUp') || k.has('w') || k.has('W') || k.has(' ') || g.touchJ;
   510→   510→    const wa = k.has('j') || k.has('J') || g.touchA;
   511→   511→
   512→   512→    /* 移动 */
   513→   513→    if (ml) { p.vx = -MOVE_SPEED; p.facing = -1; }
   514→   514→    else if (mr) { p.vx = MOVE_SPEED; p.facing = 1; }
   515→   515→    else { p.vx *= 0.7; if (Math.abs(p.vx) < 0.3) p.vx = 0; }
   516→   516→
   517→   517→    /* 跳跃 */
   518→   518→    if (wj && !g.jumpHeld && p.onGround) {
   519→   519→      p.vy = JUMP_FORCE;
   520→   520→      p.onGround = false;
   521→   521→      g.jumpHeld = true;
   522→   522→    }
   523→   523→    if (!wj) g.jumpHeld = false;
   524→   524→
   525→   525→    /* 攻击 */
   526→   526→    if (wa && !p.attacking) { p.attacking = true; p.atkTimer = ATTACK_DURATION; }
   527→   527→    if (p.attacking) { p.atkTimer--; if (p.atkTimer <= 0) p.attacking = false; }
   528→   528→
   529→   529→    /* 无敌计时 */
   530→   530→    if (p.invincible) { p.invTimer -= dt; if (p.invTimer <= 0) p.invincible = false; }
   531→   531→
   532→   532→    /* 动画 */
   533→   533→    p.frameTimer += dt;
   534→   534→    if (p.frameTimer > 120) { p.frameTimer = 0; p.frame = (p.frame + 1) % 4; }
   535→   535→
   536→   536→    /* 重力 */
   537→   537→    p.vy += GRAVITY;
   538→   538→    if (p.vy > MAX_FALL) p.vy = MAX_FALL;
   539→   539→
   540→   540→    /* 水平移动+碰撞 */
   541→   541→    p.x += p.vx;
   542→   542→    for (const pl of g.platforms) {
   543→   543→      if (pl.x + pl.w <= g.cam - 50 || pl.x >= g.cam + CANVAS_W + 50) continue;
   544→   544→      if (p.x + PLAYER_W > pl.x && p.x < pl.x + pl.w && p.y + PLAYER_H > pl.y + 4 && p.y < pl.y + pl.h) {
   545→   545→        if (p.vx > 0) p.x = pl.x - PLAYER_W;
   546→   546→        else if (p.vx < 0) p.x = pl.x + pl.w;
   547→   547→        p.vx = 0;
   548→   548→      }
   549→   549→    }
   550→   550→
   551→   551→    /* 垂直移动+碰撞 */
   552→   552→    p.y += p.vy;
   553→   553→    p.onGround = false;
   554→   554→    for (const pl of g.platforms) {
   555→   555→      if (pl.x + pl.w <= g.cam - 50 || pl.x >= g.cam + CANVAS_W + 50) continue;
   556→   556→      if (p.x + PLAYER_W > pl.x + 2 && p.x < pl.x + pl.w - 2 && p.y + PLAYER_H > pl.y && p.y + PLAYER_H < pl.y + pl.h + 10 && p.vy >= 0) {
   557→   557→        p.y = pl.y - PLAYER_H;
   558→   558→        p.vy = 0;
   559→   559→        p.onGround = true;
   560→   560→      }
   561→   561→    }
   562→   562→
   563→   563→    /* 相机跟随 */
   564→   564→    if (p.x - g.cam > CANVAS_W * 0.65) g.cam = p.x - CANVAS_W * 0.65;
   565→   565→    if (g.cam < 0) g.cam = 0;
   566→   566→    if (g.cam < p.x - 50) g.cam += SCROLL_SPEED * 0.5;
   567→   567→    g.distance = Math.floor(p.x / 100);
   568→   568→
   569→   569→    /* 程序化生成 */
   570→   570→    const threshold = g.cam + CANVAS_W + CHUNK_W;
   571→   571→    if (g.genUpTo < threshold) {
   572→   572→      g.genUpTo = generateChunk(g.platforms, g.coins, g.monsters, g.genUpTo, threshold + CHUNK_W);
   573→   573→    }
   574→   574→
   575→   575→    /* 清理远处物体 */
   576→   576→    const cx = g.cam - 300;
   577→   577→    g.platforms = g.platforms.filter(pl => pl.x + pl.w > cx);
   578→   578→    g.coins = g.coins.filter(c => c.x > cx);
   579→   579→    g.monsters = g.monsters.filter(m => m.x + m.w > cx);
   580→   580→
   581→   581→    /* 收集金币 */
   582→   582→    const pcx = p.x + PLAYER_W / 2;
   583→   583→    const pcy = p.y + PLAYER_H / 2;
   584→   584→    for (const c of g.coins) {
   585→   585→      if (c.collected) continue;
   586→   586→      const dx = pcx - c.x;
   587→   587→      const dy = pcy - c.y;
   588→   588→      if (dx * dx + dy * dy < 500) {
   589→   589→        c.collected = true;
   590→   590→        g.score += 10;
   591→   591→        g.floats.push({ x: c.x, y: c.y, text: '+10', color: '#fbbf24', life: 60 });
   592→   592→      }
   593→   593→    }
   594→   594→
   595→   595→    /* 怪物AI */
   596→   596→    for (const m of g.monsters) {
   597→   597→      if (!m.alive) continue;
   598→   598→      m.phase += 0.08;
   599→   599→      /* 找脚下平台 */
   600→   600→      let onPlat: Platform | null = null;
   601→   601→      for (const pl of g.platforms) {
   602→   602→        if (m.x + m.w > pl.x && m.x < pl.x + pl.w && Math.abs(m.y + m.h - pl.y) < 5) { onPlat = pl; break; }
   603→   603→      }
   604→   604→      m.x += m.vx;
   605→   605→      /* 边缘转向 */
   606→   606→      if (onPlat) {
   607→   607→        if (m.x <= onPlat.x + 5) { m.x = onPlat.x + 5; m.vx = Math.abs(m.vx); }
   608→   608→        else if (m.x + m.w >= onPlat.x + onPlat.w - 5) { m.x = onPlat.x + onPlat.w - m.w - 5; m.vx = -Math.abs(m.vx); }
   609→   609→      }
   610→   610→      /* 攻击判定 */
   611→   611→      if (p.attacking && p.atkTimer > ATTACK_DURATION - 10) {
   612→   612→        const ax = p.x + (p.facing > 0 ? PLAYER_W : 0);
   613→   613→        const ay = p.y + PLAYER_H / 2;
   614→   614→        const mdx = (m.x + m.w / 2) - ax;
   615→   615→        const mdy = (m.y + m.h / 2) - ay;
   616→   616→        if (Math.sqrt(mdx * mdx + mdy * mdy) < ATTACK_RANGE && p.facing * mdx >= -10) {
   617→   617→          m.hp--;
   618→   618→          if (m.hp <= 0) {
   619→   619→            m.alive = false;
   620→   620→            const pts = m.kind === 'slime' ? 20 : 50;
   621→   621→            g.score += pts;
   622→   622→            g.floats.push({ x: m.x + m.w / 2, y: m.y, text: `+${pts}`, color: m.kind === 'slime' ? '#4ade80' : '#f87171', life: 60 });
   623→   623→          }
   624→   624→        }
   625→   625→      }
   626→   626→      /* 踩怪判定 */
   627→   627→      if (m.alive && !p.invincible) {
   628→   628→        const pb = p.y + PLAYER_H;
   629→   629→        const prevB = pb - p.vy;
   630→   630→        if (p.vy > 0 && prevB <= m.y + 5 && pb >= m.y && p.x + PLAYER_W > m.x + 3 && p.x < m.x + m.w - 3) {
   631→   631→          m.hp--;
   632→   632→          p.vy = JUMP_FORCE * 0.6;
   633→   633→          if (m.hp <= 0) {
   634→   634→            m.alive = false;
   635→   635→            const pts = m.kind === 'slime' ? 20 : 50;
   636→   636→            g.score += pts;
   637→   637→            g.floats.push({ x: m.x + m.w / 2, y: m.y, text: `+${pts}`, color: m.kind === 'slime' ? '#4ade80' : '#f87171', life: 60 });
   638→   638→          }
   639→   639→          continue;
   640→   640→        }
   641→   641→      }
   642→   642→      /* 碰撞伤害 */
   643→   643→      if (m.alive && !p.invincible) {
   644→   644→        if (p.x + PLAYER_W > m.x + 2 && p.x < m.x + m.w - 2 && p.y + PLAYER_H > m.y + 2 && p.y < m.y + m.h - 2) {
   645→   645→          hurtPlayer(g);
   646→   646→        }
   647→   647→      }
   648→   648→    }
   649→   649→
   650→   650→    /* 坠落 */
   651→   651→    if (p.y > CANVAS_H + 50) {
   652→   652→      g.lives--;
   653→   653→      setLives(g.lives);
   654→   654→      if (g.lives <= 0) { endGame(g); }
   655→   655→      else { respawn(g); }
   656→   656→    }
   657→   657→
   658→   658→    /* 防止跑出左边界 */
   659→   659→    if (p.x < g.cam - 20) p.x = g.cam - 20;
   660→   660→
   661→   661→    /* 更新飘字 */
   662→   662→    g.floats = g.floats.filter(ft => { ft.life--; ft.y -= 1.2; return ft.life > 0; });
   663→   663→  };
   664→   664→
   665→   665→  /* 绘制完整画面 */
   666→   666→  const render = (g: GameCore) => {
   667→   667→    const canvas = canvasRef.current;
   668→   668→    if (!canvas) return;
   669→   669→    const ctx = canvas.getContext('2d');
   670→   670→    if (!ctx) return;
   671→   671→    const s = g.scale;
   672→   672→    ctx.setTransform(s, 0, 0, s, 0, 0);
   673→   673→    /* 天空 */
   674→   674→    const sky = ctx.createLinearGradient(0, 0, 0, CANVAS_H);
   675→   675→    sky.addColorStop(0, '#87ceeb');
   676→   676→    sky.addColorStop(0.6, '#b0e0f6');
   677→   677→    sky.addColorStop(1, '#d4f0fc');
   678→   678→    ctx.fillStyle = sky;
   679→   679→    ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   680→   680→    drawClouds(ctx, g.cam);
   681→   681→    /* 平台 */
   682→   682→    for (const pl of g.platforms) {
   683→   683→      if (pl.x + pl.w < g.cam - 10 || pl.x > g.cam + CANVAS_W + 10) continue;
   684→   684→      drawPlatform(ctx, pl, g.cam);
   685→   685→    }
   686→   686→    /* 金币 */
   687→   687→    const t = performance.now() / 1000;
   688→   688→    for (const c of g.coins) {
   689→   689→      if (c.collected || c.x < g.cam - 20 || c.x > g.cam + CANVAS_W + 20) continue;
   690→   690→      drawCoin(ctx, c.x - g.cam, c.y, t + c.phase);
   691→   691→    }
   692→   692→    /* 怪物 */
   693→   693→    for (const m of g.monsters) {
   694→   694→      if (!m.alive || m.x + m.w < g.cam - 20 || m.x > g.cam + CANVAS_W + 20) continue;
   695→   695→      const mx = m.x - g.cam;
   696→   696→      if (m.kind === 'slime') drawSlime(ctx, mx, m.y, m.w, m.h, m.phase, m.vx);
   697→   697→      else drawSkeleton(ctx, mx, m.y, m.w, m.h, m.phase, m.hp);
   698→   698→    }
   699→   699→    /* 玩家 */
   700→   700→    drawPlayer(ctx, g.player, g.cam);
   701→   701→    /* 飘字 */
   702→   702→    for (const ft of g.floats) {
   703→   703→      ctx.globalAlpha = ft.life / 60;
   704→   704→      ctx.fillStyle = ft.color;
   705→   705→      ctx.font = 'bold 16px sans-serif';
   706→   706→      ctx.textAlign = 'center';
   707→   707→      ctx.fillText(ft.text, ft.x - g.cam, ft.y);
   708→   708→      ctx.globalAlpha = 1;
   709→   709→    }
   710→   710→  };
   711→   711→
   712→   712→  /* 游戏循环 */
   713→   713→  const loop = (timestamp: number) => {
   714→   714→    const g = gRef.current;
   715→   715→    if (g.status !== 'playing') return;
   716→   716→    const dt = g.lastTime ? timestamp - g.lastTime : 16;
   717→   717→    g.lastTime = timestamp;
   718→   718→    tick(g, dt);
   719→   719→    render(g);
   720→   720→    /* 低频更新React状态 */
   721→   721→    uiUpdateRef.current++;
   722→   722→    if (uiUpdateRef.current % 6 === 0) {
   723→   723→      setScore(g.score);
   724→   724→      setDistance(g.distance);
   725→   725→    }
   726→   726→    rafRef.current = requestAnimationFrame(loop);
   727→   727→  };
   728→   728→
   729→   729→  /* 开始游戏 */
   730→   730→  const startGame = () => {
   731→   731→    if (rafRef.current) cancelAnimationFrame(rafRef.current);
   732→   732→    initGameData();
   733→   733→    rafRef.current = requestAnimationFrame(loop);
   734→   734→  };
   735→   735→
   736→   736→  /* 键盘事件 */
   737→   737→  useEffect(() => {
   738→   738→    const onDown = (e: KeyboardEvent) => {
   739→   739→      gRef.current.keys.add(e.key);
   740→   740→      if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', ' '].includes(e.key)) e.preventDefault();
   741→   741→    };
   742→   742→    const onUp = (e: KeyboardEvent) => { gRef.current.keys.delete(e.key); };
   743→   743→    window.addEventListener('keydown', onDown);
   744→   744→    window.addEventListener('keyup', onUp);
   745→   745→    return () => { window.removeEventListener('keydown', onDown); window.removeEventListener('keyup', onUp); };
   746→   746→  }, []);
   747→   747→
   748→   748→  /* 鼠标攻击 */
   749→   749→  useEffect(() => {
   750→   750→    const canvas = canvasRef.current;
   751→   751→    if (!canvas) return;
   752→   752→    const onClick = () => { if (gRef.current.status === 'playing') { const p = gRef.current.player; if (!p.attacking) { p.attacking = true; p.atkTimer = ATTACK_DURATION; } } };
   753→   753→    canvas.addEventListener('click', onClick);
   754→   754→    return () => canvas.removeEventListener('click', onClick);
   755→   755→  }, []);
   756→   756→
   757→   757→  /* 高DPI适配 */
   758→   758→  useEffect(() => {
   759→   759→    const canvas = canvasRef.current;
   760→   760→    const container = containerRef.current;
   761→   761→    if (!canvas || !container) return;
   762→   762→    const resize = () => {
   763→   763→      const cw = container.clientWidth;
   764→   764→      const dw = Math.min(cw, CANVAS_W);
   765→   765→      const dpr = window.devicePixelRatio || 1;
   766→   766→      const dh = (dw / CANVAS_W) * CANVAS_H;
   767→   767→      canvas.style.width = `${dw}px`;
   768→   768→      canvas.style.height = `${dh}px`;
   769→   769→      canvas.width = dw * dpr;
   770→   770→      canvas.height = dh * dpr;
   771→   771→      gRef.current.scale = (dw * dpr) / CANVAS_W;
   772→   772→    };
   773→   773→    resize();
   774→   774→    window.addEventListener('resize', resize);
   775→   775→    return () => window.removeEventListener('resize', resize);
   776→   776→  }, []);
   777→   777→
   778→   778→  /* 清理 */
   779→   779→  useEffect(() => { return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current); }; }, []);
   780→   780→
   781→   781→  /* 绘制待机画面 */
   782→   782→  useEffect(() => {
   783→   783→    if (gameStatus !== 'idle') return;
   784→   784→    const timer = setTimeout(() => {
   785→   785→      const canvas = canvasRef.current;
   786→   786→      if (!canvas) return;
   787→   787→      const ctx = canvas.getContext('2d');
   788→   788→      if (!ctx) return;
   789→   789→      const s = gRef.current.scale;
   790→   790→      ctx.setTransform(s, 0, 0, s, 0, 0);
   791→   791→      const sky = ctx.createLinearGradient(0, 0, 0, CANVAS_H);
   792→   792→      sky.addColorStop(0, '#87ceeb');
   793→   793→      sky.addColorStop(1, '#d4f0fc');
   794→   794→      ctx.fillStyle = sky;
   795→   795→      ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   796→   796→      ctx.fillStyle = '#4ade80';
