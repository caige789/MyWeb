   180→function getWaveZombies(wave: number, countMul: number): { type: ZombieType; delay: number }[] {
   181→  const zombies: { type: ZombieType; delay: number }[] = [];
   182→  let delay = 0;
   183→  const baseCounts: number[][] = [
   184→    [4, 1, 0, 0, 0, 0, 0],
   185→    [3, 2, 0, 1, 1, 0, 0],
   186→    [2, 2, 1, 1, 1, 1, 0],
   187→    [2, 3, 1, 1, 2, 1, 1],
   188→    [1, 3, 2, 2, 2, 2, 2],
   189→  ];
   190→  const counts = baseCounts[Math.min(wave, baseCounts.length - 1)];
   191→  const types: ZombieType[] = ['normal', 'cone', 'bucket', 'flag', 'newspaper', 'football', 'gargantuar'];
   192→  for (let i = 0; i < types.length; i++) {
   193→    const count = Math.round(counts[i] * countMul);
   194→    for (let j = 0; j < count; j++) {
   195→      zombies.push({ type: types[i], delay });
   196→      delay += 2000 + Math.random() * 3000;
   197→    }
   198→  }
   199→  for (let i = zombies.length - 1; i > 0; i--) {
   200→    const j = Math.floor(Math.random() * (i + 1));
   201→    [zombies[i], zombies[j]] = [zombies[j], zombies[i]];
   202→  }
   203→  return zombies;
   204→}

   510→  const handleCanvasMove = useCallback((clientX: number, clientY: number) => {

   525→  useEffect(() => {
   526→    if (phase !== 'playing') return;
   527→    const canvas = canvasRef.current;
   528→    if (!canvas) return;
   529→

   531→      e.preventDefault();
   532→      handleTap(e.clientX, e.clientY);
   533→    };
   534→    const onMouseMove = (e: MouseEvent) => {
   535→      handleCanvasMove(e.clientX, e.clientY);
   536→    };
   537→    const onMouseLeave = () => { handleCanvasLeave(); };

   539→    const onTouchStart = (e: TouchEvent) => {
   540→      e.preventDefault();
   541→      const t = e.touches[0];
   542→      if (t) {
   543→        touchStartRef.current = { x: t.clientX, y: t.clientY };
   544→        hasMovedRef.current = false;
   545→        handleCanvasMove(t.clientX, t.clientY);
   546→      }
   547→    };

   549→      e.preventDefault();
   550→      const t = e.touches[0];
   551→      if (t) {
   552→        const start = touchStartRef.current;
   553→        if (start) {
   554→          const dx = t.clientX - start.x;
   555→          const dy = t.clientY - start.y;
   556→          if (dx * dx + dy * dy > 400) hasMovedRef.current = true;
   557→        }
   558→        handleCanvasMove(t.clientX, t.clientY);
   559→      }
   560→    };
   561→    const onTouchEnd = (e: TouchEvent) => {
   562→      e.preventDefault();
   563→      if (!hasMovedRef.current && touchStartRef.current) {
   564→        handleTap(touchStartRef.current.x, touchStartRef.current.y);
   565→      }
   566→      cursorRef.current = null;
   567→      touchStartRef.current = null;

   571→    canvas.addEventListener('mousemove', onMouseMove);
   572→    canvas.addEventListener('mouseleave', onMouseLeave);
   573→    canvas.addEventListener('touchstart', onTouchStart, { passive: false });
   574→    canvas.addEventListener('touchmove', onTouchMove, { passive: false });
   575→    canvas.addEventListener('touchend', onTouchEnd, { passive: false });
   576→    return () => {
   577→      canvas.removeEventListener('click', onClick);
   578→      canvas.removeEventListener('mousemove', onMouseMove);
   579→      canvas.removeEventListener('mouseleave', onMouseLeave);
   580→      canvas.removeEventListener('touchstart', onTouchStart);
   581→      canvas.removeEventListener('touchmove', onTouchMove);
   582→      canvas.removeEventListener('touchend', onTouchEnd);
   583→    };

   586→  // Main game loop
   587→  useEffect(() => {
   588→    const loop = (timestamp: number) => {
   589→      const canvas = canvasRef.current;
   590→      if (!canvas) return;

   597→
   598→      if (lastTimeRef.current === 0) lastTimeRef.current = timestamp;
   599→      const deltaMs = Math.min(timestamp - lastTimeRef.current, 50);
   600→      lastTimeRef.current = timestamp;
   601→      const dt = deltaMs / 16.667;
   602→      const cfg = DIFFICULTY_CONFIG[diffRef.current];
   603→
   604→      // === GAME LOGIC (unchanged from original) ===
   605→      if (!overRef.current && !wonRef.current) {
   606→        waveElapsedRef.current += deltaMs;
   607→
   608→        // Sky sun spawning
   609→        sunDropTimerRef.current += deltaMs;
   610→        if (sunDropTimerRef.current >= SUN_DROP_INTERVAL) {
   611→          sunDropTimerRef.current -= SUN_DROP_INTERVAL;

   613→            id: sunIdRef.current++,
   614→            x: 40 + Math.random() * (LOGIC_W - 80),
   615→            y: GRID_Y - 20,
   616→            targetY: GRID_Y + 20 + Math.random() * (GRID_H - 40),
   617→            vy: 0, collected: false, alpha: 1, bouncePhase: 0,
   618→            lifeTimer: SUN_LIFE, scale: 1,
   619→          });
   620→        }
   621→
   622→        // Zombie wave spawning
   623→        const wz = waveZombiesRef.current;
   624→        const wsi = waveSpawnIdxRef.current;
   625→        if (wsi < wz.length && waveElapsedRef.current >= wz[wsi].delay) {
   626→          const zType = wz[wsi].type;
   627→          const def = ZOMBIE_DEFS[zType];

   629→            id: zombieIdRef.current++,
   630→            type: zType, name: def.name, emoji: def.emoji,
   631→            row: Math.floor(Math.random() * ROWS),

   633→            hp: def.hp, maxHp: def.hp,
   634→            baseSpeed: def.speed, speed: def.speed * cfg.zombieSpeedMul,
   635→            score: def.score, hitFlash: 0, eating: false,
   636→            walkPhase: Math.random() * Math.PI * 2,
   637→            slowTimer: 0, paperBroken: false, smashed: false,
   638→          });
   639→          waveSpawnIdxRef.current++;
   640→        }
   641→
   642→        // Wave progression
   643→        if (
   644→          waveElapsedRef.current >= WAVE_DURATION * 1000 &&
   645→          waveSpawnIdxRef.current >= wz.length &&
   646→          zombiesRef.current.length === 0
   647→        ) {
   648→          if (waveRef.current >= TOTAL_WAVES) {

   650→            submitScoreRef.current(scoreRef.current);
   651→            setDisplayScore(scoreRef.current);
   652→            setPhase('won');

   654→            waveRef.current++;
   655→            waveElapsedRef.current = 0;
   656→            waveSpawnIdxRef.current = 0;
   657→            waveZombiesRef.current = getWaveZombies(waveRef.current - 1, cfg.zombieCountMul);
   658→            setDisplayWave(waveRef.current);
   659→          }

   663→        const plants = plantsRef.current;
   664→        for (let pi = plants.length - 1; pi >= 0; pi--) {
   665→          const p = plants[pi];
   666→          if (p.dead || p.hp <= 0) { plants.splice(pi, 1); continue; }
   667→
   668→          // Cherry bomb
   669→          if (p.type === 'cherrybomb') {
   670→            p.fuseTimer += deltaMs;
   671→            if (p.fuseTimer >= CHERRY_FUSE_TIME) {
   672→              const cx = (p.col + 0.5) * CELL_W;
   673→              const cy = GRID_Y + (p.row + 0.5) * CELL_H;
   674→              const radius = CHERRY_RADIUS_CELLS * Math.max(CELL_W, CELL_H);
   675→              for (const z of zombiesRef.current) {
   676→                const zy = GRID_Y + (z.row + 0.5) * CELL_H;
   677→                const dx = z.x - cx;
   678→                const dy = zy - cy;
   679→                if (dx * dx + dy * dy < radius * radius) {
   680→                  z.hp -= CHERRY_DAMAGE;
   681→                  z.hitFlash = 10;
   682→                }
   683→              }
   684→              p.hp = 0;
   685→            }
   686→            continue;
   687→          }
   688→
   689→          // Chomper
   690→          if (p.type === 'chomper') {
   691→            if (p.digesting) {
   692→              p.digestTimer += deltaMs;
   693→              if (p.digestTimer >= CHOMP_DIGEST_TIME) {
   694→                p.digesting = false;
   695→                p.digestTimer = 0;
   696→              }
   697→              continue;
   698→            }
   699→            const chompRange = CELL_W * 1.2;
   700→            const px = (p.col + 0.5) * CELL_W;
   701→            const py = GRID_Y + (p.row + 0.5) * CELL_H;
   702→            for (const z of zombiesRef.current) {
   703→              if (z.hp <= 0 || z.row !== p.row) continue;
   704→              const zy = GRID_Y + (z.row + 0.5) * CELL_H;
   705→              const dx = z.x - px;
   706→              const dy = zy - py;
   707→              if (dx > 0 && dx < chompRange && Math.abs(dy) < CELL_H * 0.5) {
   708→                if (z.hp <= CHOMP_MAX_HP) {
   709→                  z.hp = 0;
   710→                  p.digesting = true;
   711→                  p.digestTimer = 0;
   712→                }
   713→                break;
   714→              }
   715→            }
   716→            continue;
   717→          }
   718→
   719→          // Potato mine
   720→          if (p.type === 'potatomine') {
   721→            if (!p.armed) {
   722→              p.armTimer += deltaMs;
   723→              if (p.armTimer >= MINE_ARM_TIME) {
   724→                p.armed = true;
   725→              }
   726→              continue;
   727→            }
   728→            const mx = (p.col + 0.5) * CELL_W;
   729→            const my = GRID_Y + (p.row + 0.5) * CELL_H;
   730→            const mineR = CELL_W * 0.6;
   731→            for (const z of zombiesRef.current) {
   732→              if (z.hp <= 0 || z.row !== p.row) continue;
   733→              const zy = GRID_Y + (z.row + 0.5) * CELL_H;
   734→              const dx = z.x - mx;
   735→              const dy = zy - my;
   736→              if (dx * dx + dy * dy < mineR * mineR) {
   737→                z.hp -= 1800;
   738→                z.hitFlash = 10;
   739→                p.hp = 0;
   740→                break;
   741→              }
   742→            }
   743→            continue;
   744→          }
   745→
   746→          // Sunflower sun production
   747→          if (p.type === 'sunflower') {
   748→            p.sunTimer += deltaMs;
   749→            if (p.sunTimer >= SUNFLOWER_INTERVAL) {
   750→              p.sunTimer -= SUNFLOWER_INTERVAL;
   751→              const sx = (p.col + 0.5) * CELL_W;
   752→              const sy = GRID_Y + p.row * CELL_H + CELL_H * 0.2;

   754→                id: sunIdRef.current++,
   755→                x: sx, y: sy,
   756→                targetY: sy + 20 + Math.random() * 30,
   757→                vy: 0, collected: false, alpha: 1,
   758→                bouncePhase: 0, lifeTimer: SUN_LIFE, scale: 1,
   759→              });
   760→            }
   761→          }
   762→
   763→          // Shooters
   764→          if (p.type === 'peashooter' || p.type === 'doubleshoot' || p.type === 'iceshooter') {
   765→            const hasZombie = zombiesRef.current.some(
   766→              z => z.row === p.row && z.x > (p.col + 0.5) * CELL_W && z.hp > 0
   767→            );
   768→            if (hasZombie) {
   769→              p.shootTimer += deltaMs;
   770→              if (p.shootTimer >= SHOOT_INTERVAL) {
   771→                p.shootTimer -= SHOOT_INTERVAL;
   772→                const px = (p.col + 0.8) * CELL_W;
   773→                const py = GRID_Y + (p.row + 0.5) * CELL_H;
   774→                const isIce = p.type === 'iceshooter';
   775→                peasRef.current.push({ row: p.row, x: px, y: py, isIce, isFire: false, damage: PEA_DAMAGE });
   776→                if (p.type === 'doubleshoot') {
   777→                  peasRef.current.push({ row: p.row, x: px - 8, y: py - 4, isIce: false, isFire: false, damage: PEA_DAMAGE });
   778→                }
   779→              }
   780→            }
   781→          }
   782→        }

   785→        for (const plant of plants) {
   786→          if (plant.type !== 'torchwood' || plant.hp <= 0) continue;

   788→          const tr = CELL_W * 0.6;
   789→          for (const pea of peasRef.current) {
   790→            if (pea.isFire) continue;
   791→            if (pea.row === plant.row && Math.abs(pea.x - tx) < tr) {
   792→              pea.isFire = true;
   793→              pea.isIce = false;
   794→              pea.damage = PEA_DAMAGE + TORCH_EXTRA_DAMAGE;
   795→            }
   796→          }
   797→        }
   798→
   799→        // Pea movement

   801→          peasRef.current[i].x += PEA_SPEED * dt;

   828→        for (const m of mowersRef.current) {

   832→            if (z.hp <= 0 || z.row !== m.row) continue;
   833→            if (Math.abs(z.x - m.x) < 80) {

   835→              z.smashed = true;

   840→

   847→            setDisplayScore(scoreRef.current);
   848→            zombies.splice(i, 1);
   849→            continue;
   850→          }
   851→          if (z.hitFlash > 0) z.hitFlash -= dt;
   852→          z.walkPhase += 0.05 * dt;
   853→
   854→          if (z.slowTimer > 0) {
   855→            z.slowTimer -= deltaMs;
   856→            z.speed = z.baseSpeed * cfg.zombieSpeedMul * ICE_SLOW_FACTOR;
   857→          } else {
   858→            z.speed = z.baseSpeed * cfg.zombieSpeedMul;
   859→          }
   860→
   861→          if (z.type === 'newspaper' && !z.paperBroken && z.hp <= 100) {
   862→            z.paperBroken = true;
   863→            z.baseSpeed = 0.6;
   864→            z.maxHp = 100;
   865→          }
   866→

   869→            const mower = mowersRef.current.find(m => m.row === z.row && !m.active);
   870→            if (mower) {
   871→              mower.active = true;
   872→              continue;
   873→            }
   874→          }
   875→
   876→          z.eating = false;
   877→          const zCol = Math.floor(z.x / CELL_W);
   878→          const target = plants.find(p => p.row === z.row && p.col === zCol && !p.dead && p.hp > 0);
   879→          if (target) {
   880→            z.eating = true;
   881→            const dmg = z.type === 'gargantuar' ? 1.5 : 0.5;
   882→            target.hp -= dmg * dt;
   883→          } else {
   884→            z.x -= z.speed * dt;
   885→          }
   886→
   887→          // Game over check
   888→          if (z.x < CELL_W * 0.2) {
   889→            overRef.current = true;
   890→            setDisplayScore(scoreRef.current);
   891→            submitScoreRef.current(scoreRef.current);
   892→            setPhase('lost');
   893→          }
   894→        }
   895→
   896→        // Sun updates
   897→        for (let i = sunsRef.current.length - 1; i >= 0; i--) {
   898→          const s = sunsRef.current[i];
   899→          if (s.collected) {
   900→            s.scale = Math.max(s.scale - 0.08 * dt, 0);
   901→            s.alpha -= 0.1 * dt;
   902→            if (s.alpha <= 0) sunsRef.current.splice(i, 1);
   903→            continue;
   904→          }
   905→          s.lifeTimer -= dt;
   906→          if (s.lifeTimer <= 0) { sunsRef.current.splice(i, 1); continue; }
   907→          if (s.y < s.targetY) {
   908→            s.vy += 0.15 * dt;
   909→            s.y += s.vy * dt;
   910→            if (s.y >= s.targetY) {
   911→              s.y = s.targetY;
   912→              s.vy = -s.vy * 0.3;
   913→              s.bouncePhase++;
   914→              if (s.bouncePhase >= 3) { s.y = s.targetY; s.vy = 0; }
   915→            }
   916→          }
   917→
   918→          // Cursor-based sun attraction (only when cursor is in grid area)
   919→          const cursor = cursorRef.current;
   920→          if (cursor) {
   921→            const adx = cursor.x - s.x;
   922→            const ady = cursor.y - s.y;
   923→            const adist = Math.sqrt(adx * adx + ady * ady);
   924→            if (adist < SUN_ATTRACT_RADIUS && adist > 0.1) {
   925→              const speed = SUN_ATTRACT_SPEED * dt;
   926→              s.x += (adx / adist) * speed;
   927→              s.y += (ady / adist) * speed;
   928→              if (adist < 10) {
   929→                s.collected = true;
   930→                s.scale = 2.0;
   931→                sunRef.current += SUN_VALUE;
   932→                setDisplaySun(sunRef.current);
   933→              }
   934→            }
   935→          }
   936→        }
   937→      }
   938→
   939→      // Collect all suns
   940→      if (collectAllFlagRef.current) {
   941→        collectAllFlagRef.current = false;
   942→        for (const s of sunsRef.current) {
   943→          if (!s.collected) {
   944→            s.collected = true;
   945→            s.scale = 1.8;
