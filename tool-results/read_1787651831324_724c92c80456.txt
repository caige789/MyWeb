   155→   155→/** 绘制玩家飞机 */
   156→   156→function drawPlayer(ctx: CanvasRenderingContext2D, p: Player, frame: number) {
   157→   157→  ctx.save();
   158→   158→  ctx.translate(p.x, p.y);
   159→   159→
   160→   160→  // 绘制尾焰动画
   161→   161→  const flameLen = 12 + Math.sin(frame * 0.5) * 4;
   162→   162→  const flameW = 8 + Math.sin(frame * 0.7) * 2;
   163→   163→  const flameGrad = ctx.createLinearGradient(0, PLAYER_H / 2, 0, PLAYER_H / 2 + flameLen);
   164→   164→  flameGrad.addColorStop(0, '#60a5fa');
   165→   165→  flameGrad.addColorStop(0.4, '#f97316');
   166→   166→  flameGrad.addColorStop(1, 'rgba(239,68,68,0)');
   167→   167→  ctx.fillStyle = flameGrad;
   168→   168→  ctx.beginPath();
   169→   169→  ctx.moveTo(-flameW, PLAYER_H / 2 - 2);
   170→   170→  ctx.lineTo(flameW, PLAYER_H / 2 - 2);
   171→   171→  ctx.lineTo(0, PLAYER_H / 2 + flameLen);
   172→   172→  ctx.closePath();
   173→   173→  ctx.fill();
   174→   174→
   175→   175→  // 绘制飞机机身
   176→   176→  ctx.fillStyle = '#e0f2fe';
   177→   177→  ctx.beginPath();
   178→   178→  ctx.moveTo(0, -PLAYER_H / 2);
   179→   179→  ctx.lineTo(-PLAYER_W / 2, PLAYER_H / 2 - 6);
   180→   180→  ctx.lineTo(-PLAYER_W / 4, PLAYER_H / 2);
   181→   181→  ctx.lineTo(PLAYER_W / 4, PLAYER_H / 2);
   182→   182→  ctx.lineTo(PLAYER_W / 2, PLAYER_H / 2 - 6);
   183→   183→  ctx.closePath();
   184→   184→  ctx.fill();
   185→   185→
   186→   186→  // 机翼装饰线
   187→   187→  ctx.strokeStyle = '#3b82f6';
   188→   188→  ctx.lineWidth = 1.5;
   189→   189→  ctx.beginPath();
   190→   190→  ctx.moveTo(-PLAYER_W / 2 + 4, PLAYER_H / 2 - 10);
   191→   191→  ctx.lineTo(0, -PLAYER_H / 2 + 10);
   192→   192→  ctx.lineTo(PLAYER_W / 2 - 4, PLAYER_H / 2 - 10);
   193→   193→  ctx.stroke();
   194→   194→
   195→   195→  // 座舱
   196→   196→  ctx.fillStyle = '#93c5fd';
   197→   197→  ctx.beginPath();
   198→   198→  ctx.ellipse(0, -PLAYER_H / 4, 4, 7, 0, 0, Math.PI * 2);
   199→   199→  ctx.fill();
   200→   200→
   201→   201→  // 如果有护盾，绘制护盾效果
   202→   202→  if (p.shieldTimer > 0) {
   203→   203→    const shieldAlpha = p.shieldTimer < 60 ? (Math.sin(frame * 0.3) * 0.3 + 0.4) : 0.5;
   204→   204→    ctx.strokeStyle = `rgba(74, 222, 128, ${shieldAlpha})`;
   205→   205→    ctx.lineWidth = 2.5;
   206→   206→    ctx.beginPath();
   207→   207→    ctx.ellipse(0, 0, PLAYER_W / 2 + 8, PLAYER_H / 2 + 8, 0, 0, Math.PI * 2);
   208→   208→    ctx.stroke();
   209→   209→  }
   210→   210→
   211→   211→  ctx.restore();
   212→   212→}

   262→   262→/** 绘制大型敌机（紫色六边形） */
   263→   263→function drawLargeEnemy(ctx: CanvasRenderingContext2D, e: Enemy) {
   264→   264→  ctx.save();
   265→   265→  ctx.translate(e.x, e.y);
   266→   266→  const r = e.w / 2;
   267→   267→  ctx.fillStyle = '#a855f7';
   268→   268→  ctx.beginPath();
   269→   269→  for (let i = 0; i < 6; i++) {
   270→   270→    const angle = (Math.PI / 3) * i - Math.PI / 2;
   271→   271→    const px = Math.cos(angle) * r;
   272→   272→    const py = Math.sin(angle) * r;
   273→   273→    if (i === 0) ctx.moveTo(px, py);
   274→   274→    else ctx.lineTo(px, py);
   275→   275→  }
   276→   276→  ctx.closePath();
   277→   277→  ctx.fill();
   278→   278→  // 内部装饰
   279→   279→  ctx.strokeStyle = '#d8b4fe';
   280→   280→  ctx.lineWidth = 1.5;
   281→   281→  ctx.beginPath();
   282→   282→  for (let i = 0; i < 6; i++) {
   283→   283→    const angle = (Math.PI / 3) * i - Math.PI / 2;
   284→   284→    const px = Math.cos(angle) * (r * 0.55);
   285→   285→    const py = Math.sin(angle) * (r * 0.55);
   286→   286→    if (i === 0) ctx.moveTo(px, py);
   287→   287→    else ctx.lineTo(px, py);
   288→   288→  }
   289→   289→  ctx.closePath();
   290→   290→  ctx.stroke();
   291→   291→  // 中心点
   292→   292→  ctx.fillStyle = '#e9d5ff';
   293→   293→  ctx.beginPath();
   294→   294→  ctx.arc(0, 0, 3, 0, Math.PI * 2);
   295→   295→  ctx.fill();
   296→   296→  ctx.restore();
   297→   297→}

   299→   299→/** 根据敌机类型分发绘制 */
   300→   300→function drawEnemy(ctx: CanvasRenderingContext2D, e: Enemy) {
   301→   301→  switch (e.type) {
   302→   302→    case EnemyType.Small: drawSmallEnemy(ctx, e); break;
   303→   303→    case EnemyType.Medium: drawMediumEnemy(ctx, e); break;
   304→   304→    case EnemyType.Large: drawLargeEnemy(ctx, e); break;
   305→   305→  }
   306→   306→  // 血条（非满血时显示）
   307→   307→  if (e.hp < e.maxHp) {
   308→   308→    const barW = e.w * 0.8;
   309→   309→    const barH = 3;
   310→   310→    const barX = e.x - barW / 2;
   311→   311→    const barY = e.y - e.h / 2 - 6;
   312→   312→    ctx.fillStyle = 'rgba(255,255,255,0.2)';
   313→   313→    ctx.fillRect(barX, barY, barW, barH);
   314→   314→    ctx.fillStyle = '#22c55e';
   315→   315→    ctx.fillRect(barX, barY, barW * (e.hp / e.maxHp), barH);
   316→   316→  }
   317→   317→}

   325→   325→    particles.push({
   326→   326→      x,
   327→   327→      y,
   328→   328→      vx: Math.cos(angle) * speed,
   329→   329→      vy: Math.sin(angle) * speed,
   330→   330→      life: Math.random() * 20 + 15,
   331→   331→      maxLife: 35,
   332→   332→      color,
   333→   333→      r: Math.random() * 3 + 1,
   334→   334→    });
   335→   335→  }
   336→   336→  return particles;
   337→   337→}

   340→   340→function rectCollide(

   342→   342→  bx: number, by: number, bw: number, bh: number

   433→   433→    const p = playerRef.current;
   434→   434→    const isDual = p.dualShotTimer > 0;
   435→   435→    const bulletSpeed = -8;
   436→   436→
   437→   437→    if (isDual) {
   438→   438→      // 双发子弹：左右偏移
   439→   439→      bulletsRef.current.push(
   440→   440→        { x: p.x - 10, y: p.y - PLAYER_H / 2, vy: bulletSpeed },
   441→   441→        { x: p.x + 10, y: p.y - PLAYER_H / 2, vy: bulletSpeed },
   442→   442→      );
   443→   443→    } else {
   444→   444→      bulletsRef.current.push(
   445→   445→        { x: p.x, y: p.y - PLAYER_H / 2, vy: bulletSpeed },
   446→   446→      );
   447→   447→    }
   448→   448→  }, []);
   449→   449→
   450→   450→  /** 获取生成敌机的间隔帧数 */
   451→   451→  const getSpawnInterval = useCallback(() => {
   452→   452→    const lv = levelRef.current;
   453→   453→    // 初始每45帧生成一个，最少每15帧
   454→   454→    return Math.max(45 - (lv - 1) * 4, 15);
   455→   455→  }, []);
   456→   456→
   457→   457→  /** 游戏主循环 */
   458→   458→  const gameLoop = useCallback(() => {
   459→   459→    const canvas = canvasRef.current;
   460→   460→    if (!canvas) return;
   461→   461→    const ctx = canvas.getContext('2d');
   462→   462→    if (!ctx) return;
   463→   463→
   464→   464→    const dpr = window.devicePixelRatio || 1;
   465→   465→    // 每帧重新设置缩放（确保高DPI正确）
   466→   466→  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   467→   467→
   468→   468→    const frame = frameRef.current;
   469→   469→    frameRef.current++;
   470→   470→    const p = playerRef.current;
   471→   471→
   472→   472→    // ==== 更新逻辑 ====
   473→   473→
   474→   474→    // 更新星星闪烁
   475→   475→    starsRef.current.forEach((s) => {
   476→   476→      s.phase += s.twinkleSpeed;
   477→   477→    });
   478→   478→
   479→   479→    if (statusRef.current === 'playing' && !pausedRef.current) {
   480→   480→      // 移动玩家（根据输入坐标）
   481→   481→      if (inputXRef.current !== null) {
   482→   482→        const target = inputXRef.current;
   483→   483→        const diff = target - p.x;
   484→   484→        // 平滑跟随
   485→   485→        if (Math.abs(diff) > 2) {
   486→   486→          p.x += diff * 0.15;
   487→   487→        } else {
   488→   488→          p.x = target;
   489→   489→        }
   490→   490→      }
   491→   491→      // 限制玩家在画布内
   492→   492→      p.x = Math.max(PLAYER_W / 2, Math.min(LOGIC_W - PLAYER_W / 2, p.x));
   493→   493→
   494→   494→      // 更新计时器
   495→   495→      if (p.shieldTimer > 0) p.shieldTimer--;
   496→   496→      if (p.dualShotTimer > 0) p.dualShotTimer--;
   497→   497→
   498→   498→      // 发射子弹
   499→   499→      const isMobile = isMobileRef.current;
   500→   500→      const interval = p.dualShotTimer > 0 ? DUAL_SHOOT_INTERVAL : SHOOT_INTERVAL;
   501→   501→      // 手机端自动发射，桌面端需要点击/空格
   502→   502→      if (isMobile) {
   503→   503→        if (frame - lastShootRef.current >= interval) {
   504→   504→          shoot();
   505→   505→          lastShootRef.current = frame;
   506→   506→        }
   507→   507→      } else {
   508→   508→        if (shootRequestRef.current) {
   509→   509→          if (frame - lastShootRef.current >= interval) {
   510→   510→            shoot();
   511→   511→            lastShootRef.current = frame;
   512→   512→          }
   513→   513→        }
   514→   514→      }
   515→   515→      shootRequestRef.current = false;
   516→   516→
   517→   517→      // 生成敌机
   518→   518→      const spawnInterval = getSpawnInterval();
   519→   519→      if (frame % spawnInterval === 0) {
   520→   520→        spawnEnemy();
   521→   521→      }
   522→   522→
   523→   523→      // 更新子弹位置
   524→   524→      bulletsRef.current = bulletsRef.current.filter((b) => {
   525→   525→        b.y += b.vy;
   526→   526→        return b.y > -BULLET_H;
   527→   527→      });
   528→   528→
   529→   529→      // 更新敌机位置
   530→   530→      enemiesRef.current = enemiesRef.current.filter((e) => {
   531→   531→        e.y += e.speed;
   532→   532→        return e.y < LOGIC_H + e.h;
   533→   533→      });
   534→   534→
   535→   535→      // 更新道具位置
   536→   536→      powerupsRef.current = powerupsRef.current.filter((pw) => {
   537→   537→        pw.y += 2;
   538→   538→        pw.phase += 0.1;
   539→   539→        return pw.y < LOGIC_H + 20;
   540→   540→      });
   541→   541→
   542→   542→      // 子弹-敌机碰撞检测
   543→   543→      const bullets = bulletsRef.current;
   544→   544→      const enemies = enemiesRef.current;
   545→   545→      const hitBullets = new Set<number>();
   546→   546→      const hitEnemies = new Set<number>();
   547→   547→
   548→   548→      for (let bi = 0; bi < bullets.length; bi++) {
   549→   549→        const b = bullets[bi];
   550→   550→        for (let ei = 0; ei < enemies.length; ei++) {
   551→   551→          if (hitEnemies.has(ei)) continue;
   552→   552→          const e = enemies[ei];
   553→   553→          if (rectCollide(b.x, b.y, BULLET_W, BULLET_H, e.x, e.y, e.w, e.h)) {
   554→   554→            hitBullets.add(bi);
   555→   555→            e.hp--;
   556→   556→            if (e.hp <= 0) {
   557→   557→              hitEnemies.add(ei);
   558→   558→              scoreRef.current += e.score;
   559→   559→              killCountRef.current++;
   560→   560→              setScore(scoreRef.current);
   561→   561→
   562→   562→              // 爆炸粒子
   563→   563→              const expColor = e.type === EnemyType.Small ? '#ef4444'
   564→   564→                : e.type === EnemyType.Medium ? '#f97316' : '#a855f7';
   565→   565→              const expCount = e.type === EnemyType.Large ? 20 : e.type === EnemyType.Medium ? 14 : 8;
   566→   566→              particlesRef.current.push(...createExplosion(e.x, e.y, expColor, expCount));
   567→   567→
   568→   568→              // 道具掉落
   569→   569→              if (Math.random() < POWERUP_CHANCE) {
   570→   570→                const pwType = Math.random() < 0.5 ? PowerUpType.DualShot : PowerUpType.Shield;
   571→   571→                powerupsRef.current.push({ x: e.x, y: e.y, type: pwType, phase: 0 });
   572→   572→              }
   573→   573→
   574→   574→              // 检查升级
   575→   575→              if (killCountRef.current % LEVEL_UP_KILLS === 0) {
   576→   576→                levelRef.current++;
   577→   577→                setLevel(levelRef.current);
   578→   578→              }
   579→   579→            } else {
   580→   580→              // 敌机受伤但没死，小爆炸
   581→   581→              particlesRef.current.push(...createExplosion(b.x, b.y, '#fbbf24', 3));
   582→   582→            }
   583→   583→            break;
   584→   584→          }
   585→   585→        }
   586→   586→      }
   587→   587→
   588→   588→      // 移除被击中的子弹和敌机
   589→   589→      bulletsRef.current = bullets.filter((_, i) => !hitBullets.has(i));
   590→   590→      enemiesRef.current = enemies.filter((_, i) => !hitEnemies.has(i));
   591→   591→
   592→   592→      // 敌机-玩家碰撞检测
   593→   593→      if (p.shieldTimer <= 0) {
   594→   594→        for (const e of enemiesRef.current) {
   595→   595→          if (rectCollide(p.x, p.y, PLAYER_W * 0.7, PLAYER_H * 0.7, e.x, e.y, e.w, e.h)) {
   596→   596→            // 玩家被撞，游戏结束
   597→   597→            statusRef.current = 'over';
   598→   598→            setGameStatus('over');
   599→   599→            // 玩家爆炸
   600→   600→            particlesRef.current.push(...createExplosion(p.x, p.y, '#60a5fa', 25));
   601→   601→            break;
   602→   602→          }
   603→   603→        }
   604→   604→      }
   605→   605→
   606→   606→      // 道具-玩家碰撞检测
   607→   607→      if (statusRef.current === 'playing') {
   608→   608→        powerupsRef.current = powerupsRef.current.filter((pw) => {
   609→   609→          if (rectCollide(p.x, p.y, PLAYER_W, PLAYER_H, pw.x, pw.y, 24, 24)) {
   610→   610→            if (pw.type === PowerUpType.DualShot) {
   611→   611→              p.dualShotTimer = DUAL_SHOT_DURATION;
   612→   612→            } else if (pw.type === PowerUpType.Shield) {
   613→   613→              p.shieldTimer = SHIELD_DURATION;
   614→   614→            }
   615→   615→            // 拾取粒子
   616→   616→            const pColor = pw.type === PowerUpType.DualShot ? '#3b82f6' : '#22c55e';
   617→   617→            particlesRef.current.push(...createExplosion(pw.x, pw.y, pColor, 10));
   618→   618→            return false;
   619→   619→          }
   620→   620→          return true;
   621→   621→        });
   622→   622→      }
   623→   623→
   624→   624→      // 游戏结束时提交分数
   625→   625→      if (statusRef.current === 'over') {
   626→   626→        if (scoreRef.current > highScoreRef.current) {
   627→   627→          setIsNewRecord(true);
   628→   628→          fetch('/api/games/scores', {
   629→   629→            method: 'POST',
   630→   630→            headers: { 'Content-Type': 'application/json' },
   631→   631→            body: JSON.stringify({ game: GAME_NAME, score: scoreRef.current }),
   632→   632→          })
   633→   633→            .then((r) => r.json())
   634→   634→            .then((data) => {
   635→   635→              if (data.data?.score !== undefined) {
   636→   636→                setHighScore(data.data.score);
   637→   637→                highScoreRef.current = data.data.score;
   638→   638→              }
   639→   639→            })
   640→   640→            .catch(() => {});
   641→   641→        }
   642→   642→      }
   643→   643→    }
   644→   644→
   645→   645→    // 更新粒子
   646→   646→    particlesRef.current = particlesRef.current.filter((pt) => {
   647→   647→      pt.x += pt.vx;
   648→   648→      pt.y += pt.vy;
   649→   649→      pt.life--;
   650→   650→      pt.vx *= 0.97;
   651→   651→      pt.vy *= 0.97;
   652→   652→      return pt.life > 0;
   653→   653→    });
   654→   654→
   655→   655→    // ==== 绘制逻辑 ====
   656→   656→
   657→   657→    // 深蓝色星空背景
   658→   658→    const bgGrad = ctx.createLinearGradient(0, 0, 0, LOGIC_H);
   659→   659→    bgGrad.addColorStop(0, '#0a0e27');
   660→   660→    bgGrad.addColorStop(0.5, '#0f1638');
   661→   661→    bgGrad.addColorStop(1, '#0a0e27');
   662→   662→    ctx.fillStyle = bgGrad;
   663→   663→    ctx.fillRect(0, 0, LOGIC_W, LOGIC_H);
   664→   664→
   665→   665→    // 绘制星星
   666→   666→    starsRef.current.forEach((s) => {
   667→   667→      const alpha = s.alpha * (0.5 + Math.sin(s.phase) * 0.5);
   668→   668→      ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
   669→   669→      ctx.beginPath();
   670→   670→      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
   671→   671→      ctx.fill();
   672→   672→    });
   673→   673→
   674→   674→    // 绘制道具
   675→   675→    powerupsRef.current.forEach((pw) => {
   676→   676→      const glow = 0.5 + Math.sin(pw.phase) * 0.3;
   677→   677→      const color = pw.type === PowerUpType.DualShot ? `rgba(59, 130, 246, ${glow})` : `rgba(34, 197, 94, ${glow})`;
   678→   678→      const solidColor = pw.type === PowerUpType.DualShot ? '#3b82f6' : '#22c55e';
   679→   679→      // 外圈光晕
   680→   680→      ctx.fillStyle = color;
   681→   681→      ctx.beginPath();
   682→   682→      ctx.arc(pw.x, pw.y, 14, 0, Math.PI * 2);
   683→   683→      ctx.fill();
   684→   684→      // 内圈实心
   685→   685→      ctx.fillStyle = solidColor;
   686→   686→      ctx.beginPath();
   687→   687→      ctx.arc(pw.x, pw.y, 8, 0, Math.PI * 2);
   688→   688→      ctx.fill();
   689→   689→      // 图标文字
   690→   690→      ctx.fillStyle = '#ffffff';
   691→   691→      ctx.font = 'bold 10px sans-serif';
   692→   692→      ctx.textAlign = 'center';
   693→   693→      ctx.textBaseline = 'middle';
   694→   694→      ctx.fillText(pw.type === PowerUpType.DualShot ? 'D' : 'S', pw.x, pw.y);
   695→   695→    });
   696→   696→
   697→   697→    // 绘制子弹（蓝色光束）
   698→   698→    ctx.shadowColor = '#3b82f6';
   699→   699→    ctx.shadowBlur = 6;
   700→   700→    ctx.fillStyle = '#93c5fd';
   701→   701→    bulletsRef.current.forEach((b) => {
   702→   702→      ctx.fillRect(b.x - BULLET_W / 2, b.y - BULLET_H / 2, BULLET_W, BULLET_H);
   703→   703→    });
   704→   704→    ctx.shadowBlur = 0;
   705→   705→
   706→   706→    // 绘制敌机
   707→   707→    enemiesRef.current.forEach((e) => drawEnemy(ctx, e));
   708→   708→
   709→   709→    // 绘制粒子
   710→   710→    particlesRef.current.forEach((pt) => {
   711→   711→      const alpha = pt.life / pt.maxLife;
   712→   712→      ctx.globalAlpha = alpha;
   713→   713→      ctx.fillStyle = pt.color;
   714→   714→      ctx.beginPath();
   715→   715→      ctx.arc(pt.x, pt.y, pt.r * alpha, 0, Math.PI * 2);
   716→   716→      ctx.fill();
   717→   717→    });
   718→   718→    ctx.globalAlpha = 1;
   719→   719→
   720→   720→    // 绘制玩家（仅非死亡状态）
   721→   721→    if (statusRef.current !== 'over') {
   722→   722→      drawPlayer(ctx, p, frame);
   723→   723→    }
   724→   724→
   725→   725→    // 绘制状态覆盖层
   726→   726→    if (statusRef.current === 'idle') {
   727→   727→      ctx.fillStyle = 'rgba(0, 0, 0, 0.55)';
   728→   728→      ctx.fillRect(0, 0, LOGIC_W, LOGIC_H);
   729→   729→      ctx.fillStyle = '#ffffff';
   730→   730→      ctx.font = 'bold 28px sans-serif';
   731→   731→      ctx.textAlign = 'center';
   732→   732→      ctx.textBaseline = 'middle';
   733→   733→      ctx.fillText('飞机大战', LOGIC_W / 2, LOGIC_H / 2 - 30);
   734→   734→      ctx.font = '16px sans-serif';
   735→   735→      ctx.fillStyle = '#94a3b8';
   736→   736→      ctx.fillText('点击下方按钮开始', LOGIC_W / 2, LOGIC_H / 2 + 15);
   737→   737→    }
   738→   738→
   739→   739→    if (pausedRef.current && statusRef.current === 'playing') {
   740→   740→      ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';
   741→   741→      ctx.fillRect(0, 0, LOGIC_W, LOGIC_H);
   742→   742→      ctx.fillStyle = '#ffffff';
   743→   743→      ctx.font = 'bold 32px sans-serif';
   744→   744→      ctx.textAlign = 'center';
   745→   745→      ctx.textBaseline = 'middle';
   746→   746→      ctx.fillText('已暂停', LOGIC_W / 2, LOGIC_H / 2 - 10);
   747→   747→      ctx.font = '16px sans-serif';
   748→   748→      ctx.fillStyle = '#94a3b8';
   749→   749→      ctx.fillText('按 Esc 继续', LOGIC_W / 2, LOGIC_H / 2 + 25);
   750→   750→    }
   751→   751→
   752→   752→    if (statusRef.current === 'over') {
   753→   753→      ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';
   754→   754→      ctx.fillRect(0, 0, LOGIC_W, LOGIC_H);
   755→   755→      ctx.fillStyle = '#ffffff';
   756→   756→      ctx.font = 'bold 32px sans-serif';
   757→   757→      ctx.textAlign = 'center';
   758→   758→      ctx.textBaseline = 'middle';
   759→   759→      ctx.fillText('游戏结束', LOGIC_W / 2, LOGIC_H / 2 - 40);
   760→   760→      ctx.font = '20px sans-serif';
   761→   761→      ctx.fillStyle = '#fbbf24';
   762→   762→      ctx.fillText(`得分: ${scoreRef.current}`, LOGIC_W / 2, LOGIC_H / 2 + 5);
   763→   763→      ctx.font = '14px sans-serif';
   764→   764→      ctx.fillStyle = '#94a3b8';
   765→   765→      ctx.fillText('点击重新开始', LOGIC_W / 2, LOGIC_H / 2 + 45);
   766→   766→    }
   767→   767→
   768→   768→    // 继续下一帧
   769→   769→    rafRef.current = requestAnimationFrame(gameLoop);
   770→   770→  }, [shoot, spawnEnemy, getSpawnInterval]);
   771→   771→
   772→   772→  /** 开始/重新开始游戏 */
   773→   773→  const startGame = useCallback(() => {
   774→   774→    // 重置所有游戏状态
   775→   775→    playerRef.current = { x: LOGIC_W / 2, y: LOGIC_H - 60, shieldTimer: 0, dualShotTimer: 0 };
   776→   776→    bulletsRef.current = [];
   777→   777→    enemiesRef.current = [];
   778→   778→    particlesRef.current = [];
   779→   779→    powerupsRef.current = [];
   780→   780→    scoreRef.current = 0;
   781→   781→    killCountRef.current = 0;
   782→   782→    levelRef.current = 1;
   783→   783→    frameRef.current = 0;
   784→   784→    lastShootRef.current = 0;
   785→   785→    inputXRef.current = null;
   786→   786→    shootRequestRef.current = false;
   787→   787→
   788→   788→    setScore(0);
   789→   789→    setLevel(1);
   790→   790→    setIsNewRecord(false);
   791→   791→    pausedRef.current = false;
   792→   792→    setPaused(false);
   793→   793→    statusRef.current = 'playing';
   794→   794→    setGameStatus('playing');
   795→   795→  }, []);
   796→   796→
   797→   797→  /** Canvas 点击事件（用于重新开始） */
   798→   798→  const handleCanvasClick = useCallback(() => {
   799→   799→    if (statusRef.current === 'over') {
   800→   800→      startGame();
   801→   801→    }
   802→   802→  }, [startGame]);
   803→   803→
   804→   804→  /** 检测是否为触屏设备 */
   805→   805→  useEffect(() => {
   806→   806→    isMobileRef.current = 'ontouchstart' in window || navigator.maxTouchPoints > 0;
   807→   807→  }, []);
   808→   808→
   809→   809→  /** 鼠标移动事件（桌面端控制飞机） */
   810→   810→  useEffect(() => {
   811→   811→    const handleMouseMove = (e: MouseEvent) => {
   812→   812→      if (statusRef.current !== 'playing' || pausedRef.current) return;
   813→   813→      const canvas = canvasRef.current;
   814→   814→      if (!canvas) return;
   815→   815→      const rect = canvas.getBoundingClientRect();
   816→   816→      const scaleX = LOGIC_W / rect.width;
   817→   817→      inputXRef.current = (e.clientX - rect.left) * scaleX;
   818→   818→    };
   819→   819→
   820→   820→    const handleMouseDown = () => {
   821→   821→      if (statusRef.current === 'playing' && !pausedRef.current) {
   822→   822→        shootRequestRef.current = true;
   823→   823→      }
   824→   824→    };
   825→   825→
   826→   826→    const handleKeyDown = (e: KeyboardEvent) => {
   827→   827→      if (e.code === 'Escape' && statusRef.current === 'playing') {
   828→   828→        e.preventDefault();
   829→   829→        pausedRef.current = !pausedRef.current;
   830→   830→        setPaused(pausedRef.current);
   831→   831→        return;
   832→   832→      }
   833→   833→      if (e.code === 'Space' && statusRef.current === 'playing' && !pausedRef.current) {
   834→   834→        e.preventDefault();
   835→   835→        shootRequestRef.current = true;
   836→   836→      }
   837→   837→    };
   838→   838→
   839→   839→    window.addEventListener('mousemove', handleMouseMove);
   840→   840→    window.addEventListener('mousedown', handleMouseDown);
   841→   841→    window.addEventListener('keydown', handleKeyDown);
   842→   842→    return () => {
   843→   843→      window.removeEventListener('mousemove', handleMouseMove);
   844→   844→      window.removeEventListener('mousedown', handleMouseDown);
   845→   845→      window.removeEventListener('keydown', handleKeyDown);
   846→   846→    };
   847→   847→  }, []);
   848→   848→
   849→   849→  /** 触摸事件（移动端控制飞机 + 自动发射） */
   850→   850→  useEffect(() => {
   851→   851→    const canvas = canvasRef.current;
   852→   852→    if (!canvas) return;
   853→   853→
   854→   854→    const handleTouchMove = (e: TouchEvent) => {
   855→   855→      if (statusRef.current !== 'playing' || pausedRef.current) return;

   872→   872→        const scaleX = LOGIC_W / rect.width;

   874→   874→      }
   875→   875→    };

   880→   880→      canvas.removeEventListener('touchmove', handleTouchMove);
   881→   881→      canvas.removeEventListener('touchstart', handleTouchStart);

   899→   899→      canvas.height = displayH * dpr;

   901→   901→      canvas.style.height = `${displayH}px`;

   905→   905→    window.addEventListener('resize', resizeCanvas);

   909→   909→
   910→   910→    return () => {
   911→   911→      window.removeEventListener('resize', resizeCanvas);
   912→   912→      cancelAnimationFrame(rafRef.current);
   913→   913→    };
   914→   914→  }, [gameLoop]);
   915→   915→
   916→   916→  /** 获取最高分 */

   925→   925→              (item: { game: string; score: number }) => item.game === GAME_NAME

   929→   929→              highScoreRef.current = s.score;
   930→   930→            }
   931→   931→          }
   932→   932→        }
   933→   933→      } catch {
   934→   934→        // 网络异常时静默处理
   935→   935→      } finally {
   936→   936→        setLoadingHighScore(false);
   937→   937→      }
   938→   938→    };
   939→   939→    fetchHighScore();
   940→   940→  }, []);
   941→   941→
   942→   942→  return (
   943→   943→    <div className="flex flex-col items-center w-full max-w-lg mx-auto px-2 py-2 gap-3">
   944→   944→      {/* Top bar: back | game name + score | best */}
   945→   945→      <div className="flex items-center justify-between w-full">

   972→   972→      {/* Canvas game area */}
   973→   973→      <div ref={containerRef} className="w-full flex justify-center">
   974→   974→        <canvas
   975→   975→          ref={canvasRef}
