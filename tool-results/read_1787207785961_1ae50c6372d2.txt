   144→/* ============ 每波僵尸生成配置 ============ */
   145→function getWaveZombies(wave: number, countMul: number): { type: ZombieType; delay: number }[] {
   146→  const zombies: { type: ZombieType; delay: number }[] = [];
   147→  let delay = 0;
   148→  const baseCounts: number[][] = [
   149→    [4, 1, 0, 0, 0],
   150→    [3, 2, 0, 1, 1],
   151→    [2, 2, 1, 1, 2],
   152→    [2, 3, 1, 1, 2],
   153→    [1, 3, 2, 2, 3],
   154→  ];
   155→  const counts = baseCounts[Math.min(wave, baseCounts.length - 1)];
   156→  const types: ZombieType[] = ['normal', 'cone', 'bucket', 'flag', 'newspaper'];
   157→  for (let i = 0; i < types.length; i++) {
   158→    const count = Math.round(counts[i] * countMul);
   159→    for (let j = 0; j < count; j++) {
   160→      zombies.push({ type: types[i], delay });
   161→      delay += 2000 + Math.random() * 3000;
   162→    }
   163→  }
   164→  for (let i = zombies.length - 1; i > 0; i--) {
   165→    const j = Math.floor(Math.random() * (i + 1));
   166→    [zombies[i], zombies[j]] = [zombies[j], zombies[i]];
   167→  }
   168→  return zombies;
   169→}

   185→  const peasRef = useRef<Pea[]>([]);

   187→  const sunRef = useRef(100);
   188→  const scoreRef = useRef(0);
   189→  const waveRef = useRef(1);
   190→  const waveElapsedRef = useRef(0);
   191→  const waveZombiesRef = useRef<{ type: ZombieType; delay: number }[]>([]);
   192→  const waveSpawnIdxRef = useRef(0);
   193→  const sunDropTimerRef = useRef(0);
   194→  const zombieIdRef = useRef(0);
   195→  const sunIdRef = useRef(0);
   196→  const diffRef = useRef<Difficulty>('normal');
   197→  const overRef = useRef(false);
   198→  const wonRef = useRef(false);
   199→  const selPlantRef = useRef<string | null>(null);
   200→  const lastTimeRef = useRef(0);
   201→  const rafRef = useRef(0);
   202→  const canvasRef = useRef<HTMLCanvasElement>(null);
   203→  const containerRef = useRef<HTMLDivElement>(null);
   204→  const cellWRef = useRef(LOGIC_W / COLS);
   205→  const cellHRef = useRef(LOGIC_H / ROWS);
   206→
   207→  const loopFnRef = useRef<(timestamp: number) => void>(null!);
   208→  const submitScoreRef = useRef<(s: number) => void>(() => {});
   209→
   210→  /* --- 获取最高分 --- */
   211→  useEffect(() => {
   212→    fetch('/api/games/scores')
   213→      .then(r => r.json())
   214→      .then(data => {
   215→        if (data.code === 200) {
   216→          const pvz = (data.data || []).find((s: { game: string }) => s.game === GAME_NAME);
   217→          if (pvz) setHighScore(pvz.score);
   218→        }
   219→      });
   220→  }, []);
   221→
   222→  /* --- 提交分数 --- */
   223→  const doSubmitScore = useCallback((s: number) => {
   224→    fetch('/api/games/scores', {
   225→      method: 'POST',
   226→      headers: { 'Content-Type': 'application/json' },
   227→      body: JSON.stringify({ game: GAME_NAME, score: s }),
   228→    })
   229→      .then(r => r.json())
   230→      .then(data => {
   231→        if (data.code === 200 && data.data) setHighScore(data.data.score);
   232→      });
   233→  }, []);
   234→
   235→  useEffect(() => { submitScoreRef.current = doSubmitScore; }, [doSubmitScore]);
   236→

   239→    const canvas = canvasRef.current;
   240→    const container = containerRef.current;
   241→    if (!canvas || !container) return;
   242→    const rect = container.getBoundingClientRect();
   243→    const w = Math.min(rect.width, 640);
   244→    const h = w * (ROWS / COLS);
   245→    const dpr = window.devicePixelRatio || 1;
   246→    canvas.width = w * dpr;
   247→    canvas.height = h * dpr;
   248→    canvas.style.width = w + 'px';
   249→    canvas.style.height = h + 'px';
   250→    const ctx = canvas.getContext('2d');
   251→    if (ctx) ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   252→    cellWRef.current = w / COLS;
   253→    cellHRef.current = h / ROWS;

   255→
   256→  /* --- 初始化游戏状态 --- */
   257→  const initGame = useCallback((diff: Difficulty) => {
   258→    const cfg = DIFFICULTY_CONFIG[diff];
   259→    plantsRef.current = [];
   260→    zombiesRef.current = [];
   261→    peasRef.current = [];
   262→    sunsRef.current = [];
   263→    sunRef.current = cfg.initSun;
   264→    scoreRef.current = 0;
   265→    waveRef.current = 1;
   266→    waveElapsedRef.current = 0;
   267→    waveZombiesRef.current = getWaveZombies(0, cfg.zombieCountMul);
   268→    waveSpawnIdxRef.current = 0;
   269→    sunDropTimerRef.current = 0;
   270→    zombieIdRef.current = 0;
   271→    sunIdRef.current = 0;
   272→    diffRef.current = diff;
   273→    overRef.current = false;
   274→    wonRef.current = false;
   275→    selPlantRef.current = null;
   276→    lastTimeRef.current = 0;
   277→    setSunCount(cfg.initSun);
   278→    setScore(0);
   279→    setCurrentWave(1);
   280→    setSelectedPlant(null);
   281→  }, []);
   282→
   283→  /* --- 开始游戏 --- */
   284→  const startGame = useCallback(() => {
   285→    initGame(difficulty);
   286→    setPhase('playing');
   287→  }, [difficulty, initGame]);
   288→
   289→  /* --- 坐标转换 --- */
   290→  const screenToLogic = useCallback((clientX: number, clientY: number) => {
   291→    const canvas = canvasRef.current;
   292→    if (!canvas) return { x: 0, y: 0 };
   293→    const rect = canvas.getBoundingClientRect();
   294→    return {
   295→      x: (clientX - rect.left) * (LOGIC_W / rect.width),
   296→      y: (clientY - rect.top) * (LOGIC_H / rect.height),
   297→    };
   298→  }, []);
   299→
   300→  /* --- Canvas点击/触摸处理 --- */
   301→  const handleCanvasInteraction = useCallback((clientX: number, clientY: number) => {
   302→    if (overRef.current || wonRef.current) return;
   303→    const pos = screenToLogic(clientX, clientY);
   304→    const cellW = cellWRef.current;
   305→    const cellH = cellHRef.current;
   306→
   307→    // 1. 检查是否点击了阳光
   308→    const suns = sunsRef.current;
   309→    for (let i = suns.length - 1; i >= 0; i--) {
   310→      const s = suns[i];
   311→      if (s.collected) continue;
   312→      const dx = pos.x - s.x;
   313→      const dy = pos.y - s.y;
   314→      if (dx * dx + dy * dy < 900) {
   315→        s.collected = true;
   316→        sunRef.current += SUN_VALUE;
   317→        setSunCount(sunRef.current);
   318→        return;
   319→      }
   320→    }
   321→
   322→    // 2. 在格子上种植物
   323→    const sel = selPlantRef.current;
   324→    if (sel) {
   325→      const col = Math.floor(pos.x / cellW);
   326→      const row = Math.floor(pos.y / cellH);
   327→      if (row >= 0 && row < ROWS && col >= 0 && col < COLS) {
   328→        const exists = plantsRef.current.some(p => p.row === row && p.col === col);
   329→        if (!exists) {
   330→          const def = PLANT_DEFS.find(d => d.id === sel);
   331→          if (def && sunRef.current >= def.cost) {
   332→            sunRef.current -= def.cost;
   333→            setSunCount(sunRef.current);
   334→            plantsRef.current.push({
   335→              type: def.id, row, col,
   336→              hp: def.hp, maxHp: def.hp,
   337→              shootTimer: 0, sunTimer: 0,
   338→              fuseTimer: 0, digestTimer: 0,
   339→              digesting: false, dead: false,
   340→            });
   341→          }
   342→        }
   343→      }
   344→    }
   345→  }, [screenToLogic]);
   346→
   347→  /* --- Canvas事件监听 --- 依赖phase，确保playing时Canvas已存在再绑定 */
   348→  useEffect(() => {
   349→    if (phase !== 'playing') return;
   350→    const canvas = canvasRef.current;
   351→    if (!canvas) return;
   352→    const onClick = (e: MouseEvent) => {
   353→      e.preventDefault();
   354→      handleCanvasInteraction(e.clientX, e.clientY);
   355→    };
   356→    const onTouch = (e: TouchEvent) => {
   357→      e.preventDefault();
   358→      const t = e.touches[0] || e.changedTouches[0];
   359→      if (t) handleCanvasInteraction(t.clientX, t.clientY);
   360→    };
   361→    canvas.addEventListener('click', onClick);
   362→    canvas.addEventListener('touchstart', onTouch, { passive: false });
   363→    return () => {
   364→      canvas.removeEventListener('click', onClick);
   365→      canvas.removeEventListener('touchstart', onTouch);
   366→    };
   367→  }, [phase, handleCanvasInteraction]);
   368→
   369→  /* --- 选择植物 --- */
   370→  const handleSelectPlant = useCallback((plantId: string) => {
   371→    if (selPlantRef.current === plantId) {
   372→      selPlantRef.current = null;
   373→      setSelectedPlant(null);
   374→    } else {
   375→      selPlantRef.current = plantId;
   376→      setSelectedPlant(plantId);
   377→    }
   378→  }, []);
   379→
   380→  /* --- 定义游戏主循环 --- */
   381→  useEffect(() => {
   382→    const loop = (timestamp: number) => {
   383→      const canvas = canvasRef.current;
   384→      if (!canvas) return;
   385→      const ctx = canvas.getContext('2d');
   386→      if (!ctx) return;
   387→
   388→      const dpr = window.devicePixelRatio || 1;
   389→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   390→
   391→      if (lastTimeRef.current === 0) lastTimeRef.current = timestamp;
   392→      const deltaMs = Math.min(timestamp - lastTimeRef.current, 50);
   393→      lastTimeRef.current = timestamp;
   394→      const dt = deltaMs / 16.667;
   395→
   396→      const cellW = cellWRef.current;
   397→      const cellH = cellHRef.current;
   398→      const cfg = DIFFICULTY_CONFIG[diffRef.current];
   399→
   400→      /* ===== 更新逻辑 ===== */
   401→      if (!overRef.current && !wonRef.current) {
   402→        waveElapsedRef.current += deltaMs;
   403→
   404→        // 天上掉阳光
   405→        sunDropTimerRef.current += deltaMs;
   406→        if (sunDropTimerRef.current >= SUN_DROP_INTERVAL) {
   407→          sunDropTimerRef.current -= SUN_DROP_INTERVAL;
   408→          const newSun: Sun = {
   409→            id: sunIdRef.current++,
   410→            x: 40 + Math.random() * (LOGIC_W - 80),
   411→            y: -20,
   412→            targetY: 30 + Math.random() * (LOGIC_H - 60),
   413→            vy: 0, collected: false, alpha: 1, bouncePhase: 0, lifeTimer: SUN_LIFE,
   414→          };
   415→          sunsRef.current.push(newSun);
   416→        }
   417→
   418→        // 生成当前波僵尸
   419→        const wz = waveZombiesRef.current;
   420→        const wsi = waveSpawnIdxRef.current;
   421→        if (wsi < wz.length && waveElapsedRef.current >= wz[wsi].delay) {
   422→          const zType = wz[wsi].type;
   423→          const def = ZOMBIE_DEFS[zType];
   424→          const newspaperHp = zType === 'newspaper' ? 180 : def.hp;
   425→          const newZombie: Zombie = {
   426→            id: zombieIdRef.current++,
   427→            type: zType,
   428→            name: def.name,
   429→            emoji: def.emoji,
   430→            row: Math.floor(Math.random() * ROWS),
   431→            x: LOGIC_W + 10,
   432→            hp: def.hp, maxHp: def.hp,
   433→            baseSpeed: def.speed,
   434→            speed: def.speed * cfg.zombieSpeedMul,
   435→            score: def.score,
   436→            hitFlash: 0, eating: false,
   437→            walkPhase: Math.random() * Math.PI * 2,
   438→            slowTimer: 0,
   439→            paperBroken: false,
   440→          };
   441→          zombiesRef.current.push(newZombie);
   442→          waveSpawnIdxRef.current++;
   443→        }
   444→
   445→        // 检查波次结束
   446→        if (
   447→          waveElapsedRef.current >= WAVE_DURATION * 1000 &&
   448→          waveSpawnIdxRef.current >= wz.length &&
   449→          zombiesRef.current.length === 0
   450→        ) {
   451→          if (waveRef.current >= TOTAL_WAVES) {
   452→            wonRef.current = true;
   453→            submitScoreRef.current(scoreRef.current);
   454→            setPhase('won');
   455→          } else {
   456→            waveRef.current++;
   457→            waveElapsedRef.current = 0;
   458→            waveSpawnIdxRef.current = 0;
   459→            waveZombiesRef.current = getWaveZombies(waveRef.current - 1, cfg.zombieCountMul);
   460→            setCurrentWave(waveRef.current);
   461→          }
   462→        }
   463→
   464→        // 更新植物
   465→        const plants = plantsRef.current;
   466→        for (let pi = plants.length - 1; pi >= 0; pi--) {
   467→          const p = plants[pi];
   468→          if (p.dead) { plants.splice(pi, 1); continue; }
   469→          if (p.hp <= 0) { plants.splice(pi, 1); continue; }
   470→
   471→          // 樱桃炸弹引信
   472→          if (p.type === 'cherrybomb') {
   473→            p.fuseTimer += deltaMs;
   474→            if (p.fuseTimer >= CHERRY_FUSE_TIME) {
   475→              // 范围3x3爆炸
   476→              const cx = (p.col + 0.5) * cellW;
   477→              const cy = (p.row + 0.5) * cellH;
   478→              const radius = CHERRY_RADIUS_CELLS * Math.max(cellW, cellH);
   479→              const zombies = zombiesRef.current;
   480→              for (const z of zombies) {
   481→                const zy = (z.row + 0.5) * cellH;
   482→                const dx = z.x - cx;
   483→                const dy = zy - cy;
   484→                if (dx * dx + dy * dy < radius * radius) {
   485→                  z.hp -= CHERRY_DAMAGE;
   486→                  z.hitFlash = 10;
   487→                }
   488→              }
   489→              p.hp = 0;
   490→            }
   491→            continue;
   492→          }
   493→
   494→          // 大嘴花消化中
   495→          if (p.type === 'chomper') {
   496→            if (p.digesting) {
   497→              p.digestTimer += deltaMs;
   498→              if (p.digestTimer >= CHOMP_DIGEST_TIME) {
   499→                p.digesting = false;
   500→                p.digestTimer = 0;
   501→              }
   502→              continue;
   503→            }
   504→            // 尝试吃掉前方僵尸
   505→            const chompRange = cellW * 1.2;
   506→            const px = (p.col + 0.5) * cellW;
   507→            const py = (p.row + 0.5) * cellH;
   508→            for (const z of zombiesRef.current) {
   509→              if (z.hp <= 0) continue;
   510→              if (z.row !== p.row) continue;
   511→              const zy = (z.row + 0.5) * cellH;
   512→              const dx = z.x - px;
   513→              const dy = zy - py;
   514→              if (dx > 0 && dx < chompRange && Math.abs(dy) < cellH * 0.5) {
   515→                if (z.hp <= CHOMP_MAX_HP) {
   516→                  z.hp = 0;
   517→                  p.digesting = true;
   518→                  p.digestTimer = 0;
   519→                }
   520→                break;
   521→              }
   522→            }
   523→            continue;
   524→          }
   525→
   526→          // 向日葵产阳光
   527→          if (p.type === 'sunflower') {
   528→            p.sunTimer += deltaMs;
   529→            if (p.sunTimer >= SUNFLOWER_INTERVAL) {
   530→              p.sunTimer -= SUNFLOWER_INTERVAL;
   531→              const sx = (p.col + 0.5) * cellW;
   532→              const sy = p.row * cellH + cellH * 0.2;
   533→              sunsRef.current.push({
   534→                id: sunIdRef.current++,
   535→                x: sx, y: sy,
   536→                targetY: sy + 20 + Math.random() * 30,
   537→                vy: 0, collected: false, alpha: 1, bouncePhase: 0, lifeTimer: SUN_LIFE,
   538→              });
   539→            }
   540→          }
   541→
   542→          // 射手发射豌豆
   543→          if (p.type === 'peashooter' || p.type === 'doubleshoot' || p.type === 'iceshooter') {
   544→            const hasZombie = zombiesRef.current.some(
   545→              z => z.row === p.row && z.x > (p.col + 0.5) * cellW && z.hp > 0
   546→            );
   547→            if (hasZombie) {
   548→              p.shootTimer += deltaMs;
   549→              if (p.shootTimer >= SHOOT_INTERVAL) {
   550→                p.shootTimer -= SHOOT_INTERVAL;
   551→                const px = (p.col + 0.8) * cellW;
   552→                const py = (p.row + 0.5) * cellH;
   553→                const isIce = p.type === 'iceshooter';
   554→                peasRef.current.push({ row: p.row, x: px, y: py, isIce });
   555→                if (p.type === 'doubleshoot') {
   556→                  peasRef.current.push({ row: p.row, x: px - 8, y: py - 4, isIce: false });
   557→                }
   558→              }
   559→            }
   560→          }
   561→        }
   562→
   563→        // 更新豌豆
   564→        const peas = peasRef.current;
   565→        for (let i = peas.length - 1; i >= 0; i--) {
   566→          peas[i].x += PEA_SPEED * dt;
   567→        }
   568→
   569→        // 豌豆碰撞检测
   570→        for (let i = peas.length - 1; i >= 0; i--) {
   571→          const pea = peas[i];
   572→          let hit = false;
   573→          const zombies = zombiesRef.current;
   574→          for (let zi = 0; zi < zombies.length; zi++) {
   575→            const z = zombies[zi];
   576→            if (z.hp <= 0 || z.row !== pea.row) continue;
   577→            const zy = (z.row + 0.5) * cellH;
   578→            const dx = pea.x - z.x;
   579→            const dy = pea.y - zy;
   580→            const hitR = cellW * 0.4;
   581→            if (dx * dx + dy * dy < hitR * hitR) {
   582→              z.hp -= PEA_DAMAGE;
   583→              z.hitFlash = 6;
   584→              if (pea.isIce) {
   585→                z.slowTimer = ICE_SLOW_DURATION;
   586→              }
   587→              hit = true;
   588→              break;
   589→            }
   590→          }
   591→          if (hit || pea.x > LOGIC_W + 20) {
   592→            peas.splice(i, 1);
   593→          }
   594→        }
   595→
   596→        // 更新僵尸
   597→        const zombies = zombiesRef.current;
   598→        for (let i = zombies.length - 1; i >= 0; i--) {
   599→          const z = zombies[i];
   600→          if (z.hp <= 0) {
   601→            scoreRef.current += z.score;
   602→            setScore(scoreRef.current);
   603→            zombies.splice(i, 1);
   604→            continue;
   605→          }
   606→          if (z.hitFlash > 0) z.hitFlash -= dt;
   607→          z.walkPhase += 0.05 * dt;
   608→
   609→          // 减速计时
   610→          if (z.slowTimer > 0) {
   611→            z.slowTimer -= deltaMs;
   612→            z.speed = z.baseSpeed * cfg.zombieSpeedMul * ICE_SLOW_FACTOR;
   613→          } else {
   614→            z.speed = z.baseSpeed * cfg.zombieSpeedMul;
   615→          }
   616→
   617→          // 报纸僵尸：报纸破后加速
   618→          if (z.type === 'newspaper' && !z.paperBroken && z.hp <= 100) {
   619→            z.paperBroken = true;
   620→            z.baseSpeed = 0.6;
   621→            z.maxHp = 100;
   622→          }
   623→
   624→          // 碰到植物就吃
   625→          z.eating = false;
   626→          const zCol = Math.floor(z.x / cellW);
   627→          const target = plants.find(p => p.row === z.row && p.col === zCol && !p.dead && p.hp > 0);
   628→          if (target) {
   629→            z.eating = true;
   630→            target.hp -= 0.5 * dt;
   631→          } else {
   632→            z.x -= z.speed * dt;
   633→          }
   634→
   635→          // 到达最左侧
   636→          if (z.x < cellW * 0.3) {
   637→            overRef.current = true;
   638→            submitScoreRef.current(scoreRef.current);
   639→            setPhase('lost');
   640→          }
   641→        }
   642→
   643→        // 更新阳光
   644→        const suns = sunsRef.current;
   645→        for (let i = suns.length - 1; i >= 0; i--) {
   646→          const s = suns[i];
   647→          if (s.collected) {
   648→            s.alpha -= 0.05 * dt;
   649→            if (s.alpha <= 0) suns.splice(i, 1);
   650→            continue;
   651→          }
   652→          s.lifeTimer -= dt;
   653→          if (s.lifeTimer <= 0) { suns.splice(i, 1); continue; }
   654→          if (s.y < s.targetY) {
   655→            s.vy += 0.15 * dt;
   656→            s.y += s.vy * dt;
   657→            if (s.y >= s.targetY) {
   658→              s.y = s.targetY;
   659→              s.vy = -s.vy * 0.3;
   660→              s.bouncePhase++;
   661→              if (s.bouncePhase >= 3) { s.y = s.targetY; s.vy = 0; }
   662→            }
   663→          }
   664→        }
   665→      }
   666→
   667→      /* ===== 绘制 ===== */
   668→      ctx.clearRect(0, 0, LOGIC_W, LOGIC_H);
   669→
   670→      // 草坪棋盘格
   671→      for (let r = 0; r < ROWS; r++) {
   672→        for (let c = 0; c < COLS; c++) {
   673→          ctx.fillStyle = (r + c) % 2 === 0 ? '#5cb85c' : '#4cae4c';
   674→          ctx.fillRect(c * cellW, r * cellH, cellW, cellH);
   675→        }
   676→      }
   677→
   678→      // 网格线
   679→      ctx.strokeStyle = 'rgba(0,0,0,0.06)';
   680→      ctx.lineWidth = 1;
   681→      for (let r = 0; r <= ROWS; r++) {
   682→        ctx.beginPath(); ctx.moveTo(0, r * cellH); ctx.lineTo(LOGIC_W, r * cellH); ctx.stroke();
   683→      }
   684→      for (let c = 0; c <= COLS; c++) {
   685→        ctx.beginPath(); ctx.moveTo(c * cellW, 0); ctx.lineTo(c * cellW, LOGIC_H); ctx.stroke();
   686→      }
   687→
   688→      // 选中植物时的高亮
   689→      const sel = selPlantRef.current;
   690→      if (sel) {
   691→        ctx.fillStyle = 'rgba(255,255,255,0.1)';
   692→        for (let r = 0; r < ROWS; r++) {
   693→          for (let c = 0; c < COLS; c++) {
   694→            ctx.fillRect(c * cellW, r * cellH, cellW, cellH);
   695→          }
   696→        }
   697→      }
   698→
   699→      // 绘制植物
   700→      for (const p of plantsRef.current) {
   701→        const def = PLANT_DEFS.find(d => d.id === p.type);
   702→        if (!def) continue;

   704→        const cy = (p.row + 0.5) * cellH;
   705→        const fs = Math.min(cellW, cellH) * 0.65;
   706→        ctx.font = `${fs}px serif`;
   707→        ctx.textAlign = 'center';
   708→        ctx.textBaseline = 'middle';
   709→
   710→        // 樱桃炸弹闪烁效果
   711→        if (p.type === 'cherrybomb' && p.fuseTimer > 0) {
   712→          const flash = Math.sin(p.fuseTimer * 0.02) > 0;
   713→          ctx.save();
   714→          if (flash) {
   715→            ctx.fillStyle = 'rgba(255,100,0,0.3)';
   716→            ctx.beginPath();
   717→            ctx.arc(cx, cy, cellW * 0.6, 0, Math.PI * 2);
   718→            ctx.fill();
   719→          }
   720→          ctx.restore();
   721→        }
   722→
   723→        // 大嘴花消化中显示
   724→        if (p.type === 'chomper' && p.digesting) {
   725→          ctx.globalAlpha = 0.5;
   726→        }
   727→
   728→        ctx.fillText(def.emoji, cx, cy);
   729→        ctx.globalAlpha = 1;
   730→
   731→        // 大嘴花消化进度条
   732→        if (p.type === 'chomper' && p.digesting) {
   733→          const bw = cellW * 0.7;
   734→          const bx = cx - bw / 2;
   735→          const by = (p.row + 0.15) * cellH;
   736→          ctx.fillStyle = 'rgba(0,0,0,0.3)';
   737→          ctx.fillRect(bx, by, bw, 3);
   738→          const ratio = Math.min(1, p.digestTimer / CHOMP_DIGEST_TIME);
   739→          ctx.fillStyle = '#9c27b0';
   740→          ctx.fillRect(bx, by, bw * ratio, 3);
   741→        }
   742→
   743→        // 血量条（除樱桃炸弹外）
   744→        if (p.type !== 'cherrybomb' && p.hp < p.maxHp) {
   745→          const bw = cellW * 0.7;
   746→          const bx = cx - bw / 2;
   747→          const by = (p.row + 0.92) * cellH;
   748→          ctx.fillStyle = 'rgba(0,0,0,0.3)';
   749→          ctx.fillRect(bx, by, bw, 4);
   750→          const ratio = Math.max(0, p.hp / p.maxHp);
   751→          ctx.fillStyle = ratio > 0.5 ? '#4caf50' : ratio > 0.25 ? '#ff9800' : '#f44336';
   752→          ctx.fillRect(bx, by, bw * ratio, 4);
   753→        }
   754→      }
   755→
   756→      // 绘制豌豆
   757→      for (const pea of peasRef.current) {
   758→        if (pea.isIce) {
   759→          ctx.fillStyle = '#42a5f5';
   760→          ctx.beginPath();
   761→          ctx.arc(pea.x, pea.y, 5, 0, Math.PI * 2);
   762→          ctx.fill();
   763→          ctx.fillStyle = '#bbdefb';
   764→          ctx.beginPath();
   765→          ctx.arc(pea.x - 1, pea.y - 1, 2, 0, Math.PI * 2);
   766→          ctx.fill();
   767→        } else {
   768→          ctx.fillStyle = '#66bb6a';
   769→          ctx.beginPath();
   770→          ctx.arc(pea.x, pea.y, 4, 0, Math.PI * 2);
   771→          ctx.fill();
   772→          ctx.fillStyle = '#a5d6a7';
   773→          ctx.beginPath();
   774→          ctx.arc(pea.x - 1, pea.y - 1, 1.5, 0, Math.PI * 2);
   775→          ctx.fill();
   776→        }
   777→      }
   778→
   779→      // 绘制阳光
   780→      for (const s of sunsRef.current) {
   781→        ctx.save();
   782→        ctx.globalAlpha = Math.max(0, s.alpha);
   783→        const glow = ctx.createRadialGradient(s.x, s.y, 2, s.x, s.y, 22);
   784→        glow.addColorStop(0, 'rgba(255,235,59,0.8)');
   785→        glow.addColorStop(0.5, 'rgba(255,193,7,0.4)');
   786→        glow.addColorStop(1, 'rgba(255,193,7,0)');
   787→        ctx.fillStyle = glow;
   788→        ctx.beginPath(); ctx.arc(s.x, s.y, 22, 0, Math.PI * 2); ctx.fill();
   789→        ctx.fillStyle = '#ffeb3b';
   790→        ctx.beginPath(); ctx.arc(s.x, s.y, 12, 0, Math.PI * 2); ctx.fill();
   791→        ctx.fillStyle = '#fff9c4';
   792→        ctx.beginPath(); ctx.arc(s.x - 3, s.y - 3, 4, 0, Math.PI * 2); ctx.fill();
   793→        ctx.fillStyle = '#f57f17';
   794→        ctx.font = 'bold 10px sans-serif';
   795→        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
   796→        ctx.fillText('25', s.x, s.y + 1);
   797→        ctx.restore();
   798→      }
   799→
   800→      // 绘制僵尸
   801→      for (const z of zombiesRef.current) {
   802→        if (z.hp <= 0) continue;
   803→        const cy = (z.row + 0.5) * cellH;
   804→        const wobble = Math.sin(z.walkPhase) * 3;
   805→        const fs = Math.min(cellW, cellH) * 0.7;
   806→        ctx.save();
   807→        if (z.hitFlash > 0) {
   808→          ctx.filter = 'hue-rotate(-60deg) saturate(3)';
   809→        }
   810→        // 减速效果 - 蓝色边框
   811→        if (z.slowTimer > 0) {
   812→          ctx.shadowColor = '#42a5f5';
   813→          ctx.shadowBlur = 8;
   814→        }
   815→        ctx.font = `${fs}px serif`;
   816→        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
   817→        ctx.fillText(z.emoji, z.x + wobble, cy);
   818→        ctx.restore();
   819→
   820→        if (z.hp < z.maxHp) {
   821→          const bw = cellW * 0.7;
   822→          const bx = z.x - bw / 2;
   823→          const by = (z.row + 0.92) * cellH;
   824→          ctx.fillStyle = 'rgba(0,0,0,0.3)';
   825→          ctx.fillRect(bx, by, bw, 4);
   826→          const ratio = Math.max(0, z.hp / z.maxHp);
   827→          ctx.fillStyle = ratio > 0.5 ? '#e53935' : '#b71c1c';
   828→          ctx.fillRect(bx, by, bw * ratio, 4);
   829→        }
   830→      }
   831→
   832→      // 游戏结束/胜利遮罩
   833→      if (overRef.current || wonRef.current) {
   834→        ctx.fillStyle = 'rgba(0,0,0,0.6)';
   835→        ctx.fillRect(0, 0, LOGIC_W, LOGIC_H);
   836→        ctx.fillStyle = '#fff';
   837→        ctx.font = 'bold 28px sans-serif';
   838→        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
   839→        if (wonRef.current) {
   840→          ctx.fillText('胜利!', LOGIC_W / 2, LOGIC_H / 2 - 20);
   841→        } else {
   842→          ctx.fillText('游戏结束', LOGIC_W / 2, LOGIC_H / 2 - 20);
   843→        }
   844→        ctx.font = '18px sans-serif';
   845→        ctx.fillText(`得分: ${scoreRef.current}`, LOGIC_W / 2, LOGIC_H / 2 + 20);
   846→      }
   847→
   848→      rafRef.current = requestAnimationFrame(loop);
   849→    };
   850→
   851→    loopFnRef.current = loop;
   852→  }, []);
   853→
   854→  /* --- 启动/停止游戏循环 --- */
   855→  useEffect(() => {
   856→    if (phase === 'playing') {
   857→      lastTimeRef.current = 0;
   858→      // 等DOM渲染后再设置Canvas（确保canvas元素已挂载）
   859→      requestAnimationFrame(() => {
   860→        setupCanvas();
   861→        rafRef.current = requestAnimationFrame(loopFnRef.current);
   862→      });
   863→    }
   864→    return () => {
   865→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   866→    };
   867→  }, [phase, setupCanvas]);
   868→
   869→  /* --- 窗口resize --- */
   870→  useEffect(() => {
   871→    const onResize = () => { if (phase === 'playing') setupCanvas(); };
   872→    window.addEventListener('resize', onResize);
   873→    return () => window.removeEventListener('resize', onResize);
   874→  }, [phase, setupCanvas]);
   875→
   876→  /* ============ 设置面板 ============ */
   877→  if (phase === 'settings') {
   878→    return (
   879→      <div className="flex flex-col items-center w-full max-w-xl mx-auto">
   880→        <div className="w-full flex items-center gap-3 mb-6">
   881→          <Button variant="ghost" size="icon" onClick={onBack}>
   882→            <ArrowLeft className="h-5 w-5" />
   883→          </Button>
   884→          <h1 className="text-2xl font-bold">植物大战僵尸</h1>
   885→        </div>
   886→
   887→        <div className="w-full bg-card border rounded-lg p-4 md:p-6 space-y-5">
   888→          <div>
   889→            <h2 className="text-lg font-semibold mb-3">难度选择</h2>
   890→            <div className="flex gap-3">
   891→              {(['easy', 'normal', 'hard'] as Difficulty[]).map(d => (
   892→                <button
   893→                  key={d}
   894→                  onClick={() => setDifficulty(d)}
   895→                  className={`flex-1 py-3 px-4 rounded-lg border-2 text-sm font-medium transition-all ${
   896→                    difficulty === d
   897→                      ? 'border-primary bg-primary/10 text-primary'
   898→                      : 'border-muted bg-background hover:border-primary/50'
   899→                  }`}
   900→                >
   901→                  {DIFFICULTY_CONFIG[d].label}
   902→                  <span className="block text-xs text-muted-foreground mt-1">
   903→                    初始阳光 {DIFFICULTY_CONFIG[d].initSun}
   904→                  </span>
   905→                </button>
   906→              ))}
   907→            </div>
   908→          </div>
   909→
   910→          <div>
   911→            <h2 className="text-lg font-semibold mb-3">操作说明</h2>
   912→            <ol className="list-decimal list-inside space-y-2 text-sm text-muted-foreground">
   913→              <li>先点击底部植物卡片选择要种的植物</li>
   914→              <li>再点击草坪上的格子来种植</li>
   915→              <li>点击掉落的阳光来收集</li>
   916→              <li>消灭所有波次的僵尸即可获胜</li>
   917→            </ol>
   918→          </div>
   919→
   920→          <div>
   921→            <h2 className="text-lg font-semibold mb-3">植物介绍</h2>
   922→            <div className="grid grid-cols-2 sm:grid-cols-3 gap-2">
   923→              {PLANT_DEFS.map(p => (
   924→                <div key={p.id} className="flex items-center gap-2 text-sm bg-muted/50 rounded-lg p-2">
   925→                  <span className="text-2xl">{p.emoji}</span>
   926→                  <div>
   927→                    <div className="font-medium">{p.name}</div>
   928→                    <div className="text-xs text-muted-foreground">{p.desc} ({p.cost}阳光)</div>
   929→                  </div>
   930→                </div>
   931→              ))}
   932→            </div>
   933→          </div>
   934→
   935→          <div>
   936→            <h2 className="text-lg font-semibold mb-3">僵尸介绍</h2>
   937→            <div className="grid grid-cols-3 sm:grid-cols-5 gap-2">
   938→              {(Object.values(ZOMBIE_DEFS) as ZombieDef[]).map(z => (
   939→                <div key={z.type} className="flex flex-col items-center gap-1 text-sm bg-muted/50 rounded-lg p-2">
   940→                  <span className="text-2xl">{z.emoji}</span>
   941→                  <div className="font-medium text-xs">{z.name}</div>
   942→                  <div className="text-xs text-muted-foreground">血量 {z.hp}</div>

   945→            </div>

   947→
   948→          {highScore > 0 && (

   950→              <Trophy className="h-4 w-4" />
   951→              <span className="text-sm font-medium">最高分: {highScore}</span>
   952→            </div>
   953→          )}
   954→
   955→          <Button className="w-full" size="lg" onClick={startGame}>
   956→            开始游戏
   957→          </Button>
   958→        </div>
   959→      </div>

   961→  }
   962→
   963→  /* ============ 游戏界面 ============ */
   964→  return (
   965→    <div className="flex flex-col items-center w-full max-w-xl mx-auto">
   966→      {/* 顶部栏 */}
   967→      <div className="w-full flex items-center justify-between mb-2 px-1">
   968→        <Button variant="ghost" size="icon" onClick={onBack} className="shrink-0">
   969→          <ArrowLeft className="h-5 w-5" />
   970→        </Button>
   971→        <h1 className="text-lg font-bold">植物大战僵尸</h1>

   978→      {/* 阳光数量显示 */}
   979→      <div className="w-full flex items-center justify-center gap-2 mb-3 py-1.5 bg-amber-50 dark:bg-amber-950/30 border border-amber-200 dark:border-amber-800 rounded-lg">
   980→        <span className="text-xl">☀️</span>
   981→        <span className="text-lg font-bold text-amber-700 dark:text-amber-400">{sunCount}</span>

   984→      {/* Canvas游戏区域 */}
   985→      <div ref={containerRef} className="w-full" style={{ maxWidth: 640 }}>
   986→        <canvas
   987→          ref={canvasRef}
   988→          className="w-full border-2 border-green-700 dark:border-green-800 rounded-lg cursor-pointer block"
