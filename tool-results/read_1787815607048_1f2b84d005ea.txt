   154→function createStars(count: number): Star[] {
   155→  const stars: Star[] = [];
   156→  for (let i = 0; i < count; i++) {
   157→    stars.push({
   158→      x: Math.random() * LOGIC_W,
   159→      y: Math.random() * LOGIC_H,
   160→      r: Math.random() * 1.5 + 0.5,
   161→      alpha: Math.random() * 0.6 + 0.4,
   162→      twinkleSpeed: Math.random() * 0.03 + 0.01,
   163→      phase: Math.random() * Math.PI * 2,
   164→    });
   165→  }
   166→  return stars;
   167→}
   168→
   169→function createExplosion(x: number, y: number, color: string, count: number): Particle[] {
   170→  const particles: Particle[] = [];
   171→  for (let i = 0; i < count; i++) {
   172→    const angle = Math.random() * Math.PI * 2;
   173→    const speed = Math.random() * 3 + 1;
   174→    particles.push({
   175→      x, y,
   176→      vx: Math.cos(angle) * speed,
   177→      vy: Math.sin(angle) * speed,
   178→      life: Math.random() * 20 + 15,
   179→      maxLife: 35,
   180→      color,
   181→      r: Math.random() * 3 + 1,
   182→    });
   183→  }
   184→  return particles;
   185→}

   187→function rectCollide(
   188→  ax: number, ay: number, aw: number, ah: number,
   189→  bx: number, by: number, bw: number, bh: number,
   190→): boolean {
   191→  return Math.abs(ax - bx) < (aw + bw) / 2 && Math.abs(ay - by) < (ah + bh) / 2;
   192→}

   196→function drawPlayer(ctx: CanvasRenderingContext2D, p: Player, frame: number) {
   197→  ctx.save();
   198→  ctx.translate(p.x, p.y);
   199→
   200→  // Invincibility blink
   201→  if (p.invincibleTimer > 0 && Math.floor(frame / 3) % 2 === 0) {
   202→    ctx.globalAlpha = 0.35;
   203→  }
   204→
   205→  // Tail flame animation
   206→  const flameLen = 12 + Math.sin(frame * 0.5) * 4;
   207→  const flameW = 8 + Math.sin(frame * 0.7) * 2;
   208→  const flameGrad = ctx.createLinearGradient(0, PLAYER_H / 2, 0, PLAYER_H / 2 + flameLen);
   209→  flameGrad.addColorStop(0, '#38bdf8');
   210→  flameGrad.addColorStop(0.4, '#f97316');
   211→  flameGrad.addColorStop(1, 'rgba(239,68,68,0)');
   212→  ctx.fillStyle = flameGrad;
   213→  ctx.beginPath();
   214→  ctx.moveTo(-flameW, PLAYER_H / 2 - 2);
   215→  ctx.lineTo(flameW, PLAYER_H / 2 - 2);
   216→  ctx.lineTo(0, PLAYER_H / 2 + flameLen);
   217→  ctx.closePath();
   218→  ctx.fill();
   219→
   220→  // Aircraft body
   221→  ctx.fillStyle = '#e0f2fe';
   222→  ctx.beginPath();
   223→  ctx.moveTo(0, -PLAYER_H / 2);
   224→  ctx.lineTo(-PLAYER_W / 2, PLAYER_H / 2 - 6);
   225→  ctx.lineTo(-PLAYER_W / 4, PLAYER_H / 2);
   226→  ctx.lineTo(PLAYER_W / 4, PLAYER_H / 2);
   227→  ctx.lineTo(PLAYER_W / 2, PLAYER_H / 2 - 6);
   228→  ctx.closePath();
   229→  ctx.fill();
   230→
   231→  // Wing decoration lines
   232→  ctx.strokeStyle = '#38bdf8';
   233→  ctx.lineWidth = 1.5;
   234→  ctx.beginPath();
   235→  ctx.moveTo(-PLAYER_W / 2 + 4, PLAYER_H / 2 - 10);
   236→  ctx.lineTo(0, -PLAYER_H / 2 + 10);
   237→  ctx.lineTo(PLAYER_W / 2 - 4, PLAYER_H / 2 - 10);
   238→  ctx.stroke();
   239→
   240→  // Cockpit
   241→  ctx.fillStyle = '#7dd3fc';
   242→  ctx.beginPath();
   243→  ctx.ellipse(0, -PLAYER_H / 4, 4, 7, 0, 0, Math.PI * 2);
   244→  ctx.fill();
   245→
   246→  // Shield bubble
   247→  if (p.shieldTimer > 0) {
   248→    const shieldAlpha = p.shieldTimer < 60 ? (Math.sin(frame * 0.3) * 0.3 + 0.4) : 0.5;
   249→    ctx.strokeStyle = `rgba(6, 182, 212, ${shieldAlpha})`;
   250→    ctx.lineWidth = 2.5;
   251→    ctx.beginPath();
   252→  ctx.ellipse(0, 0, PLAYER_W / 2 + 8, PLAYER_H / 2 + 8, 0, 0, Math.PI * 2);
   253→    ctx.stroke();
   254→  }
   255→
   256→  ctx.restore();
   257→}

   259→function drawEnemy(ctx: CanvasRenderingContext2D, e: Enemy) {
   260→  ctx.save();
   261→  ctx.translate(e.x, e.y);
   262→
   263→  if (e.type === EnemyType.Normal) {
   264→    // Red inverted triangle
   265→    ctx.fillStyle = '#ef4444';
   266→    ctx.beginPath();
   267→    ctx.moveTo(0, e.h / 2);
   268→    ctx.lineTo(-e.w / 2, -e.h / 2);
   269→    ctx.lineTo(e.w / 2, -e.h / 2);
   270→    ctx.closePath();
   271→    ctx.fill();
   272→    ctx.strokeStyle = '#fca5a5';
   273→    ctx.lineWidth = 1;
   274→    ctx.beginPath();
   275→    ctx.moveTo(0, e.h / 2 - 6);
   276→    ctx.lineTo(-e.w / 4, -e.h / 4);
   277→    ctx.moveTo(0, e.h / 2 - 6);
   278→    ctx.lineTo(e.w / 4, -e.h / 4);
   279→    ctx.stroke();
   280→  } else if (e.type === EnemyType.Fast) {
   281→    // Yellow arrow shape
   282→    ctx.fillStyle = '#f59e0b';
   283→    ctx.beginPath();
   284→    ctx.moveTo(0, e.h / 2);
   285→    ctx.lineTo(-e.w / 2, -e.h / 3);
   286→    ctx.lineTo(-e.w / 4, -e.h / 2);
   287→    ctx.lineTo(e.w / 4, -e.h / 2);
   288→    ctx.lineTo(e.w / 2, -e.h / 3);
   289→    ctx.closePath();
   290→    ctx.fill();
   291→    ctx.fillStyle = '#fcd34d';
   292→    ctx.beginPath();
   293→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   294→    ctx.fill();
   295→  } else if (e.type === EnemyType.Tank) {
   296→    // Purple hexagon
   297→    const r = e.w / 2;
   298→    ctx.fillStyle = '#a855f7';
   299→    ctx.beginPath();
   300→    for (let i = 0; i < 6; i++) {
   301→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   302→      const px = Math.cos(angle) * r;
   303→      const py = Math.sin(angle) * r;
   304→      if (i === 0) ctx.moveTo(px, py);
   305→      else ctx.lineTo(px, py);
   306→    }
   307→    ctx.closePath();
   308→    ctx.fill();
   309→    ctx.strokeStyle = '#d8b4fe';
   310→    ctx.lineWidth = 1.5;
   311→    ctx.beginPath();
   312→    for (let i = 0; i < 6; i++) {
   313→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   314→      const px = Math.cos(angle) * (r * 0.55);
   315→      const py = Math.sin(angle) * (r * 0.55);
   316→      if (i === 0) ctx.moveTo(px, py);
   317→      else ctx.lineTo(px, py);
   318→    }
   319→    ctx.closePath();
   320→    ctx.stroke();
   321→    ctx.fillStyle = '#e9d5ff';
   322→    ctx.beginPath();
   323→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   324→    ctx.fill();
   325→  }
   326→
   327→  ctx.restore();
   328→
   329→  // HP bar (only when damaged)
   330→  if (e.hp < e.maxHp) {
   331→    const barW = e.w * 0.8;
   332→    const barH = 3;
   333→    const barX = e.x - barW / 2;
   334→    const barY = e.y - e.h / 2 - 6;
   335→    ctx.fillStyle = 'rgba(255,255,255,0.2)';
   336→    ctx.fillRect(barX, barY, barW, barH);
   337→    ctx.fillStyle = '#22c55e';
   338→    ctx.fillRect(barX, barY, barW * (e.hp / e.maxHp), barH);
   339→  }
   340→}

   342→function drawBoss(ctx: CanvasRenderingContext2D, boss: Boss, frame: number) {
   343→  ctx.save();
   344→  ctx.translate(boss.x, boss.y);
   345→
   346→  // Flash white on hit
   347→  if (boss.flashTimer > 0) {
   348→    ctx.globalAlpha = 0.5 + Math.sin(frame) * 0.5;
   349→    // White overlay circle
   350→    ctx.fillStyle = 'rgba(255,255,255,0.6)';
   351→    ctx.beginPath();
   352→    ctx.ellipse(0, 0, BOSS_W / 2 + 4, BOSS_H / 2 + 4, 0, 0, Math.PI * 2);
   353→    ctx.fill();
   354→    ctx.globalAlpha = 1;
   355→  }
   356→
   357→  // Boss emoji
   358→  ctx.font = `${BOSS_H}px sans-serif`;
   359→  ctx.textAlign = 'center';
   360→  ctx.textBaseline = 'middle';
   361→  ctx.fillText(boss.emoji, 0, 2);
   362→
   363→  ctx.restore();
   364→
   365→  // Boss HP bar at top of canvas
   366→  const barW = LOGIC_W * 0.6;
   367→  const barH = 8;
   368→  const barX = (LOGIC_W - barW) / 2;
   369→  const barY = 6;
   370→
   371→  ctx.fillStyle = 'rgba(255,255,255,0.15)';
   372→  ctx.fillRect(barX - 1, barY - 1, barW + 2, barH + 2);
   373→
   374→  const hpRatio = boss.hp / boss.maxHp;
   375→  const hpColor = hpRatio > 0.5 ? '#ef4444' : hpRatio > 0.25 ? '#f97316' : '#fbbf24';
   376→  ctx.fillStyle = hpColor;
   377→  ctx.fillRect(barX, barY, barW * hpRatio, barH);
   378→
   379→  ctx.strokeStyle = 'rgba(255,255,255,0.4)';
   380→  ctx.lineWidth = 1;
   381→  ctx.strokeRect(barX - 1, barY - 1, barW + 2, barH + 2);
   382→
   383→  ctx.fillStyle = '#ffffff';
   384→  ctx.font = 'bold 11px sans-serif';
   385→  ctx.textAlign = 'center';
   386→  ctx.textBaseline = 'top';
   387→  ctx.fillText('BOSS', LOGIC_W / 2, barY + barH + 2);
   388→}

   390→function drawPowerUp(ctx: CanvasRenderingContext2D, pw: PowerUp) {
   391→  const glow = 0.3 + Math.sin(pw.phase) * 0.2;
   392→  const color = POWERUP_COLORS[pw.type];
   393→
   394→  // Outer glow
   395→  ctx.save();
   396→  ctx.globalAlpha = glow;
   397→  ctx.fillStyle = color;
   398→  ctx.beginPath();
   399→  ctx.arc(pw.x, pw.y, 16, 0, Math.PI * 2);
   400→  ctx.fill();
   401→  ctx.restore();
   402→
   403→  // Inner circle
   404→  ctx.fillStyle = color;
   405→  ctx.beginPath();
   406→  ctx.arc(pw.x, pw.y, 12, 0, Math.PI * 2);
   407→  ctx.fill();
   408→
   409→  // Emoji icon
   410→  ctx.font = '14px sans-serif';
   411→  ctx.textAlign = 'center';
   412→  ctx.textBaseline = 'middle';
   413→  ctx.fillText(POWERUP_EMOJIS[pw.type], pw.x, pw.y);
   414→}

   417→  ctx.save();

   517→
   518→    const tankChance = Math.min(0.1 + lv * 0.04, 0.35);
   519→    const fastChance = Math.min(0.15 + lv * 0.03, 0.4);
   520→
   521→    if (rand < tankChance) {
   522→      type = EnemyType.Tank;
   523→    } else if (rand < tankChance + fastChance) {
   524→      type = EnemyType.Fast;
   525→    } else {
   526→      type = EnemyType.Normal;
   527→    }
   528→
   529→    const cfg = ENEMY_CONFIG[type];
   530→    const speedMul = 1 + (lv - 1) * 0.12;
   531→    const x = Math.random() * (LOGIC_W - cfg.w - 20) + cfg.w / 2 + 10;
   532→
   533→    enemiesRef.current.push({
   534→      x, y: -cfg.h,
   535→      type,
   536→      hp: cfg.hp,
   537→      maxHp: cfg.hp,
   538→      speed: cfg.speed * speedMul,
   539→      w: cfg.w,
   540→      h: cfg.h,
   541→      score: cfg.score,
   542→    });
   543→  }, []);
   544→
   545→  /** Spawn a boss */
   546→  const spawnBoss = useCallback(() => {
   547→    const idx = bossSpawnedCountRef.current;
   548→    const emoji = BOSS_EMOJIS[idx % BOSS_EMOJIS.length];
   549→    const hp = 30 + idx * 15;
   550→    const score = 200 + idx * 100;
   551→
   552→    bossRef.current = {
   553→      x: LOGIC_W / 2,
   554→      y: -BOSS_H,
   555→      hp,
   556→      maxHp: hp,
   557→      bossIndex: idx,
   558→      emoji,
   559→      flashTimer: 0,
   560→      shootTimer: 0,
   561→      dirX: 1.5,
   562→      score,
   563→      entered: false,
   564→    };
   565→    bossSpawnedCountRef.current++;
   566→    bossKillCounterRef.current = 0;
   567→  }, []);
   568→
   569→  /** Fire player bullets based on weapon level */
   570→  const shoot = useCallback(() => {
   571→    const p = playerRef.current;
   572→    const speed = 8;
   573→    const bx = p.x;
   574→    const by = p.y - PLAYER_H / 2;
   575→    const bullets = bulletsRef.current;
   576→
   577→    switch (p.weaponLevel) {
   578→      case 1:
   579→        bullets.push({ x: bx, y: by, vx: 0, vy: -speed });
   580→        break;
   581→      case 2:
   582→        bullets.push(
   583→          { x: bx - 8, y: by, vx: 0, vy: -speed },
   584→          { x: bx + 8, y: by, vx: 0, vy: -speed },
   585→        );
   586→        break;
   587→      case 3: {
   588→        const a = Math.PI / 12; // 15 degrees
   589→        const svx = Math.sin(a) * speed;
   590→        const svy = -Math.cos(a) * speed;
   591→        bullets.push(
   592→          { x: bx, y: by, vx: 0, vy: -speed },
   593→          { x: bx, y: by, vx: svx, vy: svy },
   594→          { x: bx, y: by, vx: -svx, vy: svy },
   595→        );
   596→        break;
   597→      }
   598→      case 4: {
   599→        const a1 = Math.PI / 18; // 10 degrees
   600→        const a2 = Math.PI / 9;  // 20 degrees
   601→        bullets.push(
   602→          { x: bx, y: by, vx: 0, vy: -speed },
   603→          { x: bx, y: by, vx: Math.sin(a1) * speed, vy: -Math.cos(a1) * speed },
   604→          { x: bx, y: by, vx: -Math.sin(a1) * speed, vy: -Math.cos(a1) * speed },
   605→          { x: bx, y: by, vx: Math.sin(a2) * speed, vy: -Math.cos(a2) * speed },
   606→          { x: bx, y: by, vx: -Math.sin(a2) * speed, vy: -Math.cos(a2) * speed },
   607→        );
   608→        break;
   609→      }
   610→      default:
   611→        break;
   612→    }
   613→  }, []);
   614→
   615→  /** Get shoot interval based on weapon level */
   616→  const getShootInterval = useCallback(() => {
   617→    return Math.max(BASE_SHOOT_INTERVAL - playerRef.current.weaponLevel + 1, 6);
   618→  }, []);
   619→
   620→  /** Get enemy spawn interval based on difficulty */
   621→  const getSpawnInterval = useCallback(() => {
   622→    const lv = levelRef.current;
   623→    return Math.max(45 - (lv - 1) * 4, 15);
   624→  }, []);
   625→
   626→  /** Activate bomb powerup effect */
   627→  const activateBomb = useCallback(() => {
   628→    screenFlashRef.current = SCREEN_FLASH_FRAMES;
   629→
   630→    // Destroy all regular enemies
   631→    for (const e of enemiesRef.current) {
   632→      scoreRef.current += e.score;
   633→      const expColor = e.type === EnemyType.Fast ? '#f59e0b'
   634→        : e.type === EnemyType.Tank ? '#a855f7' : '#ef4444';
   635→      particlesRef.current.push(...createExplosion(e.x, e.y, expColor, 10));
   636→    }
   637→    setScore(scoreRef.current);
   638→    enemiesRef.current = [];
   639→
   640→    // Clear enemy bullets
   641→    enemyBulletsRef.current = [];
   642→
   643→    // Damage boss by 30% if present
   644→    const boss = bossRef.current;
   645→    if (boss) {
   646→      const dmg = Math.ceil(boss.maxHp * 0.3);
   647→      boss.hp = Math.max(1, boss.hp - dmg);
   648→      boss.flashTimer = 10;
   649→      particlesRef.current.push(...createExplosion(boss.x, boss.y, '#fbbf24', 20));
   650→
   651→      if (boss.hp <= 1) {
   652→        // Boss killed by bomb
   653→        scoreRef.current += boss.score;
   654→        setScore(scoreRef.current);
   655→        particlesRef.current.push(...createExplosion(boss.x, boss.y, '#f97316', 30));
   656→        // Drop powerups
   657→        const dropTypes = [PowerUpType.Weapon, PowerUpType.Heal, PowerUpType.Shield];
   658→        for (const dt of dropTypes) {
   659→          powerupsRef.current.push({
   660→            x: boss.x + (Math.random() - 0.5) * 60,
   661→            y: boss.y + (Math.random() - 0.5) * 30,
   662→            type: dt,
   663→            phase: Math.random() * Math.PI * 2,
   664→          });
   665→        }
   666→        bossRef.current = null;
   667→      }
   668→    }
   669→  }, []);
   670→
   671→  /** Main game loop */
   672→  const gameLoop = useCallback(() => {
   673→    const canvas = canvasRef.current;
   674→    if (!canvas) return;
   675→    const ctx = canvas.getContext('2d');
   676→    if (!ctx) return;
   677→
   678→    ctx.setTransform(dprRef.current * scaleRef.current, 0, 0, dprRef.current * scaleRef.current, 0, 0);
   679→
   680→    const frame = frameRef.current;
   681→    frameRef.current++;
   682→    const p = playerRef.current;
   683→    const boss = bossRef.current;
   684→
   685→    // ==== UPDATE LOGIC ====
   686→
   687→    // Twinkling stars
   688→    starsRef.current.forEach((s) => { s.phase += s.twinkleSpeed; });
   689→
   690→    if (statusRef.current === 'playing' && !pausedRef.current) {
   691→      // Move player toward input
   692→      if (inputXRef.current !== null) {
   693→        const target = inputXRef.current;
   694→        const diff = target - p.x;
   695→        if (Math.abs(diff) > 2) {
   696→          p.x += diff * 0.15;
   697→        } else {
   698→          p.x = target;
   699→        }
   700→      }
   701→      p.x = Math.max(PLAYER_W / 2, Math.min(LOGIC_W - PLAYER_W / 2, p.x));
   702→
   703→      // Update timers
   704→      if (p.shieldTimer > 0) p.shieldTimer--;
   705→      if (p.invincibleTimer > 0) p.invincibleTimer--;
   706→      if (screenFlashRef.current > 0) screenFlashRef.current--;
   707→
   708→      // Shooting
   709→      const isMobile = isMobileRef.current;
   710→      const interval = getShootInterval();
   711→      if (isMobile) {
   712→        if (frame - lastShootRef.current >= interval) {
   713→          shoot();
   714→          lastShootRef.current = frame;
   715→        }
   716→      } else {
   717→        if (shootRequestRef.current) {
   718→          if (frame - lastShootRef.current >= interval) {
   719→            shoot();
   720→            lastShootRef.current = frame;
   721→          }
   722→        }
   723→      }
   724→      shootRequestRef.current = false;
   725→
   726→      // Spawn regular enemies (reduced rate during boss)
   727→      const spawnInterval = boss ? getSpawnInterval() * 1.5 : getSpawnInterval();
   728→      if (frame % Math.round(spawnInterval) === 0) {
   729→        spawnEnemy();
   730→      }
   731→
   732→      // Check boss spawn (every 30 kills, no boss currently active)
   733→      if (bossKillCounterRef.current >= BOSS_SPAWN_KILLS && !boss) {
   734→        spawnBoss();
   735→      }
   736→
   737→      // Update player bullets
   738→      bulletsRef.current = bulletsRef.current.filter((b) => {
   739→        b.x += b.vx;
   740→        b.y += b.vy;
   741→        return b.y > -BULLET_H && b.x > -BULLET_W && b.x < LOGIC_W + BULLET_W;
   742→      });
   743→
   744→      // Update enemy bullets
   745→      enemyBulletsRef.current = enemyBulletsRef.current.filter((b) => {
   746→        b.x += b.vx;
   747→        b.y += b.vy;
   748→        return b.y < LOGIC_H + 10 && b.x > -10 && b.x < LOGIC_W + 10;
   749→      });
   750→
   751→      // Update regular enemies
   752→      enemiesRef.current = enemiesRef.current.filter((e) => {
   753→        e.y += e.speed;
   754→        return e.y < LOGIC_H + e.h;
   755→      });
   756→
   757→      // Update boss
   758→      if (boss) {
   759→        if (!boss.entered) {
   760→          boss.y += 1.5;
   761→          if (boss.y >= 80) {
   762→            boss.y = 80;
   763→            boss.entered = true;
   764→          }
   765→        } else {
   766→          // Side-to-side movement
   767→          boss.x += boss.dirX;
   768→          if (boss.x > LOGIC_W - BOSS_W / 2 - 10) boss.dirX = -Math.abs(boss.dirX);
   769→          if (boss.x < BOSS_W / 2 + 10) boss.dirX = Math.abs(boss.dirX);
   770→
   771→          // Boss shooting
   772→          boss.shootTimer++;
   773→          if (boss.shootTimer >= BOSS_SHOOT_INTERVAL) {
   774→            boss.shootTimer = 0;
   775→            for (let i = -1; i <= 1; i++) {
   776→              const angle = Math.PI / 2 + i * BOSS_SPREAD_ANGLE;
   777→              enemyBulletsRef.current.push({
   778→                x: boss.x,
   779→                y: boss.y + BOSS_H / 2,
   780→                vx: Math.cos(angle) * BOSS_BULLET_SPEED,
   781→                vy: Math.sin(angle) * BOSS_BULLET_SPEED,
   782→              });
   783→            }
   784→          }
   785→        }
   786→        if (boss.flashTimer > 0) boss.flashTimer--;
   787→      }
   788→
   789→      // Update powerups
   790→      powerupsRef.current = powerupsRef.current.filter((pw) => {
   791→        pw.y += 2;
   792→        pw.phase += 0.1;
   793→        return pw.y < LOGIC_H + 20;
   794→      });
   795→
   796→      // === COLLISION: player bullets vs enemies ===
   797→      const bullets = bulletsRef.current;
   798→      const enemies = enemiesRef.current;
   799→      const hitBullets = new Set<number>();
   800→      const hitEnemies = new Set<number>();
   801→
   802→      for (let bi = 0; bi < bullets.length; bi++) {
   803→        const b = bullets[bi];
   804→        for (let ei = 0; ei < enemies.length; ei++) {
   805→          if (hitEnemies.has(ei)) continue;
   806→          const e = enemies[ei];
   807→          if (rectCollide(b.x, b.y, BULLET_W, BULLET_H, e.x, e.y, e.w, e.h)) {
   808→            hitBullets.add(bi);
   809→            e.hp--;
   810→            if (e.hp <= 0) {
   811→              hitEnemies.add(ei);
   812→              scoreRef.current += e.score;
   813→              killCountRef.current++;
   814→              bossKillCounterRef.current++;
   815→              setScore(scoreRef.current);
   816→
   817→              const expColor = e.type === EnemyType.Fast ? '#f59e0b'
   818→                : e.type === EnemyType.Tank ? '#a855f7' : '#ef4444';
   819→              const expCount = e.type === EnemyType.Tank ? 18 : e.type === EnemyType.Normal ? 8 : 6;
   820→              particlesRef.current.push(...createExplosion(e.x, e.y, expColor, expCount));
   821→
   822→              // Powerup drop (20%)
   823→              if (Math.random() < POWERUP_CHANCE) {
   824→                const types = [PowerUpType.Weapon, PowerUpType.Heal, PowerUpType.Bomb, PowerUpType.Shield];
   825→                const weights = [0.35, 0.25, 0.15, 0.25];
   826→                let r = Math.random();
   827→                let chosen = PowerUpType.Shield;
   828→                for (let ti = 0; ti < types.length; ti++) {
   829→                  r -= weights[ti];
   830→                  if (r <= 0) { chosen = types[ti]; break; }
   831→                }
   832→                powerupsRef.current.push({ x: e.x, y: e.y, type: chosen, phase: 0 });
   833→              }
   834→
   835→              // Difficulty level up
   836→              if (killCountRef.current % LEVEL_UP_KILLS === 0) {
   837→                levelRef.current++;
   838→                setLevel(levelRef.current);
   839→              }
   840→            } else {
   841→              particlesRef.current.push(...createExplosion(b.x, b.y, '#fbbf24', 3));
   842→            }
   843→            break;
   844→          }
   845→        }
   846→      }
   847→
   848→      bulletsRef.current = bullets.filter((_, i) => !hitBullets.has(i));
   849→      enemiesRef.current = enemies.filter((_, i) => !hitEnemies.has(i));
   850→
   851→      // === COLLISION: player bullets vs boss ===
   852→      if (boss && statusRef.current === 'playing') {
   853→        const bossBullets = bulletsRef.current;
   854→        const bossHitBullets = new Set<number>();
   855→        for (let bi = 0; bi < bossBullets.length; bi++) {
   856→          const b = bossBullets[bi];
   857→          if (rectCollide(b.x, b.y, BULLET_W, BULLET_H, boss.x, boss.y, BOSS_W, BOSS_H)) {
   858→            bossHitBullets.add(bi);
   859→            boss.hp--;
   860→            boss.flashTimer = 4;
   861→            particlesRef.current.push(...createExplosion(b.x, b.y, '#fbbf24', 2));
   862→
   863→            if (boss.hp <= 0) {
   864→              // Boss defeated!
   865→              scoreRef.current += boss.score;
   866→              setScore(scoreRef.current);
   867→              particlesRef.current.push(...createExplosion(boss.x, boss.y, '#f97316', 35));
   868→
   869→              // Drop 3 powerups
   870→              const dropTypes = [PowerUpType.Weapon, PowerUpType.Heal, PowerUpType.Shield];
   871→              for (const dt of dropTypes) {
   872→                powerupsRef.current.push({
   873→                  x: boss.x + (Math.random() - 0.5) * 80,
   874→                  y: boss.y + (Math.random() - 0.5) * 30,
   875→                  type: dt,
   876→                  phase: Math.random() * Math.PI * 2,
   877→                });
   878→              }
   879→              bossRef.current = null;
   880→              break;
   881→            }
   882→          }
   883→        }
   884→        if (bossRef.current !== null) {
   885→          bulletsRef.current = bossBullets.filter((_, i) => !bossHitBullets.has(i));
   886→        } else {
   887→          // Boss was just killed, remove all hit bullets
   888→          bulletsRef.current = bossBullets.filter((_, i) => !bossHitBullets.has(i));
   889→        }
   890→      }
   891→
   892→      // === COLLISION: enemy bullets vs player ===
   893→      if (statusRef.current === 'playing') {
   894→        for (const eb of enemyBulletsRef.current) {
   895→          if (rectCollide(p.x, p.y, PLAYER_W * 0.5, PLAYER_H * 0.5, eb.x, eb.y, 8, 8)) {
   896→            damagePlayer();
   897→            break;
   898→          }
   899→        }
   900→      }
   901→
   902→      // === COLLISION: enemies vs player ===
   903→      if (statusRef.current === 'playing') {
   904→        for (const e of enemiesRef.current) {
   905→          if (rectCollide(p.x, p.y, PLAYER_W * 0.6, PLAYER_H * 0.6, e.x, e.y, e.w, e.h)) {
   906→            damagePlayer();
   907→            // Destroy the enemy on contact
   908→            e.hp = 0;
   909→            break;
   910→          }
   911→        }
   912→        enemiesRef.current = enemiesRef.current.filter((e) => e.hp > 0);
   913→      }
   914→
   915→      // === COLLISION: boss vs player ===
   916→      if (boss && statusRef.current === 'playing') {
   917→        if (rectCollide(p.x, p.y, PLAYER_W * 0.6, PLAYER_H * 0.6, boss.x, boss.y, BOSS_W, BOSS_H)) {
   918→          damagePlayer();
   919→        }
   920→      }
   921→
   922→      // === COLLISION: powerups vs player ===
   923→      if (statusRef.current === 'playing') {
   924→        powerupsRef.current = powerupsRef.current.filter((pw) => {
   925→          if (rectCollide(p.x, p.y, PLAYER_W, PLAYER_H, pw.x, pw.y, 24, 24)) {
   926→            switch (pw.type) {
   927→              case PowerUpType.Weapon:
   928→                if (p.weaponLevel < MAX_WEAPON_LEVEL) {
   929→                  p.weaponLevel++;
   930→                  setWeaponLevel(p.weaponLevel);
   931→                }
   932→                break;
   933→              case PowerUpType.Heal:
   934→                if (p.lives < MAX_LIVES) {
   935→                  p.lives++;
   936→                  setLives(p.lives);
   937→                }
   938→                break;
   939→              case PowerUpType.Bomb:
   940→                activateBomb();
   941→                break;
   942→              case PowerUpType.Shield:
   943→                p.shieldTimer = SHIELD_DURATION;
   944→                break;
   945→            }
   946→            const pColor = POWERUP_COLORS[pw.type];
   947→            particlesRef.current.push(...createExplosion(pw.x, pw.y, pColor, 10));
   948→            return false;
   949→          }
   950→          return true;
   951→        });
   952→      }
   953→    }
   954→
   955→    // Submit score on game over (outside the playing block to avoid TS narrowing)
   956→    if (statusRef.current === 'over' && !scoreSubmittedRef.current) {
   957→      scoreSubmittedRef.current = true;
   958→      if (scoreRef.current > highScoreRef.current) {
   959→        setIsNewRecord(true);
   960→        fetch('/api/games/scores', {
   961→          method: 'POST',
   962→          headers: { 'Content-Type': 'application/json' },
   963→          body: JSON.stringify({ game: GAME_NAME, score: scoreRef.current }),
   964→        })
   965→          .then((r) => r.json())
   966→          .then((data) => {
   967→            if (data.data?.score !== undefined) {
   968→              setHighScore(data.data.score);
   969→              highScoreRef.current = data.data.score;
   970→            }
   971→          })
   972→          .catch(() => {});
   973→      }
   974→    }
   975→
   976→    // Update particles
   977→    particlesRef.current = particlesRef.current.filter((pt) => {
   978→      pt.x += pt.vx;
   979→      pt.y += pt.vy;
   980→      pt.life--;
   981→      pt.vx *= 0.97;
   982→      pt.vy *= 0.97;
   983→      return pt.life > 0;
   984→    });
   985→
   986→    // ==== DRAW LOGIC ====
   987→
   988→    // Deep space background
   989→    const bgGrad = ctx.createLinearGradient(0, 0, 0, LOGIC_H);
   990→    bgGrad.addColorStop(0, '#0a0e27');
   991→    bgGrad.addColorStop(0.5, '#0f1638');
   992→    bgGrad.addColorStop(1, '#0a0e27');
   993→    ctx.fillStyle = bgGrad;
   994→    ctx.fillRect(0, 0, LOGIC_W, LOGIC_H);
   995→
   996→    // Stars
   997→    starsRef.current.forEach((s) => {
   998→      const alpha = s.alpha * (0.5 + Math.sin(s.phase) * 0.5);
   999→      ctx.fillStyle = `rgba(255, 255, 255, ${alpha})`;
  1000→      ctx.beginPath();
  1001→      ctx.arc(s.x, s.y, s.r, 0, Math.PI * 2);
  1002→      ctx.fill();
  1003→    });
  1004→
  1005→    // Powerups
  1006→    powerupsRef.current.forEach((pw) => drawPowerUp(ctx, pw));
  1007→
  1008→    // Player bullets (cyan glow)
  1009→    ctx.shadowColor = '#38bdf8';
  1010→    ctx.shadowBlur = 6;
  1011→    ctx.fillStyle = '#7dd3fc';
  1012→    bulletsRef.current.forEach((b) => {
  1013→      ctx.fillRect(b.x - BULLET_W / 2, b.y - BULLET_H / 2, BULLET_W, BULLET_H);
  1014→    });
  1015→    ctx.shadowBlur = 0;
  1016→
  1017→    // Enemy bullets
  1018→    enemyBulletsRef.current.forEach((b) => drawEnemyBullet(ctx, b));
  1019→
  1020→    // Regular enemies
  1021→    enemiesRef.current.forEach((e) => drawEnemy(ctx, e));
  1022→
  1023→    // Boss
  1024→    if (bossRef.current) {
  1025→      drawBoss(ctx, bossRef.current, frame);
  1026→    }
  1027→
  1028→    // Particles
  1029→    particlesRef.current.forEach((pt) => {
  1030→      const alpha = pt.life / pt.maxLife;
  1031→      ctx.globalAlpha = alpha;
  1032→      ctx.fillStyle = pt.color;
  1033→      ctx.beginPath();
  1034→      ctx.arc(pt.x, pt.y, pt.r * alpha, 0, Math.PI * 2);
  1035→      ctx.fill();
  1036→    });
  1037→    ctx.globalAlpha = 1;
  1038→
  1039→    // Player (not drawn when dead)
  1040→    if (statusRef.current !== 'over') {
  1041→      drawPlayer(ctx, p, frame);
  1042→    }
  1043→
  1044→    // Canvas HUD (lives + weapon level)
  1045→    if (statusRef.current === 'playing' || statusRef.current === 'over') {
  1046→      drawCanvasHUD(ctx, p, !!bossRef.current);
  1047→    }
  1048→
