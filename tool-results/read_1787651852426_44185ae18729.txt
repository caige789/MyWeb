   137→/** 生成某一关的砖块布局 */
   138→function generateBricks(level: number, canvasW: number): Brick[] {
   139→  const bricks: Brick[] = [];
   140→  const brickW = (canvasW - (BRICK_COLS + 1) * BRICK_GAP) / BRICK_COLS;
   141→
   142→  if (level === 1) {
   143→    // 第一关：5行标准排列
   144→    for (let row = 0; row < 5; row++) {
   145→      const typeIndex = row;
   146→      for (let col = 0; col < BRICK_COLS; col++) {
   147→        bricks.push({
   148→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   149→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   150→          w: brickW,
   151→          h: BRICK_H,
   152→          typeIndex,
   153→          alive: true,
   154→        });
   155→      }
   156→    }
   157→  } else if (level === 2) {
   158→    // 第二关：6行，部分位置空缺形成菱形
   159→    for (let row = 0; row < 6; row++) {
   160→      const typeIndex = Math.min(row, BRICK_TYPES.length - 1);
   161→      for (let col = 0; col < BRICK_COLS; col++) {
   162→        // 菱形空缺：中间两行各跳过2个砖块
   163→        if (row === 2 && (col === 0 || col === 7)) continue;
   164→        if (row === 3 && (col === 3 || col === 4)) continue;
   165→        bricks.push({
   166→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   167→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   168→          w: brickW,
   169→          h: BRICK_H,
   170→          typeIndex,
   171→          alive: true,
   172→        });
   173→      }
   174→    }
   175→  } else {
   176→    // 第三关：7行，交错排列，高密度
   177→    for (let row = 0; row < 7; row++) {
   178→      const typeIndex = Math.min(row, BRICK_TYPES.length - 1);
   179→      for (let col = 0; col < BRICK_COLS; col++) {
   180→        // 交错空缺
   181→        if (row % 2 === 0 && col % 2 === 0) continue;
   182→        if (row === 1 && col === 4) continue;
   183→        if (row === 4 && col === 2) continue;
   184→        bricks.push({
   185→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   186→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   187→          w: brickW,
   188→          h: BRICK_H,
   189→          typeIndex,
   190→          alive: true,
   191→        });
   192→      }
   193→    }
   194→  }
   195→  return bricks;
   196→}

   198→/** 绘制圆角矩形路径 */
   199→function roundRect(
   200→  ctx: CanvasRenderingContext2D,
   201→  x: number,
   202→  y: number,
   203→  w: number,
   204→  h: number,
   205→  r: number
   206→) {
   207→  const rr = Math.min(r, w / 2, h / 2);
   208→  ctx.beginPath();
   209→  ctx.moveTo(x + rr, y);
   210→  ctx.lineTo(x + w - rr, y);
   211→  ctx.quadraticCurveTo(x + w, y, x + w, y + rr);
   212→  ctx.lineTo(x + w, y + h - rr);
   213→  ctx.quadraticCurveTo(x + w, y + h, x + w - rr, y + h);
   214→  ctx.lineTo(x + rr, y + h);
   215→  ctx.quadraticCurveTo(x, y + h, x, y + h - rr);
   216→  ctx.lineTo(x, y + rr);
   217→  ctx.quadraticCurveTo(x, y, x + rr, y);
   218→  ctx.closePath();
   219→}

   221→export default function BreakoutGame({ onBack }: BreakoutGameProps) {
   222→  const canvasRef = useRef<HTMLCanvasElement>(null);
   223→  const containerRef = useRef<HTMLDivElement>(null);
   224→  const animFrameRef = useRef<number>(0);

   230→  const bricksRef = useRef<Brick[]>([]);
   231→  const powerUpsRef = useRef<PowerUp[]>([]);
   232→  const particlesRef = useRef<Particle[]>([]);
   233→  const trailsRef = useRef<TrailPoint[]>([]);
   234→  const scoreRef = useRef(0);
   235→  const livesRef = useRef(MAX_LIVES);
   236→  const levelRef = useRef(1);
   237→  const gameStatusRef = useRef<GameStatus>('idle');
   238→  const paddleExtendTimerRef = useRef<number | null>(null);
   239→  const keysRef = useRef<Set<string>>(new Set());
   240→
   241→  // UI 渲染用的 state
   242→  const [score, setScore] = useState(0);
   243→  const [lives, setLives] = useState(MAX_LIVES);
   244→  const [level, setLevel] = useState(1);
   245→  const [gameStatus, setGameStatus] = useState<GameStatus>('idle');
   246→  const [highScore, setHighScore] = useState(0);
   247→  const [loadingHighScore, setLoadingHighScore] = useState(true);
   248→  const [isNewRecord, setIsNewRecord] = useState(false);
   249→  const [paused, setPaused] = useState(false);
   250→  const pausedRef = useRef(false);
   251→
   252→  // 获取挡板 Y 坐标
   253→  const getPaddleY = useCallback(() => {
   254→    return CANVAS_H - PADDLE_Y_OFFSET;
   255→  }, []);
   256→
   257→  // 创建一个新球
   258→  const createBall = useCallback(
   259→    (canvasW: number) => {
   260→      const speed = BALL_SPEED + (levelRef.current - 1) * 0.5;
   261→      const angle = -Math.PI / 2 + (Math.random() - 0.5) * 0.8;
   262→      return {
   263→        x: canvasW / 2,
   264→        y: getPaddleY() - PADDLE_H / 2 - BALL_RADIUS - 1,
   265→        dx: Math.cos(angle) * speed,
   266→        dy: Math.sin(angle) * speed,
   267→        radius: BALL_RADIUS,
   268→        speed,
   269→      };
   270→    },
   271→    [getPaddleY]
   272→  );
   273→
   274→  // 生成砖块碎片粒子
   275→  const spawnParticles = useCallback((brick: Brick) => {
   276→    const brickType = BRICK_TYPES[brick.typeIndex];
   277→    const cx = brick.x + brick.w / 2;
   278→    const cy = brick.y + brick.h / 2;
   279→    for (let i = 0; i < 8; i++) {
   280→      const angle = Math.random() * Math.PI * 2;
   281→      const speed = 1 + Math.random() * 3;
   282→      particlesRef.current.push({
   283→        x: cx + (Math.random() - 0.5) * brick.w * 0.6,
   284→        y: cy + (Math.random() - 0.5) * brick.h * 0.6,
   285→        dx: Math.cos(angle) * speed,
   286→        dy: Math.sin(angle) * speed,
   287→        life: 30 + Math.random() * 20,
   288→        maxLife: 50,
   289→        color: brickType.color,
   290→        size: 2 + Math.random() * 3,
   291→      });
   292→    }
   293→  }, []);
   294→
   295→  // 随机生成道具
   296→  const trySpawnPowerUp = useCallback((brick: Brick) => {
   297→    if (Math.random() > POWERUP_CHANCE) return;
   298→    const types = [PowerUpType.Extend, PowerUpType.Multi, PowerUpType.Slow];
   299→    const type = types[Math.floor(Math.random() * types.length)];
   300→    powerUpsRef.current.push({
   301→      x: brick.x + brick.w / 2,
   302→      y: brick.y + brick.h / 2,
   303→      type,
   304→      frame: 0,
   305→    });
   306→  }, []);
   307→
   308→  // 应用道具效果
   309→  const applyPowerUp = useCallback(
   310→    (pu: PowerUp) => {
   311→      if (pu.type === PowerUpType.Extend) {
   312→        paddleRef.current.w = Math.min(PADDLE_W * 1.6, canvasWRef.current - 20);
   313→        if (paddleExtendTimerRef.current) clearTimeout(paddleExtendTimerRef.current);
   314→        paddleExtendTimerRef.current = window.setTimeout(() => {
   315→          paddleRef.current.w = PADDLE_W;
   316→          paddleExtendTimerRef.current = null;
   317→        }, 10000);
   318→      } else if (pu.type === PowerUpType.Multi) {
   319→        // 每个现有球分裂出2个新球
   320→        const newBalls: Ball[] = [];
   321→        ballsRef.current.forEach((b) => {
   322→          for (let i = 0; i < 2; i++) {
   323→            const angle = Math.atan2(b.dy, b.dx) + (i === 0 ? 0.5 : -0.5);
   324→            newBalls.push({
   325→              ...b,
   326→              dx: Math.cos(angle) * b.speed,
   327→              dy: Math.sin(angle) * b.speed,
   328→            });
   329→          }
   330→        });
   331→        ballsRef.current.push(...newBalls);
   332→      } else if (pu.type === PowerUpType.Slow) {
   333→        ballsRef.current.forEach((b) => {
   334→          const slowFactor = 0.65;
   335→          b.dx *= slowFactor;
   336→          b.dy *= slowFactor;
   337→          b.speed *= slowFactor;
   338→        });
   339→      }
   340→    },
   341→    []
   342→  );
   343→
   344→  // 初始化某一关
   345→  const initLevel = useCallback(
   346→    (lvl: number) => {
   347→      const canvasW = canvasWRef.current;
   348→      bricksRef.current = generateBricks(lvl, canvasW);
   349→      ballsRef.current = [createBall(canvasW)];
   350→      powerUpsRef.current = [];
   351→      particlesRef.current = [];
   352→      trailsRef.current = [];
   353→      paddleRef.current.w = PADDLE_W;
   354→      paddleRef.current.x = canvasW / 2 - PADDLE_W / 2;
   355→      if (paddleExtendTimerRef.current) {
   356→        clearTimeout(paddleExtendTimerRef.current);
   357→        paddleExtendTimerRef.current = null;
   358→      }
   359→      levelRef.current = lvl;
   360→      setLevel(lvl);
   361→    },
   362→    [createBall]
   363→  );
   364→
   365→  // 游戏逻辑更新
   366→  const update = useCallback(() => {
   367→    const canvasW = canvasWRef.current;
   368→    const paddle = paddleRef.current;
   369→    const paddleY = getPaddleY();
   370→    const balls = ballsRef.current;
   371→    const bricks = bricksRef.current;
   372→    const powerUps = powerUpsRef.current;
   373→    const particles = particlesRef.current;
   374→    const trails = trailsRef.current;
   375→    const keys = keysRef.current;
   376→
   377→    // 键盘控制挡板移动
   378→    const moveSpeed = 7;
   379→    if (keys.has('ArrowLeft') || keys.has('a') || keys.has('A')) {
   380→      paddle.x = Math.max(0, paddle.x - moveSpeed);
   381→    }
   382→    if (keys.has('ArrowRight') || keys.has('d') || keys.has('D')) {
   383→      paddle.x = Math.min(canvasW - paddle.w, paddle.x + moveSpeed);
   384→    }
   385→
   386→    // 更新弹球
   387→    const ballsToRemove: number[] = [];
   388→    balls.forEach((ball, idx) => {
   389→      // 记录拖尾
   390→      trails.push({ x: ball.x, y: ball.y, alpha: 1 });
   391→
   392→      ball.x += ball.dx;
   393→      ball.y += ball.dy;
   394→
   395→      // 左右墙壁碰撞
   396→      if (ball.x - ball.radius <= 0) {
   397→        ball.x = ball.radius;
   398→        ball.dx = Math.abs(ball.dx);
   399→      }
   400→      if (ball.x + ball.radius >= canvasW) {
   401→        ball.x = canvasW - ball.radius;
   402→        ball.dx = -Math.abs(ball.dx);
   403→      }
   404→
   405→      // 顶部墙壁碰撞
   406→      if (ball.y - ball.radius <= 0) {
   407→        ball.y = ball.radius;
   408→        ball.dy = Math.abs(ball.dy);
   409→      }
   410→
   411→      // 底部出界
   412→      if (ball.y - ball.radius > CANVAS_H) {
   413→        ballsToRemove.push(idx);
   414→        return;
   415→      }
   416→
   417→      // 挡板碰撞检测
   418→      const pLeft = paddle.x;
   419→      const pRight = paddle.x + paddle.w;
   420→      const pTop = paddleY - PADDLE_H / 2;
   421→      const pBottom = paddleY + PADDLE_H / 2;
   422→      if (
   423→        ball.dy > 0 &&
   424→        ball.y + ball.radius >= pTop &&
   425→        ball.y + ball.radius <= pBottom + 4 &&
   426→        ball.x >= pLeft - ball.radius &&
   427→        ball.x <= pRight + ball.radius
   428→      ) {
   429→        ball.y = pTop - ball.radius;
   430→        // 根据击中挡板的位置改变反弹角度
   431→        const hitPos = (ball.x - (paddle.x + paddle.w / 2)) / (paddle.w / 2);
   432→        const maxAngle = Math.PI / 3;
   433→        const angle = hitPos * maxAngle;
   434→        const spd = ball.speed;
   435→        ball.dx = Math.sin(angle) * spd;
   436→        ball.dy = -Math.cos(angle) * spd;
   437→      }
   438→
   439→      // 砖块碰撞检测
   440→      bricks.forEach((brick) => {
   441→        if (!brick.alive) return;
   442→        const bLeft = brick.x;
   443→        const bRight = brick.x + brick.w;
   444→        const bTop = brick.y;
   445→        const bBottom = brick.y + brick.h;
   446→
   447→        // AABB 碰撞检测
   448→        if (
   449→          ball.x + ball.radius > bLeft &&
   450→          ball.x - ball.radius < bRight &&
   451→          ball.y + ball.radius > bTop &&
   452→          ball.y - ball.radius < bBottom
   453→        ) {
   454→          brick.alive = false;
   455→          scoreRef.current += BRICK_TYPES[brick.typeIndex].score;
   456→          setScore(scoreRef.current);
   457→          spawnParticles(brick);
   458→          trySpawnPowerUp(brick);
   459→
   460→          // 判断碰撞方向
   461→          const overlapLeft = ball.x + ball.radius - bLeft;
   462→          const overlapRight = bRight - (ball.x - ball.radius);
   463→          const overlapTop = ball.y + ball.radius - bTop;
   464→          const overlapBottom = bBottom - (ball.y - ball.radius);
   465→          const minOverlapX = Math.min(overlapLeft, overlapRight);
   466→          const minOverlapY = Math.min(overlapTop, overlapBottom);
   467→
   468→          if (minOverlapX < minOverlapY) {
   469→            ball.dx = -ball.dx;
   470→          } else {
   471→            ball.dy = -ball.dy;
   472→          }
   473→        }
   474→      });
   475→    });
   476→
   477→    // 移除出界的球（从后往前删）
   478→    for (let i = ballsToRemove.length - 1; i >= 0; i--) {
   479→      balls.splice(ballsToRemove[i], 1);
   480→    }
   481→
   482→    // 所有球都出界，失去一条命
   483→    if (balls.length === 0) {
   484→      livesRef.current -= 1;
   485→      setLives(livesRef.current);
   486→      if (livesRef.current <= 0) {
   487→        gameStatusRef.current = 'over';
   488→        setGameStatus('over');
   489→      } else {
   490→        ballsRef.current = [createBall(canvasW)];
   491→      }
   492→    }
   493→
   494→    // 检查是否清关
   495→    if (bricks.every((b) => !b.alive) && gameStatusRef.current === 'playing') {
   496→      if (levelRef.current >= MAX_LEVEL) {
   497→        gameStatusRef.current = 'over';
   498→        setGameStatus('over');
   499→      } else {
   500→        gameStatusRef.current = 'levelUp';
   501→        setGameStatus('levelUp');
   502→        setTimeout(() => {
   503→          initLevel(levelRef.current + 1);
   504→          gameStatusRef.current = 'playing';
   505→          setGameStatus('playing');
   506→        }, 1500);
   507→      }
   508→    }
   509→
   510→    // 更新道具下落
   511→    const puToRemove: number[] = [];
   512→    powerUps.forEach((pu, idx) => {
   513→      pu.y += POWERUP_SPEED;
   514→      pu.frame += 1;
   515→
   516→      // 检测是否被挡板接住
   517→      if (
   518→        pu.y + POWERUP_SIZE / 2 >= paddleY - PADDLE_H / 2 &&
   519→        pu.y - POWERUP_SIZE / 2 <= paddleY + PADDLE_H / 2 &&
   520→        pu.x + POWERUP_SIZE / 2 >= paddle.x &&
   521→        pu.x - POWERUP_SIZE / 2 <= paddle.x + paddle.w
   522→      ) {
   523→        applyPowerUp(pu);
   524→        puToRemove.push(idx);
   525→      }
   526→
   527→      // 出界移除
   528→      if (pu.y > CANVAS_H + 20) {
   529→        puToRemove.push(idx);
   530→      }
   531→    });
   532→    for (let i = puToRemove.length - 1; i >= 0; i--) {
   533→      powerUps.splice(puToRemove[i], 1);
   534→    }
   535→
   536→    // 更新粒子
   537→    for (let i = particles.length - 1; i >= 0; i--) {
   538→      const p = particles[i];
   539→      p.x += p.dx;
   540→      p.y += p.dy;
   541→      p.dy += 0.08; // 重力
   542→      p.life -= 1;
   543→      if (p.life <= 0) {
   544→        particles.splice(i, 1);
   545→      }
   546→    }
   547→
   548→    // 更新拖尾衰减
   549→    for (let i = trails.length - 1; i >= 0; i--) {
   550→      trails[i].alpha -= 0.08;
   551→      if (trails[i].alpha <= 0) {
   552→        trails.splice(i, 1);
   553→      }
   554→    }
   555→  }, [createBall, spawnParticles, trySpawnPowerUp, applyPowerUp, getPaddleY, initLevel]);
   556→
   557→  // 绘制画面
   558→  const draw = useCallback(() => {
   559→    const canvas = canvasRef.current;
   560→    if (!canvas) return;
   561→    const ctx = canvas.getContext('2d');
   562→    if (!ctx) return;
   563→
   564→    const w = canvasWRef.current;
   565→    const h = CANVAS_H;
   566→    const paddle = paddleRef.current;
   567→    const paddleY = getPaddleY();
   568→    const balls = ballsRef.current;
   569→    const bricks = bricksRef.current;
   570→    const powerUps = powerUpsRef.current;
   571→    const particles = particlesRef.current;
   572→    const trails = trailsRef.current;
   573→
   574→    // 清空画布并绘制深色背景
   575→    ctx.fillStyle = '#0f0f1a';
   576→    ctx.fillRect(0, 0, w, h);
   577→
   578→    // 绘制背景网格线
   579→    ctx.strokeStyle = 'rgba(255,255,255,0.03)';
   580→    ctx.lineWidth = 0.5;
   581→    for (let i = 0; i < w; i += 40) {
   582→      ctx.beginPath();
   583→      ctx.moveTo(i, 0);
   584→      ctx.lineTo(i, h);
   585→      ctx.stroke();
   586→    }
   587→    for (let j = 0; j < h; j += 40) {
   588→      ctx.beginPath();
   589→      ctx.moveTo(0, j);
   590→      ctx.lineTo(w, j);
   591→      ctx.stroke();
   592→    }
   593→
   594→    // 绘制拖尾
   595→    trails.forEach((t) => {
   596→      ctx.beginPath();
   597→      ctx.arc(t.x, t.y, BALL_RADIUS * 0.6, 0, Math.PI * 2);
   598→      ctx.fillStyle = `rgba(255,255,255,${t.alpha * 0.3})`;
   599→      ctx.fill();
   600→    });
   601→
   602→    // 绘制砖块
   603→    bricks.forEach((brick) => {
   604→      if (!brick.alive) return;
   605→      const bt = BRICK_TYPES[brick.typeIndex];
   606→      const grad = ctx.createLinearGradient(brick.x, brick.y, brick.x, brick.y + brick.h);
   607→      grad.addColorStop(0, bt.gradient[0]);
   608→      grad.addColorStop(1, bt.gradient[1]);
   609→      ctx.fillStyle = grad;
   610→      roundRect(ctx, brick.x, brick.y, brick.w, brick.h, 4);
   611→      ctx.fill();
   612→      // 高光线条
   613→      ctx.strokeStyle = 'rgba(255,255,255,0.2)';
   614→      ctx.lineWidth = 1;
   615→      roundRect(ctx, brick.x + 1, brick.y + 1, brick.w - 2, brick.h / 2, 3);
   616→      ctx.stroke();
   617→    });
   618→
   619→    // 绘制挡板
   620→    const paddleGrad = ctx.createLinearGradient(
   621→      paddle.x, paddleY - PADDLE_H / 2,
   622→      paddle.x, paddleY + PADDLE_H / 2
   623→    );
   624→    paddleGrad.addColorStop(0, '#60a5fa');
   625→    paddleGrad.addColorStop(0.5, '#3b82f6');
   626→    paddleGrad.addColorStop(1, '#2563eb');
   627→    ctx.fillStyle = paddleGrad;
   628→    roundRect(ctx, paddle.x, paddleY - PADDLE_H / 2, paddle.w, PADDLE_H, 6);
   629→    ctx.fill();
   630→    // 挡板高光
   631→    ctx.fillStyle = 'rgba(255,255,255,0.25)';
   632→    roundRect(ctx, paddle.x + 4, paddleY - PADDLE_H / 2 + 1, paddle.w - 8, PADDLE_H / 3, 3);
   633→    ctx.fill();
   634→
   635→    // 绘制弹球
   636→    balls.forEach((ball) => {
   637→      // 光晕
   638→      ctx.shadowColor = '#ffffff';
   639→      ctx.shadowBlur = 12;
   640→      ctx.fillStyle = '#ffffff';
   641→      ctx.beginPath();
   642→      ctx.arc(ball.x, ball.y, ball.radius, 0, Math.PI * 2);
   643→      ctx.fill();
   644→      // 高光点
   645→      ctx.fillStyle = 'rgba(255,255,255,0.9)';
   646→      ctx.beginPath();
   647→      ctx.arc(ball.x - 1.5, ball.y - 1.5, ball.radius * 0.35, 0, Math.PI * 2);
   648→      ctx.fill();
   649→      ctx.shadowBlur = 0;
   650→    });
   651→
   652→    // 绘制道具
   653→    powerUps.forEach((pu) => {
   654→      const color = POWERUP_COLORS[pu.type];
   655→      const blink = Math.sin(pu.frame * 0.3) * 0.3 + 0.7;
   656→      ctx.globalAlpha = blink;
   657→      ctx.fillStyle = color;
   658→      roundRect(ctx, pu.x - POWERUP_SIZE / 2, pu.y - POWERUP_SIZE / 2, POWERUP_SIZE, POWERUP_SIZE, 5);
   659→      ctx.fill();
   660→      ctx.fillStyle = '#ffffff';
   661→      ctx.font = 'bold 13px sans-serif';
   662→      ctx.textAlign = 'center';
   663→      ctx.textBaseline = 'middle';
   664→      ctx.fillText(pu.type, pu.x, pu.y + 1);
   665→      ctx.globalAlpha = 1;
   666→    });
   667→
   668→    // 绘制粒子
   669→    particles.forEach((p) => {
   670→      const alpha = p.life / p.maxLife;
   671→      ctx.globalAlpha = alpha;
   672→      ctx.fillStyle = p.color;
   673→      ctx.fillRect(p.x - p.size / 2, p.y - p.size / 2, p.size, p.size);
   674→    });
   675→    ctx.globalAlpha = 1;
   676→
   677→    // 过关提示
   678→    if (gameStatusRef.current === 'levelUp') {
   679→      ctx.fillStyle = 'rgba(0,0,0,0.5)';
   680→      ctx.fillRect(0, 0, w, h);
   681→      ctx.fillStyle = '#fbbf24';
   682→      ctx.font = 'bold 36px sans-serif';
   683→      ctx.textAlign = 'center';
   684→      ctx.textBaseline = 'middle';
   685→      ctx.fillText(`第 ${levelRef.current} 关通过！`, w / 2, h / 2 - 20);
   686→      ctx.fillStyle = '#ffffff';
   687→      ctx.font = '20px sans-serif';
   688→      ctx.fillText('准备进入下一关...', w / 2, h / 2 + 25);
   689→    }
   690→
   691→    // Pause overlay
   692→    if (pausedRef.current && gameStatusRef.current === 'playing') {
   693→      ctx.fillStyle = 'rgba(0,0,0,0.6)';
   694→      ctx.fillRect(0, 0, w, h);
   695→      ctx.fillStyle = '#ffffff';
   696→      ctx.font = 'bold 32px sans-serif';
   697→      ctx.textAlign = 'center';
   698→      ctx.textBaseline = 'middle';
   699→      ctx.fillText('已暂停', w / 2, h / 2 - 10);
   700→      ctx.font = '16px sans-serif';
   701→      ctx.fillStyle = '#a1a1aa';
   702→      ctx.fillText('按 Esc 继续', w / 2, h / 2 + 25);
   703→    }
   704→
   705→    // 游戏未开始提示
   706→    if (gameStatusRef.current === 'idle') {
   707→      ctx.fillStyle = 'rgba(0,0,0,0.55)';
   708→      ctx.fillRect(0, 0, w, h);
   709→      ctx.fillStyle = '#ffffff';
   710→      ctx.font = 'bold 28px sans-serif';
   711→      ctx.textAlign = 'center';
   712→      ctx.textBaseline = 'middle';
   713→      ctx.fillText('打砖块', w / 2, h / 2 - 30);
   714→      ctx.font = '16px sans-serif';
   715→      ctx.fillStyle = '#a1a1aa';
   716→      ctx.fillText('按「开始游戏」开始', w / 2, h / 2 + 15);
   717→    }
   718→
   719→    // 游戏结束遮罩
   720→    if (gameStatusRef.current === 'over') {
   721→      ctx.fillStyle = 'rgba(0,0,0,0.65)';
   722→      ctx.fillRect(0, 0, w, h);
   723→      const isWin = livesRef.current > 0;
   724→      ctx.fillStyle = isWin ? '#4ade80' : '#ef4444';
   725→      ctx.font = 'bold 36px sans-serif';
   726→      ctx.textAlign = 'center';
   727→      ctx.textBaseline = 'middle';
   728→      ctx.fillText(isWin ? '恭喜通关！' : '游戏结束', w / 2, h / 2 - 40);
   729→      ctx.fillStyle = '#fbbf24';
   730→      ctx.font = 'bold 24px sans-serif';
   731→      ctx.fillText(`最终得分: ${scoreRef.current}`, w / 2, h / 2 + 10);
   732→      if (isWin) {
   733→        ctx.fillStyle = '#a1a1aa';
   734→        ctx.font = '16px sans-serif';
   735→        ctx.fillText('所有关卡已通关！', w / 2, h / 2 + 45);
   736→      }
   737→    }
   738→  }, [getPaddleY]);
   739→
   740→  // 游戏主循环
   741→  const gameLoop = useCallback(() => {
   742→    if (gameStatusRef.current === 'playing' && !pausedRef.current) {
   743→      update();
   744→    }
   745→    draw();
   746→    animFrameRef.current = requestAnimationFrame(gameLoop);
   747→  }, [update, draw]);
   748→
   749→  // 结束游戏并提交分数
   750→  const endGame = useCallback(async () => {
   751→    if (animFrameRef.current) {
   752→      cancelAnimationFrame(animFrameRef.current);
   753→      animFrameRef.current = 0;
   754→    }
   755→
   756→    if (scoreRef.current > highScore) {
   757→      setIsNewRecord(true);
   758→      try {
   759→        const res = await fetch('/api/games/scores', {
   760→          method: 'POST',
   761→          headers: { 'Content-Type': 'application/json' },
   762→          body: JSON.stringify({ game: GAME_NAME, score: scoreRef.current }),
   763→        });
   764→        if (res.ok) {
   765→          const data = await res.json();
   766→          if (data.data?.score !== undefined) {
   767→            setHighScore(data.data.score);
   768→          }
   769→        }
   770→      } catch {
   771→        // 网络异常静默处理
   772→      }
   773→    } else {
   774→      setIsNewRecord(false);
   775→    }
   776→  }, [highScore]);
   777→
   778→  // 游戏结束触发提交
   779→  useEffect(() => {
   780→    if (gameStatus === 'over') {
   781→      endGame();
   782→    }
   783→  }, [gameStatus, endGame]);
   784→
   785→  // 开始游戏
   786→  const startGame = useCallback(() => {
   787→    scoreRef.current = 0;
   788→    livesRef.current = MAX_LIVES;
   789→    setScore(0);
   790→    setLives(MAX_LIVES);
   791→    setIsNewRecord(false);
   792→    initLevel(1);
   793→    gameStatusRef.current = 'playing';
   794→    setGameStatus('playing');
   795→
   796→    // 启动游戏循环（如果尚未启动）
   797→    if (animFrameRef.current) {
   798→      cancelAnimationFrame(animFrameRef.current);
   799→    }
   800→    animFrameRef.current = requestAnimationFrame(gameLoop);
   801→  }, [initLevel, gameLoop]);
   802→
   803→  // 键盘事件
   804→  useEffect(() => {
   805→    const onKeyDown = (e: KeyboardEvent) => {
   806→      if (e.key === 'Escape' && gameStatusRef.current === 'playing') {
   807→        e.preventDefault();
   808→        pausedRef.current = !pausedRef.current;
   809→        setPaused(pausedRef.current);
   810→        return;
   811→      }
   812→      keysRef.current.add(e.key);
   813→      if (['ArrowLeft', 'ArrowRight', 'ArrowUp', 'ArrowDown'].includes(e.key)) {
   814→        e.preventDefault();
   815→      }
   816→    };
   817→    const onKeyUp = (e: KeyboardEvent) => {
   818→      keysRef.current.delete(e.key);
   819→    };
   820→    window.addEventListener('keydown', onKeyDown);
   821→    window.addEventListener('keyup', onKeyUp);
   822→    return () => {
   823→      window.removeEventListener('keydown', onKeyDown);
   824→      window.removeEventListener('keyup', onKeyUp);
   825→    };
   826→  }, []);
   827→
   828→  // 鼠标移动控制挡板
   829→  useEffect(() => {
   830→    const canvas = canvasRef.current;
   831→    if (!canvas) return;
   832→
   833→    const onMouseMove = (e: MouseEvent) => {
   834→      if (gameStatusRef.current !== 'playing') return;
   835→      const rect = canvas.getBoundingClientRect();
   836→      const scaleX = canvasWRef.current / rect.width;
   837→      const mouseX = (e.clientX - rect.left) * scaleX;
   838→      const paddle = paddleRef.current;
   839→      paddle.x = Math.max(
   840→        0,
   841→        Math.min(canvasWRef.current - paddle.w, mouseX - paddle.w / 2)
   842→      );
   843→    };
   844→
   845→    canvas.addEventListener('mousemove', onMouseMove);
   846→    return () => canvas.removeEventListener('mousemove', onMouseMove);
   847→  }, []);
   848→
   849→  // 触摸滑动控制挡板
   850→  useEffect(() => {
   851→    const canvas = canvasRef.current;
   852→    if (!canvas) return;
   853→
   854→    const onTouchMove = (e: TouchEvent) => {
   855→      if (gameStatusRef.current !== 'playing') return;
   856→      e.preventDefault();
   857→      const touch = e.touches[0];
   858→      const rect = canvas.getBoundingClientRect();
   859→      const scaleX = canvasWRef.current / rect.width;
   860→      const touchX = (touch.clientX - rect.left) * scaleX;
   861→      const paddle = paddleRef.current;
   862→      paddle.x = Math.max(
   863→        0,
   864→        Math.min(canvasWRef.current - paddle.w, touchX - paddle.w / 2)
   865→      );
   866→    };
   867→
   868→    canvas.addEventListener('touchmove', onTouchMove, { passive: false });
   869→    return () => canvas.removeEventListener('touchmove', onTouchMove);
   870→  }, []);
   871→
   872→  // 获取最高分
   873→  useEffect(() => {
   874→    const fetchHighScore = async () => {
   875→      try {
   876→        const res = await fetch('/api/games/scores');
   877→        if (res.ok) {
   878→          const data = await res.json();
   879→          if (Array.isArray(data.data)) {
   880→            const found = data.data.find(
   881→              (item: { game: string; score: number }) => item.game === GAME_NAME
   882→            );
   883→            if (found) {
   884→              setHighScore(found.score);
   885→            }
   886→          }
   887→        }
   888→      } catch {
   889→        // 静默处理
   890→      } finally {
   891→        setLoadingHighScore(false);
   892→      }
   893→    };
   894→    fetchHighScore();
   895→  }, []);
   896→
   897→  // Canvas 尺寸自适应 + 高DPI适配
   898→  useEffect(() => {
   899→    const updateSize = () => {
   900→      const container = containerRef.current;
   901→      const canvas = canvasRef.current;
   902→      if (!container || !canvas) return;
   903→
   904→      const containerWidth = container.clientWidth;
   905→      const w = Math.min(containerWidth, CANVAS_MAX_W);
   906→      const dpr = window.devicePixelRatio || 1;
   907→
   908→      canvas.width = w * dpr;
   909→      canvas.height = CANVAS_H * dpr;
   910→      canvas.style.width = `${w}px`;
   911→      canvas.style.height = `${CANVAS_H}px`;
   912→
   913→      const ctx = canvas.getContext('2d');
   914→      if (ctx) ctx.scale(dpr, dpr);
   915→
   916→      canvasWRef.current = w;
   917→    };
   918→
   919→    updateSize();
   920→    window.addEventListener('resize', updateSize);
   921→    return () => window.removeEventListener('resize', updateSize);
   922→  }, []);
   923→
   924→  // 组件卸载时清理
   925→  useEffect(() => {
   926→    return () => {
   927→      if (animFrameRef.current) cancelAnimationFrame(animFrameRef.current);
   928→      if (paddleExtendTimerRef.current) clearTimeout(paddleExtendTimerRef.current);
   929→    };
   930→  }, []);
   931→
   932→  // 生成生命值展示
   933→  const heartsDisplay = Array.from({ length: MAX_LIVES }, (_, i) => (
   934→    <span
   935→      key={i}
   936→      className={i < lives ? 'opacity-100' : 'opacity-25'}
   937→    >
   938→      ❤️
   939→    </span>
   940→  ));
   941→
   942→  return (
   943→    <div className="flex flex-col items-center w-full max-w-md mx-auto px-2 py-2 gap-2">
   944→      {/* Top bar */}
   945→      <div className="flex items-center justify-between w-full">
   946→        <div className="flex items-center gap-1">
   947→          <Button
   948→            variant="ghost"
   949→            size="sm"
   950→            onClick={onBack}
   951→            className="text-muted-foreground hover:text-foreground"
   952→          >
   953→            <ArrowLeft className="h-4 w-4 mr-1" />
   954→            返回大厅
   955→          </Button>
   956→          {gameStatus === 'playing' && <GameControlsHelp info={breakoutControlsInfo} variant="button" />}
   957→        </div>
   958→        <h2 className="text-lg font-bold tracking-wide">🧱 打砖块</h2>
   959→        <div className="w-8" />
   960→      </div>
   961→
   962→      {/* Score and status bar */}
   963→      <div className="grid grid-cols-4 gap-2 w-full text-center">
   964→        <div className="flex flex-col items-center gap-0.5">
   965→          <span className="text-xs text-muted-foreground">分数</span>
   966→          <span className="text-xl font-bold text-primary tabular-nums">{score}</span>
   967→        </div>
   968→        <div className="flex flex-col items-center gap-0.5">
   969→          <span className="text-xs text-muted-foreground flex items-center gap-1">
   970→            <Trophy className="h-3 w-3 text-yellow-500" />
   971→            最高分
   972→          </span>
   973→          <span className="text-xl font-bold text-yellow-500 tabular-nums">
   974→            {loadingHighScore ? '...' : highScore}
   975→          </span>
   976→        </div>
   977→        <div className="flex flex-col items-center gap-0.5">
   978→          <span className="text-xs text-muted-foreground">生命</span>
   979→          <div className="text-base leading-tight">{heartsDisplay}</div>
   980→        </div>
   981→        <div className="flex flex-col items-center gap-0.5">
   982→          <span className="text-xs text-muted-foreground">关卡</span>
   983→          <span className="text-xl font-bold tabular-nums">
   984→            {level}/{MAX_LEVEL}
   985→          </span>
   986→        </div>
   987→      </div>
   988→
   989→      {/* Canvas game area */}
   990→      <div
   991→        ref={containerRef}
   992→        className="w-full flex justify-center"
   993→      >
   994→        <canvas
   995→          ref={canvasRef}
   996→          className="rounded-lg border border-border/30 shadow-lg"
   997→          style={{ touchAction: 'none' }}
   998→        />
   999→      </div>
  1000→
  1001→      {/* Action buttons */}
  1002→      <div className="flex flex-col items-center gap-3 w-full">
  1003→        {gameStatus === 'over' && isNewRecord && (
  1004→          <p className="text-sm font-semibold text-yellow-500 animate-pulse">
  1005→            🎉 新纪录！
  1006→          </p>
  1007→        )}
  1008→
  1009→        <Button
  1010→          onClick={startGame}
  1011→          size="lg"
  1012→          className="w-full max-w-xs text-base font-semibold"
  1013→          disabled={gameStatus === 'playing' || gameStatus === 'levelUp'}
  1014→        >
  1015→          <RotateCcw className="h-4 w-4 mr-2" />
  1016→          {gameStatus === 'playing'
  1017→            ? '游戏中...'
  1018→            : gameStatus === 'levelUp'
  1019→              ? '过关中...'
  1020→              : '开始游戏'}
  1021→        </Button>
  1022→      </div>
  1023→    </div>
  1024→  );
  1025→}
  1026→
