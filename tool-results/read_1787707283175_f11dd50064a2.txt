   133→/** Current piece info */
   134→interface Piece {
   135→  type: string;
   136→  shape: number[][];
   137→  x: number;
   138→  y: number;
   139→}
   140→
   141→type GameStatus = 'idle' | 'playing' | 'over';
   142→
   143→/* ==================== Utility functions ==================== */
   144→
   145→function randomType(): string {
   146→  return PIECE_TYPES[Math.floor(Math.random() * PIECE_TYPES.length)];
   147→}
   148→
   149→function cloneShape(shape: number[][]): number[][] {
   150→  return shape.map(row => [...row]);
   151→}
   152→
   153→function rotateShape(shape: number[][]): number[][] {
   154→  const n = shape.length;
   155→  const rotated: number[][] = [];
   156→  for (let i = 0; i < n; i++) {
   157→    rotated[i] = [];
   158→    for (let j = 0; j < n; j++) {
   159→      rotated[i][j] = shape[n - 1 - j][i];
   160→    }
   161→  }
   162→  return rotated;
   163→}
   164→
   165→function createPiece(type: string): Piece {
   166→  return {
   167→    type,
   168→    shape: cloneShape(SHAPES[type]),
   169→    x: Math.floor((COLS - SHAPES[type][0].length) / 2),
   170→    y: 0,
   171→  };
   172→}
   173→
   174→function isValid(
   175→  shape: number[][],
   176→  board: (string | null)[][],
   177→  px: number,
   178→  py: number
   179→): boolean {
   180→  for (let r = 0; r < shape.length; r++) {
   181→    for (let c = 0; c < shape[r].length; c++) {
   182→      if (shape[r][c]) {
   183→        const nx = px + c;
   184→        const ny = py + r;
   185→        if (nx < 0 || nx >= COLS || ny >= ROWS) return false;
   186→        if (ny >= 0 && board[ny][nx] !== null) return false;
   187→      }
   188→    }
   189→  }
   190→  return true;
   191→}

   195→export default function TetrisGame({ onBack }: TetrisGameProps) {
   196→  const isMobile = useIsMobile();
   197→
   198→  /* --- Refs --- */
   199→  const canvasRef = useRef<HTMLCanvasElement>(null);
   200→  const containerRef = useRef<HTMLDivElement>(null);
   201→  const rafRef = useRef<number | null>(null);
   202→  const boardRef = useRef<(string | null)[][]>(
   203→    Array.from({ length: ROWS }, () => Array(COLS).fill(null))
   204→  );
   205→  const currentRef = useRef<Piece | null>(null);
   206→  const nextRef = useRef<string>(randomType());
   207→  const scoreRef = useRef<number>(0);
   208→  const levelRef = useRef<number>(1);
   209→  const linesRef = useRef<number>(0);
   210→  const statusRef = useRef<GameStatus>('idle');
   211→  const lastDropRef = useRef<number>(0);
   212→  const speedRef = useRef<SpeedKey>('normal');
   213→  const flashRowsRef = useRef<number[]>([]);
   214→  const flashStartRef = useRef<number>(0);
   215→  const gameOverCheckedRef = useRef<boolean>(false);
   216→  const scaleRef = useRef<number>(1);
   217→
   218→  /* --- React state (for UI rendering) --- */
   219→  const [score, setScore] = useState(0);
   220→  const [level, setLevel] = useState(1);
   221→  const [highScore, setHighScore] = useState(0);
   222→  const [gameStatus, setGameStatus] = useState<GameStatus>('idle');
   223→  const [speed, setSpeed] = useState<SpeedKey>('normal');
   224→  const [isNewRecord, setIsNewRecord] = useState(false);
   225→  const [loadingHighScore, setLoadingHighScore] = useState(true);
   226→  const [paused, setPaused] = useState(false);
   227→  const pausedRef = useRef(false);
   228→  const [nextPiece, setNextPiece] = useState(randomType());
   229→
   230→  /* ==================== Drop interval ==================== */
   231→  const getDropInterval = useCallback((): number => {
   232→    const base = SPEED_CONFIG[speedRef.current].base;
   233→    const lv = levelRef.current;
   234→    return Math.max(80, base * Math.pow(0.9, lv - 1));
   235→  }, []);
   236→
   237→  /* ==================== Reset board ==================== */
   238→  const resetBoard = useCallback(() => {
   239→    boardRef.current = Array.from({ length: ROWS }, () =>
   240→      Array(COLS).fill(null)
   241→    );
   242→  }, []);
   243→
   244→  /* ==================== Spawn piece ==================== */
   245→  const spawnPiece = useCallback(() => {
   246→    const type = nextRef.current;
   247→    nextRef.current = randomType();
   248→    setNextPiece(nextRef.current);
   249→    const piece = createPiece(type);
   250→    currentRef.current = piece;
   251→    if (!isValid(piece.shape, boardRef.current, piece.x, piece.y)) {
   252→      statusRef.current = 'over';
   253→      setGameStatus('over');
   254→      gameOverCheckedRef.current = false;
   255→    }
   256→  }, []);
   257→
   258→  /* ==================== Lock piece to board ==================== */
   259→  const lockPiece = useCallback(() => {
   260→    const piece = currentRef.current;
   261→    if (!piece) return;
   262→    const board = boardRef.current;
   263→    for (let r = 0; r < piece.shape.length; r++) {
   264→      for (let c = 0; c < piece.shape[r].length; c++) {
   265→        if (piece.shape[r][c]) {
   266→          const ny = piece.y + r;
   267→          const nx = piece.x + c;
   268→          if (ny >= 0 && ny < ROWS && nx >= 0 && nx < COLS) {
   269→            board[ny][nx] = piece.type;
   270→          }
   271→        }
   272→      }
   273→    }
   274→  }, []);
   275→
   276→  /* ==================== Clear lines ==================== */
   277→  const clearLines = useCallback((): number => {
   278→    const board = boardRef.current;
   279→    const fullRows: number[] = [];
   280→    for (let r = 0; r < ROWS; r++) {
   281→      if (board[r].every(cell => cell !== null)) {
   282→        fullRows.push(r);
   283→      }
   284→    }
   285→    if (fullRows.length === 0) return 0;
   286→    flashRowsRef.current = fullRows;
   287→    flashStartRef.current = performance.now();
   288→    return fullRows.length;
   289→  }, []);
   290→
   291→  /** Remove full rows and add empty rows at top */
   292→  const removeLines = useCallback(() => {
   293→    const board = boardRef.current;
   294→    const fullRows = flashRowsRef.current;
   295→    const newBoard = board.filter((_, idx) => !fullRows.includes(idx));
   296→    while (newBoard.length < ROWS) {
   297→      newBoard.unshift(Array(COLS).fill(null));
   298→    }
   299→    boardRef.current = newBoard;
   300→    flashRowsRef.current = [];
   301→  }, []);
   302→
   303→  /* ==================== Move down ==================== */
   304→  const moveDown = useCallback(() => {
   305→    const piece = currentRef.current;
   306→    if (!piece || statusRef.current !== 'playing') return;
   307→    const board = boardRef.current;
   308→    if (isValid(piece.shape, board, piece.x, piece.y + 1)) {
   309→      piece.y += 1;
   310→    } else {
   311→      lockPiece();
   312→      const cleared = clearLines();
   313→      if (cleared > 0) {
   314→        const pts = LINE_SCORES[cleared] || cleared * 100;
   315→        scoreRef.current += pts;
   316→        linesRef.current += cleared;
   317→        const newLv = Math.floor(linesRef.current / 10) + 1;
   318→        levelRef.current = newLv;
   319→        setScore(scoreRef.current);
   320→        setLevel(newLv);
   321→      }
   322→      if (cleared === 0) {
   323→        spawnPiece();
   324→      }
   325→    }
   326→  }, [lockPiece, clearLines, spawnPiece]);
   327→
   328→  /* ==================== Move left ==================== */
   329→  const moveLeft = useCallback(() => {
   330→    const piece = currentRef.current;
   331→    if (!piece || statusRef.current !== 'playing') return;
   332→    if (isValid(piece.shape, boardRef.current, piece.x - 1, piece.y)) {
   333→      piece.x -= 1;
   334→    }
   335→  }, []);
   336→
   337→  /* ==================== Move right ==================== */
   338→  const moveRight = useCallback(() => {
   339→    const piece = currentRef.current;
   340→    if (!piece || statusRef.current !== 'playing') return;
   341→    if (isValid(piece.shape, boardRef.current, piece.x + 1, piece.y)) {
   342→      piece.x += 1;
   343→    }
   344→  }, []);
   345→
   346→  /* ==================== Rotate ==================== */
   347→  const rotate = useCallback(() => {
   348→    const piece = currentRef.current;
   349→    if (!piece || statusRef.current !== 'playing') return;
   350→    if (piece.type === 'O') return;
   351→    const rotated = rotateShape(piece.shape);
   352→    if (isValid(rotated, boardRef.current, piece.x, piece.y)) {
   353→      piece.shape = rotated;
   354→      return;
   355→    }
   356→    const kicks = [1, -1, 2, -2];
   357→    for (const kick of kicks) {
   358→      if (isValid(rotated, boardRef.current, piece.x + kick, piece.y)) {
   359→        piece.shape = rotated;
   360→        piece.x += kick;
   361→        return;
   362→      }
   363→    }
   364→  }, []);
   365→
   366→  /* ==================== Hard drop ==================== */
   367→  const hardDrop = useCallback(() => {
   368→    const piece = currentRef.current;
   369→    if (!piece || statusRef.current !== 'playing') return;
   370→    const board = boardRef.current;
   371→    const originalY = piece.y;
   372→    let dropY = piece.y;
   373→    while (isValid(piece.shape, board, piece.x, dropY + 1)) {
   374→      dropY += 1;
   375→    }
   376→    piece.y = dropY;
   377→    scoreRef.current += (dropY - originalY) * 2;
   378→    setScore(scoreRef.current);
   379→    lockPiece();
   380→    const cleared = clearLines();
   381→    if (cleared > 0) {
   382→      const pts = LINE_SCORES[cleared] || cleared * 100;
   383→      scoreRef.current += pts;
   384→      linesRef.current += cleared;
   385→      const newLv = Math.floor(linesRef.current / 10) + 1;
   386→      levelRef.current = newLv;
   387→      setScore(scoreRef.current);
   388→      setLevel(newLv);
   389→    }
   390→    if (cleared === 0) {
   391→      spawnPiece();
   392→    }
   393→  }, [lockPiece, clearLines, spawnPiece]);
   394→
   395→  /* ==================== Ghost Y ==================== */
   396→  const getGhostY = useCallback((): number => {
   397→    const piece = currentRef.current;
   398→    if (!piece) return 0;
   399→    const board = boardRef.current;
   400→    let ghostY = piece.y;
   401→    while (isValid(piece.shape, board, piece.x, ghostY + 1)) {
   402→      ghostY += 1;
   403→    }
   404→    return ghostY;
   405→  }, []);
   406→
   407→  /* ==================== Canvas rendering ==================== */
   408→  const draw = useCallback(
   409→    (ctx: CanvasRenderingContext2D, now: number) => {
   410→      const s = scaleRef.current;
   411→      const dpr = window.devicePixelRatio || 1;
   412→      const cw = COLS * CELL;
   413→      const ch = ROWS * CELL;
   414→      const board = boardRef.current;
   415→      const piece = currentRef.current;
   416→
   417→      // Set transform to map game coordinates to bitmap pixels
   418→      ctx.setTransform(s * dpr, 0, 0, s * dpr, 0, 0);
   419→      ctx.clearRect(0, 0, cw, ch);
   420→
   421→      // Deep background
   422→      ctx.fillStyle = '#1a1a2e';
   423→      ctx.fillRect(0, 0, cw, ch);
   424→
   425→      // Grid lines
   426→      ctx.strokeStyle = 'rgba(255,255,255,0.06)';
   427→      ctx.lineWidth = 0.5;
   428→      for (let r = 0; r <= ROWS; r++) {
   429→        ctx.beginPath();
   430→        ctx.moveTo(0, r * CELL);
   431→        ctx.lineTo(cw, r * CELL);
   432→        ctx.stroke();
   433→      }
   434→      for (let c = 0; c <= COLS; c++) {
   435→        ctx.beginPath();
   436→        ctx.moveTo(c * CELL, 0);
   437→        ctx.lineTo(c * CELL, ch);
   438→        ctx.stroke();
   439→      }
   440→
   441→      // Draw locked cells
   442→      for (let r = 0; r < ROWS; r++) {
   443→        for (let c = 0; c < COLS; c++) {
   444→          if (board[r][c] !== null) {
   445→            drawCell(ctx, c, r, board[r][c]!, 1);
   446→          }
   447→        }
   448→      }
   449→
   450→      // Line clear flash animation
   451→      if (flashRowsRef.current.length > 0) {
   452→        const elapsed = now - flashStartRef.current;
   453→        const flashDuration = 300;
   454→        if (elapsed < flashDuration) {
   455→          const alpha = 0.5 + 0.5 * Math.sin((elapsed / flashDuration) * Math.PI * 4);
   456→          ctx.fillStyle = `rgba(255,255,255,${alpha})`;
   457→          for (const row of flashRowsRef.current) {
   458→            ctx.fillRect(0, row * CELL, cw, CELL);
   459→          }
   460→        } else {
   461→          removeLines();
   462→          spawnPiece();
   463→        }
   464→      }
   465→
   466→      // Draw ghost piece
   467→      if (piece && statusRef.current === 'playing' && flashRowsRef.current.length === 0) {
   468→        const ghostY = getGhostY();
   469→        if (ghostY !== piece.y) {
   470→          for (let r = 0; r < piece.shape.length; r++) {
   471→            for (let c = 0; c < piece.shape[r].length; c++) {
   472→              if (piece.shape[r][c]) {
   473→                const px = (piece.x + c) * CELL;
   474→                const py = (ghostY + r) * CELL;
   475→                ctx.strokeStyle = COLORS[piece.type];
   476→                ctx.globalAlpha = 0.3;
   477→                ctx.lineWidth = 2;
   478→                ctx.strokeRect(px + 1, py + 1, CELL - 2, CELL - 2);
   479→                ctx.globalAlpha = 0.08;
   480→                ctx.fillStyle = COLORS[piece.type];
   481→                ctx.fillRect(px + 1, py + 1, CELL - 2, CELL - 2);
   482→                ctx.globalAlpha = 1;
   483→              }
   484→            }
   485→          }
   486→        }
   487→      }
   488→
   489→      // Draw current piece
   490→      if (piece && statusRef.current === 'playing') {
   491→        for (let r = 0; r < piece.shape.length; r++) {
   492→          for (let c = 0; c < piece.shape[r].length; c++) {
   493→            if (piece.shape[r][c]) {
   494→              drawCell(ctx, piece.x + c, piece.y + r, piece.type, 1);
   495→            }
   496→          }
   497→        }
   498→      }
   499→    },
   500→    [getGhostY, removeLines, spawnPiece]
   501→  );
   502→
   503→  /** Draw a single cell with highlight and stroke */
   504→  function drawCell(
   505→    ctx: CanvasRenderingContext2D,
   506→    x: number,
   507→    y: number,
   508→    type: string,
   509→    alpha: number
   510→  ) {
   511→    const px = x * CELL;
   512→    const py = y * CELL;
   513→    ctx.globalAlpha = alpha;
   514→    ctx.fillStyle = COLORS[type];
   515→    ctx.fillRect(px + 1, py + 1, CELL - 2, CELL - 2);
   516→    ctx.strokeStyle = STROKE_COLORS[type];
   517→    ctx.lineWidth = 1.5;
   518→    ctx.strokeRect(px + 1, py + 1, CELL - 2, CELL - 2);
   519→    ctx.fillStyle = HIGHLIGHT_COLORS[type];
   520→    ctx.globalAlpha = alpha * 0.45;
   521→    ctx.fillRect(px + 3, py + 3, CELL - 6, 5);
   522→    ctx.fillRect(px + 3, py + 3, 3, CELL - 6);
   523→    ctx.globalAlpha = 1;
   524→  }
   525→
   526→  /* ==================== Main game loop ==================== */
   527→  const gameLoop = useCallback(
   528→    (now: number) => {
   529→      if (statusRef.current === 'playing' && !pausedRef.current) {
   530→        if (flashRowsRef.current.length === 0) {
   531→          if (now - lastDropRef.current >= getDropInterval()) {
   532→            moveDown();
   533→            lastDropRef.current = now;
   534→          }
   535→        }
   536→      }
   537→
   538→      if (
   539→        statusRef.current === 'over' &&
   540→        !gameOverCheckedRef.current
   541→      ) {
   542→        gameOverCheckedRef.current = true;
   543→        submitScore(scoreRef.current);
   544→      }
   545→
   546→      const canvas = canvasRef.current;
   547→      if (canvas) {
   548→        const ctx = canvas.getContext('2d');
   549→        if (ctx) draw(ctx, now);
   550→      }
   551→
   552→      rafRef.current = requestAnimationFrame(gameLoop);
   553→    },
   554→    [draw, getDropInterval, moveDown]
   555→  );

   557→  /* ==================== Submit score ==================== */
   558→  const submitScore = useCallback(async (finalScore: number) => {
   559→    try {
   560→      const res = await fetch('/api/games/scores', {
   561→        method: 'POST',
   562→        headers: { 'Content-Type': 'application/json' },
   563→        body: JSON.stringify({ game: GAME_NAME, score: finalScore }),
   564→      });
   565→      const data = await res.json();
   566→      if (data.data) {
   567→        const newHigh = data.data.score;
   568→        setHighScore(newHigh);
   569→        if (newHigh === finalScore && finalScore > 0) {
   570→          setIsNewRecord(true);
   571→        }
   572→      }
   573→    } catch {
   574→      // Silent handling
   575→    }
   576→  }, []);
   577→
   578→  /* ==================== Load high score ==================== */
   579→  useEffect(() => {
   580→    async function load() {
   581→      try {
   582→        const res = await fetch('/api/games/scores');
   583→        const data = await res.json();
   584→        if (data.data && Array.isArray(data.data)) {
   585→          const entry = data.data.find(
   586→            (e: { game: string; score: number }) => e.game === GAME_NAME
   587→          );
   588→          if (entry) setHighScore(entry.score);
   589→        }
   590→      } catch {
   591→        // Silent handling
   592→      } finally {
   593→        setLoadingHighScore(false);
   594→      }
   595→    }
   596→    load();
   597→  }, []);
   598→
   599→  /* ==================== Canvas sizing (container-based, high-DPI) ==================== */
   600→  useEffect(() => {
   601→    const canvas = canvasRef.current;
   602→    if (!canvas) return;
   603→
   604→    const updateSize = () => {
   605→      const w = canvas.clientWidth;
   606→      const h = canvas.clientHeight;
   607→      if (w === 0 || h === 0) return;
   608→      const dpr = window.devicePixelRatio || 1;
   609→      canvas.width = w * dpr;
   610→      canvas.height = h * dpr;
   611→      scaleRef.current = w / (COLS * CELL);
   612→    };
   613→
   614→    // Use rAF to ensure browser has laid out the element
   615→    const raf = requestAnimationFrame(updateSize);
   616→
   617→    const ro = new ResizeObserver(updateSize);
   618→    ro.observe(canvas);
   619→    return () => {
   620→      cancelAnimationFrame(raf);
   621→      ro.disconnect();
   622→    };
   623→  }, [gameStatus]);
   624→
   625→  /* ==================== Game loop start/stop ==================== */
   626→  useEffect(() => {
   627→    if (gameStatus === 'playing') {
   628→      lastDropRef.current = performance.now();
   629→      rafRef.current = requestAnimationFrame(gameLoop);
   630→    }
   631→    return () => {
   632→      if (rafRef.current) {
   633→        cancelAnimationFrame(rafRef.current);
   634→        rafRef.current = null;
   635→      }
   636→    };
   637→  }, [gameStatus, gameLoop]);
   638→
   639→  /* ==================== Keyboard events ==================== */
   640→  useEffect(() => {
   641→    if (gameStatus !== 'playing') return;
   642→    const handleKeyDown = (e: KeyboardEvent) => {
   643→      if (e.key === 'Escape') {
   644→        e.preventDefault();
   645→        pausedRef.current = !pausedRef.current;
   646→        setPaused(pausedRef.current);
   647→        return;
   648→      }
   649→      if (pausedRef.current) return;
   650→      switch (e.key) {
   651→        case 'ArrowLeft':
   652→          e.preventDefault();
   653→          moveLeft();
   654→          break;
   655→        case 'ArrowRight':
   656→          e.preventDefault();
   657→          moveRight();
   658→          break;
   659→        case 'ArrowDown':
   660→          e.preventDefault();
   661→          moveDown();
   662→          lastDropRef.current = performance.now();
   663→          break;
   664→        case 'ArrowUp':
   665→          e.preventDefault();
   666→          rotate();
   667→          break;
   668→        case ' ':
   669→          e.preventDefault();
   670→          hardDrop();
   671→          break;
   672→      }
   673→    };
   674→    window.addEventListener('keydown', handleKeyDown);
   675→    return () => window.removeEventListener('keydown', handleKeyDown);
   676→  }, [gameStatus, moveLeft, moveRight, moveDown, rotate, hardDrop]);
   677→
   678→  /* ==================== Start game ==================== */
   679→  const startGame = useCallback(() => {
   680→    resetBoard();
   681→    scoreRef.current = 0;
   682→    levelRef.current = 1;
   683→    linesRef.current = 0;
   684→    speedRef.current = speed;
   685→    flashRowsRef.current = [];
   686→    gameOverCheckedRef.current = false;
   687→    pausedRef.current = false;
   688→    setPaused(false);
   689→    setIsNewRecord(false);
   690→    setScore(0);
   691→    setLevel(1);
   692→    nextRef.current = randomType();
   693→    setNextPiece(nextRef.current);
   694→    spawnPiece();
   695→    statusRef.current = 'playing';
   696→    setGameStatus('playing');
   697→  }, [speed, resetBoard, spawnPiece]);
   698→
   699→  /* ==================== Restart ==================== */
   700→  const restartGame = useCallback(() => {
   701→    startGame();
   702→  }, [startGame]);
   703→
   704→  /* ==================== Next piece preview data ==================== */
   705→  const previewShape = SHAPES[nextPiece];
   706→  const previewColor = COLORS[nextPiece];
   707→
   708→  /* ==================== IDLE SCREEN ==================== */
   709→  if (gameStatus === 'idle') {
   710→    return (
   711→      <div className="flex flex-col items-center w-full min-h-[60vh] gap-3 px-2 py-2">
   712→        <div className="w-full max-w-md">
   713→          <Button variant="ghost" onClick={onBack} className="min-h-[44px] mb-2">
   714→            <ArrowLeft className="h-5 w-5 mr-2" />
   715→            返回
   716→          </Button>
   717→          <Card className="bg-card border-border">
   718→            <CardContent className="p-4 space-y-4">
   719→              <div className="text-center">
   720→                <h2 className="text-xl sm:text-2xl font-bold flex items-center justify-center gap-2">
   721→                  <span className="text-xl sm:text-2xl">🟦</span> 俄罗斯方块
   722→                </h2>
   723→                {!loadingHighScore && highScore > 0 && (
   724→                  <p className="text-xs sm:text-sm text-muted-foreground mt-1 flex items-center justify-center gap-1">
   725→                    <Trophy className="h-3.5 w-3.5 text-yellow-500" />
   726→                    最高分: {highScore}
   727→                  </p>
   728→                )}
   729→              </div>
   730→
   731→              <div className="space-y-2">
   732→                <p className="text-sm font-medium">速度设置</p>
   733→                <div className="grid grid-cols-3 gap-2">
   734→                  {(Object.entries(SPEED_CONFIG) as [SpeedKey, { label: string; base: number }][]).map(
   735→                    ([key, cfg]) => (
   736→                      <Button
   737→                        key={key}
   738→                        variant={speed === key ? 'default' : 'outline'}
   739→                        className="w-full min-h-[44px]"
   740→                        onClick={() => setSpeed(key)}
   741→                      >
   742→                        {cfg.label}
   743→                      </Button>
   744→                    )
   745→                  )}
   746→                </div>
   747→              </div>
   748→
   749→              <div className="flex items-center justify-between">
   750→                <p className="text-sm font-medium">操作说明</p>
   751→                <GameControlsHelp info={tetrisControlsInfo} variant="text" />
   752→              </div>
   753→
   754→              <Button onClick={startGame} className="w-full min-h-[44px]" size="lg">
   755→                开始游戏
   756→              </Button>
   757→            </CardContent>
   758→          </Card>
   759→        </div>
   760→      </div>
   761→    );
   762→  }
   763→
   764→  /* ==================== PLAYING / GAME OVER SCREEN ==================== */
   765→  return (
   766→    <div className="flex flex-col items-center w-full min-h-[60vh] gap-2 px-2 py-2">
   767→      {/* Compact status bar */}
   768→      <div className="w-full max-w-[300px] flex items-center justify-between gap-1">
   769→        <Button variant="ghost" onClick={onBack} className="min-h-[44px] shrink-0 px-2">
   770→          <ArrowLeft className="h-4 w-4 mr-0.5" />
   771→          <span className="hidden sm:inline">返回</span>
   772→        </Button>
   773→        <div className="flex items-center gap-1.5 shrink-0">
   774→          <h2 className="text-sm sm:text-base font-bold flex items-center gap-1 truncate">
   775→            <span>🟦</span>
   776→            <span className="hidden sm:inline">俄罗斯方块</span>
   777→          </h2>
   778→          <GameControlsHelp info={tetrisControlsInfo} variant="button" />
   779→        </div>
   780→        <div className="flex items-center gap-1.5 sm:gap-2 shrink-0">
   781→          <div className="flex items-center gap-1">
   782→            <Trophy className="h-3.5 w-3.5 text-yellow-500" />
   783→            <span className="text-xs sm:text-sm font-semibold tabular-nums">{score}</span>
   784→          </div>
   785→          <span className="text-xs text-muted-foreground tabular-nums">Lv.{level}</span>
   786→          {/* Next piece mini-preview */}
   787→          <div className="grid grid-cols-4 gap-[1px] w-5 h-5 shrink-0" title="下一个">
   788→            {previewShape.flat().map((cell, i) => (
   789→              <div
   790→                key={i}
   791→                className="rounded-[1px]"
   792→                style={{
   793→                  backgroundColor: cell ? previewColor : 'transparent',
   794→                }}
   795→              />
   796→            ))}
   797→          </div>
   798→        </div>
   799→      </div>
   800→
   801→      {/* Canvas container with HTML overlays */}
   802→      <div ref={containerRef} className="w-full max-w-[300px] mx-auto relative">
   803→        <canvas
   804→          ref={canvasRef}
   805→          className="w-full rounded-lg border border-border/50"
   806→          style={{ aspectRatio: `${COLS}/${ROWS}`, touchAction: 'none' }}
   807→        />
   808→
   809→        {/* Paused overlay */}
   810→        {paused && (
   811→          <div className="absolute inset-0 bg-black/60 rounded-lg flex flex-col items-center justify-center gap-2 z-10">
   812→            <p className="text-white text-lg sm:text-xl font-bold">已暂停</p>
   813→            <p className="text-white/60 text-xs sm:text-sm">按 Esc 继续</p>
   814→          </div>
   815→        )}
   816→
   817→        {/* Game over overlay */}
   818→        {gameStatus === 'over' && (
   819→          <div className="absolute inset-0 bg-black/65 rounded-lg flex flex-col items-center justify-center gap-2 z-10 px-4">
   820→            <p className="text-red-400 text-lg sm:text-xl font-bold">游戏结束</p>
   821→            <p className="text-white text-sm sm:text-base">得分: {score}</p>
   822→            {isNewRecord && (
   823→              <p className="text-yellow-400 text-xs sm:text-sm font-semibold animate-pulse">
   824→                🎉 新纪录！
   825→              </p>
   826→            )}
   827→            <Button
   828→              onClick={restartGame}
   829→              className="mt-2 min-h-[44px] font-semibold"
   830→              size="lg"
   831→            >
   832→              <RotateCcw className="h-4 w-4 mr-2" />
   833→              再来一局
   834→            </Button>
   835→          </div>
   836→        )}
   837→      </div>
   838→
   839→      {/* Mobile virtual buttons */}
   840→      {isMobile && (
   841→        <div className="w-full max-w-[300px] px-1">
   842→          <div className="flex items-center justify-center gap-1.5 flex-wrap">
   843→            <Button
   844→              variant="outline"
   845→              className="min-h-[44px] min-w-[44px] text-lg font-bold px-3"
   846→              onTouchStart={(e) => {
   847→                e.preventDefault();
   848→                if (!pausedRef.current) moveLeft();
   849→              }}
   850→              disabled={gameStatus === 'over' || paused}
   851→            >
   852→              ←
   853→            </Button>
   854→            <Button
   855→              variant="outline"
   856→              className="min-h-[44px] min-w-[44px] text-lg font-bold px-3"
   857→              onTouchStart={(e) => {
   858→                e.preventDefault();
   859→                if (!pausedRef.current) {
   860→                  moveDown();
   861→                  lastDropRef.current = performance.now();
   862→                }
   863→              }}
   864→              disabled={gameStatus === 'over' || paused}
   865→            >
   866→              ↓
   867→            </Button>
   868→            <Button
   869→              variant="outline"
   870→              className="min-h-[44px] min-w-[44px] text-lg font-bold px-3"
   871→              onTouchStart={(e) => {
   872→                e.preventDefault();
   873→                if (!pausedRef.current) moveRight();
   874→              }}
   875→              disabled={gameStatus === 'over' || paused}
   876→            >
   877→              →
   878→            </Button>
   879→            <Button
   880→              variant="outline"
   881→              className="min-h-[44px] min-w-[44px] text-sm font-bold px-3"
   882→              onTouchStart={(e) => {
   883→                e.preventDefault();
   884→                if (!pausedRef.current) rotate();
   885→              }}
   886→              disabled={gameStatus === 'over' || paused}
   887→            >
   888→              旋转
   889→            </Button>
   890→            <Button
   891→              variant="outline"
   892→              className="min-h-[44px] min-w-[44px] text-sm font-bold px-3"
   893→              onTouchStart={(e) => {
   894→                e.preventDefault();
   895→                if (!pausedRef.current) hardDrop();
   896→              }}
   897→              disabled={gameStatus === 'over' || paused}
   898→            >
   899→              硬降
   900→            </Button>
   901→          </div>
   902→        </div>
   903→      )}
   904→    </div>
   905→  );
   906→}
