   260→function createStars(count: number): Star[] {
   261→  const stars: Star[] = [];
   262→  for (let i = 0; i < count; i++) {
   263→    stars.push({
   264→      x: Math.random() * LOGIC_W,
   265→      y: Math.random() * LOGIC_H,
   266→      r: Math.random() * 1.5 + 0.5,
   267→      alpha: Math.random() * 0.6 + 0.4,
   268→      twinkleSpeed: Math.random() * 0.03 + 0.01,
   269→      phase: Math.random() * Math.PI * 2,
   270→    });
   271→  }
   272→  return stars;
   273→}

   275→function createExplosion(x: number, y: number, color: string, count: number): Particle[] {
   276→  const particles: Particle[] = [];
   277→  for (let i = 0; i < count; i++) {
   278→    const angle = Math.random() * Math.PI * 2;
   279→    const speed = Math.random() * 3 + 1;
   280→    particles.push({
   281→      x, y,
   282→      vx: Math.cos(angle) * speed,
   283→      vy: Math.sin(angle) * speed,
   284→      life: Math.random() * 20 + 15,
   285→      maxLife: 35,
   286→      color,
   287→      r: Math.random() * 3 + 1,
   288→    });
   289→  }
   290→  return particles;
   291→}

   310→function drawPlayer(ctx: CanvasRenderingContext2D, p: Player, frame: number, skinEmoji: string, skinAttackType: string) {
   311→  ctx.save();
   312→  ctx.translate(p.x, p.y);
   313→
   314→  if (p.invincibleTimer > 0 && Math.floor(frame / 3) % 2 === 0) {
   315→    ctx.globalAlpha = 0.35;
   316→  }
   317→
   318→  // Tail flame
   319→  let flameColor1 = '#38bdf8';
   320→  let flameColor2 = '#f97316';
   321→  if (skinAttackType === 'Fire') { flameColor1 = '#f97316'; flameColor2 = '#ef4444'; }
   322→  else if (skinAttackType === 'Ice') { flameColor1 = '#06b6d4'; flameColor2 = '#a5f3fc'; }
   323→  else if (skinAttackType === 'Star') { flameColor1 = '#fbbf24'; flameColor2 = '#f472b6'; }
   324→  else if (skinAttackType === 'Shadow') { flameColor1 = '#7c3aed'; flameColor2 = '#1e1b4b'; }
   325→  else if (skinAttackType === 'Lightning') { flameColor1 = '#fbbf24'; flameColor2 = '#fde047'; }
   326→
   327→  const flameLen = 12 + Math.sin(frame * 0.5) * 4;
   328→  const flameW = 8 + Math.sin(frame * 0.7) * 2;
   329→  const flameGrad = ctx.createLinearGradient(0, PLAYER_H / 2, 0, PLAYER_H / 2 + flameLen);
   330→  flameGrad.addColorStop(0, flameColor1);
   331→  flameGrad.addColorStop(0.4, flameColor2);
   332→  flameGrad.addColorStop(1, 'rgba(0,0,0,0)');
   333→  ctx.fillStyle = flameGrad;
   334→  ctx.beginPath();
   335→  ctx.moveTo(-flameW, PLAYER_H / 2 - 2);
   336→  ctx.lineTo(flameW, PLAYER_H / 2 - 2);
   337→  ctx.lineTo(0, PLAYER_H / 2 + flameLen);
   338→  ctx.closePath();
   339→  ctx.fill();
   340→
   341→  // Skin emoji
   342→  ctx.font = '32px sans-serif';
   343→  ctx.textAlign = 'center';
   344→  ctx.textBaseline = 'middle';
   345→  ctx.fillText(skinEmoji, 0, -2);
   346→
   347→  // Shield bubble
   348→  if (p.shieldTimer > 0) {
   349→    const shieldAlpha = p.shieldTimer < 60 ? (Math.sin(frame * 0.3) * 0.3 + 0.4) : 0.5;
   350→    ctx.strokeStyle = `rgba(6, 182, 212, ${shieldAlpha})`;
   351→    ctx.lineWidth = 2.5;
   352→    ctx.beginPath();
   353→    ctx.ellipse(0, 0, PLAYER_W / 2 + 8, PLAYER_H / 2 + 8, 0, 0, Math.PI * 2);
   354→    ctx.stroke();
   355→  }
   356→
   357→  ctx.restore();
   358→}

   360→function drawEnemy(ctx: CanvasRenderingContext2D, e: Enemy) {
   361→  ctx.save();
   362→  ctx.translate(e.x, e.y);
   363→
   364→  if (e.type === EnemyType.Normal) {
   365→    ctx.fillStyle = '#ef4444';
   366→    ctx.beginPath();
   367→    ctx.moveTo(0, e.h / 2);
   368→    ctx.lineTo(-e.w / 2, -e.h / 2);
   369→    ctx.lineTo(e.w / 2, -e.h / 2);
   370→    ctx.closePath();
   371→    ctx.fill();
   372→    ctx.strokeStyle = '#fca5a5';
   373→    ctx.lineWidth = 1;
   374→    ctx.beginPath();
   375→    ctx.moveTo(0, e.h / 2 - 6);
   376→    ctx.lineTo(-e.w / 4, -e.h / 4);
   377→    ctx.moveTo(0, e.h / 2 - 6);
   378→    ctx.lineTo(e.w / 4, -e.h / 4);
   379→    ctx.stroke();
   380→  } else if (e.type === EnemyType.Fast) {
   381→    ctx.fillStyle = '#f59e0b';
   382→    ctx.beginPath();
   383→    ctx.moveTo(0, e.h / 2);
   384→    ctx.lineTo(-e.w / 2, -e.h / 3);
   385→    ctx.lineTo(-e.w / 4, -e.h / 2);
   386→    ctx.lineTo(e.w / 4, -e.h / 2);
   387→    ctx.lineTo(e.w / 2, -e.h / 3);
   388→    ctx.closePath();
   389→    ctx.fill();
   390→    ctx.fillStyle = '#fcd34d';
   391→    ctx.beginPath();
   392→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   393→    ctx.fill();
   394→  } else if (e.type === EnemyType.Tank) {
   395→    const r = e.w / 2;
   396→    ctx.fillStyle = '#a855f7';
   397→    ctx.beginPath();
   398→    for (let i = 0; i < 6; i++) {
   399→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   400→      const px = Math.cos(angle) * r;
   401→      const py = Math.sin(angle) * r;
   402→      if (i === 0) ctx.moveTo(px, py);
   403→      else ctx.lineTo(px, py);
   404→    }
   405→    ctx.closePath();
   406→    ctx.fill();
   407→    ctx.strokeStyle = '#d8b4fe';
   408→    ctx.lineWidth = 1.5;
   409→    ctx.beginPath();
   410→    for (let i = 0; i < 6; i++) {
   411→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   412→      const px = Math.cos(angle) * (r * 0.55);
   413→      const py = Math.sin(angle) * (r * 0.55);
   414→      if (i === 0) ctx.moveTo(px, py);
   415→      else ctx.lineTo(px, py);
   416→    }
   417→    ctx.closePath();
   418→    ctx.stroke();
   419→    ctx.fillStyle = '#e9d5ff';
   420→    ctx.beginPath();
   421→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   422→    ctx.fill();
   423→  }
   424→
   425→  // Slow indicator
   426→  if (e.slowTimer > 0) {
   427→    ctx.strokeStyle = 'rgba(6, 182, 212, 0.6)';
   428→    ctx.lineWidth = 2;
   429→    ctx.beginPath();
   430→    ctx.arc(0, 0, e.w / 2 + 4, 0, Math.PI * 2);
   431→    ctx.stroke();
   432→  }
   433→
   434→  ctx.restore();
   435→
   436→  // HP bar
   437→  if (e.hp < e.maxHp) {
   438→    const barW = e.w * 0.8;
   439→    const barH = 3;
   440→    const barX = e.x - barW / 2;
   441→    const barY = e.y - e.h / 2 - 6;
   442→    ctx.fillStyle = 'rgba(255,255,255,0.2)';
   443→    ctx.fillRect(barX, barY, barW, barH);
   444→    ctx.fillStyle = '#22c55e';
   445→    ctx.fillRect(barX, barY, barW * (e.hp / e.maxHp), barH);
   446→  }
   447→}

   449→function drawBoss(ctx: CanvasRenderingContext2D, boss: Boss, frame: number) {
   450→  ctx.save();
   451→  ctx.translate(boss.x, boss.y);
   452→
   453→  if (boss.flashTimer > 0) {
   454→    ctx.globalAlpha = 0.5 + Math.sin(frame) * 0.5;
   455→    ctx.fillStyle = 'rgba(255,255,255,0.6)';

   531→  ctx.beginPath();
