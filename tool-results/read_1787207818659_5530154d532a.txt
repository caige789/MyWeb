   144→   144→/* ============ 每波僵尸生成配置 ============ */
   145→   145→function getWaveZombies(wave: number, countMul: number): { type: ZombieType; delay: number }[] {
   146→   146→  const zombies: { type: ZombieType; delay: number }[] = [];
   147→   147→  let delay = 0;
   148→   148→  const baseCounts: number[][] = [
   149→   149→    [4, 1, 0, 0, 0],
   150→   150→    [3, 2, 0, 1, 1],
   151→   151→    [2, 2, 1, 1, 2],
   152→   152→    [2, 3, 1, 1, 2],
   153→   153→    [1, 3, 2, 2, 3],
   154→   154→  ];
   155→   155→  const counts = baseCounts[Math.min(wave, baseCounts.length - 1)];
   156→   156→  const types: ZombieType[] = ['normal', 'cone', 'bucket', 'flag', 'newspaper'];
   157→   157→  for (let i = 0; i < types.length; i++) {
   158→   158→    const count = Math.round(counts[i] * countMul);
   159→   159→    for (let j = 0; j < count; j++) {
   160→   160→      zombies.push({ type: types[i], delay });
   161→   161→      delay += 2000 + Math.random() * 3000;
   162→   162→    }
   163→   163→  }
   164→   164→  for (let i = zombies.length - 1; i > 0; i--) {
   165→   165→    const j = Math.floor(Math.random() * (i + 1));
   166→   166→    [zombies[i], zombies[j]] = [zombies[j], zombies[i]];
   167→   167→  }
   168→   168→  return zombies;
   169→   169→}

   389→   389→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   390→   390→
