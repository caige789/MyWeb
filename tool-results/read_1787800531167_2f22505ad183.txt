   424→   424→      ctx.font = `bold ${Math.round(cellSize * 1.5)}px sans-serif`;
   425→   425→      ctx.textAlign = 'center';
   426→   426→      ctx.textBaseline = 'middle';
   427→   427→      ctx.fillText('游戏结束', size / 2, size / 2 - cellSize);
   428→   428→      ctx.font = `${Math.round(cellSize * 0.9)}px sans-serif`;
