     1→'use client';
     2→
     3→import { useRef, useEffect, useState, useCallback } from 'react';
     4→import { Button } from '@/components/ui/button';
     5→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card';

     8→
     9→interface FlappyBirdGameProps {
    10→  onBack: () => void;
    11→}
    12→
    13→type Phase = 'settings' | 'playing' | 'gameover';
    14→type Difficulty = 'easy' | 'normal' | 'hard';
    15→type PowerUpType = 'magnet' | 'shield' | 'feather';
    16→type PipeType = 'normal' | 'moving' | 'narrow';
    17→
    18→const GAME_NAME = 'flappy';
    19→const CANVAS_W = 360;
    20→const CANVAS_H = 520;
    21→const GROUND_H = 60;
    22→const BIRD_X = 80;
    23→const BIRD_R = 16;
    24→const PIPE_W = 52;
    25→const PIPE_GAP_EASY = 180;
    26→const PIPE_GAP_NORMAL = 140;
    27→const PIPE_GAP_HARD = 100;
    28→const PIPE_SPEED_EASY = 2;
    29→const PIPE_SPEED_NORMAL = 2.8;
    30→const PIPE_SPEED_HARD = 3.8;
    31→const BASE_GRAVITY = 0.42;
    32→const FLAP_VEL = -7;
    33→const PIPE_INTERVAL_EASY = 180;
    34→const PIPE_INTERVAL_NORMAL = 130;
    35→const PIPE_INTERVAL_HARD = 95;
    36→const MIN_GAP = 90;
    37→const COIN_R = 10;
    38→const POWERUP_R = 14;
    39→const MAGNET_DURATION = 300; // frames (~5s at 60fps)
    40→const FEATHER_DURATION = 300;
    41→const MAGNET_RANGE = 100;
    42→const LEVEL_UP_INTERVAL = 10; // pipes per level
    43→
    44→const POWERUP_EMOJI: Record<PowerUpType, string> = {
    45→  magnet: '\u2B50',
    46→  shield: '\uD83D\uDEE1\uFE0F',
    47→  feather: '\uD83E\uDDAB',
    48→};
    49→
    50→const POWERUP_LABEL: Record<PowerUpType, string> = {
    51→  magnet: 'Magnet',
    52→  shield: 'Shield',
    53→  feather: 'Feather',
    54→};
    55→
    56→interface Pipe {
    57→  x: number;
    58→  gapY: number;
    59→  passed: boolean;
    60→  pipeType: PipeType;
    61→  moveDir: number;
    62→  moveSpeed: number;
    63→  moveMin: number;
    64→  moveMax: number;
    65→}
    66→
    67→interface Coin {
    68→  x: number;
    69→  y: number;
    70→  collected: boolean;
    71→}
    72→
    73→interface PowerUp {
    74→  x: number;
    75→  y: number;
    76→  type: PowerUpType;
    77→  collected: boolean;
    78→}
    79→
    80→const DIFF_CONFIG: Record<Difficulty, { gap: number; speed: number; interval: number; label: string }> = {
    81→  easy: { gap: PIPE_GAP_EASY, speed: PIPE_SPEED_EASY, interval: PIPE_INTERVAL_EASY, label: '\u7B80\u5355' },
    82→  normal: { gap: PIPE_GAP_NORMAL, speed: PIPE_SPEED_NORMAL, interval: PIPE_INTERVAL_NORMAL, label: '\u666E\u901A' },
    83→  hard: { gap: PIPE_GAP_HARD, speed: PIPE_SPEED_HARD, interval: PIPE_INTERVAL_HARD, label: '\u56F0\u96BE' },
    84→};
    85→
    86→function getSkyGradient(ctx: CanvasRenderingContext2D, h: number) {
    87→  const g = ctx.createLinearGradient(0, 0, 0, h);
    88→  g.addColorStop(0, '#87CEEB');
    89→  g.addColorStop(0.7, '#B0E0F6');
    90→  g.addColorStop(1, '#E0F4FF');
    91→  return g;
    92→}

   121→export default function FlappyBirdGame({ onBack }: FlappyBirdGameProps) {
   122→  const canvasRef = useRef<HTMLCanvasElement>(null);
   123→  const containerRef = useRef<HTMLDivElement>(null);
   124→  const scaleRef = useRef(1);
   125→  const rafRef = useRef<number>(0);
   126→  const [phase, setPhase] = useState<Phase>('settings');
   127→  const [difficulty, setDifficulty] = useState<Difficulty>('normal');
   128→  const [score, setScore] = useState(0);
   129→  const [bestScore, setBestScore] = useState(0);
   130→  const [loadingBest, setLoadingBest] = useState(true);
   131→
   132→  // game refs
   133→  const birdYRef = useRef(CANVAS_H / 2);
   134→  const velRef = useRef(0);
   135→  const pipesRef = useRef<Pipe[]>([]);
   136→  const coinsRef = useRef<Coin[]>([]);
   137→  const powerupsRef = useRef<PowerUp[]>([]);
   138→  const scoreRef = useRef(0);
   139→  const frameRef = useRef(0);
   140→  const diffRef = useRef<Difficulty>('normal');
   141→  const phaseRef = useRef<Phase>('settings');
   142→  const passedCountRef = useRef(0); // total pipes passed
   143→  const levelRef = useRef(1);
   144→  const currentGapRef = useRef(0);
   145→  const currentSpeedRef = useRef(0);
   146→  // power-up state
   147→  const magnetTimerRef = useRef(0);
   148→  const shieldActiveRef = useRef(false);
   149→  const featherTimerRef = useRef(0);
   150→  // floating text effects
   151→  const floatTextsRef = useRef<Array<{ x: number; y: number; text: string; life: number; color: string }>>([]);
   152→
   153→  const cloudsRef = useRef<Array<{ x: number; y: number; r: number }>>([
   154→    { x: 50, y: 60, r: 30 }, { x: 180, y: 40, r: 25 }, { x: 300, y: 80, r: 35 },
   155→  ]);
   156→
   157→  const fetchBestScore = useCallback(async () => {
   158→    try {
   159→      const res = await fetch('/api/games/scores');
   160→      const json = await res.json();
   161→      if (json.data) {
   162→        const entry = json.data.find((s: { game: string; score: number }) => s.game === GAME_NAME);
   163→        setBestScore(entry ? entry.score : 0);
   164→      }
   165→    } catch { /* ignore */ }
   166→    setLoadingBest(false);
   167→  }, []);
   168→
   169→  useEffect(() => { fetchBestScore(); }, [fetchBestScore]);
   170→
   171→  const submitScore = useCallback(async (s: number) => {
   172→    try {
   173→      await fetch('/api/games/scores', {
   174→        method: 'POST',
   175→        headers: { 'Content-Type': 'application/json' },
   176→        body: JSON.stringify({ game: GAME_NAME, score: s }),
   177→      });
   178→    } catch { /* ignore */ }
   179→  }, []);
   180→
   181→  const flap = useCallback(() => {
   182→    if (phaseRef.current !== 'playing') return;
   183→    velRef.current = FLAP_VEL;
   184→  }, []);
   185→
   186→  const addFloatText = useCallback((x: number, y: number, text: string, color: string) => {
   187→    floatTextsRef.current.push({ x, y, text, life: 40, color });
   188→  }, []);
   189→
   190→  const initGame = useCallback((diff: Difficulty) => {
   191→    diffRef.current = diff;
   192→    const cfg = DIFF_CONFIG[diff];
   193→    birdYRef.current = CANVAS_H / 2;
   194→    velRef.current = 0;
   195→    pipesRef.current = [];
   196→    coinsRef.current = [];
   197→    powerupsRef.current = [];
   198→    scoreRef.current = 0;
   199→    frameRef.current = 0;
   200→    passedCountRef.current = 0;
   201→    levelRef.current = 1;
   202→    currentGapRef.current = cfg.gap;
   203→    currentSpeedRef.current = cfg.speed;
   204→    magnetTimerRef.current = 0;
   205→    shieldActiveRef.current = false;
   206→    featherTimerRef.current = 0;
   207→    floatTextsRef.current = [];
   208→    setScore(0);
   209→    phaseRef.current = 'playing';
   210→    setPhase('playing');
   211→  }, []);
   212→
   213→  // game loop
   214→  useEffect(() => {
   215→    if (phase !== 'playing') return;
   216→    const canvas = canvasRef.current;
   217→    if (!canvas) return;
   218→    const ctx = canvas.getContext('2d');
   219→    if (!ctx) return;
   220→    const dpr = Math.min(window.devicePixelRatio || 1, 2);
   221→
   222→    function resize() {
   223→      if (!canvas) return;
   224→      const dw = canvas.clientWidth;
   225→      if (dw <= 0) return;
   226→      scaleRef.current = dw / CANVAS_W;
   227→      canvas.width = dw * dpr;
   228→      canvas.height = (dw / CANVAS_W * CANVAS_H) * dpr;
   229→    }
   230→    resize();
   231→    const ro = new ResizeObserver(resize);
   232→    if (containerRef.current) ro.observe(containerRef.current);
   233→    const onVis = () => { if (!document.hidden) resize(); };
   234→    document.addEventListener('visibilitychange', onVis);
   235→
   236→    const playH = CANVAS_H - GROUND_H;
   237→
   238→    function loop() {
   239→      if (!ctx) return;
   240→      if (phaseRef.current !== 'playing') return;
   241→      frameRef.current++;
   242→      ctx.setTransform(dpr * scaleRef.current, 0, 0, dpr * scaleRef.current, 0, 0);
   243→
   244→      const curGap = currentGapRef.current;
   245→      const curSpeed = currentSpeedRef.current;
   246→      const gravity = featherTimerRef.current > 0 ? BASE_GRAVITY * 0.5 : BASE_GRAVITY;
   247→
   248→      // decrement power-up timers
   249→      if (magnetTimerRef.current > 0) magnetTimerRef.current--;
   250→      if (featherTimerRef.current > 0) featherTimerRef.current--;
   251→
   252→      // physics
   253→      velRef.current += gravity;
   254→      birdYRef.current += velRef.current;
   255→
   256→      // spawn pipe
   257→      const cfg = DIFF_CONFIG[diffRef.current];
   258→      if (frameRef.current % cfg.interval === 0) {
   259→        // determine pipe type: 20% special
   260→        const roll = Math.random();
   261→        let pipeType: PipeType = 'normal';
   262→        if (roll < 0.12) {
   263→          pipeType = 'moving';
   264→        } else if (roll < 0.20) {
   265→          pipeType = 'narrow';
   266→        }
   267→
   268→        let gap = curGap;
   269→        if (pipeType === 'narrow') {
   270→          gap = Math.max(MIN_GAP, curGap - 20);
   271→        }
   272→
   273→        const minGapY = gap / 2 + 30;
   274→        const maxGapY = playH - gap / 2 - 30;
   275→        const gapY = minGapY + Math.random() * (maxGapY - minGapY);
   276→
   277→        pipesRef.current.push({
   278→          x: CANVAS_W + PIPE_W,
   279→          gapY,
   280→          passed: false,
   281→          pipeType,
   282→          moveDir: 1,
   283→          moveSpeed: 0.5 + Math.random() * 0.5,
   284→          moveMin: gap / 2 + 30,
   285→          moveMax: playH - gap / 2 - 30,
   286→        });
   287→
   288→        // spawn coins in the gap area (1-3 coins)
   289→        const coinCount = 1 + Math.floor(Math.random() * 3);
   290→        for (let i = 0; i < coinCount; i++) {
   291→          const coinX = CANVAS_W + PIPE_W / 2 + (Math.random() - 0.5) * 10;
   292→          const coinYRange = gap - 20;
   293→          const coinY = gapY - coinYRange / 2 + (coinCount === 1
   294→            ? coinYRange / 2
   295→            : (i / (coinCount - 1)) * coinYRange);
   296→          coinsRef.current.push({ x: coinX, y: coinY, collected: false });
   297→        }
   298→
   299→        // spawn power-up (12% chance)
   300→        if (Math.random() < 0.12) {
   301→          const types: PowerUpType[] = ['magnet', 'shield', 'feather'];
   302→          const puType = types[Math.floor(Math.random() * types.length)];
   303→          const puY = gapY + (Math.random() - 0.5) * (gap * 0.4);
   304→          powerupsRef.current.push({
   305→            x: CANVAS_W + PIPE_W / 2 + 30 + Math.random() * 20,
   306→            y: puY,
   307→            type: puType,
   308→            collected: false,
   309→          });
   310→        }
   311→      }
   312→
   313→      // move pipes and handle moving pipe animation
   314→      for (const p of pipesRef.current) {
   315→        p.x -= curSpeed;
   316→
   317→        // moving pipes oscillate
   318→        if (p.pipeType === 'moving') {
   319→          p.gapY += p.moveDir * p.moveSpeed;
   320→          if (p.gapY >= p.moveMax) { p.moveDir = -1; p.gapY = p.moveMax; }
   321→          if (p.gapY <= p.moveMin) { p.moveDir = 1; p.gapY = p.moveMin; }
   322→        }
   323→
   324→        // scoring
   325→        if (!p.passed && p.x + PIPE_W < BIRD_X) {
   326→          p.passed = true;
   327→          let pts = 1;
   328→          if (p.pipeType === 'narrow') pts = 3;
   329→          scoreRef.current += pts;
   330→          passedCountRef.current++;
   331→          setScore(scoreRef.current);
   332→
   333→          if (pts > 1) {
   334→            addFloatText(BIRD_X, birdYRef.current - 20, `+${pts}`, '#FF6B6B');
   335→          }
   336→
   337→          // level up check
   338→          if (passedCountRef.current % LEVEL_UP_INTERVAL === 0) {
   339→            levelRef.current++;
   340→            const newGap = Math.max(MIN_GAP, DIFF_CONFIG[diffRef.current].gap - (levelRef.current - 1) * 5);
   341→            const newSpeed = DIFF_CONFIG[diffRef.current].speed + (levelRef.current - 1) * 0.3;
   342→            currentGapRef.current = newGap;
   343→            currentSpeedRef.current = newSpeed;
   344→            addFloatText(CANVAS_W / 2, CANVAS_H / 2 - 30, `Level ${levelRef.current}!`, '#FFD700');
   345→          }
   346→        }
   347→      }
   348→      pipesRef.current = pipesRef.current.filter(p => p.x + PIPE_W > -10);
   349→
   350→      // move coins
   351→      for (const c of coinsRef.current) {
   352→        c.x -= curSpeed;
   353→
   354→        // magnet attraction
   355→        if (!c.collected && magnetTimerRef.current > 0) {
   356→          const dx = BIRD_X - c.x;
   357→          const dy = birdYRef.current - c.y;
   358→          const dist = Math.sqrt(dx * dx + dy * dy);
   359→          if (dist < MAGNET_RANGE && dist > 1) {
   360→            c.x += (dx / dist) * 4;
   361→            c.y += (dy / dist) * 4;
   362→          }
   363→        }
   364→
   365→        // coin collection
   366→        if (!c.collected) {
   367→          const dx = BIRD_X - c.x;
   368→          const dy = birdYRef.current - c.y;
   369→          if (dx * dx + dy * dy < (BIRD_R + COIN_R) * (BIRD_R + COIN_R)) {
   370→            c.collected = true;
   371→            scoreRef.current += 5;
   372→            setScore(scoreRef.current);
   373→            addFloatText(c.x, c.y - 10, '+5', '#FFD700');
   374→          }
   375→        }
   376→      }
   377→      coinsRef.current = coinsRef.current.filter(c => c.x > -20);
   378→
   379→      // move and collect power-ups
   380→      for (const pu of powerupsRef.current) {
   381→        pu.x -= curSpeed;
   382→
   383→        if (!pu.collected) {
   384→          const dx = BIRD_X - pu.x;
   385→          const dy = birdYRef.current - pu.y;
   386→          if (dx * dx + dy * dy < (BIRD_R + POWERUP_R) * (BIRD_R + POWERUP_R)) {
   387→            pu.collected = true;
   388→            switch (pu.type) {
   389→              case 'magnet':
   390→                magnetTimerRef.current = MAGNET_DURATION;
   391→                addFloatText(pu.x, pu.y - 15, POWERUP_LABEL.magnet, '#FFD700');
   392→                break;
   393→              case 'shield':
   394→                shieldActiveRef.current = true;
   395→                addFloatText(pu.x, pu.y - 15, POWERUP_LABEL.shield, '#00E5FF');
   396→                break;
   397→              case 'feather':
   398→                featherTimerRef.current = FEATHER_DURATION;
   399→                addFloatText(pu.x, pu.y - 15, POWERUP_LABEL.feather, '#E040FB');
   400→                break;
   401→            }
   402→          }
   403→        }
   404→      }
   405→      powerupsRef.current = powerupsRef.current.filter(pu => pu.x > -20);
   406→
   407→      // update floating texts
   408→      for (const ft of floatTextsRef.current) {
   409→        ft.y -= 1;
   410→        ft.life--;
   411→      }
   412→      floatTextsRef.current = floatTextsRef.current.filter(ft => ft.life > 0);
   413→
   414→      // collision: ground / ceiling
   415→      if (birdYRef.current + BIRD_R > playH || birdYRef.current - BIRD_R < 0) {
   416→        endGame(); return;
   417→      }
   418→      // collision: pipes
   419→      for (const p of pipesRef.current) {
   420→        const pipeGap = p.pipeType === 'narrow'
   421→          ? Math.max(MIN_GAP, currentGapRef.current - 20)
   422→          : currentGapRef.current;
   423→        if (BIRD_X + BIRD_R > p.x && BIRD_X - BIRD_R < p.x + PIPE_W) {
   424→          const topPipeBottom = p.gapY - pipeGap / 2;
   425→          const botPipeTop = p.gapY + pipeGap / 2;
   426→          if (birdYRef.current - BIRD_R < topPipeBottom || birdYRef.current + BIRD_R > botPipeTop) {
   427→            if (shieldActiveRef.current) {
   428→              shieldActiveRef.current = false;
   429→              addFloatText(BIRD_X, birdYRef.current - 25, 'Shield!', '#00E5FF');
   430→              // push bird back to safe position
   431→              const midY = (topPipeBottom + botPipeTop) / 2;
   432→              birdYRef.current = midY;
   433→              velRef.current = 0;
   434→            } else {
   435→              endGame(); return;
   436→            }
   437→          }
   438→        }
   439→      }
   440→
   441→      // ===== DRAWING =====
   442→      // sky
   443→      ctx.fillStyle = getSkyGradient(ctx, CANVAS_H);
   444→      ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   445→
   446→      // clouds
   447→      ctx.fillStyle = 'rgba(255,255,255,0.8)';
   448→      for (const c of cloudsRef.current) {
   449→        c.x -= 0.3;
   450→        if (c.x + c.r * 2 < 0) c.x = CANVAS_W + c.r;
   451→        ctx.beginPath(); ctx.arc(c.x, c.y, c.r, 0, Math.PI * 2); ctx.fill();
   452→        ctx.beginPath(); ctx.arc(c.x - c.r * 0.6, c.y + 5, c.r * 0.7, 0, Math.PI * 2); ctx.fill();
   453→        ctx.beginPath(); ctx.arc(c.x + c.r * 0.7, c.y + 3, c.r * 0.65, 0, Math.PI * 2); ctx.fill();
   454→      }
   455→
   456→      // pipes
   457→      for (const p of pipesRef.current) {
   458→        const pipeGap = p.pipeType === 'narrow'
   459→          ? Math.max(MIN_GAP, currentGapRef.current - 20)
   460→          : currentGapRef.current;
   461→        const topBottom = p.gapY - pipeGap / 2;
   462→        const botTop = p.gapY + pipeGap / 2;
   463→
   464→        let pipeColor = '#2E8B57';
   465→        let capColor = '#3CB371';
   466→        if (p.pipeType === 'moving') {
   467→          pipeColor = '#E67E22';
   468→          capColor = '#F39C12';
   469→        } else if (p.pipeType === 'narrow') {
   470→          pipeColor = '#C0392B';
   471→          capColor = '#E74C3C';
   472→        }
   473→
   474→        // top pipe
   475→        ctx.fillStyle = pipeColor;
   476→        ctx.fillRect(p.x, 0, PIPE_W, topBottom);
   477→        ctx.strokeStyle = 'rgba(0,0,0,0.3)';
   478→        ctx.lineWidth = 2;
   479→        ctx.strokeRect(p.x, 0, PIPE_W, topBottom);
   480→        // bottom pipe
   481→        ctx.fillStyle = pipeColor;
   482→        ctx.fillRect(p.x, botTop, PIPE_W, playH - botTop);
   483→        ctx.strokeRect(p.x, botTop, PIPE_W, playH - botTop);
   484→        // pipe caps
   485→        ctx.fillStyle = capColor;
   486→        ctx.fillRect(p.x - 4, topBottom - 20, PIPE_W + 8, 20);
   487→        ctx.strokeRect(p.x - 4, topBottom - 20, PIPE_W + 8, 20);
   488→        ctx.fillStyle = capColor;
   489→        ctx.fillRect(p.x - 4, botTop, PIPE_W + 8, 20);
   490→        ctx.strokeRect(p.x - 4, botTop, PIPE_W + 8, 20);
   491→
   492→        // moving pipe indicator arrows
   493→        if (p.pipeType === 'moving') {
   494→          ctx.fillStyle = 'rgba(255,255,255,0.6)';
   495→          ctx.font = '14px sans-serif';
   496→          ctx.textAlign = 'center';
   497→          ctx.textBaseline = 'middle';
   498→          ctx.fillText('\u2195', p.x + PIPE_W / 2, topBottom - 35);
   499→          ctx.fillText('\u2195', p.x + PIPE_W / 2, botTop + 35);
   500→        }
   501→
   502→        // narrow pipe bonus indicator
   503→        if (p.pipeType === 'narrow') {
   504→          ctx.fillStyle = 'rgba(255,255,255,0.7)';
   505→          ctx.font = 'bold 12px sans-serif';
   506→          ctx.textAlign = 'center';
   507→          ctx.textBaseline = 'middle';
   508→          ctx.fillText('x3', p.x + PIPE_W / 2, topBottom - 35);
   509→        }
   510→      }
   511→
   512→      // coins
   513→      for (const c of coinsRef.current) {
   514→        if (c.collected) continue;
   515→        ctx.beginPath();
   516→        ctx.arc(c.x, c.y, COIN_R, 0, Math.PI * 2);
