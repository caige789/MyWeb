   159→function getWaveZombies(wave: number, countMul: number): { type: ZombieType; delay: number }[] {
   160→  const zombies: { type: ZombieType; delay: number }[] = [];
   161→  let delay = 0;
   162→  const baseCounts: number[][] = [
   163→    [4, 1, 0, 0, 0, 0, 0],
   164→    [3, 2, 0, 1, 1, 0, 0],
   165→    [2, 2, 1, 1, 1, 1, 0],
   166→    [2, 3, 1, 1, 2, 1, 1],
   167→    [1, 3, 2, 2, 2, 2, 2],
   168→  ];
   169→  const counts = baseCounts[Math.min(wave, baseCounts.length - 1)];
   170→  const types: ZombieType[] = ['normal', 'cone', 'bucket', 'flag', 'newspaper', 'football', 'gargantuar'];
   171→  for (let i = 0; i < types.length; i++) {
   172→    const count = Math.round(counts[i] * countMul);
   173→    for (let j = 0; j < count; j++) {
   174→      zombies.push({ type: types[i], delay });
   175→      delay += 2000 + Math.random() * 3000;
   176→    }
   177→  }
   178→  for (let i = zombies.length - 1; i > 0; i--) {
   179→    const j = Math.floor(Math.random() * (i + 1));
   180→    [zombies[i], zombies[j]] = [zombies[j], zombies[i]];
   181→  }
   182→  return zombies;
   183→}

   513→            score: def.score, hitFlash: 0, eating: false,
   514→            walkPhase: Math.random() * Math.PI * 2,
   515→            slowTimer: 0, paperBroken: false, smashed: false,
   516→          });
   517→          waveSpawnIdxRef.current++;
   518→        }
   519→
   520→        // Check wave end
   521→        if (
   522→          waveElapsedRef.current >= WAVE_DURATION * 1000 &&
   523→          waveSpawnIdxRef.current >= wz.length &&
   524→          zombiesRef.current.length === 0
   525→        ) {
   526→          if (waveRef.current >= TOTAL_WAVES) {
   527→            wonRef.current = true;
   528→            submitScoreRef.current(scoreRef.current);
   529→            setDisplayScore(scoreRef.current);
   530→            setPhase('won');
   531→          } else {
   532→            waveRef.current++;
   533→            waveElapsedRef.current = 0;
   534→            waveSpawnIdxRef.current = 0;
   535→            waveZombiesRef.current = getWaveZombies(waveRef.current - 1, cfg.zombieCountMul);
   536→            setDisplayWave(waveRef.current);
   537→          }
   538→        }
   539→
   540→        // Update plants
   541→        const plants = plantsRef.current;
   542→        for (let pi = plants.length - 1; pi >= 0; pi--) {
   543→          const p = plants[pi];
   544→          if (p.dead || p.hp <= 0) { plants.splice(pi, 1); continue; }
   545→
   546→          // Cherry bomb
   547→          if (p.type === 'cherrybomb') {
   548→            p.fuseTimer += deltaMs;
   549→            if (p.fuseTimer >= CHERRY_FUSE_TIME) {
   550→              const cx = (p.col + 0.5) * LOGIC_W / COLS;
   551→              const cy = (p.row + 0.5) * LOGIC_H / ROWS;
   552→              const radius = CHERRY_RADIUS_CELLS * Math.max(LOGIC_W / COLS, LOGIC_H / ROWS);
   553→              for (const z of zombiesRef.current) {
   554→                const zy = (z.row + 0.5) * LOGIC_H / ROWS;
   555→                const dx = z.x - cx;
   556→                const dy = zy - cy;
   557→                if (dx * dx + dy * dy < radius * radius) {
   558→                  z.hp -= CHERRY_DAMAGE;
   559→                  z.hitFlash = 10;
   560→                }
   561→              }
   562→              p.hp = 0;
   563→            }
   564→            continue;
   565→          }
   566→
   567→          // Chomper
   568→          if (p.type === 'chomper') {
   569→            if (p.digesting) {
   570→              p.digestTimer += deltaMs;
   571→              if (p.digestTimer >= CHOMP_DIGEST_TIME) {
   572→                p.digesting = false;
   573→                p.digestTimer = 0;
   574→              }
   575→              continue;
   576→            }
   577→            const chompRange = (LOGIC_W / COLS) * 1.2;
   578→            const px = (p.col + 0.5) * LOGIC_W / COLS;
   579→            const py = (p.row + 0.5) * LOGIC_H / ROWS;
   580→            for (const z of zombiesRef.current) {
   581→              if (z.hp <= 0 || z.row !== p.row) continue;
   582→              const zy = (z.row + 0.5) * LOGIC_H / ROWS;
   583→              const dx = z.x - px;
   584→              const dy = zy - py;
   585→              if (dx > 0 && dx < chompRange && Math.abs(dy) < LOGIC_H / ROWS * 0.5) {
   586→                if (z.hp <= CHOMP_MAX_HP) {
   587→                  z.hp = 0;
   588→                  p.digesting = true;
   589→                  p.digestTimer = 0;
   590→                }
   591→                break;
   592→              }
   593→            }
   594→            continue;
   595→          }
   596→
   597→          // Potato mine - arm timer
   598→          if (p.type === 'potatomine') {
   599→            if (!p.armed) {
   600→              p.armTimer += deltaMs;
   601→              if (p.armTimer >= MINE_ARM_TIME) {
   602→                p.armed = true;
   603→              }
   604→              continue;
   605→            }
   606→            // Explode when zombie steps on it
   607→            const mx = (p.col + 0.5) * LOGIC_W / COLS;
   608→            const my = (p.row + 0.5) * LOGIC_H / ROWS;
   609→            const mineR = LOGIC_W / COLS * 0.6;
   610→            for (const z of zombiesRef.current) {
   611→              if (z.hp <= 0 || z.row !== p.row) continue;
   612→              const zy = (z.row + 0.5) * LOGIC_H / ROWS;
   613→              const dx = z.x - mx;
   614→              const dy = zy - my;
   615→              if (dx * dx + dy * dy < mineR * mineR) {
   616→                z.hp -= 1800;
   617→                z.hitFlash = 10;
   618→                p.hp = 0;
   619→                break;
   620→              }
   621→            }
   622→            continue;
   623→          }
   624→
   625→          // Sunflower
   626→          if (p.type === 'sunflower') {
   627→            p.sunTimer += deltaMs;
   628→            if (p.sunTimer >= SUNFLOWER_INTERVAL) {
   629→              p.sunTimer -= SUNFLOWER_INTERVAL;
   630→              const sx = (p.col + 0.5) * LOGIC_W / COLS;
   631→              const sy = p.row * LOGIC_H / ROWS + LOGIC_H / ROWS * 0.2;
   632→              sunsRef.current.push({
   633→                id: sunIdRef.current++,
   634→                x: sx, y: sy,
   635→                targetY: sy + 20 + Math.random() * 30,
   636→                vy: 0, collected: false, alpha: 1,
   637→                bouncePhase: 0, lifeTimer: SUN_LIFE, scale: 1,
   638→              });
   639→            }
   640→          }
   641→
   642→          // Shooters
   643→          if (p.type === 'peashooter' || p.type === 'doubleshoot' || p.type === 'iceshooter') {
   644→            const hasZombie = zombiesRef.current.some(
   645→              z => z.row === p.row && z.x > (p.col + 0.5) * LOGIC_W / COLS && z.hp > 0
   646→            );
   647→            if (hasZombie) {
   648→              p.shootTimer += deltaMs;
   649→              if (p.shootTimer >= SHOOT_INTERVAL) {
   650→                p.shootTimer -= SHOOT_INTERVAL;
   651→                const px = (p.col + 0.8) * LOGIC_W / COLS;
   652→                const py = (p.row + 0.5) * LOGIC_H / ROWS;
   653→                const isIce = p.type === 'iceshooter';
   654→                peasRef.current.push({ row: p.row, x: px, y: py, isIce, isFire: false, damage: PEA_DAMAGE });
   655→                if (p.type === 'doubleshoot') {
   656→                  peasRef.current.push({ row: p.row, x: px - 8, y: py - 4, isIce: false, isFire: false, damage: PEA_DAMAGE });
   657→                }
   658→              }
   659→            }
   660→          }
   661→        }
   662→
   663→        // Check torchwood - convert passing peas to fire
   664→        for (const plant of plants) {
   665→          if (plant.type !== 'torchwood' || plant.hp <= 0) continue;
   666→          const tx = (plant.col + 0.5) * LOGIC_W / COLS;
   667→          const tr = LOGIC_W / COLS * 0.6;
   668→          for (const pea of peasRef.current) {
   669→            if (pea.isFire) continue;
   670→            if (pea.row === plant.row && Math.abs(pea.x - tx) < tr) {
   671→              pea.isFire = true;
   672→              pea.isIce = false;
   673→              pea.damage = PEA_DAMAGE + TORCH_EXTRA_DAMAGE;
   674→            }
   675→          }
   676→        }
   677→
   678→        // Update peas
   679→        for (let i = peasRef.current.length - 1; i >= 0; i--) {
   680→          peasRef.current[i].x += PEA_SPEED * dt;
   681→        }
   682→
   683→        // Pea-zombie collision
   684→        for (let i = peasRef.current.length - 1; i >= 0; i--) {
   685→          const pea = peasRef.current[i];
   686→          let hit = false;
   687→          const hitR = LOGIC_W / COLS * 0.4;
   688→          for (const z of zombiesRef.current) {
   689→            if (z.hp <= 0 || z.row !== pea.row) continue;
   690→            const zy = (z.row + 0.5) * LOGIC_H / ROWS;
   691→            const dx = pea.x - z.x;
   692→            const dy = pea.y - zy;
   693→            if (dx * dx + dy * dy < hitR * hitR) {
   694→              z.hp -= pea.damage;
   695→              z.hitFlash = 6;
   696→              if (pea.isIce) z.slowTimer = ICE_SLOW_DURATION;
   697→              hit = true;
   698→              break;
   699→            }
   700→          }
   701→          if (hit || pea.x > LOGIC_W + 20) {
   702→            peasRef.current.splice(i, 1);
   703→          }
   704→        }
   705→
   706→        // Update mowers
   707→        for (const m of mowersRef.current) {
   708→          if (!m.active) continue;
   709→          m.x += m.speed * dt;
   710→          // Kill zombies in path
   711→          for (const z of zombiesRef.current) {
   712→            if (z.hp <= 0 || z.row !== m.row) continue;
   713→            if (Math.abs(z.x - m.x) < MOWER_KILL_RADIUS) {
   714→              z.hp = 0;
   715→              z.smashed = true;
   716→            }
   717→          }
   718→        }
   719→        mowersRef.current = mowersRef.current.filter(m => !m.active || m.x < LOGIC_W + 50);
   720→
   721→        // Update zombies
   722→        const zombies = zombiesRef.current;
   723→        for (let i = zombies.length - 1; i >= 0; i--) {
   724→          const z = zombies[i];
   725→          if (z.hp <= 0) {
   726→            scoreRef.current += z.score;
   727→            setDisplayScore(scoreRef.current);
   728→            zombies.splice(i, 1);
   729→            continue;
   730→          }
   731→          if (z.hitFlash > 0) z.hitFlash -= dt;
   732→          z.walkPhase += 0.05 * dt;
   733→
   734→          if (z.slowTimer > 0) {
   735→            z.slowTimer -= deltaMs;
   736→            z.speed = z.baseSpeed * cfg.zombieSpeedMul * ICE_SLOW_FACTOR;
   737→          } else {
   738→            z.speed = z.baseSpeed * cfg.zombieSpeedMul;
   739→          }
   740→
   741→          if (z.type === 'newspaper' && !z.paperBroken && z.hp <= 100) {
   742→            z.paperBroken = true;
   743→            z.baseSpeed = 0.6;
   744→            z.maxHp = 100;
   745→          }
   746→
   747→          // Check if zombie is in a mower's row and mower hasn't fired
   748→          if (z.x < LOGIC_W / COLS * 0.5) {
   749→            const mower = mowersRef.current.find(m => m.row === z.row && !m.active);
   750→            if (mower) {
   751→              mower.active = true;
   752→              continue;
   753→            }
   754→          }
   755→
   756→          z.eating = false;
   757→          const zCol = Math.floor(z.x / (LOGIC_W / COLS));
   758→          const target = plants.find(p => p.row === z.row && p.col === zCol && !p.dead && p.hp > 0);
   759→          if (target) {
   760→            z.eating = true;
   761→            const dmg = z.type === 'gargantuar' ? 1.5 : 0.5;
   762→            target.hp -= dmg * dt;
   763→          } else {
   764→            z.x -= z.speed * dt;
   765→          }
   766→
   767→          if (z.x < LOGIC_W / COLS * 0.2) {
   768→            overRef.current = true;
   769→            setDisplayScore(scoreRef.current);
   770→            submitScoreRef.current(scoreRef.current);
   771→            setPhase('lost');
   772→          }
   773→        }
   774→
   775→        // Update suns
   776→        for (let i = sunsRef.current.length - 1; i >= 0; i--) {
   777→          const s = sunsRef.current[i];
   778→          if (s.collected) {
   779→            s.scale = Math.max(s.scale - 0.08 * dt, 0);
   780→            s.alpha -= 0.08 * dt;
   781→            if (s.alpha <= 0) sunsRef.current.splice(i, 1);
   782→            continue;
   783→          }
   784→          s.lifeTimer -= dt;
   785→          if (s.lifeTimer <= 0) { sunsRef.current.splice(i, 1); continue; }
   786→          if (s.y < s.targetY) {
   787→            s.vy += 0.15 * dt;
   788→            s.y += s.vy * dt;
   789→            if (s.y >= s.targetY) {
   790→              s.y = s.targetY;
   791→              s.vy = -s.vy * 0.3;
   792→              s.bouncePhase++;
   793→              if (s.bouncePhase >= 3) { s.y = s.targetY; s.vy = 0; }
   794→            }
   795→          }
   796→        }
   797→      }
   798→
   799→      // ===== DRAW =====
   800→      const lw = LOGIC_W / COLS;
   801→      const lh = LOGIC_H / ROWS;
   802→      ctx.clearRect(0, 0, LOGIC_W, LOGIC_H);
   803→
   804→      // Grass checkerboard
   805→      for (let r = 0; r < ROWS; r++) {
   806→        for (let c = 0; c < COLS; c++) {
   807→          ctx.fillStyle = (r + c) % 2 === 0 ? '#5cb85c' : '#4cae4c';
   808→          ctx.fillRect(c * lw, r * lh, lw, lh);
   809→        }
   810→      }
   811→
   812→      // Grid lines
   813→      ctx.strokeStyle = 'rgba(0,0,0,0.06)';
   814→      ctx.lineWidth = 1;
   815→      for (let r = 0; r <= ROWS; r++) {
   816→        ctx.beginPath(); ctx.moveTo(0, r * lh); ctx.lineTo(LOGIC_W, r * lh); ctx.stroke();
   817→      }
   818→      for (let c = 0; c <= COLS; c++) {
   819→        ctx.beginPath(); ctx.moveTo(c * lw, 0); ctx.lineTo(c * lw, LOGIC_H); ctx.stroke();
   820→      }
   821→
   822→      // Mowers
   823→      for (const m of mowersRef.current) {
   824→        const my = (m.row + 0.5) * lh;
   825→        ctx.fillStyle = '#666';
   826→        ctx.fillRect(m.x - 12, my - 8, 24, 16);
   827→        ctx.fillStyle = '#f44336';
   828→        ctx.fillRect(m.x + 6, my - 5, 8, 10);
   829→      }
   830→
   831→      // Cursor highlight
   832→      const cursor = cursorRef.current;
   833→      const sel = selPlantRef.current;
   834→      if (cursor && (sel || isShovelRef.current)) {
   835→        const hCol = Math.floor(cursor.x / lw);
   836→        const hRow = Math.floor(cursor.y / lh);
   837→        if (hRow >= 0 && hRow < ROWS && hCol >= 0 && hCol < COLS) {
   838→          const exists = plantsRef.current.some(p => p.row === hRow && p.col === hCol);
   839→          if (isShovelRef.current) {
   840→            ctx.fillStyle = exists ? 'rgba(244,67,54,0.35)' : 'rgba(255,255,255,0.1)';
   841→          } else {
   842→            ctx.fillStyle = exists ? 'rgba(244,67,54,0.35)' : 'rgba(255,255,255,0.25)';
   843→          }
   844→          ctx.fillRect(hCol * lw, hRow * lh, lw, lh);
   845→          ctx.strokeStyle = exists ? '#f44336' : '#fff';
   846→          ctx.lineWidth = 2;
   847→          ctx.strokeRect(hCol * lw + 1, hRow * lh + 1, lw - 2, lh - 2);
   848→
   849→          // Draw plant at cursor position (drag preview)
   850→          if (sel && !exists) {
   851→            const pDef = PLANT_DEFS.find(d => d.id === sel);
   852→            if (pDef) {
   853→              const fs = Math.min(lw, lh) * 0.6;
   854→              ctx.globalAlpha = 0.7;
   855→              ctx.font = `${fs}px serif`;
   856→              ctx.textAlign = 'center';
   857→              ctx.textBaseline = 'middle';
   858→              ctx.fillText(pDef.emoji, cursor.x, cursor.y);
   859→              ctx.globalAlpha = 1;
   860→            }
   861→          }
   862→        }
   863→      }
   864→
   865→      // Draw plants
   866→      for (const p of plantsRef.current) {
   867→        const def = PLANT_DEFS.find(d => d.id === p.type);
   868→        if (!def) continue;
   869→        const cx = (p.col + 0.5) * lw;
   870→        const cy = (p.row + 0.5) * lh;
   871→        const fs = Math.min(lw, lh) * 0.65;
   872→        ctx.font = `${fs}px serif`;
   873→        ctx.textAlign = 'center';
   874→        ctx.textBaseline = 'middle';
   875→
   876→        if (p.type === 'cherrybomb' && p.fuseTimer > 0) {
   877→          const flash = Math.sin(p.fuseTimer * 0.02) > 0;
   878→          if (flash) {
   879→            ctx.fillStyle = 'rgba(255,100,0,0.3)';
   880→            ctx.beginPath();
   881→            ctx.arc(cx, cy, lw * 0.6, 0, Math.PI * 2);
   882→            ctx.fill();
   883→          }
   884→        }
   885→
   886→        if (p.type === 'chomper' && p.digesting) ctx.globalAlpha = 0.5;
   887→        if (p.type === 'potatomine' && !p.armed) ctx.globalAlpha = 0.4;
   888→
   889→        ctx.fillText(def.emoji, cx, cy);
   890→        ctx.globalAlpha = 1;
   891→
   892→        // Chomper digest bar
   893→        if (p.type === 'chomper' && p.digesting) {
   894→          const bw = lw * 0.7;
   895→          const bx = cx - bw / 2;
   896→          const by = (p.row + 0.15) * lh;
   897→          ctx.fillStyle = 'rgba(0,0,0,0.3)';
   898→          ctx.fillRect(bx, by, bw, 3);
   899→          ctx.fillStyle = '#9c27b0';
   900→          ctx.fillRect(bx, by, bw * Math.min(1, p.digestTimer / CHOMP_DIGEST_TIME), 3);
   901→        }
   902→
   903→        // Potato mine arm progress
   904→        if (p.type === 'potatomine' && !p.armed) {
   905→          const bw = lw * 0.7;
   906→          const bx = cx - bw / 2;
   907→          const by = (p.row + 0.15) * lh;
   908→          ctx.fillStyle = 'rgba(0,0,0,0.3)';
   909→          ctx.fillRect(bx, by, bw, 3);
   910→          ctx.fillStyle = '#ff9800';
   911→          ctx.fillRect(bx, by, bw * Math.min(1, p.armTimer / MINE_ARM_TIME), 3);
   912→        }
   913→
   914→        // HP bar
   915→        if (p.type !== 'cherrybomb' && p.type !== 'potatomine' && p.hp < p.maxHp) {
   916→          const bw = lw * 0.7;
   917→          const bx = cx - bw / 2;
   918→          const by = (p.row + 0.92) * lh;
   919→          ctx.fillStyle = 'rgba(0,0,0,0.3)';
   920→          ctx.fillRect(bx, by, bw, 4);
   921→          const ratio = Math.max(0, p.hp / p.maxHp);
   922→          ctx.fillStyle = ratio > 0.5 ? '#4caf50' : ratio > 0.25 ? '#ff9800' : '#f44336';
   923→          ctx.fillRect(bx, by, bw * ratio, 4);
   924→        }
   925→      }
   926→
   927→      // Draw peas
   928→      for (const pea of peasRef.current) {
   929→        if (pea.isFire) {
   930→          ctx.fillStyle = '#ff5722';
   931→          ctx.beginPath(); ctx.arc(pea.x, pea.y, 6, 0, Math.PI * 2); ctx.fill();
   932→          ctx.fillStyle = '#ffeb3b';
   933→          ctx.beginPath(); ctx.arc(pea.x, pea.y, 3, 0, Math.PI * 2); ctx.fill();
   934→        } else if (pea.isIce) {
   935→          ctx.fillStyle = '#42a5f5';
   936→          ctx.beginPath(); ctx.arc(pea.x, pea.y, 5, 0, Math.PI * 2); ctx.fill();
   937→          ctx.fillStyle = '#bbdefb';
   938→          ctx.beginPath(); ctx.arc(pea.x - 1, pea.y - 1, 2, 0, Math.PI * 2); ctx.fill();
   939→        } else {
   940→          ctx.fillStyle = '#66bb6a';
   941→          ctx.beginPath(); ctx.arc(pea.x, pea.y, 4, 0, Math.PI * 2); ctx.fill();
   942→        }
   943→      }
   944→
   945→      // Draw suns (simplified for performance)
   946→      for (const s of sunsRef.current) {
   947→        ctx.save();
   948→        ctx.globalAlpha = Math.max(0, s.alpha);
   949→        const sc = s.scale || 1;
   950→        // Outer glow
   951→        ctx.fillStyle = 'rgba(255,235,59,0.3)';
   952→        ctx.beginPath(); ctx.arc(s.x, s.y, 20 * sc, 0, Math.PI * 2); ctx.fill();
   953→        // Core
   954→        ctx.fillStyle = '#ffeb3b';
   955→        ctx.beginPath(); ctx.arc(s.x, s.y, 12 * sc, 0, Math.PI * 2); ctx.fill();
   956→        // Highlight
   957→        ctx.fillStyle = '#fff9c4';
   958→        ctx.beginPath(); ctx.arc(s.x - 3, s.y - 3, 4 * sc, 0, Math.PI * 2); ctx.fill();
   959→        // Value text
   960→        ctx.fillStyle = '#f57f17';
   961→        ctx.font = 'bold 10px sans-serif';
   962→        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
   963→        ctx.fillText('25', s.x, s.y + 1);
   964→        // Flashing when about to expire
   965→        if (s.lifeTimer < 120 && !s.collected) {
   966→          ctx.globalAlpha = Math.sin(s.lifeTimer * 0.1) > 0 ? 1 : 0.3;
   967→        }
   968→        ctx.restore();
   969→      }
   970→
   971→      // Draw zombies
   972→      for (const z of zombiesRef.current) {
   973→        if (z.hp <= 0) continue;
   974→        const cy = (z.row + 0.5) * lh;

   976→        const fs = Math.min(lw, lh) * 0.7;
   977→        ctx.save();
   978→        if (z.hitFlash > 0) {
   979→          ctx.globalAlpha = 0.5 + Math.sin(z.hitFlash * 2) * 0.5;
   980→        }
   981→        // Slow indicator
   982→        if (z.slowTimer > 0) {
   983→          ctx.fillStyle = 'rgba(66,165,245,0.3)';
   984→          ctx.beginPath(); ctx.arc(z.x + wobble, cy, fs * 0.6, 0, Math.PI * 2); ctx.fill();
   985→        }
   986→        ctx.font = `${fs}px serif`;
   987→        ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
   988→        ctx.fillText(z.emoji, z.x + wobble, cy);
   989→        ctx.restore();
   990→
   991→        // HP bar
