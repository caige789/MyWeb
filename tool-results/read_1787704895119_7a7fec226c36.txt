   137→   137→/** 生成某一关的砖块布局 */
   138→   138→function generateBricks(level: number, canvasW: number): Brick[] {
   139→   139→  const bricks: Brick[] = [];
   140→   140→  const brickW = (canvasW - (BRICK_COLS + 1) * BRICK_GAP) / BRICK_COLS;
   141→   141→
   142→   142→  if (level === 1) {
   143→   143→    // 第一关：5行标准排列
   144→   144→    for (let row = 0; row < 5; row++) {
   145→   145→      const typeIndex = row;
   146→   146→      for (let col = 0; col < BRICK_COLS; col++) {
   147→   147→        bricks.push({
   148→   148→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   149→   149→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   150→   150→          w: brickW,
   151→   151→          h: BRICK_H,
   152→   152→          typeIndex,
   153→   153→          alive: true,
   154→   154→        });
   155→   155→      }
   156→   156→    }
   157→   157→  } else if (level === 2) {
   158→   158→    // 第二关：6行，部分位置空缺形成菱形
   159→   159→    for (let row = 0; row < 6; row++) {
   160→   160→      const typeIndex = Math.min(row, BRICK_TYPES.length - 1);
   161→   161→      for (let col = 0; col < BRICK_COLS; col++) {
   162→   162→        // 菱形空缺：中间两行各跳过2个砖块
   163→   163→        if (row === 2 && (col === 0 || col === 7)) continue;
   164→   164→        if (row === 3 && (col === 3 || col === 4)) continue;
   165→   165→        bricks.push({
   166→   166→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   167→   167→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   168→   168→          w: brickW,
   169→   169→          h: BRICK_H,
   170→   170→          typeIndex,
   171→   171→          alive: true,
   172→   172→        });
   173→   173→      }
   174→   174→    }
   175→   175→  } else {
   176→   176→    // 第三关：7行，交错排列，高密度
   177→   177→    for (let row = 0; row < 7; row++) {
   178→   178→      const typeIndex = Math.min(row, BRICK_TYPES.length - 1);
   179→   179→      for (let col = 0; col < BRICK_COLS; col++) {
   180→   180→        // 交错空缺
   181→   181→        if (row % 2 === 0 && col % 2 === 0) continue;
   182→   182→        if (row === 1 && col === 4) continue;
   183→   183→        if (row === 4 && col === 2) continue;
   184→   184→        bricks.push({
   185→   185→          x: BRICK_GAP + col * (brickW + BRICK_GAP),
   186→   186→          y: BRICK_TOP_OFFSET + row * (BRICK_H + BRICK_GAP),
   187→   187→          w: brickW,
   188→   188→          h: BRICK_H,
   189→   189→          typeIndex,
   190→   190→          alive: true,
   191→   191→        });
   192→   192→      }
   193→   193→    }
   194→   194→  }
   195→   195→  return bricks;
   196→   196→}

   198→   198→/** 绘制圆角矩形路径 */
   199→   199→function roundRect(
   200→   200→  ctx: CanvasRenderingContext2D,
   201→   201→  x: number,
   202→   202→  y: number,
   203→   203→  w: number,
   204→   204→  h: number,
   205→   205→  r: number
   206→   206→) {
   207→   207→  const rr = Math.min(r, w / 2, h / 2);
   208→   208→  ctx.beginPath();
   209→   209→  ctx.moveTo(x + rr, y);
   210→   210→  ctx.lineTo(x + w - rr, y);
   211→   211→  ctx.quadraticCurveTo(x + w, y, x + w, y + rr);
   212→   212→  ctx.lineTo(x + w, y + h - rr);
   213→   213→  ctx.quadraticCurveTo(x + w, y + h, x + w - rr, y + h);
   214→   214→  ctx.lineTo(x + rr, y + h);
   215→   215→  ctx.quadraticCurveTo(x, y + h, x, y + h - rr);
   216→   216→  ctx.lineTo(x, y + rr);
   217→   217→  ctx.quadraticCurveTo(x, y, x + rr, y);
   218→   218→  ctx.closePath();
   219→   219→}

   392→   392→      ball.x += ball.dx;
   393→   393→      ball.y += ball.dy;
   394→   394→

   401→   401→        ball.x = canvasW - ball.radius;
   402→   402→        ball.dx = -Math.abs(ball.dx);
   403→   403→      }
   404→   404→
   405→   405→      // 顶部墙壁碰撞
   406→   406→      if (ball.y - ball.radius <= 0) {
   407→   407→        ball.y = ball.radius;
   408→   408→        ball.dy = Math.abs(ball.dy);
   409→   409→      }
   410→   410→
   411→   411→      // 底部出界
   412→   412→      if (ball.y - ball.radius > CANVAS_H) {
   413→   413→        ballsToRemove.push(idx);
   414→   414→        return;
   415→   415→      }
   416→   416→
   417→   417→      // 挡板碰撞检测
   418→   418→      const pLeft = paddle.x;
   419→   419→      const pRight = paddle.x + paddle.w;
   420→   420→      const pTop = paddleY - PADDLE_H / 2;
   421→   421→      const pBottom = paddleY + PADDLE_H / 2;
   422→   422→      if (
   423→   423→        ball.dy > 0 &&
   424→   424→        ball.y + ball.radius >= pTop &&
   425→   425→        ball.y + ball.radius <= pBottom + 4 &&
   426→   426→        ball.x >= pLeft - ball.radius &&
   427→   427→        ball.x <= pRight + ball.radius
   428→   428→      ) {
   429→   429→        ball.y = pTop - ball.radius;
   430→   430→        // 根据击中挡板的位置改变反弹角度
   431→   431→        const hitPos = (ball.x - (paddle.x + paddle.w / 2)) / (paddle.w / 2);
   432→   432→        const maxAngle = Math.PI / 3;
   433→   433→        const angle = hitPos * maxAngle;
   434→   434→        const spd = ball.speed;
   435→   435→        ball.dx = Math.sin(angle) * spd;
   436→   436→        ball.dy = -Math.cos(angle) * spd;
   437→   437→      }
   438→   438→
   439→   439→      // 砖块碰撞检测
   440→   440→      bricks.forEach((brick) => {
   441→   441→        if (!brick.alive) return;
   442→   442→        const bLeft = brick.x;
   443→   443→        const bRight = brick.x + brick.w;
   444→   444→        const bTop = brick.y;
   445→   445→        const bBottom = brick.y + brick.h;
   446→   446→
   447→   447→        // AABB 碰撞检测
   448→   448→        if (
   449→   449→          ball.x + ball.radius > bLeft &&
   450→   450→          ball.x - ball.radius < bRight &&
   451→   451→          ball.y + ball.radius > bTop &&
   452→   452→          ball.y - ball.radius < bBottom
   453→   453→        ) {
   454→   454→          brick.alive = false;
   455→   455→          scoreRef.current += BRICK_TYPES[brick.typeIndex].score;
   456→   456→          setScore(scoreRef.current);
   457→   457→          spawnParticles(brick);
   458→   458→          trySpawnPowerUp(brick);
   459→   459→
   460→   460→          // 判断碰撞方向
   461→   461→          const overlapLeft = ball.x + ball.radius - bLeft;
   462→   462→          const overlapRight = bRight - (ball.x - ball.radius);
   463→   463→          const overlapTop = ball.y + ball.radius - bTop;
   464→   464→          const overlapBottom = bBottom - (ball.y - ball.radius);
   465→   465→          const minOverlapX = Math.min(overlapLeft, overlapRight);
   466→   466→          const minOverlapY = Math.min(overlapTop, overlapBottom);
   467→   467→
   468→   468→          if (minOverlapX < minOverlapY) {
   469→   469→            ball.dx = -ball.dx;
   470→   470→          } else {
   471→   471→            ball.dy = -ball.dy;
   472→   472→          }
   473→   473→        }
   474→   474→      });
   475→   475→    });
   476→   476→
   477→   477→    // 移除出界的球（从后往前删）
   478→   478→    for (let i = ballsToRemove.length - 1; i >= 0; i--) {
   479→   479→      balls.splice(ballsToRemove[i], 1);
   480→   480→    }

   486→   486→      if (livesRef.current <= 0) {
   487→   487→        gameStatusRef.current = 'over';
   488→   488→        setGameStatus('over');
   489→   489→      } else {
   490→   490→        ballsRef.current = [createBall(canvasW)];
   491→   491→      }
   492→   492→    }
   493→   493→

   496→   496→      if (levelRef.current >= MAX_LEVEL) {
   497→   497→        gameStatusRef.current = 'over';
   498→   498→        setGameStatus('over');
   499→   499→      } else {

   507→   507→      }
   508→   508→    }
   509→   509→
   510→   510→    // 更新道具下落
   511→   511→    const puToRemove: number[] = [];
   512→   512→    powerUps.forEach((pu, idx) => {
   513→   513→      pu.y += POWERUP_SPEED;
   514→   514→      pu.frame += 1;
   515→   515→
   516→   516→      // 检测是否被挡板接住

   519→   519→        pu.y - POWERUP_SIZE / 2 <= paddleY + PADDLE_H / 2 &&
   520→   520→        pu.x + POWERUP_SIZE / 2 >= paddle.x &&
   521→   521→        pu.x - POWERUP_SIZE / 2 <= paddle.x + paddle.w
   522→   522→      ) {
   523→   523→        applyPowerUp(pu);
   524→   524→        puToRemove.push(idx);
   525→   525→      }
   526→   526→
   527→   527→      // 出界移除
   528→   528→      if (pu.y > CANVAS_H + 20) {
   529→   529→        puToRemove.push(idx);
   530→   530→      }
   531→   531→    });
   532→   532→    for (let i = puToRemove.length - 1; i >= 0; i--) {
   533→   533→      powerUps.splice(puToRemove[i], 1);
   534→   534→    }
   535→   535→
   536→   536→    // 更新粒子
   537→   537→    for (let i = particles.length - 1; i >= 0; i--) {
   538→   538→      const p = particles[i];
   539→   539→      p.x += p.dx;
   540→   540→      p.y += p.dy;
   541→   541→      p.dy += 0.08; // 重力
   542→   542→      p.life -= 1;
   543→   543→      if (p.life <= 0) {
   544→   544→        particles.splice(i, 1);
   545→   545→      }
   546→   546→    }
   547→   547→
   548→   548→    // 更新拖尾衰减

   550→   550→      trails[i].alpha -= 0.08;
   551→   551→      if (trails[i].alpha <= 0) {
   552→   552→        trails.splice(i, 1);
   553→   553→      }
   554→   554→    }
   555→   555→  }, [createBall, spawnParticles, trySpawnPowerUp, applyPowerUp, getPaddleY, initLevel]);

   806→   806→  // 键盘事件
