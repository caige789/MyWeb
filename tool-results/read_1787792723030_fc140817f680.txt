   154→   154→function createStars(count: number): Star[] {
   155→   155→  const stars: Star[] = [];
   156→   156→  for (let i = 0; i < count; i++) {
   157→   157→    stars.push({
   158→   158→      x: Math.random() * LOGIC_W,
   159→   159→      y: Math.random() * LOGIC_H,
   160→   160→      r: Math.random() * 1.5 + 0.5,
   161→   161→      alpha: Math.random() * 0.6 + 0.4,
   162→   162→      twinkleSpeed: Math.random() * 0.03 + 0.01,
   163→   163→      phase: Math.random() * Math.PI * 2,
   164→   164→    });
   165→   165→  }
   166→   166→  return stars;
   167→   167→}

   174→   174→    particles.push({

   176→   176→      vx: Math.cos(angle) * speed,
   177→   177→      vy: Math.sin(angle) * speed,

   180→   180→      color,

   196→   196→function drawPlayer(ctx: CanvasRenderingContext2D, p: Player, frame: number) {
   197→   197→  ctx.save();
   198→   198→  ctx.translate(p.x, p.y);
   199→   199→
   200→   200→  // Invincibility blink
   201→   201→  if (p.invincibleTimer > 0 && Math.floor(frame / 3) % 2 === 0) {
   202→   202→    ctx.globalAlpha = 0.35;
   203→   203→  }
   204→   204→
   205→   205→  // Tail flame animation
   206→   206→  const flameLen = 12 + Math.sin(frame * 0.5) * 4;
   207→   207→  const flameW = 8 + Math.sin(frame * 0.7) * 2;
   208→   208→  const flameGrad = ctx.createLinearGradient(0, PLAYER_H / 2, 0, PLAYER_H / 2 + flameLen);
   209→   209→  flameGrad.addColorStop(0, '#38bdf8');
   210→   210→  flameGrad.addColorStop(0.4, '#f97316');
   211→   211→  flameGrad.addColorStop(1, 'rgba(239,68,68,0)');
   212→   212→  ctx.fillStyle = flameGrad;
   213→   213→  ctx.beginPath();
   214→   214→  ctx.moveTo(-flameW, PLAYER_H / 2 - 2);
   215→   215→  ctx.lineTo(flameW, PLAYER_H / 2 - 2);
   216→   216→  ctx.lineTo(0, PLAYER_H / 2 + flameLen);
   217→   217→  ctx.closePath();
   218→   218→  ctx.fill();
   219→   219→
   220→   220→  // Aircraft body
   221→   221→  ctx.fillStyle = '#e0f2fe';
   222→   222→  ctx.beginPath();
   223→   223→  ctx.moveTo(0, -PLAYER_H / 2);
   224→   224→  ctx.lineTo(-PLAYER_W / 2, PLAYER_H / 2 - 6);
   225→   225→  ctx.lineTo(-PLAYER_W / 4, PLAYER_H / 2);
   226→   226→  ctx.lineTo(PLAYER_W / 4, PLAYER_H / 2);
   227→   227→  ctx.lineTo(PLAYER_W / 2, PLAYER_H / 2 - 6);
   228→   228→  ctx.closePath();
   229→   229→  ctx.fill();
   230→   230→
   231→   231→  // Wing decoration lines
   232→   232→  ctx.strokeStyle = '#38bdf8';
   233→   233→  ctx.lineWidth = 1.5;
   234→   234→  ctx.beginPath();
   235→   235→  ctx.moveTo(-PLAYER_W / 2 + 4, PLAYER_H / 2 - 10);
   236→   236→  ctx.lineTo(0, -PLAYER_H / 2 + 10);
   237→   237→  ctx.lineTo(PLAYER_W / 2 - 4, PLAYER_H / 2 - 10);
   238→   238→  ctx.stroke();
   239→   239→
   240→   240→  // Cockpit
   241→   241→  ctx.fillStyle = '#7dd3fc';
   242→   242→  ctx.beginPath();
   243→   243→  ctx.ellipse(0, -PLAYER_H / 4, 4, 7, 0, 0, Math.PI * 2);
   244→   244→  ctx.fill();
   245→   245→
   246→   246→  // Shield bubble
   247→   247→  if (p.shieldTimer > 0) {
   248→   248→    const shieldAlpha = p.shieldTimer < 60 ? (Math.sin(frame * 0.3) * 0.3 + 0.4) : 0.5;
   249→   249→    ctx.strokeStyle = `rgba(6, 182, 212, ${shieldAlpha})`;
   250→   250→    ctx.lineWidth = 2.5;
   251→   251→    ctx.beginPath();
   252→   252→  ctx.ellipse(0, 0, PLAYER_W / 2 + 8, PLAYER_H / 2 + 8, 0, 0, Math.PI * 2);
   253→   253→    ctx.stroke();
   254→   254→  }
   255→   255→
   256→   256→  ctx.restore();
   257→   257→}

   259→   259→function drawEnemy(ctx: CanvasRenderingContext2D, e: Enemy) {
   260→   260→  ctx.save();
   261→   261→  ctx.translate(e.x, e.y);
   262→   262→
   263→   263→  if (e.type === EnemyType.Normal) {
   264→   264→    // Red inverted triangle
   265→   265→    ctx.fillStyle = '#ef4444';
   266→   266→    ctx.beginPath();
   267→   267→    ctx.moveTo(0, e.h / 2);
   268→   268→    ctx.lineTo(-e.w / 2, -e.h / 2);
   269→   269→    ctx.lineTo(e.w / 2, -e.h / 2);
   270→   270→    ctx.closePath();
   271→   271→    ctx.fill();
   272→   272→    ctx.strokeStyle = '#fca5a5';
   273→   273→    ctx.lineWidth = 1;
   274→   274→    ctx.beginPath();
   275→   275→    ctx.moveTo(0, e.h / 2 - 6);
   276→   276→    ctx.lineTo(-e.w / 4, -e.h / 4);
   277→   277→    ctx.moveTo(0, e.h / 2 - 6);
   278→   278→    ctx.lineTo(e.w / 4, -e.h / 4);
   279→   279→    ctx.stroke();
   280→   280→  } else if (e.type === EnemyType.Fast) {
   281→   281→    // Yellow arrow shape
   282→   282→    ctx.fillStyle = '#f59e0b';
   283→   283→    ctx.beginPath();
   284→   284→    ctx.moveTo(0, e.h / 2);
   285→   285→    ctx.lineTo(-e.w / 2, -e.h / 3);
   286→   286→    ctx.lineTo(-e.w / 4, -e.h / 2);
   287→   287→    ctx.lineTo(e.w / 4, -e.h / 2);
   288→   288→    ctx.lineTo(e.w / 2, -e.h / 3);
   289→   289→    ctx.closePath();
   290→   290→    ctx.fill();
   291→   291→    ctx.fillStyle = '#fcd34d';
   292→   292→    ctx.beginPath();
   293→   293→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   294→   294→    ctx.fill();
   295→   295→  } else if (e.type === EnemyType.Tank) {
   296→   296→    // Purple hexagon
   297→   297→    const r = e.w / 2;
   298→   298→    ctx.fillStyle = '#a855f7';
   299→   299→    ctx.beginPath();
   300→   300→    for (let i = 0; i < 6; i++) {
   301→   301→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   302→   302→      const px = Math.cos(angle) * r;
   303→   303→      const py = Math.sin(angle) * r;
   304→   304→      if (i === 0) ctx.moveTo(px, py);
   305→   305→      else ctx.lineTo(px, py);
   306→   306→    }
   307→   307→    ctx.closePath();
   308→   308→    ctx.fill();
   309→   309→    ctx.strokeStyle = '#d8b4fe';
   310→   310→    ctx.lineWidth = 1.5;
   311→   311→    ctx.beginPath();
   312→   312→    for (let i = 0; i < 6; i++) {
   313→   313→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   314→   314→      const px = Math.cos(angle) * (r * 0.55);
   315→   315→      const py = Math.sin(angle) * (r * 0.55);
   316→   316→      if (i === 0) ctx.moveTo(px, py);
   317→   317→      else ctx.lineTo(px, py);
   318→   318→    }
   319→   319→    ctx.closePath();
   320→   320→    ctx.stroke();
   321→   321→    ctx.fillStyle = '#e9d5ff';
   322→   322→    ctx.beginPath();
   323→   323→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   324→   324→    ctx.fill();
   325→   325→  }
   326→   326→
   327→   327→  ctx.restore();
   328→   328→
   329→   329→  // HP bar (only when damaged)
   330→   330→  if (e.hp < e.maxHp) {
   331→   331→    const barW = e.w * 0.8;
   332→   332→    const barH = 3;
   333→   333→    const barX = e.x - barW / 2;
   334→   334→    const barY = e.y - e.h / 2 - 6;
   335→   335→    ctx.fillStyle = 'rgba(255,255,255,0.2)';
   336→   336→    ctx.fillRect(barX, barY, barW, barH);
   337→   337→    ctx.fillStyle = '#22c55e';
   338→   338→    ctx.fillRect(barX, barY, barW * (e.hp / e.maxHp), barH);
   339→   339→  }
   340→   340→}

   342→   342→function drawBoss(ctx: CanvasRenderingContext2D, boss: Boss, frame: number) {

   380→   380→  ctx.lineWidth = 1;
   381→   381→  ctx.strokeRect(barX - 1, barY - 1, barW + 2, barH + 2);

   383→   383→  ctx.fillStyle = '#ffffff';
   384→   384→  ctx.font = 'bold 11px sans-serif';
   385→   385→  ctx.textAlign = 'center';
   386→   386→  ctx.textBaseline = 'top';
   387→   387→  ctx.fillText('BOSS', LOGIC_W / 2, barY + barH + 2);
   388→   388→}
