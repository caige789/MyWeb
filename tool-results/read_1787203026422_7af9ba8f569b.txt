   148→   148→
   149→   149→/* ============ 主组件 ============ */
   150→   150→export default function PvzGame({ onBack }: PvzGameProps) {
   151→   151→  /* --- UI状态 --- */
   152→   152→  const [phase, setPhase] = useState<'settings' | 'playing' | 'won' | 'lost'>('settings');
   153→   153→  const [difficulty, setDifficulty] = useState<Difficulty>('normal');
   154→   154→  const [sunCount, setSunCount] = useState(100);
   155→   155→  const [score, setScore] = useState(0);
   156→   156→  const [currentWave, setCurrentWave] = useState(1);
   157→   157→  const [selectedPlant, setSelectedPlant] = useState<string | null>(null);
   158→   158→  const [highScore, setHighScore] = useState(0);
   159→   159→
   160→   160→  /* --- 游戏状态Refs（分开声明，避免immutability lint问题） --- */
   161→   161→  const plantsRef = useRef<Plant[]>([]);
   162→   162→  const zombiesRef = useRef<Zombie[]>([]);
   163→   163→  const peasRef = useRef<Pea[]>([]);
   164→   164→  const sunsRef = useRef<Sun[]>([]);
   165→   165→  const sunRef = useRef(100);
   166→   166→  const scoreRef = useRef(0);
   167→   167→  const waveRef = useRef(1);
   168→   168→  const waveElapsedRef = useRef(0);
   169→   169→  const waveZombiesRef = useRef<{ type: ZombieType; delay: number }[]>([]);
   170→   170→  const waveSpawnIdxRef = useRef(0);
   171→   171→  const sunDropTimerRef = useRef(0);
   172→   172→  const zombieIdRef = useRef(0);
   173→   173→  const sunIdRef = useRef(0);
   174→   174→  const diffRef = useRef<Difficulty>('normal');
   175→   175→  const overRef = useRef(false);
   176→   176→  const wonRef = useRef(false);
   177→   177→  const selPlantRef = useRef<string | null>(null);
   178→   178→  const lastTimeRef = useRef(0);
   179→   179→  const rafRef = useRef(0);
   180→   180→  const canvasRef = useRef<HTMLCanvasElement>(null);
   181→   181→  const containerRef = useRef<HTMLDivElement>(null);
   182→   182→  const cellWRef = useRef(LOGIC_W / COLS);
   183→   183→  const cellHRef = useRef(LOGIC_H / ROWS);
   184→   184→
   185→   185→  /* --- 游戏循环函数Ref（避免useCallback自引用问题） --- */
   186→   186→  const loopFnRef = useRef<(timestamp: number) => void>(null!);
   187→   187→  const submitScoreRef = useRef<(s: number) => void>(() => {});
   188→   188→
   189→   189→  /* --- 获取最高分 --- */
   190→   190→  useEffect(() => {
   191→   191→    fetch('/api/games/scores')
   192→   192→      .then(r => r.json())
   193→   193→      .then(data => {
   194→   194→        if (data.code === 200) {
   195→   195→          const pvz = (data.data || []).find((s: { game: string }) => s.game === GAME_NAME);
   196→   196→          if (pvz) setHighScore(pvz.score);
   197→   197→        }
   198→   198→      });
   199→   199→  }, []);
   200→   200→
   201→   201→  /* --- 提交分数 --- */
   202→   202→  const doSubmitScore = useCallback((s: number) => {
   203→   203→    fetch('/api/games/scores', {
   204→   204→      method: 'POST',
   205→   205→      headers: { 'Content-Type': 'application/json' },
   206→   206→      body: JSON.stringify({ game: GAME_NAME, score: s }),
   207→   207→    })
   208→   208→      .then(r => r.json())
   209→   209→      .then(data => {
   210→   210→        if (data.code === 200 && data.data) setHighScore(data.data.score);
   211→   211→      });
   212→   212→  }, []);
   213→   213→
   214→   214→  useEffect(() => { submitScoreRef.current = doSubmitScore; }, [doSubmitScore]);
   215→   215→
   216→   216→  /* --- 高DPI Canvas设置 --- */
   217→   217→  const setupCanvas = useCallback(() => {
   218→   218→    const canvas = canvasRef.current;
   219→   219→    const container = containerRef.current;
   220→   220→    if (!canvas || !container) return;
   221→   221→    const rect = container.getBoundingClientRect();
   222→   222→    const w = Math.min(rect.width, 640);
   223→   223→    const h = w * (ROWS / COLS);
   224→   224→    const dpr = window.devicePixelRatio || 1;
   225→   225→    canvas.width = w * dpr;
   226→   226→    canvas.height = h * dpr;
   227→   227→    canvas.style.width = w + 'px';
   228→   228→    canvas.style.height = h + 'px';
   229→   229→    const ctx = canvas.getContext('2d');
   230→   230→    if (ctx) ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   231→   231→    cellWRef.current = w / COLS;
   232→   232→    cellHRef.current = h / ROWS;
   233→   233→  }, []);
   234→   234→
   235→   235→  /* --- 初始化游戏状态 --- */
   236→   236→  const initGame = useCallback((diff: Difficulty) => {
   237→   237→    const cfg = DIFFICULTY_CONFIG[diff];
   238→   238→    plantsRef.current = [];
   239→   239→    zombiesRef.current = [];
   240→   240→    peasRef.current = [];
   241→   241→    sunsRef.current = [];
   242→   242→    sunRef.current = cfg.initSun;
   243→   243→    scoreRef.current = 0;
   244→   244→    waveRef.current = 1;
   245→   245→    waveElapsedRef.current = 0;
   246→   246→    waveZombiesRef.current = getWaveZombies(0, cfg.zombieCountMul);
   247→   247→    waveSpawnIdxRef.current = 0;
   248→   248→    sunDropTimerRef.current = 0;
   249→   249→    zombieIdRef.current = 0;
   250→   250→    sunIdRef.current = 0;
   251→   251→    diffRef.current = diff;
   252→   252→    overRef.current = false;
   253→   253→    wonRef.current = false;
   254→   254→    selPlantRef.current = null;
   255→   255→    lastTimeRef.current = 0;
   256→   256→    setSunCount(cfg.initSun);
   257→   257→    setScore(0);
   258→   258→    setCurrentWave(1);
   259→   259→    setSelectedPlant(null);
   260→   260→  }, []);
   261→   261→
   262→   262→  /* --- 开始游戏 --- */
   263→   263→  const startGame = useCallback(() => {
   264→   264→    initGame(difficulty);
   265→   265→    setPhase('playing');
   266→   266→  }, [difficulty, initGame]);
   267→   267→
   268→   268→  /* --- 坐标转换 --- */
   269→   269→  const screenToLogic = useCallback((clientX: number, clientY: number) => {
   270→   270→    const canvas = canvasRef.current;
   271→   271→    if (!canvas) return { x: 0, y: 0 };
   272→   272→    const rect = canvas.getBoundingClientRect();
   273→   273→    return {
   274→   274→      x: (clientX - rect.left) * (LOGIC_W / rect.width),
   275→   275→      y: (clientY - rect.top) * (LOGIC_H / rect.height),
   276→   276→    };
   277→   277→  }, []);
   278→   278→
   279→   279→  /* --- Canvas点击/触摸处理 --- */
   280→   280→  const handleCanvasInteraction = useCallback((clientX: number, clientY: number) => {
   281→   281→    if (overRef.current || wonRef.current) return;
   282→   282→    const pos = screenToLogic(clientX, clientY);
   283→   283→    const cellW = cellWRef.current;
   284→   284→    const cellH = cellHRef.current;
   285→   285→
   286→   286→    // 1. 检查是否点击了阳光
   287→   287→    const suns = sunsRef.current;
   288→   288→    for (let i = suns.length - 1; i >= 0; i--) {
   289→   289→      const s = suns[i];
   290→   290→      if (s.collected) continue;
   291→   291→      const dx = pos.x - s.x;
   292→   292→      const dy = pos.y - s.y;
   293→   293→      if (dx * dx + dy * dy < 900) {
   294→   294→        s.collected = true;
   295→   295→        sunRef.current += SUN_VALUE;
   296→   296→        setSunCount(sunRef.current);
   297→   297→        return;
   298→   298→      }
   299→   299→    }
   300→   300→
   301→   301→    // 2. 在格子上种植物
   302→   302→    const sel = selPlantRef.current;
   303→   303→    if (sel) {
   304→   304→      const col = Math.floor(pos.x / cellW);
   305→   305→      const row = Math.floor(pos.y / cellH);
   306→   306→      if (row >= 0 && row < ROWS && col >= 0 && col < COLS) {
   307→   307→        const exists = plantsRef.current.some(p => p.row === row && p.col === col);
   308→   308→        if (!exists) {
   309→   309→          const def = PLANT_DEFS.find(d => d.id === sel);
   310→   310→          if (def && sunRef.current >= def.cost) {
   311→   311→            sunRef.current -= def.cost;
   312→   312→            setSunCount(sunRef.current);
   313→   313→            plantsRef.current.push({
   314→   314→              type: def.id, row, col,
   315→   315→              hp: def.hp, maxHp: def.hp,
   316→   316→              shootTimer: 0, sunTimer: 0,
   317→   317→            });
   318→   318→          }
   319→   319→        }
   320→   320→      }
   321→   321→    }
   322→   322→  }, [screenToLogic]);
   323→   323→
   324→   324→  /* --- Canvas事件监听 --- */
   325→   325→  useEffect(() => {
   326→   326→    const canvas = canvasRef.current;
   327→   327→    if (!canvas) return;
   328→   328→    const onClick = (e: MouseEvent) => {
   329→   329→      e.preventDefault();
   330→   330→      handleCanvasInteraction(e.clientX, e.clientY);
   331→   331→    };
   332→   332→    const onTouch = (e: TouchEvent) => {
   333→   333→      e.preventDefault();
   334→   334→      const t = e.touches[0] || e.changedTouches[0];
   335→   335→      if (t) handleCanvasInteraction(t.clientX, t.clientY);
   336→   336→    };
   337→   337→    canvas.addEventListener('click', onClick);
   338→   338→    canvas.addEventListener('touchstart', onTouch, { passive: false });
   339→   339→    return () => {
   340→   340→      canvas.removeEventListener('click', onClick);
   341→   341→      canvas.removeEventListener('touchstart', onTouch);
   342→   342→    };
   343→   343→  }, [handleCanvasInteraction]);
   344→   344→
   345→   345→  /* --- 选择植物 --- */
   346→   346→  const handleSelectPlant = useCallback((plantId: string) => {
   347→   347→    if (selPlantRef.current === plantId) {
   348→   348→      selPlantRef.current = null;
   349→   349→      setSelectedPlant(null);
   350→   350→    } else {
   351→   351→      selPlantRef.current = plantId;
   352→   352→      setSelectedPlant(plantId);
   353→   353→    }
   354→   354→  }, []);
   355→   355→
   356→   356→  /* --- 定义游戏主循环 --- */
   357→   357→  useEffect(() => {
   358→   358→    const loop = (timestamp: number) => {
   359→   359→      const canvas = canvasRef.current;
   360→   360→      if (!canvas) return;
   361→   361→      const ctx = canvas.getContext('2d');
   362→   362→      if (!ctx) return;
   363→   363→
   364→   364→      // 高DPI每帧重设变换
   365→   365→      const dpr = window.devicePixelRatio || 1;
   366→   366→      ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
   367→   367→
   368→   368→      // delta时间计算
   369→   369→      if (lastTimeRef.current === 0) lastTimeRef.current = timestamp;
   370→   370→      const deltaMs = Math.min(timestamp - lastTimeRef.current, 50);
   371→   371→      lastTimeRef.current = timestamp;
   372→   372→      const dt = deltaMs / 16.667;
   373→   373→
   374→   374→      const cellW = cellWRef.current;
   375→   375→      const cellH = cellHRef.current;
   376→   376→      const cfg = DIFFICULTY_CONFIG[diffRef.current];
   377→   377→
   378→   378→      /* ===== 更新逻辑 ===== */
   379→   379→      if (!overRef.current && !wonRef.current) {
   380→   380→        waveElapsedRef.current += deltaMs;
   381→   381→
   382→   382→        // 天上掉阳光
   383→   383→        sunDropTimerRef.current += deltaMs;
   384→   384→        if (sunDropTimerRef.current >= SUN_DROP_INTERVAL) {
   385→   385→          sunDropTimerRef.current -= SUN_DROP_INTERVAL;
   386→   386→          const newSun: Sun = {
   387→   387→            id: sunIdRef.current++,
   388→   388→            x: 40 + Math.random() * (LOGIC_W - 80),
   389→   389→            y: -20,
   390→   390→            targetY: 30 + Math.random() * (LOGIC_H - 60),
   391→   391→            vy: 0, collected: false, alpha: 1, bouncePhase: 0, lifeTimer: SUN_LIFE,
   392→   392→          };
   393→   393→          sunsRef.current.push(newSun);
   394→   394→        }
   395→   395→
   396→   396→        // 生成当前波僵尸
   397→   397→        const wz = waveZombiesRef.current;
   398→   398→        const wsi = waveSpawnIdxRef.current;
   399→   399→        if (wsi < wz.length && waveElapsedRef.current >= wz[wsi].delay) {
   400→   400→          const def = ZOMBIE_DEFS[wz[wsi].type];
   401→   401→          const newZombie: Zombie = {
   402→   402→            id: zombieIdRef.current++,
   403→   403→            type: wz[wsi].type,
   404→   404→            emoji: def.emoji,
   405→   405→            row: Math.floor(Math.random() * ROWS),
   406→   406→            x: LOGIC_W + 10,
   407→   407→            hp: def.hp, maxHp: def.hp,
   408→   408→            speed: def.speed * cfg.zombieSpeedMul,
   409→   409→            score: def.score,
   410→   410→            hitFlash: 0, eating: false,
   411→   411→            walkPhase: Math.random() * Math.PI * 2,
   412→   412→          };
   413→   413→          zombiesRef.current.push(newZombie);
   414→   414→          waveSpawnIdxRef.current++;
   415→   415→        }
   416→   416→
   417→   417→        // 检查波次结束
   418→   418→        if (
   419→   419→          waveElapsedRef.current >= WAVE_DURATION * 1000 &&
   420→   420→          waveSpawnIdxRef.current >= wz.length &&
   421→   421→          zombiesRef.current.length === 0
   422→   422→        ) {
   423→   423→          if (waveRef.current >= TOTAL_WAVES) {
   424→   424→            wonRef.current = true;
   425→   425→            submitScoreRef.current(scoreRef.current);
   426→   426→            setPhase('won');
   427→   427→          } else {
   428→   428→            waveRef.current++;
   429→   429→            waveElapsedRef.current = 0;
   430→   430→            waveSpawnIdxRef.current = 0;
   431→   431→            waveZombiesRef.current = getWaveZombies(waveRef.current - 1, cfg.zombieCountMul);
   432→   432→            setCurrentWave(waveRef.current);
   433→   433→          }
   434→   434→        }
   435→   435→
   436→   436→        // 更新植物
   437→   437→        const plants = plantsRef.current;
   438→   438→        for (let pi = plants.length - 1; pi >= 0; pi--) {
   439→   439→          const p = plants[pi];
   440→   440→          if (p.hp <= 0) { plants.splice(pi, 1); continue; }
   441→   441→
   442→   442→          // 向日葵产阳光
   443→   443→          if (p.type === 'sunflower') {
   444→   444→            p.sunTimer += deltaMs;
   445→   445→            if (p.sunTimer >= SUNFLOWER_INTERVAL) {
   446→   446→              p.sunTimer -= SUNFLOWER_INTERVAL;
   447→   447→              const sx = (p.col + 0.5) * cellW;
   448→   448→              const sy = p.row * cellH + cellH * 0.2;
   449→   449→              sunsRef.current.push({
   450→   450→                id: sunIdRef.current++,
   451→   451→                x: sx, y: sy,
   452→   452→                targetY: sy + 20 + Math.random() * 30,
   453→   453→                vy: 0, collected: false, alpha: 1, bouncePhase: 0, lifeTimer: SUN_LIFE,
   454→   454→              });
   455→   455→            }
   456→   456→          }
   457→   457→
   458→   458→          // 射手发射豌豆
   459→   459→          if (p.type === 'peashooter' || p.type === 'doubleshoot') {
   460→   460→            const hasZombie = zombiesRef.current.some(
   461→   461→              z => z.row === p.row && z.x > (p.col + 0.5) * cellW && z.hp > 0
   462→   462→            );
   463→   463→            if (hasZombie) {
   464→   464→              p.shootTimer += deltaMs;
   465→   465→              if (p.shootTimer >= SHOOT_INTERVAL) {
   466→   466→                p.shootTimer -= SHOOT_INTERVAL;
   467→   467→                const px = (p.col + 0.8) * cellW;
   468→   468→                const py = (p.row + 0.5) * cellH;
   469→   469→                peasRef.current.push({ row: p.row, x: px, y: py });
   470→   470→                if (p.type === 'doubleshoot') {
   471→   471→                  peasRef.current.push({ row: p.row, x: px - 8, y: py - 4 });
   472→   472→                }
   473→   473→              }
   474→   474→            }
   475→   475→          }
   476→   476→        }
   477→   477→
   478→   478→        // 更新豌豆
   479→   479→        const peas = peasRef.current;
   480→   480→        for (let i = peas.length - 1; i >= 0; i--) {
   481→   481→          peas[i].x += PEA_SPEED * dt;
   482→   482→        }
   483→   483→
   484→   484→        // 豌豆碰撞检测
   485→   485→        for (let i = peas.length - 1; i >= 0; i--) {
   486→   486→          const pea = peas[i];
   487→   487→          let hit = false;
   488→   488→          const zombies = zombiesRef.current;
   489→   489→          for (let zi = 0; zi < zombies.length; zi++) {
   490→   490→            const z = zombies[zi];
   491→   491→            if (z.hp <= 0 || z.row !== pea.row) continue;
   492→   492→            const zy = (z.row + 0.5) * cellH;
   493→   493→            const dx = pea.x - z.x;
   494→   494→            const dy = pea.y - zy;
   495→   495→            const hitR = cellW * 0.4;
   496→   496→            if (dx * dx + dy * dy < hitR * hitR) {
   497→   497→              z.hp -= PEA_DAMAGE;
   498→   498→              z.hitFlash = 6;
   499→   499→              hit = true;
   500→   500→              break;
   501→   501→            }
   502→   502→          }
   503→   503→          if (hit || pea.x > LOGIC_W + 20) {
   504→   504→            peas.splice(i, 1);
   505→   505→          }
   506→   506→        }
   507→   507→
   508→   508→        // 更新僵尸
   509→   509→        const zombies = zombiesRef.current;
   510→   510→        for (let i = zombies.length - 1; i >= 0; i--) {
   511→   511→          const z = zombies[i];
   512→   512→          if (z.hp <= 0) {
   513→   513→            scoreRef.current += z.score;
   514→   514→            setScore(scoreRef.current);
   515→   515→            zombies.splice(i, 1);
   516→   516→            continue;
   517→   517→          }
   518→   518→          if (z.hitFlash > 0) z.hitFlash -= dt;
   519→   519→          z.walkPhase += 0.05 * dt;
   520→   520→
   521→   521→          // 碰到植物就吃
   522→   522→          z.eating = false;
   523→   523→          const zCol = Math.floor(z.x / cellW);
   524→   524→ const target = plants.find(p => p.row === z.row && p.col === zCol);
   525→   525→          if (target) {
   526→   526→            z.eating = true;
   527→   527→            target.hp -= 0.5 * dt;
   528→   528→          } else {
   529→   529→            z.x -= z.speed * dt;
   530→   530→          }
   531→   531→
   532→   532→          // 到达最左侧
   533→   533→          if (z.x < cellW * 0.3) {
   534→   534→            overRef.current = true;
   535→   535→            submitScoreRef.current(scoreRef.current);
   536→   536→            setPhase('lost');
   537→   537→          }
   538→   538→        }
   539→   539→
   540→   540→        // 更新阳光
   541→   541→        const suns = sunsRef.current;
   542→   542→        for (let i = suns.length - 1; i >= 0; i--) {
   543→   543→          const s = suns[i];
   544→   544→          if (s.collected) {
   545→   545→            s.alpha -= 0.05 * dt;
   546→   546→            if (s.alpha <= 0) suns.splice(i, 1);
   547→   547→            continue;
   548→   548→          }
   549→   549→          s.lifeTimer -= dt;
   550→   550→          if (s.lifeTimer <= 0) { suns.splice(i, 1); continue; }
   551→   551→          if (s.y < s.targetY) {
   552→   552→            s.vy += 0.15 * dt;
   553→   553→            s.y += s.vy * dt;
   554→   554→            if (s.y >= s.targetY) {
   555→   555→              s.y = s.targetY;
   556→   556→              s.vy = -s.vy * 0.3;
   557→   557→              s.bouncePhase++;
   558→   558→              if (s.bouncePhase >= 3) { s.y = s.targetY; s.vy = 0; }
   559→   559→            }
   560→   560→          }
   561→   561→        }
   562→   562→      }
   563→   563→
   564→   564→      /* ===== 绘制 ===== */
   565→   565→      ctx.clearRect(0, 0, LOGIC_W, LOGIC_H);
   566→   566→
   567→   567→      // 草坪棋盘格
   568→   568→      for (let r = 0; r < ROWS; r++) {
   569→   569→        for (let c = 0; c < COLS; c++) {
   570→   570→          ctx.fillStyle = (r + c) % 2 === 0 ? '#5cb85c' : '#4cae4c';
   571→   571→          ctx.fillRect(c * cellW, r * cellH, cellW, cellH);
   572→   572→        }
   573→   573→      }
   574→   574→
   575→   575→      // 网格线
   576→   576→      ctx.strokeStyle = 'rgba(0,0,0,0.06)';
   577→   577→      ctx.lineWidth = 1;
   578→   578→      for (let r = 0; r <= ROWS; r++) {
   579→   579→        ctx.beginPath(); ctx.moveTo(0, r * cellH); ctx.lineTo(LOGIC_W, r * cellH); ctx.stroke();
   580→   580→      }
   581→   581→      for (let c = 0; c <= COLS; c++) {
   582→   582→        ctx.beginPath(); ctx.moveTo(c * cellW, 0); ctx.lineTo(c * cellW, LOGIC_H); ctx.stroke();
   583→   583→      }
