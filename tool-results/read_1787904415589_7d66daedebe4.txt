    63→// Generate maze using recursive backtracking (iterative stack to avoid stack overflow)
    64→function generateMaze(rows: number, cols: number): number[][] {
    65→  // Initialize grid with all walls
    66→  const grid: number[][] = Array.from({ length: rows }, () =>
    67→    Array.from({ length: cols }, () => WALL)
    68→  );
    69→
    70→  // Mark starting cell as path
    71→  const startR = 1;
    72→  const startC = 1;
    73→  grid[startR][startC] = PATH;
    74→
    75→  // Stack-based DFS
    76→  const stack: [number, number][] = [[startR, startC]];
    77→  const directions = [
    78→    [-2, 0], // up
    79→    [2, 0],  // down
    80→    [0, -2], // left
    81→    [0, 2],  // right
    82→  ];
    83→
    84→  while (stack.length > 0) {
    85→    const [cr, cc] = stack[stack.length - 1];
    86→
    87→    // Shuffle directions
    88→    const shuffled = [...directions].sort(() => Math.random() - 0.5);
    89→
    90→    let found = false;
    91→    for (const [dr, dc] of shuffled) {
    92→      const nr = cr + dr;
    93→      const nc = cc + dc;
    94→
    95→      if (nr > 0 && nr < rows - 1 && nc > 0 && nc < cols - 1 && grid[nr][nc] === WALL) {
    96→        // Carve the wall between current and neighbor
    97→        grid[cr + dr / 2][cc + dc / 2] = PATH;
    98→        // Mark neighbor as path
    99→        grid[nr][nc] = PATH;
   100→        stack.push([nr, nc]);
   101→        found = true;
   102→        break;
   103→      }
   104→    }
   105→
   106→    if (!found) {
   107→      stack.pop();
   108→    }
   109→  }
   110→
   111→  // Ensure start and end are open
   112→  grid[1][1] = PATH;
   113→  grid[rows - 2][cols - 2] = PATH;
   114→
   115→  return grid;
   116→}

   478→
   479→    if (Math.abs(dx) > Math.abs(dy)) {
   480→      movePlayer(0, dx > 0 ? 1 : -1);
   481→    } else {
   482→      movePlayer(dy > 0 ? 1 : -1, 0);
   483→    }
   484→    touchStartRef.current = null;
   485→  }, [movePlayer]);
   486→
   487→  // Canvas size setup (high DPI)

   489→    const container = containerRef.current;
   490→    const canvas = canvasRef.current;
   491→    if (!container || !canvas) return;
   492→
   493→    const update = () => {
   494→      const dw = canvas.clientWidth;
   495→      if (dw <= 0) return;
   496→      const displaySize = Math.min(dw, LOGIC_SIZE);
   497→      const dpr = Math.min(window.devicePixelRatio || 1, 2);
   498→      const scale = displaySize / LOGIC_SIZE;
   499→
   500→      canvas.width = displaySize * dpr;
   501→      canvas.height = displaySize * dpr;
   502→
   503→      dprRef.current = dpr;
   504→      scaleRef.current = scale;
   505→      draw();
   506→    };
   507→
   508→    const ro = new ResizeObserver(update);
   509→    ro.observe(container);
   510→    const onVis = () => { if (document.visibilityState === 'visible') update(); };
   511→    document.addEventListener('visibilitychange', onVis);
   512→
   513→    update();
   514→    return () => { ro.disconnect(); document.removeEventListener('visibilitychange', onVis); };
   515→  }, [draw]);
   516→
   517→  // Cleanup
   518→  useEffect(() => {
   519→    return () => cancelAnimationFrame(animFrameRef.current);
   520→  }, []);
   521→
   522→  // GamePlayer pause/resume events
   523→  useEffect(() => {
   524→    const onPause = () => { pausedRef.current = true; setPaused(true); };

   527→    window.addEventListener('game-resume', onResume);

   530→      window.removeEventListener('game-resume', onResume);

   532→  }, []);

   540→

   546→  };
   547→

   549→  if (phase === 'settings') {

   552→        <div className="flex items-center justify-between w-full px-2 pt-2 pb-1">
   553→          <Button variant="ghost" size="sm" onClick={onBack} className="text-muted-foreground hover:text-foreground">
