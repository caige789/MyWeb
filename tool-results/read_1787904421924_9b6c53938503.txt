    63→    63→// Generate maze using recursive backtracking (iterative stack to avoid stack overflow)
    64→    64→function generateMaze(rows: number, cols: number): number[][] {
    65→    65→  // Initialize grid with all walls
    66→    66→  const grid: number[][] = Array.from({ length: rows }, () =>
    67→    67→    Array.from({ length: cols }, () => WALL)
    68→    68→  );
    69→    69→
    70→    70→  // Mark starting cell as path
    71→    71→  const startR = 1;
    72→    72→  const startC = 1;
    73→    73→  grid[startR][startC] = PATH;
    74→    74→
    75→    75→  // Stack-based DFS
    76→    76→  const stack: [number, number][] = [[startR, startC]];
    77→    77→  const directions = [
    78→    78→    [-2, 0], // up
    79→    79→    [2, 0],  // down
    80→    80→    [0, -2], // left
    81→    81→    [0, 2],  // right
    82→    82→  ];
    83→    83→
    84→    84→  while (stack.length > 0) {
    85→    85→    const [cr, cc] = stack[stack.length - 1];
    86→    86→
    87→    87→    // Shuffle directions
    88→    88→    const shuffled = [...directions].sort(() => Math.random() - 0.5);
    89→    89→
    90→    90→    let found = false;
    91→    91→    for (const [dr, dc] of shuffled) {
    92→    92→      const nr = cr + dr;
    93→    93→      const nc = cc + dc;
    94→    94→
    95→    95→      if (nr > 0 && nr < rows - 1 && nc > 0 && nc < cols - 1 && grid[nr][nc] === WALL) {
    96→    96→        // Carve the wall between current and neighbor
    97→    97→        grid[cr + dr / 2][cc + dc / 2] = PATH;
    98→    98→        // Mark neighbor as path
    99→    99→        grid[nr][nc] = PATH;
   100→   100→        stack.push([nr, nc]);
   101→   101→        found = true;
   102→   102→        break;
   103→   103→      }
   104→   104→    }
   105→   105→
   106→   106→    if (!found) {
   107→   107→      stack.pop();
   108→   108→    }
   109→   109→  }
   110→   110→
   111→   111→  // Ensure start and end are open
   112→   112→  grid[1][1] = PATH;
   113→   113→  grid[rows - 2][cols - 2] = PATH;
   114→   114→
   115→   115→  return grid;
   116→   116→}
