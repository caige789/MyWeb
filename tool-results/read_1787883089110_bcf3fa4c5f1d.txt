   182→function createStars(count: number): Star[] {
   183→  const stars: Star[] = [];
   184→  for (let i = 0; i < count; i++) {
   185→    stars.push({
   186→      x: Math.random() * LOGIC_W,
   187→      y: Math.random() * LOGIC_H,
   188→      r: Math.random() * 1.5 + 0.5,
   189→      alpha: Math.random() * 0.6 + 0.4,
   190→      twinkleSpeed: Math.random() * 0.03 + 0.01,
   191→      phase: Math.random() * Math.PI * 2,
   192→    });
   193→  }
   194→  return stars;
   195→}
   196→
   197→function createExplosion(x: number, y: number, color: string, count: number): Particle[] {
   198→  const particles: Particle[] = [];
   199→  for (let i = 0; i < count; i++) {
   200→    const angle = Math.random() * Math.PI * 2;
   201→    const speed = Math.random() * 3 + 1;
   202→    particles.push({
   203→      x, y,
   204→      vx: Math.cos(angle) * speed,
   205→      vy: Math.sin(angle) * speed,
   206→      life: Math.random() * 20 + 15,
   207→      maxLife: 35,
   208→      color,
   209→      r: Math.random() * 3 + 1,
   210→    });
   211→  }
   212→  return particles;
   213→}

   215→function rectCollide(
   216→  ax: number, ay: number, aw: number, ah: number,
   217→  bx: number, by: number, bw: number, bh: number,
   218→): boolean {
   219→  return Math.abs(ax - bx) < (aw + bw) / 2 && Math.abs(ay - by) < (ah + bh) / 2;
   220→}

   233→function drawPlayer(ctx: CanvasRenderingContext2D, p: Player, frame: number) {
   234→  ctx.save();
   235→  ctx.translate(p.x, p.y);
   236→
   237→  // Invincibility blink
   238→  if (p.invincibleTimer > 0 && Math.floor(frame / 3) % 2 === 0) {
   239→    ctx.globalAlpha = 0.35;
   240→  }
   241→
   242→  // Tail flame animation
   243→  const flameLen = 12 + Math.sin(frame * 0.5) * 4;
   244→  const flameW = 8 + Math.sin(frame * 0.7) * 2;
   245→  const flameGrad = ctx.createLinearGradient(0, PLAYER_H / 2, 0, PLAYER_H / 2 + flameLen);
   246→  flameGrad.addColorStop(0, '#38bdf8');
   247→  flameGrad.addColorStop(0.4, '#f97316');
   248→  flameGrad.addColorStop(1, 'rgba(239,68,68,0)');
   249→  ctx.fillStyle = flameGrad;
   250→  ctx.beginPath();
   251→  ctx.moveTo(-flameW, PLAYER_H / 2 - 2);
   252→  ctx.lineTo(flameW, PLAYER_H / 2 - 2);
   253→  ctx.lineTo(0, PLAYER_H / 2 + flameLen);
   254→  ctx.closePath();
   255→  ctx.fill();
   256→
   257→  // Aircraft body
   258→  ctx.fillStyle = '#e0f2fe';
   259→  ctx.beginPath();
   260→  ctx.moveTo(0, -PLAYER_H / 2);
   261→  ctx.lineTo(-PLAYER_W / 2, PLAYER_H / 2 - 6);
   262→  ctx.lineTo(-PLAYER_W / 4, PLAYER_H / 2);
   263→  ctx.lineTo(PLAYER_W / 4, PLAYER_H / 2);
   264→  ctx.lineTo(PLAYER_W / 2, PLAYER_H / 2 - 6);
   265→  ctx.closePath();
   266→  ctx.fill();
   267→
   268→  // Wing decoration lines
   269→  ctx.strokeStyle = '#38bdf8';
   270→  ctx.lineWidth = 1.5;
   271→  ctx.beginPath();
   272→  ctx.moveTo(-PLAYER_W / 2 + 4, PLAYER_H / 2 - 10);
   273→  ctx.lineTo(0, -PLAYER_H / 2 + 10);
   274→  ctx.lineTo(PLAYER_W / 2 - 4, PLAYER_H / 2 - 10);
   275→  ctx.stroke();
   276→
   277→  // Cockpit
   278→  ctx.fillStyle = '#7dd3fc';
   279→  ctx.beginPath();
   280→  ctx.ellipse(0, -PLAYER_H / 4, 4, 7, 0, 0, Math.PI * 2);
   281→  ctx.fill();
   282→
   283→  // Shield bubble
   284→  if (p.shieldTimer > 0) {
   285→    const shieldAlpha = p.shieldTimer < 60 ? (Math.sin(frame * 0.3) * 0.3 + 0.4) : 0.5;
   286→    ctx.strokeStyle = `rgba(6, 182, 212, ${shieldAlpha})`;
   287→    ctx.lineWidth = 2.5;
   288→    ctx.beginPath();
   289→  ctx.ellipse(0, 0, PLAYER_W / 2 + 8, PLAYER_H / 2 + 8, 0, 0, Math.PI * 2);
   290→    ctx.stroke();
   291→  }
   292→
   293→  ctx.restore();
   294→}

   296→function drawEnemy(ctx: CanvasRenderingContext2D, e: Enemy) {
   297→  ctx.save();
   298→  ctx.translate(e.x, e.y);
   299→
   300→  if (e.type === EnemyType.Normal) {
   301→    ctx.fillStyle = '#ef4444';
   302→    ctx.beginPath();
   303→    ctx.moveTo(0, e.h / 2);
   304→    ctx.lineTo(-e.w / 2, -e.h / 2);
   305→    ctx.lineTo(e.w / 2, -e.h / 2);
   306→    ctx.closePath();
   307→    ctx.fill();
   308→    ctx.strokeStyle = '#fca5a5';
   309→    ctx.lineWidth = 1;
   310→    ctx.beginPath();
   311→    ctx.moveTo(0, e.h / 2 - 6);
   312→    ctx.lineTo(-e.w / 4, -e.h / 4);
   313→    ctx.moveTo(0, e.h / 2 - 6);
   314→    ctx.lineTo(e.w / 4, -e.h / 4);
   315→    ctx.stroke();
   316→  } else if (e.type === EnemyType.Fast) {
   317→    ctx.fillStyle = '#f59e0b';
   318→    ctx.beginPath();
   319→    ctx.moveTo(0, e.h / 2);
   320→    ctx.lineTo(-e.w / 2, -e.h / 3);
   321→    ctx.lineTo(-e.w / 4, -e.h / 2);
   322→    ctx.lineTo(e.w / 4, -e.h / 2);
   323→    ctx.lineTo(e.w / 2, -e.h / 3);
   324→    ctx.closePath();
   325→    ctx.fill();
   326→    ctx.fillStyle = '#fcd34d';
   327→    ctx.beginPath();
   328→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   329→    ctx.fill();
   330→  } else if (e.type === EnemyType.Tank) {
   331→    const r = e.w / 2;
   332→    ctx.fillStyle = '#a855f7';
   333→    ctx.beginPath();
   334→    for (let i = 0; i < 6; i++) {
   335→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   336→      const px = Math.cos(angle) * r;
   337→      const py = Math.sin(angle) * r;
   338→      if (i === 0) ctx.moveTo(px, py);
   339→      else ctx.lineTo(px, py);
   340→    }
   341→    ctx.closePath();
   342→    ctx.fill();
   343→    ctx.strokeStyle = '#d8b4fe';
   344→    ctx.lineWidth = 1.5;
   345→    ctx.beginPath();
   346→    for (let i = 0; i < 6; i++) {
   347→      const angle = (Math.PI / 3) * i - Math.PI / 2;
   348→      const px = Math.cos(angle) * (r * 0.55);
   349→      const py = Math.sin(angle) * (r * 0.55);
   350→      if (i === 0) ctx.moveTo(px, py);
   351→      else ctx.lineTo(px, py);
   352→    }
   353→    ctx.closePath();
   354→    ctx.stroke();
   355→    ctx.fillStyle = '#e9d5ff';
   356→    ctx.beginPath();
   357→    ctx.arc(0, 0, 3, 0, Math.PI * 2);
   358→    ctx.fill();
   359→  }
   360→
   361→  ctx.restore();
   362→
   363→  // HP bar for multi-hp enemies
   364→  if (e.hp < e.maxHp) {
   365→    const barW = e.w * 0.8;
   366→    const barH = 3;
   367→    const barX = e.x - barW / 2;
   368→    const barY = e.y - e.h / 2 - 6;
   369→    ctx.fillStyle = 'rgba(255,255,255,0.2)';
   370→    ctx.fillRect(barX, barY, barW, barH);
   371→    ctx.fillStyle = '#22c55e';
   372→    ctx.fillRect(barX, barY, barW * (e.hp / e.maxHp), barH);
   373→  }
   374→}

   376→function drawBoss(ctx: CanvasRenderingContext2D, boss: Boss, frame: number) {
   377→  ctx.save();
   378→  ctx.translate(boss.x, boss.y);
   379→
   380→  // Flash white on hit
   381→  if (boss.flashTimer > 0) {
   382→    ctx.globalAlpha = 0.5 + Math.sin(frame) * 0.5;
   383→    ctx.fillStyle = 'rgba(255,255,255,0.6)';
   384→    ctx.beginPath();
   385→    ctx.ellipse(0, 0, BOSS_W / 2 + 4, BOSS_H / 2 + 4, 0, 0, Math.PI * 2);
   386→    ctx.fill();
   387→    ctx.globalAlpha = 1;
   388→  }
   389→
   390→  // Enrage glow for Type 5
   391→  if (boss.enraged) {
   392→    const enrageAlpha = 0.15 + Math.sin(frame * 0.2) * 0.1;
   393→    ctx.fillStyle = `rgba(255, 0, 0, ${enrageAlpha})`;
   394→    ctx.beginPath();
   395→    ctx.ellipse(0, 0, BOSS_W / 2 + 12, BOSS_H / 2 + 12, 0, 0, Math.PI * 2);
   396→    ctx.fill();
   397→  }
   398→
   399→  // Boss emoji
   400→  ctx.font = `${BOSS_H}px sans-serif`;
   401→  ctx.textAlign = 'center';
   402→  ctx.textBaseline = 'middle';
   403→  ctx.fillText(boss.emoji, 0, 2);
   404→
   405→  ctx.restore();
   406→
   407→  // Boss HP bar at top of canvas
   408→  const barW = LOGIC_W * 0.6;
   409→  const barH = 8;
   410→  const barX = (LOGIC_W - barW) / 2;
   411→  const barY = 6;
   412→
   413→  ctx.fillStyle = 'rgba(255,255,255,0.15)';
   414→  ctx.fillRect(barX - 1, barY - 1, barW + 2, barH + 2);
   415→
   416→  const hpRatio = boss.hp / boss.maxHp;
   417→  const hpColor = hpRatio > 0.5 ? '#ef4444' : hpRatio > 0.25 ? '#f97316' : '#fbbf24';
   418→  ctx.fillStyle = hpColor;
   419→  ctx.fillRect(barX, barY, barW * hpRatio, barH);
   420→
   421→  ctx.strokeStyle = 'rgba(255,255,255,0.4)';
   422→  ctx.lineWidth = 1;
   423→  ctx.strokeRect(barX - 1, barY - 1, barW + 2, barH + 2);
   424→
   425→  // Boss type label
   426→  ctx.fillStyle = '#ffffff';
   427→  ctx.font = 'bold 11px sans-serif';
   428→  ctx.textAlign = 'center';
   429→  ctx.textBaseline = 'top';
   430→  ctx.fillText(`BOSS - ${BOSS_TYPE_NAMES[boss.bossType]}`, LOGIC_W / 2, barY + barH + 2);
   431→
   432→  // Shield bar for Type 3 boss
   433→  if (boss.bossType === 3 && boss.shieldMaxHp > 0) {
   434→    const shieldBarY = barY + barH + 16;
   435→    const shieldRatio = boss.shieldHp / boss.shieldMaxHp;
   436→    ctx.fillStyle = 'rgba(0,0,0,0.3)';
   437→    ctx.fillRect(barX - 1, shieldBarY - 1, barW + 2, 5 + 2);
   438→    ctx.fillStyle = shieldRatio > 0 ? '#3b82f6' : '#1e3a5f';
   439→    ctx.fillRect(barX, shieldBarY, barW * shieldRatio, 5);
   440→    ctx.strokeStyle = 'rgba(100,180,255,0.4)';
   441→    ctx.strokeRect(barX - 1, shieldBarY - 1, barW + 2, 5 + 2);
   442→    ctx.fillStyle = '#93c5fd';
   443→    ctx.font = '9px sans-serif';
   444→    ctx.textAlign = 'center';
   445→    ctx.textBaseline = 'top';
   446→    ctx.fillText('SHIELD', LOGIC_W / 2, shieldBarY + 6);
   447→  }
   448→}

   508→  ctx.font = 'bold 11px sans-serif';
   509→  ctx.textAlign = 'left';
   510→  ctx.textBaseline = 'top';
   511→  ctx.fillText(`STAGE ${stage}`, 6, topOffset + extraOffset + 18);

   522→  ctx.fillText(`WPN ${wpnNames[p.weaponLevel]}`, LOGIC_W - 6, topOffset + extraOffset);

   524→}

   994→                    x: boss.x,
   995→                    y: boss.y + BOSS_H / 2,
   996→                    vx: Math.cos(angle) * BOSS_BULLET_SPEED,

   998→                  });
   999→                }
  1000→              }
  1001→              break;
  1002→            }
  1003→            case 2: {
  1004→              // Rapid Fire: 5-way narrow spread, every 3rd shot is aimed

  1006→              if (boss.shootTimer >= interval) {
  1007→                boss.shootTimer = 0;
  1008→                boss.specialTimer++;
  1009→
  1010→                if (boss.specialTimer % 3 === 0) {
  1011→                  // Aimed shot at player
  1012→                  const dx = p.x - boss.x;
  1013→                  const dy = p.y - boss.y;
  1014→                  const dist = Math.sqrt(dx * dx + dy * dy);
  1015→                  if (dist > 0) {

  1017→                      x: boss.x,
  1018→                      y: boss.y + BOSS_H / 2,
  1019→                      vx: (dx / dist) * (BOSS_BULLET_SPEED * 1.3),
  1020→                      vy: (dy / dist) * (BOSS_BULLET_SPEED * 1.3),
  1021→                    });
  1022→                  }
  1023→                } else {
  1024→                  // 5-way narrow spread
  1025→                  const narrowAngle = Math.PI / 12;
  1026→                  for (let i = -2; i <= 2; i++) {
  1027→                    const angle = Math.PI / 2 + i * narrowAngle;
  1028→                    enemyBulletsRef.current.push({
  1029→                      x: boss.x,
