   109→/** 在[fromX, toX)区间程序化生成地图 */
   110→function generateChunk(
   111→  platforms: Platform[], coins: Coin[], monsters: Monster[],
   112→  fromX: number, toX: number
   113→): number {
   114→  let x = fromX;
   115→  while (x < toX) {
   116→    const r = Math.random();
   117→    if (r < 0.35) {
   118→      /* 地面段 */
   119→      const w = 120 + Math.random() * 200;
   120→      platforms.push({ x, y: GROUND_Y, w, h: 60 });
   121→      if (Math.random() < 0.5) {
   122→        monsters.push(makeMonster(Math.random() < 0.6 ? 'slime' : 'skeleton', x + 30, GROUND_Y));
   123→      }
   124→      const nc = Math.floor(Math.random() * 4) + 1;
   125→      for (let i = 0; i < nc; i++) {
   126→        coins.push({ x: x + 20 + i * 30, y: GROUND_Y - 40, collected: false, phase: Math.random() * 6.28 });
   127→      }
   128→      x += w + 30 + Math.random() * 50;
   129→    } else if (r < 0.65) {
   130→      /* 浮空平台 */
   131→      const w = 80 + Math.random() * 100;
   132→      const py = GROUND_Y - 80 - Math.random() * 120;
   133→      platforms.push({ x, y: py, w, h: 20 });
   134→      coins.push({ x: x + w / 2, y: py - 35, collected: false, phase: Math.random() * 6.28 });
   135→      if (Math.random() < 0.3) monsters.push(makeMonster('slime', x + 10, py));
   136→      x += w + 50 + Math.random() * 80;
   137→    } else if (r < 0.8) {
   138→      /* 阶梯 */
   139→      const steps = 2 + Math.floor(Math.random() * 2);
   140→      let sx = x;
   141→      for (let s = 0; s < steps; s++) {
   142→        const sw = 60 + Math.random() * 40;
   143→        const py = GROUND_Y - 50 - s * 55;
   144→        platforms.push({ x: sx, y: py, w: sw, h: 18 });
   145→        coins.push({ x: sx + sw / 2, y: py - 30, collected: false, phase: Math.random() * 6.28 });
   146→        sx += sw + 20;
   147→      }
   148→      x = sx + 40;
   149→    } else {
   150→      /* 深渊+地面 */
   151→      x += 60 + Math.random() * 60;
   152→      const w = 150 + Math.random() * 200;
   153→      platforms.push({ x, y: GROUND_Y, w, h: 60 });
   154→      if (Math.random() < 0.6) {
   155→        monsters.push(makeMonster(Math.random() < 0.6 ? 'slime' : 'skeleton', x + 20, GROUND_Y));
   156→      }
   157→      const nc = Math.floor(Math.random() * 3) + 1;
   158→      for (let i = 0; i < nc; i++) {
   159→        coins.push({ x: x + 30 + i * 35, y: GROUND_Y - 40, collected: false, phase: Math.random() * 6.28 });
   160→      }
   161→      x += w + 30 + Math.random() * 50;
   162→    }
   163→  }
   164→  return x;
   165→}

   302→    ctx.globalAlpha = 0.4;
   303→    ctx.strokeStyle = '#93c5fd';
   304→    ctx.lineWidth = 2;
   305→    ctx.beginPath();
   306→    ctx.arc(0, 0, 22, angle - 0.5, angle + 0.5);
   307→    ctx.stroke();
   308→    ctx.globalAlpha = 1;
   309→  }

   313→/** 绘制玩家角色 */
   314→function drawPlayer(ctx: CanvasRenderingContext2D, p: PlayerState, cam: number) {
   315→  if (p.invincible && Math.sin(performance.now() / 60) > 0) return;
   316→  const x = p.x - cam;
   317→  const y = p.y;
   318→  const running = Math.abs(p.vx) > 0.5;
   319→  const air = !p.onGround;
   320→  ctx.save();
   321→  ctx.translate(x + PLAYER_W / 2, y + PLAYER_H / 2);
   322→  ctx.scale(p.facing, 1);
   323→  ctx.translate(-PLAYER_W / 2, -PLAYER_H / 2);
   324→  /* 腿 */
   325→  const ls = running && !air ? Math.sin(p.frame * Math.PI / 2) * 6 : 0;
   326→  ctx.fillStyle = '#1e40af';
   327→  ctx.fillRect(6, 30, 6, 10 + (air ? -3 : ls));
   328→  ctx.fillRect(16, 30, 6, 10 + (air ? 3 : -ls));
   329→  ctx.fillStyle = '#78350f';
   330→  ctx.fillRect(4, 38 + (air ? -3 : ls), 9, 3);
   331→  ctx.fillRect(15, 38 + (air ? 3 : -ls), 9, 3);
   332→  /* 身体 */
   333→  ctx.fillStyle = '#3b82f6';
   334→  ctx.fillRect(5, 16, 18, 16);
   335→  ctx.fillStyle = '#92400e';
   336→  ctx.fillRect(5, 28, 18, 3);
   337→  ctx.fillStyle = '#fbbf24';
   338→  ctx.fillRect(12, 27, 5, 5);
   339→  /* 手臂 */
   340→  const as = running && !air ? Math.sin(p.frame * Math.PI / 2) * 8 : 0;
   341→  if (p.attacking) {
   342→    ctx.fillStyle = '#2563eb';
   343→    ctx.fillRect(18, 18, 16, 5);
   344→    drawSword(ctx, 34, 16, p.atkTimer);
   345→  } else {
   346→    ctx.fillStyle = '#2563eb';
   347→    ctx.fillRect(18, 18, 5, 12 + as);
   348→  }
   349→  ctx.fillStyle = '#2563eb';
   350→  ctx.fillRect(0, 18, 5, 12 - as);
   351→  /* 头 */
   352→  ctx.fillStyle = '#fbbf24';
   353→  ctx.fillRect(5, 2, 18, 15);
   354→  ctx.fillStyle = '#78350f';
   355→  ctx.fillRect(4, 0, 20, 5);
   356→  ctx.fillRect(3, 2, 4, 8);
   357→  ctx.fillStyle = '#1a1a2e';
   358→  ctx.fillRect(16, 6, 3, 3);
   359→  ctx.fillStyle = '#dc2626';
   360→  ctx.fillRect(15, 12, 5, 1);
   361→  ctx.restore();
   362→}

   402→  const gRef = useRef<GameCore>({
   403→    status: 'idle', player: makePlayer(),
   404→    platforms: [], coins: [], monsters: [], floats: [],
   405→    score: 0, lives: 3, cam: 0, genUpTo: 0, distance: 0,
   406→    lastTime: 0, keys: new Set(),
   407→    touchL: false, touchR: false, touchJ: false, touchA: false,
   408→    jumpHeld: false, scale: 1,
   409→  });

   411→  /* 检测移动端 */

   424→        if (res.code === 0 && res.data) {
   425→          const found = res.data.find((s: { game: string; score: number }) => s.game === GAME_NAME);
   426→          if (found) setHighScore(found.score);
   427→        }
   428→      })
   429→      .catch(() => { /* 静默处理 */ });
   430→  }, []);
   431→
   432→  /* 提交分数 */
   433→  const submitScore = (finalScore: number) => {
   434→    fetch('/api/games/scores', {
   435→      method: 'POST',
   436→      headers: { 'Content-Type': 'application/json' },
   437→      body: JSON.stringify({ game: GAME_NAME, score: finalScore }),
   438→    })
   439→      .then(r => r.json())
   440→      .then(res => {
   441→        if (res.code === 0 && res.data && finalScore >= res.data.score) {
   442→          setIsNewRecord(true);
   443→          setHighScore(finalScore);
   444→        }
   445→      })
   446→      .catch(() => { /* 静默处理 */ });
   447→  };

   468→    for (let i = 0; i < 5; i++) {
   469→      g.coins.push({ x: 200 + i * 40, y: GROUND_Y - 40, collected: false, phase: Math.random() * 6.28 });
   470→    }
   471→    g.genUpTo = generateChunk(g.platforms, g.coins, g.monsters, 400, 400 + CHUNK_W * 3);
   472→    setScore(0);
   473→    setLives(3);
   474→    setIsNewRecord(false);
   475→    setGameStatus('playing');
   476→  };
   477→
   478→  /* 游戏结束 */
   479→  const endGame = (g: GameCore) => {
   480→    g.status = 'over';
   481→    setGameStatus('over');
   482→    submitScore(g.score);
   483→  };
   484→
   485→  /* 玩家受伤 */
   486→  const hurtPlayer = (g: GameCore) => {
   487→    if (g.player.invincible) return;
   488→    g.lives--;
   489→    setLives(g.lives);
   490→    if (g.lives <= 0) {
   491→      endGame(g);
   492→    } else {
   493→      g.player.invincible = true;
   494→      g.player.invTimer = INVINCIBLE_TIME;
   495→    }
   496→  };
   497→
   498→  /* 重生 */
   499→  const respawn = (g: GameCore) => {
   500→    let best: Platform | null = null;
   501→    let bd = Infinity;
   502→    for (const pl of g.platforms) {
   503→      if (pl.x + pl.w > g.cam && pl.x < g.cam + CANVAS_W) {
   504→        const d = Math.abs(pl.x + pl.w / 2 - (g.cam + CANVAS_W / 3));
   505→        if (d < bd) { bd = d; best = pl; }
   506→      }
   507→    }
   508→    if (best) {
   509→      g.player.x = best.x + best.w / 2 - PLAYER_W / 2;
   510→      g.player.y = best.y - PLAYER_H;
   511→    } else {
   512→      g.player.x = g.cam + 100;
   513→      g.player.y = GROUND_Y - PLAYER_H;
   514→    }
   515→    g.player.vx = 0;
   516→    g.player.vy = 0;
   517→    g.player.onGround = false;
   518→    g.player.invincible = true;
   519→    g.player.invTimer = INVINCIBLE_TIME;
   520→  };
   521→
   522→  /* Core update logic (pure function, receives g as param) */
   523→  const tick = (g: GameCore, dt: number) => {
   524→    if (pausedRef.current) return;
   525→    const p = g.player;
   526→    const k = g.keys;
   527→
   528→    /* 读取输入 */
   529→    const ml = k.has('ArrowLeft') || k.has('a') || k.has('A') || g.touchL;
   530→    const mr = k.has('ArrowRight') || k.has('d') || k.has('D') || g.touchR;
   531→    const wj = k.has('ArrowUp') || k.has('w') || k.has('W') || k.has(' ') || g.touchJ;
   532→    const wa = k.has('j') || k.has('J') || g.touchA;
   533→
   534→    /* 移动 */
   535→    if (ml) { p.vx = -MOVE_SPEED; p.facing = -1; }
   536→    else if (mr) { p.vx = MOVE_SPEED; p.facing = 1; }
   537→    else { p.vx *= 0.7; if (Math.abs(p.vx) < 0.3) p.vx = 0; }
   538→
   539→    /* 跳跃 */
   540→    if (wj && !g.jumpHeld && p.onGround) {
   541→      p.vy = JUMP_FORCE;
   542→      p.onGround = false;
   543→      g.jumpHeld = true;
   544→    }
   545→    if (!wj) g.jumpHeld = false;
   546→
   547→    /* 攻击 */
   548→    if (wa && !p.attacking) { p.attacking = true; p.atkTimer = ATTACK_DURATION; }
   549→    if (p.attacking) { p.atkTimer--; if (p.atkTimer <= 0) p.attacking = false; }
   550→
   551→    /* 无敌计时 */
   552→    if (p.invincible) { p.invTimer -= dt; if (p.invTimer <= 0) p.invincible = false; }
   553→
   554→    /* 动画 */
   555→    p.frameTimer += dt;
   556→    if (p.frameTimer > 120) { p.frameTimer = 0; p.frame = (p.frame + 1) % 4; }
   557→
   558→    /* 重力 */
   559→    p.vy += GRAVITY;
   560→    if (p.vy > MAX_FALL) p.vy = MAX_FALL;
   561→
   562→    /* 水平移动+碰撞 */
   563→    p.x += p.vx;
   564→    for (const pl of g.platforms) {
   565→      if (pl.x + pl.w <= g.cam - 50 || pl.x >= g.cam + CANVAS_W + 50) continue;
   566→      if (p.x + PLAYER_W > pl.x && p.x < pl.x + pl.w && p.y + PLAYER_H > pl.y + 4 && p.y < pl.y + pl.h) {
   567→        if (p.vx > 0) p.x = pl.x - PLAYER_W;
   568→        else if (p.vx < 0) p.x = pl.x + pl.w;
   569→        p.vx = 0;
   570→      }
   571→    }
   572→
   573→    /* 垂直移动+碰撞 */
   574→    p.y += p.vy;
   575→    p.onGround = false;
   576→    for (const pl of g.platforms) {
   577→      if (pl.x + pl.w <= g.cam - 50 || pl.x >= g.cam + CANVAS_W + 50) continue;
   578→      if (p.x + PLAYER_W > pl.x + 2 && p.x < pl.x + pl.w - 2 && p.y + PLAYER_H > pl.y && p.y + PLAYER_H < pl.y + pl.h + 10 && p.vy >= 0) {
   579→        p.y = pl.y - PLAYER_H;
   580→        p.vy = 0;
   581→        p.onGround = true;
   582→      }
   583→    }
   584→
   585→    /* 相机跟随 */
   586→    if (p.x - g.cam > CANVAS_W * 0.65) g.cam = p.x - CANVAS_W * 0.65;
   587→    if (g.cam < 0) g.cam = 0;
   588→    if (g.cam < p.x - 50) g.cam += SCROLL_SPEED * 0.5;
   589→    g.distance = Math.floor(p.x / 100);
   590→
   591→    /* 程序化生成 */
   592→    const threshold = g.cam + CANVAS_W + CHUNK_W;
   593→    if (g.genUpTo < threshold) {
   594→      g.genUpTo = generateChunk(g.platforms, g.coins, g.monsters, g.genUpTo, threshold + CHUNK_W);
   595→    }
   596→
   597→    /* 清理远处物体 */
   598→    const cx = g.cam - 300;
   599→    g.platforms = g.platforms.filter(pl => pl.x + pl.w > cx);
   600→    g.coins = g.coins.filter(c => c.x > cx);
   601→    g.monsters = g.monsters.filter(m => m.x + m.w > cx);
   602→
   603→    /* 收集金币 */
   604→    const pcx = p.x + PLAYER_W / 2;
   605→    const pcy = p.y + PLAYER_H / 2;
   606→    for (const c of g.coins) {
   607→      if (c.collected) continue;
   608→      const dx = pcx - c.x;
   609→      const dy = pcy - c.y;
   610→      if (dx * dx + dy * dy < 500) {
   611→        c.collected = true;
   612→        g.score += 10;
   613→        g.floats.push({ x: c.x, y: c.y, text: '+10', color: '#fbbf24', life: 60 });
   614→      }
   615→    }
   616→
   617→    /* 怪物AI */
   618→    for (const m of g.monsters) {
   619→      if (!m.alive) continue;
   620→      m.phase += 0.08;
   621→      /* 找脚下平台 */
   622→      let onPlat: Platform | null = null;
   623→      for (const pl of g.platforms) {
   624→        if (m.x + m.w > pl.x && m.x < pl.x + pl.w && Math.abs(m.y + m.h - pl.y) < 5) { onPlat = pl; break; }
   625→      }
   626→      m.x += m.vx;
   627→      /* 边缘转向 */
   628→      if (onPlat) {
   629→        if (m.x <= onPlat.x + 5) { m.x = onPlat.x + 5; m.vx = Math.abs(m.vx); }
   630→        else if (m.x + m.w >= onPlat.x + onPlat.w - 5) { m.x = onPlat.x + onPlat.w - m.w - 5; m.vx = -Math.abs(m.vx); }
   631→      }
   632→      /* 攻击判定 */
   633→      if (p.attacking && p.atkTimer > ATTACK_DURATION - 10) {
   634→        const ax = p.x + (p.facing > 0 ? PLAYER_W : 0);
   635→        const ay = p.y + PLAYER_H / 2;
   636→        const mdx = (m.x + m.w / 2) - ax;
   637→        const mdy = (m.y + m.h / 2) - ay;
   638→        if (Math.sqrt(mdx * mdx + mdy * mdy) < ATTACK_RANGE && p.facing * mdx >= -10) {
   639→          m.hp--;
   640→          if (m.hp <= 0) {
   641→            m.alive = false;
   642→            const pts = m.kind === 'slime' ? 20 : 50;
   643→            g.score += pts;
   644→            g.floats.push({ x: m.x + m.w / 2, y: m.y, text: `+${pts}`, color: m.kind === 'slime' ? '#4ade80' : '#f87171', life: 60 });
   645→          }
   646→        }
   647→      }
   648→      /* 踩怪判定 */
   649→      if (m.alive && !p.invincible) {
   650→        const pb = p.y + PLAYER_H;
   651→        const prevB = pb - p.vy;
   652→        if (p.vy > 0 && prevB <= m.y + 5 && pb >= m.y && p.x + PLAYER_W > m.x + 3 && p.x < m.x + m.w - 3) {
   653→          m.hp--;
   654→          p.vy = JUMP_FORCE * 0.6;
   655→          if (m.hp <= 0) {
   656→            m.alive = false;
   657→            const pts = m.kind === 'slime' ? 20 : 50;
   658→            g.score += pts;
   659→            g.floats.push({ x: m.x + m.w / 2, y: m.y, text: `+${pts}`, color: m.kind === 'slime' ? '#4ade80' : '#f87171', life: 60 });
   660→          }
   661→          continue;
   662→        }
   663→      }
   664→      /* 碰撞伤害 */
   665→      if (m.alive && !p.invincible) {
   666→        if (p.x + PLAYER_W > m.x + 2 && p.x < m.x + m.w - 2 && p.y + PLAYER_H > m.y + 2 && p.y < m.y + m.h - 2) {
   667→          hurtPlayer(g);
   668→        }
   669→      }
   670→    }
   671→
   672→    /* 坠落 */
   673→    if (p.y > CANVAS_H + 50) {
   674→      g.lives--;
   675→      setLives(g.lives);
   676→      if (g.lives <= 0) { endGame(g); }
   677→      else { respawn(g); }
   678→    }
   679→
   680→    /* 防止跑出左边界 */
   681→    if (p.x < g.cam - 20) p.x = g.cam - 20;
   682→
   683→    /* 更新飘字 */
   684→    g.floats = g.floats.filter(ft => { ft.life--; ft.y -= 1.2; return ft.life > 0; });
   685→  };
   686→
   687→  /* 绘制完整画面 */
   688→  const render = (g: GameCore) => {
   689→    const canvas = canvasRef.current;
   690→    if (!canvas) return;
   691→    const ctx = canvas.getContext('2d');
   692→    if (!ctx) return;
   693→    const s = g.scale;
   694→    ctx.setTransform(s, 0, 0, s, 0, 0);
   695→    /* 天空 */
   696→    const sky = ctx.createLinearGradient(0, 0, 0, CANVAS_H);
   697→    sky.addColorStop(0, '#87ceeb');
   698→    sky.addColorStop(0.6, '#b0e0f6');
   699→    sky.addColorStop(1, '#d4f0fc');
   700→    ctx.fillStyle = sky;
   701→    ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   702→    drawClouds(ctx, g.cam);
   703→    /* 平台 */
   704→    for (const pl of g.platforms) {
   705→      if (pl.x + pl.w < g.cam - 10 || pl.x > g.cam + CANVAS_W + 10) continue;
   706→      drawPlatform(ctx, pl, g.cam);
   707→    }
   708→    /* 金币 */
   709→    const t = performance.now() / 1000;
   710→    for (const c of g.coins) {
   711→      if (c.collected || c.x < g.cam - 20 || c.x > g.cam + CANVAS_W + 20) continue;
   712→      drawCoin(ctx, c.x - g.cam, c.y, t + c.phase);
   713→    }
   714→    /* 怪物 */
   715→    for (const m of g.monsters) {
   716→      if (!m.alive || m.x + m.w < g.cam - 20 || m.x > g.cam + CANVAS_W + 20) continue;
   717→      const mx = m.x - g.cam;
   718→      if (m.kind === 'slime') drawSlime(ctx, mx, m.y, m.w, m.h, m.phase, m.vx);
   719→      else drawSkeleton(ctx, mx, m.y, m.w, m.h, m.phase, m.hp);
   720→    }
   721→    /* 玩家 */
   722→    drawPlayer(ctx, g.player, g.cam);
   723→    /* Float text */
   724→    for (const ft of g.floats) {
   725→      ctx.globalAlpha = ft.life / 60;
   726→      ctx.fillStyle = ft.color;
   727→      ctx.font = 'bold 16px sans-serif';
   728→      ctx.textAlign = 'center';
   729→      ctx.fillText(ft.text, ft.x - g.cam, ft.y);
   730→      ctx.globalAlpha = 1;
   731→    }
   732→    /* Paused overlay */
   733→    if (pausedRef.current) {
   734→      ctx.fillStyle = 'rgba(0,0,0,0.55)';
   735→      ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   736→      ctx.fillStyle = '#ffffff';
   737→      ctx.font = 'bold 32px sans-serif';
   738→      ctx.textAlign = 'center';
   739→      ctx.fillText('已暂停', CANVAS_W / 2, CANVAS_H / 2 - 10);
   740→      ctx.font = '16px sans-serif';
   741→      ctx.fillStyle = 'rgba(255,255,255,0.6)';
   742→      ctx.fillText('按 Esc 继续', CANVAS_W / 2, CANVAS_H / 2 + 20);
   743→      ctx.textAlign = 'start';
   744→    }
   745→  };
   746→
   747→  /* Game loop */
   748→  const loop = (timestamp: number) => {
   749→    const g = gRef.current;
   750→    if (g.status !== 'playing') return;
   751→    if (pausedRef.current) {
   752→      g.lastTime = 0;
   753→      render(g);
   754→      rafRef.current = requestAnimationFrame(loop);
   755→      return;
   756→    }
   757→    const dt = g.lastTime ? timestamp - g.lastTime : 16;
   758→    g.lastTime = timestamp;
   759→    tick(g, dt);
   760→    render(g);
   761→    /* 低频更新React状态 */
   762→    uiUpdateRef.current++;
   763→    if (uiUpdateRef.current % 6 === 0) {
   764→      setScore(g.score);
   765→      setDistance(g.distance);
   766→    }
   767→    rafRef.current = requestAnimationFrame(loop);
   768→  };
   769→
   770→  /* 开始游戏 */
   771→  const startGame = () => {
   772→    if (rafRef.current) cancelAnimationFrame(rafRef.current);
   773→    initGameData();
   774→    rafRef.current = requestAnimationFrame(loop);
   775→  };
   776→
   777→  /* Keyboard events */
   778→  useEffect(() => {
   779→    const onDown = (e: KeyboardEvent) => {
   780→      if (e.key === 'Escape' && gRef.current.status === 'playing') {
   781→        e.preventDefault();
   782→        pausedRef.current = !pausedRef.current;
   783→        setPaused(pausedRef.current);
   784→        return;
   785→      }
   786→      if (pausedRef.current) return;
   787→      gRef.current.keys.add(e.key);
   788→      if (['ArrowUp', 'ArrowDown', 'ArrowLeft', 'ArrowRight', ' '].includes(e.key)) e.preventDefault();
   789→    };
   790→    const onUp = (e: KeyboardEvent) => { gRef.current.keys.delete(e.key); };
   791→    window.addEventListener('keydown', onDown);
   792→    window.addEventListener('keyup', onUp);
   793→    return () => { window.removeEventListener('keydown', onDown); window.removeEventListener('keyup', onUp); };
   794→  }, []);
   795→
   796→  /* Mouse attack */
   797→  useEffect(() => {
   798→    const canvas = canvasRef.current;
   799→    if (!canvas) return;
   800→    const onClick = () => { if (gRef.current.status === 'playing' && !pausedRef.current) { const p = gRef.current.player; if (!p.attacking) { p.attacking = true; p.atkTimer = ATTACK_DURATION; } } };
   801→    canvas.addEventListener('click', onClick);
   802→    return () => canvas.removeEventListener('click', onClick);
   803→  }, []);
   804→
   805→  /* 高DPI适配 */
   806→  useEffect(() => {
   807→    const canvas = canvasRef.current;
   808→    const container = containerRef.current;
   809→    if (!canvas || !container) return;
   810→    const resize = () => {
   811→      const cw = container.clientWidth;
   812→      const dw = Math.min(cw, CANVAS_W);
   813→      const dpr = window.devicePixelRatio || 1;
   814→      const dh = (dw / CANVAS_W) * CANVAS_H;
   815→      canvas.style.width = `${dw}px`;
   816→      canvas.style.height = `${dh}px`;
   817→      canvas.width = dw * dpr;
   818→      canvas.height = dh * dpr;
   819→      gRef.current.scale = (dw * dpr) / CANVAS_W;
   820→    };
   821→    resize();
   822→    window.addEventListener('resize', resize);
   823→    return () => window.removeEventListener('resize', resize);
   824→  }, []);
   825→
   826→  /* 清理 */
   827→  useEffect(() => { return () => { if (rafRef.current) cancelAnimationFrame(rafRef.current); }; }, []);
   828→
   829→  /* 绘制待机画面 */
   830→  useEffect(() => {
   831→    if (gameStatus !== 'idle') return;
   832→    const timer = setTimeout(() => {
   833→      const canvas = canvasRef.current;
   834→      if (!canvas) return;
   835→      const ctx = canvas.getContext('2d');
   836→      if (!ctx) return;
   837→      const s = gRef.current.scale;
   838→      ctx.setTransform(s, 0, 0, s, 0, 0);
   839→      const sky = ctx.createLinearGradient(0, 0, 0, CANVAS_H);
   840→      sky.addColorStop(0, '#87ceeb');
   841→      sky.addColorStop(1, '#d4f0fc');
   842→      ctx.fillStyle = sky;
   843→      ctx.fillRect(0, 0, CANVAS_W, CANVAS_H);
   844→      ctx.fillStyle = '#4ade80';
   845→      ctx.fillRect(0, GROUND_Y, CANVAS_W, 8);
   846→      ctx.fillStyle = '#8B6914';
   847→      ctx.fillRect(0, GROUND_Y + 8, CANVAS_W, 52);
   848→      ctx.fillStyle = '#1e40af';
   849→      ctx.font = 'bold 36px sans-serif';
   850→      ctx.textAlign = 'center';
   851→      ctx.fillText('勇者冒险', CANVAS_W / 2, CANVAS_H / 2 - 40);
   852→      ctx.font = '16px sans-serif';
   853→      ctx.fillStyle = '#475569';
   854→      ctx.fillText('横版跑酷闯关游戏', CANVAS_W / 2, CANVAS_H / 2 - 10);
   855→      ctx.font = '14px sans-serif';
   856→      ctx.fillStyle = '#64748b';
   857→      ctx.fillText(isMobile ? '点击下方按钮开始游戏' : '点击「?」查看操作说明', CANVAS_W / 2, CANVAS_H / 2 + 30);
   858→    }, 100);
   859→    return () => clearTimeout(timer);
   860→  }, [gameStatus, isMobile]);
   861→
   862→  /* 距离用state显示 */
   863→
   864→  /* 触摸处理函数 */
   865→  const tb = {
   866→    left: {
   867→      onTouchStart: (e: React.TouchEvent) => { e.preventDefault(); gRef.current.touchL = true; },
   868→      onTouchEnd: (e: React.TouchEvent) => { e.preventDefault(); gRef.current.touchL = false; },
   869→      onTouchCancel: () => { gRef.current.touchL = false; },
   870→    },
   871→    right: {
   872→      onTouchStart: (e: React.TouchEvent) => { e.preventDefault(); gRef.current.touchR = true; },
   873→      onTouchEnd: (e: React.TouchEvent) => { e.preventDefault(); gRef.current.touchR = false; },
   874→      onTouchCancel: () => { gRef.current.touchR = false; },
   875→    },
   876→    jump: {
   877→      onTouchStart: (e: React.TouchEvent) => { e.preventDefault(); gRef.current.touchJ = true; },
   878→      onTouchEnd: (e: React.TouchEvent) => { e.preventDefault(); gRef.current.touchJ = false; },
   879→      onTouchCancel: () => { gRef.current.touchJ = false; },
   880→    },
   881→    attack: {
   882→      onTouchStart: (e: React.TouchEvent) => { e.preventDefault(); gRef.current.touchA = true; },
   883→      onTouchEnd: (e: React.TouchEvent) => { e.preventDefault(); gRef.current.touchA = false; },
   884→      onTouchCancel: () => { gRef.current.touchA = false; },
   885→    },
   886→  };
   887→
   888→  return (
   889→    <div className="min-h-screen flex flex-col items-center p-3 sm:p-4">
   890→      <Card className="w-full max-w-2xl">
   891→        <CardHeader className="pb-2">
   892→          <div className="flex items-center justify-between">
   893→            <div className="flex items-center gap-1">
   894→              <Button variant="ghost" size="sm" onClick={onBack}>
   895→                <ArrowLeft className="h-4 w-4 mr-1" />
   896→                返回
   897→              </Button>
   898→              {gameStatus === 'playing' && <GameControlsHelp info={adventureControlsInfo} variant="button" />}
   899→            </div>
   900→            <CardTitle className="text-base sm:text-lg">勇者冒险</CardTitle>
   901→            <span className="text-amber-500 font-bold text-sm">
   902→              <Trophy className="h-3.5 w-3.5 inline mr-0.5" />
   903→              {highScore}
   904→            </span>
   905→          </div>
   906→          <div className="flex items-center justify-between text-sm px-1">
   907→            <span className="font-bold">分数: {score}</span>
   908→            <span className="text-xs text-muted-foreground">距离: {distance}m</span>
   909→            <span className="text-base">
   910→              {Array.from({ length: 3 }, (_, i) => (
   911→                <span key={i} style={{ opacity: i < lives ? 1 : 0.2 }}>❤️</span>
   912→              ))}
   913→            </span>
   914→          </div>
   915→        </CardHeader>
   916→        <CardContent className="pb-3">
   917→          <div ref={containerRef} className="w-full relative" style={{ touchAction: 'none' }}>
   918→            <canvas
   919→              ref={canvasRef}
   920→              className="w-full rounded-lg border border-border block"
   921→              style={{ maxHeight: '400px' }}
   922→            />
   923→            {gameStatus === 'over' && (
   924→              <div className="absolute inset-0 bg-black/60 rounded-lg flex flex-col items-center justify-center gap-3">
   925→                <div className="text-2xl font-bold text-white">游戏结束</div>
   926→                <div className="text-amber-400 text-lg font-bold">得分: {score}</div>
   927→                {isNewRecord && (
   928→                  <div className="text-yellow-300 text-sm font-bold animate-bounce">
   929→                    🎉 新纪录！
   930→                  </div>
   931→                )}
   932→                <div className="text-gray-300 text-xs">最高分: {highScore}</div>
   933→                <Button onClick={startGame} className="mt-2" size="sm">
   934→                  <RotateCcw className="h-4 w-4 mr-1" />
   935→                  再来一局
   936→                </Button>
   937→              </div>
   938→            )}
   939→            {gameStatus === 'idle' && (
   940→              <div className="absolute inset-0 bg-black/30 rounded-lg flex flex-col items-center justify-center gap-2">
   941→                <Button onClick={startGame} size="lg">
   942→                  开始游戏
   943→                </Button>
   944→              </div>
   945→            )}
   946→          </div>
   947→          {isMobile && gameStatus === 'playing' && !paused && (
   948→            <div className="relative w-full mt-3 flex items-center justify-between" style={{ height: '64px', touchAction: 'none' }}>
   949→              <button
   950→                {...tb.left}
   951→                className="rounded-xl bg-blue-600 active:bg-blue-700 text-white font-bold text-xl w-20 h-14 flex items-center justify-center shadow-lg select-none"
   952→                style={{ touchAction: 'none', userSelect: 'none', WebkitUserSelect: 'none' }}
   953→              >◀</button>
