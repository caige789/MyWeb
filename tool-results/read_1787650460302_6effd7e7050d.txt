   137→/** 当前方块信息 */
   138→interface Piece {
   139→  type: string;

   145→/** 游戏状态类型 */

   148→/* ==================== 工具函数 ==================== */
   149→
   150→/** 随机生成一个方块类型 */
   151→function randomType(): string {
   152→  return PIECE_TYPES[Math.floor(Math.random() * PIECE_TYPES.length)];
   153→}
   154→
   155→/** 深拷贝二维数组 */
   156→function cloneShape(shape: number[][]): number[][] {
   157→  return shape.map(row => [...row]);
   158→}
   159→
   160→/** 顺时针旋转矩阵90度 */
   161→function rotateShape(shape: number[][]): number[][] {
   162→  const n = shape.length;
   163→  const rotated: number[][] = [];
   164→  for (let i = 0; i < n; i++) {
   165→    rotated[i] = [];
   166→    for (let j = 0; j < n; j++) {
   167→      rotated[i][j] = shape[n - 1 - j][i];
   168→    }
   169→  }
   170→  return rotated;
   171→}
   172→
   173→/** 创建新方块 */
   174→function createPiece(type: string): Piece {
   175→  return {
   176→    type,
   177→    shape: cloneShape(SHAPES[type]),
   178→    x: Math.floor((COLS - SHAPES[type][0].length) / 2),
   179→    y: 0,
   180→  };
   181→}
   182→
   183→/** 碰撞检测：检查方块在指定位置是否合法 */
   184→function isValid(
   185→  shape: number[][],
   186→  board: (string | null)[][],
   187→  px: number,
   188→  py: number
   189→): boolean {
   190→  for (let r = 0; r < shape.length; r++) {
   191→    for (let c = 0; c < shape[r].length; c++) {
   192→      if (shape[r][c]) {
   193→        const nx = px + c;
   194→        const ny = py + r;
   195→        if (nx < 0 || nx >= COLS || ny >= ROWS) return false;
   196→        if (ny >= 0 && board[ny][nx] !== null) return false;
   197→      }
   198→    }
   199→  }
   200→  return true;
   201→}

   224→  const gameOverCheckedRef = useRef<boolean>(false);

   226→  const displayWidthRef = useRef<number>(COLS * CELL);
   227→
   228→  /* --- React状态（用于UI渲染） --- */

   280→    const piece = currentRef.current;
   281→    if (!piece) return;

   283→    for (let r = 0; r < piece.shape.length; r++) {
   284→      for (let c = 0; c < piece.shape[r].length; c++) {

   290→          }

   292→      }

   298→    const board = boardRef.current;
   299→    const fullRows: number[] = [];
   300→    for (let r = 0; r < ROWS; r++) {
   301→      if (board[r].every(cell => cell !== null)) {
   302→        fullRows.push(r);
   303→      }
   304→    }
   305→    if (fullRows.length === 0) return 0;
   306→    // 触发闪白动画

   311→  }, []);
   312→
   313→  /** 实际移除已满行并添加空行到顶部 */
   314→  const removeLines = useCallback(() => {
   315→    const board = boardRef.current;
   316→    const fullRows = flashRowsRef.current;
   317→    // 从下往上移除满行
   318→    const newBoard = board.filter((_, idx) => !fullRows.includes(idx));
   319→    // 顶部补空行
   320→    while (newBoard.length < ROWS) {
   321→      newBoard.unshift(Array(COLS).fill(null));
   322→    }
   323→    boardRef.current = newBoard;
   324→    flashRowsRef.current = [];
   325→  }, []);
   326→
   327→  /* ==================== 方块下移 ==================== */
   328→  const moveDown = useCallback(() => {
   329→    const piece = currentRef.current;
   330→    if (!piece || statusRef.current !== 'playing') return;
   331→    const board = boardRef.current;
   332→    if (isValid(piece.shape, board, piece.x, piece.y + 1)) {
   333→      piece.y += 1;
   334→    } else {
   335→      // 方块触底，固定并处理消行
   336→      lockPiece();
   337→      const cleared = clearLines();
   338→      if (cleared > 0) {
   339→        const pts = LINE_SCORES[cleared] || cleared * 100;
   340→        scoreRef.current += pts;
   341→        linesRef.current += cleared;
   342→        // 每10行升一级
   343→        const newLv = Math.floor(linesRef.current / 10) + 1;
   344→        levelRef.current = newLv;
   345→        setScore(scoreRef.current);
   346→        setLevel(newLv);
   347→      }
   348→      // 如果没有消行动画，直接生成新方块
   349→      if (cleared === 0) {
   350→        spawnPiece();
   351→      }
   352→      // 消行动画中会在主循环的渲染阶段处理
   353→    }
   354→  }, [lockPiece, clearLines, spawnPiece]);
   355→
   356→  /* ==================== 左移 ==================== */
   357→  const moveLeft = useCallback(() => {
   358→    const piece = currentRef.current;
   359→    if (!piece || statusRef.current !== 'playing') return;
   360→    if (isValid(piece.shape, boardRef.current, piece.x - 1, piece.y)) {
   361→      piece.x -= 1;
   362→    }
   363→  }, []);
   364→
   365→  /* ==================== 右移 ==================== */
   366→  const moveRight = useCallback(() => {
   367→    const piece = currentRef.current;
   368→    if (!piece || statusRef.current !== 'playing') return;
   369→    if (isValid(piece.shape, boardRef.current, piece.x + 1, piece.y)) {
   370→      piece.x += 1;
   371→    }
   372→  }, []);
   373→
   374→  /* ==================== 旋转 ==================== */
   375→  const rotate = useCallback(() => {
   376→    const piece = currentRef.current;
   377→    if (!piece || statusRef.current !== 'playing') return;
   378→    if (piece.type === 'O') return; // O方块不需要旋转
   379→    const rotated = rotateShape(piece.shape);
   380→    // 尝试基本旋转
   381→    if (isValid(rotated, boardRef.current, piece.x, piece.y)) {
   382→      piece.shape = rotated;
   383→      return;
   384→    }
   385→    // 墙踢：尝试左右偏移
   386→    const kicks = [1, -1, 2, -2];
   387→    for (const kick of kicks) {
   388→      if (isValid(rotated, boardRef.current, piece.x + kick, piece.y)) {
   389→        piece.shape = rotated;
   390→        piece.x += kick;
   391→        return;
   392→      }
   393→    }
   394→  }, []);
   395→
   396→  /* ==================== 硬降 ==================== */
   397→  const hardDrop = useCallback(() => {
   398→    const piece = currentRef.current;
   399→    if (!piece || statusRef.current !== 'playing') return;
   400→    const board = boardRef.current;
   401→    const originalY = piece.y;
   402→    let dropY = piece.y;
   403→    while (isValid(piece.shape, board, piece.x, dropY + 1)) {
   404→      dropY += 1;
   405→    }
   406→    piece.y = dropY;
   407→    // 加上硬降距离的额外分数
   408→    scoreRef.current += (dropY - originalY) * 2;
   409→    setScore(scoreRef.current);
   410→    // 直接触底固定
   411→    lockPiece();
   412→    const cleared = clearLines();
   413→    if (cleared > 0) {
   414→      const pts = LINE_SCORES[cleared] || cleared * 100;
   415→      scoreRef.current += pts;
   416→      linesRef.current += cleared;
   417→      const newLv = Math.floor(linesRef.current / 10) + 1;
   418→      levelRef.current = newLv;
   419→      setScore(scoreRef.current);
   420→      setLevel(newLv);
   421→    }
   422→    if (cleared === 0) {
   423→      spawnPiece();
   424→    }
   425→  }, [lockPiece, clearLines, spawnPiece]);
   426→
   427→  /* ==================== 计算Ghost位置 ==================== */
   428→  const getGhostY = useCallback((): number => {
   429→    const piece = currentRef.current;
   430→    if (!piece) return 0;
   431→    const board = boardRef.current;
   432→    let ghostY = piece.y;
   433→    while (isValid(piece.shape, board, piece.x, ghostY + 1)) {
   434→      ghostY += 1;
   435→    }
   436→    return ghostY;
   437→  }, []);
   438→
   439→  /* ==================== Canvas rendering ==================== */
   440→  const draw = useCallback(
   441→    (ctx: CanvasRenderingContext2D, now: number) => {
   442→      const s = scaleRef.current;
   443→      const cw = COLS * CELL;
   444→      const ch = ROWS * CELL;
   445→      const board = boardRef.current;
   446→      const piece = currentRef.current;
   447→
   448→      // 清空画布
   449→      ctx.clearRect(0, 0, ctx.canvas.width, ctx.canvas.height);
   450→      ctx.save();
   451→      ctx.scale(s, s);
   452→
   453→      // 深色背景
   454→      ctx.fillStyle = '#1a1a2e';
   455→      ctx.fillRect(0, 0, cw, ch);
   456→
   457→      // 网格线
   458→      ctx.strokeStyle = 'rgba(255,255,255,0.06)';
   459→      ctx.lineWidth = 0.5;
   460→      for (let r = 0; r <= ROWS; r++) {
   461→        ctx.beginPath();
   462→        ctx.moveTo(0, r * CELL);
   463→        ctx.lineTo(cw, r * CELL);
   464→        ctx.stroke();
   465→      }
   466→      for (let c = 0; c <= COLS; c++) {
   467→        ctx.beginPath();
   468→        ctx.moveTo(c * CELL, 0);
   469→        ctx.lineTo(c * CELL, ch);
   470→        ctx.stroke();
   471→      }
   472→
   473→      // 绘制已固定的方块
   474→      for (let r = 0; r < ROWS; r++) {
   475→        for (let c = 0; c < COLS; c++) {
   476→          if (board[r][c] !== null) {
   477→            drawCell(ctx, c, r, board[r][c]!, 1);
   478→          }
   479→        }
   480→      }
   481→
   482→      // 消行闪白动画
   483→      if (flashRowsRef.current.length > 0) {
   484→        const elapsed = now - flashStartRef.current;
   485→        const flashDuration = 300; // 300ms闪白
   486→        if (elapsed < flashDuration) {
   487→          const alpha = 0.5 + 0.5 * Math.sin((elapsed / flashDuration) * Math.PI * 4);
   488→          ctx.fillStyle = `rgba(255,255,255,${alpha})`;
   489→          for (const row of flashRowsRef.current) {
   490→            ctx.fillRect(0, row * CELL, cw, CELL);
   491→          }
   492→        } else {
   493→          // 动画结束，执行实际消行
   494→          removeLines();
   495→          spawnPiece();
   496→        }
   497→      }
   498→
   499→      // 绘制Ghost方块
   500→      if (piece && statusRef.current === 'playing' && flashRowsRef.current.length === 0) {
   501→        const ghostY = getGhostY();
   502→        if (ghostY !== piece.y) {
   503→          for (let r = 0; r < piece.shape.length; r++) {
   504→            for (let c = 0; c < piece.shape[r].length; c++) {
   505→              if (piece.shape[r][c]) {
   506→                const px = (piece.x + c) * CELL;
   507→                const py = (ghostY + r) * CELL;
   508→                ctx.strokeStyle = COLORS[piece.type];
   509→                ctx.globalAlpha = 0.3;
   510→                ctx.lineWidth = 2;
   511→                ctx.strokeRect(px + 1, py + 1, CELL - 2, CELL - 2);
   512→                ctx.globalAlpha = 0.08;
   513→                ctx.fillStyle = COLORS[piece.type];
   514→                ctx.fillRect(px + 1, py + 1, CELL - 2, CELL - 2);
   515→                ctx.globalAlpha = 1;
   516→              }
   517→            }
   518→          }
   519→        }
   520→      }
   521→
   522→      // 绘制当前方块
   523→      if (piece && statusRef.current === 'playing') {
   524→        for (let r = 0; r < piece.shape.length; r++) {
   525→          for (let c = 0; c < piece.shape[r].length; c++) {
   526→            if (piece.shape[r][c]) {
   527→              drawCell(ctx, piece.x + c, piece.y + r, piece.type, 1);
   528→            }
   529→          }
   530→        }
   531→      }
   532→
   533→      ctx.restore();
   534→
   535→      /* --- 绘制下一个方块预览（在主Canvas右侧） --- */
   536→      const previewX = cw + 16;
   537→      const previewY = 8;
   538→      const nType = nextRef.current;
   539→      const nShape = SHAPES[nType];
   540→      const pw = 4 * PREVIEW_CELL;
   541→      const ph = 4 * PREVIEW_CELL;
   542→
   543→      // 预览区背景
   544→      ctx.fillStyle = 'rgba(255,255,255,0.05)';
   545→      ctx.beginPath();
   546→      ctx.roundRect(previewX - 8, previewY - 4, pw + 16, ph + 24, 8);
   547→      ctx.fill();
   548→
   549→      // "NEXT" 标签
   550→      ctx.fillStyle = 'rgba(255,255,255,0.6)';
   551→      ctx.font = `${11 * s}px sans-serif`;
   552→      ctx.fillText('下一个', previewX, previewY + 10);
   553→
   554→      // 计算居中偏移
   555→      const ox = previewX + (pw - nShape[0].length * PREVIEW_CELL) / 2;
   556→      const oy = previewY + 20 + (ph - nShape.length * PREVIEW_CELL) / 2;
   557→      for (let r = 0; r < nShape.length; r++) {
   558→        for (let c = 0; c < nShape[r].length; c++) {
   559→          if (nShape[r][c]) {
   560→            const px = ox + c * PREVIEW_CELL;
   561→            const py = oy + r * PREVIEW_CELL;
   562→            ctx.fillStyle = COLORS[nType];
   563→            ctx.fillRect(px + 1, py + 1, PREVIEW_CELL - 2, PREVIEW_CELL - 2);
   564→            ctx.strokeStyle = STROKE_COLORS[nType];
   565→            ctx.lineWidth = 1.5;
   566→            ctx.strokeRect(px + 1, py + 1, PREVIEW_CELL - 2, PREVIEW_CELL - 2);
   567→            // 高光效果
   568→            ctx.fillStyle = HIGHLIGHT_COLORS[nType];
   569→            ctx.globalAlpha = 0.4;
   570→            ctx.fillRect(px + 2, py + 2, PREVIEW_CELL - 6, 4);
   571→            ctx.globalAlpha = 1;
   572→          }
   573→        }
   574→      }
   575→
   576→      /* --- Paused overlay --- */
   577→      if (pausedRef.current && statusRef.current === 'playing') {
   578→        ctx.fillStyle = 'rgba(0,0,0,0.6)';
   579→        ctx.fillRect(0, 0, ctx.canvas.width, ctx.canvas.height);
   580→        ctx.fillStyle = '#ffffff';
   581→        ctx.font = `bold ${28 * s}px sans-serif`;
   582→        ctx.textAlign = 'center';
   583→        ctx.fillText('已暂停', ctx.canvas.width / 2, ctx.canvas.height / 2 - 10 * s);
   584→        ctx.fillStyle = 'rgba(255,255,255,0.6)';
   585→        ctx.font = `${14 * s}px sans-serif`;
   586→        ctx.fillText('按 Esc 继续', ctx.canvas.width / 2, ctx.canvas.height / 2 + 16 * s);
   587→        ctx.textAlign = 'start';
   588→      }
   589→      /* --- Game over overlay --- */
   590→      if (statusRef.current === 'over') {
   591→        ctx.fillStyle = 'rgba(0,0,0,0.65)';
   592→        ctx.fillRect(0, 0, ctx.canvas.width, ctx.canvas.height);
   593→        ctx.fillStyle = '#ff4757';
   594→        ctx.font = `bold ${28 * s}px sans-serif`;
   595→        ctx.textAlign = 'center';
   596→        ctx.fillText('游戏结束', ctx.canvas.width / 2, ctx.canvas.height / 2 - 20 * s);
   597→        ctx.fillStyle = '#ffffff';
   598→        ctx.font = `${16 * s}px sans-serif`;
   599→        ctx.fillText(`得分: ${scoreRef.current}`, ctx.canvas.width / 2, ctx.canvas.height / 2 + 16 * s);
   600→        if (isNewRecord) {
   601→          ctx.fillStyle = '#ffd700';
   602→          ctx.font = `bold ${14 * s}px sans-serif`;
   603→          ctx.fillText('🎉 新纪录！', ctx.canvas.width / 2, ctx.canvas.height / 2 + 44 * s);
   604→        }
   605→        ctx.textAlign = 'start';
   606→      }
   607→    },
   608→    [getGhostY, removeLines, spawnPiece, isNewRecord]
   609→  );

   611→  /** 绘制单个方块单元格（带高光和描边效果） */
   612→  function drawCell(
   613→    ctx: CanvasRenderingContext2D,
   614→    x: number,
   615→    y: number,
   616→    type: string,
   617→    alpha: number
   618→  ) {
   619→    const px = x * CELL;
   620→    const py = y * CELL;
   621→    ctx.globalAlpha = alpha;
   622→    // 主体填充
   623→    ctx.fillStyle = COLORS[type];
   624→    ctx.fillRect(px + 1, py + 1, CELL - 2, CELL - 2);
   625→    // 描边
   626→    ctx.strokeStyle = STROKE_COLORS[type];
   627→    ctx.lineWidth = 1.5;
   628→    ctx.strokeRect(px + 1, py + 1, CELL - 2, CELL - 2);
   629→    // 顶部高光条
   630→    ctx.fillStyle = HIGHLIGHT_COLORS[type];
   631→    ctx.globalAlpha = alpha * 0.45;
   632→    ctx.fillRect(px + 3, py + 3, CELL - 6, 5);
   633→    // 左侧高光条
   634→    ctx.fillRect(px + 3, py + 3, 3, CELL - 6);
   635→    ctx.globalAlpha = 1;
   636→  }

   638→  /* ==================== Main game loop ==================== */
   639→  const gameLoop = useCallback(
   640→    (now: number) => {
   641→      if (statusRef.current === 'playing' && !pausedRef.current) {
   642→        // 处理自动下落（消行动画期间暂停下落）
   643→        if (flashRowsRef.current.length === 0) {
   644→          if (now - lastDropRef.current >= getDropInterval()) {
   645→            moveDown();
   646→            lastDropRef.current = now;
   647→          }
   648→        }
   649→      }
   650→
   651→      // Submit score on game over
   652→      if (
   653→        statusRef.current === 'over' &&
   654→        !gameOverCheckedRef.current
   655→      ) {
   656→        gameOverCheckedRef.current = true;
   657→        submitScore(scoreRef.current);
   658→      }
   659→
   660→      // 绘制
   661→      const canvas = canvasRef.current;
   662→      if (canvas) {
   663→        const ctx = canvas.getContext('2d');
   664→        if (ctx) draw(ctx, now);
   665→      }
   666→
   667→      rafRef.current = requestAnimationFrame(gameLoop);
   668→    },
   669→    [draw, getDropInterval, moveDown]
   670→  );
   671→
   672→  /* ==================== 提交分数到API ==================== */
   673→  const submitScore = useCallback(async (finalScore: number) => {
   674→    try {
   675→      const res = await fetch('/api/games/scores', {
   676→        method: 'POST',
   677→        headers: { 'Content-Type': 'application/json' },
   678→        body: JSON.stringify({ game: GAME_NAME, score: finalScore }),
   679→      });
   680→      const data = await res.json();
   681→      if (data.data) {
   682→        const newHigh = data.data.score;
   683→        setHighScore(newHigh);
   684→        if (newHigh === finalScore && finalScore > 0) {
   685→          setIsNewRecord(true);
   686→        }
   687→      }
   688→    } catch {
   689→      // 静默处理
   690→    }
   691→  }, []);
   692→
   693→  /* ==================== 加载最高分 ==================== */
   694→  useEffect(() => {
   695→    async function load() {
   696→      try {
   697→        const res = await fetch('/api/games/scores');
   698→        const data = await res.json();
   699→        if (data.data && Array.isArray(data.data)) {
   700→          const entry = data.data.find(
   701→            (e: { game: string; score: number }) => e.game === GAME_NAME
   702→          );
   703→          if (entry) setHighScore(entry.score);
   704→        }
   705→      } catch {
   706→        // 静默处理
   707→      } finally {
   708→        setLoadingHighScore(false);
   709→      }
   710→    }
   711→    load();
   712→  }, []);
   713→
   714→  /* ==================== Canvas尺寸设置（高DPI） ==================== */
   715→  useEffect(() => {
   716→    const canvas = canvasRef.current;
   717→    if (!canvas) return;
   718→    const s = computeScale();
   719→    const dpr = window.devicePixelRatio || 1;
   720→    // 主区域宽度 + 预览区宽度
   721→    const logicW = COLS * CELL + 4 * PREVIEW_CELL + 40;
   722→    const logicH = ROWS * CELL;
   723→    canvas.width = logicW * s * dpr;
   724→    canvas.height = logicH * s * dpr;
   725→    canvas.style.width = `${logicW * s}px`;
   726→    canvas.style.height = `${logicH * s}px`;
   727→    const ctx = canvas.getContext('2d');
   728→    if (ctx) ctx.scale(dpr, dpr);
   729→  }, [computeScale, isMobile, gameStatus]);
   730→
   731→  /* ==================== 游戏循环启停 ==================== */
   732→  useEffect(() => {
   733→    if (gameStatus === 'playing') {
   734→      lastDropRef.current = performance.now();
   735→      rafRef.current = requestAnimationFrame(gameLoop);
   736→    }
   737→    return () => {
   738→      if (rafRef.current) {
   739→        cancelAnimationFrame(rafRef.current);
   740→        rafRef.current = null;
   741→      }
   742→    };
   743→  }, [gameStatus, gameLoop]);
   744→
   745→  /* ==================== Keyboard events ==================== */
   746→  useEffect(() => {
   747→    if (gameStatus !== 'playing') return;
   748→    const handleKeyDown = (e: KeyboardEvent) => {
   749→      if (e.key === 'Escape') {
   750→        e.preventDefault();
   751→        pausedRef.current = !pausedRef.current;
   752→        setPaused(pausedRef.current);
   753→        return;
   754→      }
   755→      if (pausedRef.current) return;
   756→      switch (e.key) {
   757→        case 'ArrowLeft':
   758→          e.preventDefault();
   759→          moveLeft();
   760→          break;
   761→        case 'ArrowRight':
   762→          e.preventDefault();
   763→          moveRight();
   764→          break;
   765→        case 'ArrowDown':
   766→          e.preventDefault();
   767→          moveDown();
   768→          lastDropRef.current = performance.now();
   769→          break;
   770→        case 'ArrowUp':
   771→          e.preventDefault();
   772→          rotate();
   773→          break;
   774→        case ' ':
   775→          e.preventDefault();
   776→          hardDrop();
   777→          break;
   778→      }
   779→    };
   780→    window.addEventListener('keydown', handleKeyDown);
   781→    return () => window.removeEventListener('keydown', handleKeyDown);
   782→  }, [gameStatus, moveLeft, moveRight, moveDown, rotate, hardDrop]);
   783→
   784→  /* ==================== 开始游戏 ==================== */
   785→  const startGame = useCallback(() => {
   786→    resetBoard();
   787→    scoreRef.current = 0;
   788→    levelRef.current = 1;
   789→    linesRef.current = 0;
   790→    speedRef.current = speed;
   791→    flashRowsRef.current = [];
   792→    gameOverCheckedRef.current = false;
   793→    pausedRef.current = false;
   794→    setPaused(false);
   795→    setIsNewRecord(false);
   796→    setScore(0);
   797→    setLevel(1);
   798→    nextRef.current = randomType();
   799→    spawnPiece();
   800→    statusRef.current = 'playing';
   801→    setGameStatus('playing');
   802→  }, [speed, resetBoard, spawnPiece]);
   803→
   804→  /* ==================== 重新开始 ==================== */
   805→  const restartGame = useCallback(() => {
   806→    startGame();
   807→  }, [startGame]);
   808→
   809→  /* ==================== 设置面板 ==================== */
   810→  if (gameStatus === 'idle') {
   811→    return (
   812→      <div className="flex flex-col items-center w-full min-h-[70vh] gap-4 px-2 py-2">
   813→        <div className="w-full max-w-md">
   814→          <Button variant="ghost" onClick={onBack} className="mb-2">
   815→            <ArrowLeft className="h-5 w-5 mr-2" />
   816→            返回
   817→          </Button>
   818→          <Card className="bg-card border-border">
   819→            <CardContent className="p-4 space-y-5">
   820→              <div className="text-center">
   821→                <h2 className="text-2xl font-bold flex items-center justify-center gap-2">
   822→                  <span className="text-2xl">🟦</span> 俄罗斯方块
   823→                </h2>
   824→                {!loadingHighScore && highScore > 0 && (
   825→                  <p className="text-sm text-muted-foreground mt-1 flex items-center justify-center gap-1">
   826→                    <Trophy className="h-4 w-4 text-yellow-500" />
   827→                    最高分: {highScore}
   828→                  </p>
   829→                )}
   830→              </div>
   831→
   832→              <div className="space-y-2">
   833→                <p className="text-sm font-medium">速度设置</p>
   834→                <div className="grid grid-cols-3 gap-2">
   835→                  {(Object.entries(SPEED_CONFIG) as [SpeedKey, { label: string; base: number }][]).map(
   836→                    ([key, cfg]) => (
   837→                      <Button
   838→                        key={key}
   839→                        variant={speed === key ? 'default' : 'outline'}
   840→                        size="sm"
   841→                        className="w-full"
   842→                        onClick={() => setSpeed(key)}
   843→                      >
   844→                        {cfg.label}
   845→                      </Button>
   846→                    )
   847→                  )}
   848→                </div>
   849→              </div>
   850→
   851→              <div className="space-y-2">
   852→                <div className="flex items-center justify-between">
   853→                  <p className="text-sm font-medium">操作说明</p>
   854→                  <GameControlsHelp info={tetrisControlsInfo} variant="text" />
   855→                </div>
   856→              </div>
   857→
   858→              <Button onClick={startGame} className="w-full" size="lg">
   859→                开始游戏
   860→              </Button>
   861→            </CardContent>
   862→          </Card>
   863→        </div>
   864→      </div>
   865→    );
   866→  }
   867→
   868→  /* ==================== 游戏进行中/结束界面 ==================== */
   869→  return (
   870→    <div className="flex flex-col items-center w-full min-h-[70vh] gap-3 p-2">
   871→      {/* Top status bar */}
   872→      <div className="w-full max-w-md flex items-center justify-between px-2">
   873→        <Button variant="ghost" size="sm" onClick={onBack}>
   874→          <ArrowLeft className="h-4 w-4 mr-1" />
   875→          返回
   876→        </Button>
   877→        <div className="flex items-center gap-2">
   878→          <h2 className="text-lg font-bold flex items-center gap-1.5">
   879→            <span>🟦</span> 俄罗斯方块
   880→          </h2>
   881→          <GameControlsHelp info={tetrisControlsInfo} variant="button" />
   882→        </div>
   883→        <div className="flex items-center gap-3 text-sm">
   884→          <span className="flex items-center gap-1">
   885→            <Trophy className="h-4 w-4 text-yellow-500" />
   886→            {score}
   887→          </span>
   888→          <span className="text-muted-foreground">Lv.{level}</span>
   889→        </div>
   890→      </div>
   891→
   892→      {/* Canvas game area */}
   893→      <canvas
   894→        ref={canvasRef}
   895→        className="rounded-lg border border-border/50"
   896→        style={{
   897→          width: '100%',
   898→          maxWidth: 300,
   899→          aspectRatio: '1/2',
   900→          touchAction: 'none',
   901→        }}
   902→      />
   903→
   904→      {/* Mobile virtual buttons */}
   905→      {isMobile && (

   909→              variant="outline"
   910→              size="lg"
   911→              className="min-h-12 min-w-14 text-xl font-bold"
   912→              onTouchStart={(e) => {
   913→                e.preventDefault();
   914→                if (!pausedRef.current) moveLeft();
   915→              }}
   916→              disabled={gameStatus === 'over' || paused}
   917→            >
   918→              ←

   920→            <Button

   924→              onTouchStart={(e) => {

   937→              size="lg"

   940→                e.preventDefault();
   941→                if (!pausedRef.current) moveRight();

   944→            >
   945→              →
   946→            </Button>
   947→            <Button

   949→              size="lg"
   950→              className="min-h-12 min-w-14 text-base font-bold"

   987→      )}
   988→    </div>
   989→  );
